`google_iap_settings` -- real, typed bindings generated directly from the
`hashicorp/google` provider schema, in every SDK language.

Every tab below is a complete, runnable program, not a fragment,
real enough to save and run exactly as shown.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	iap "github.com/ubiquex/ubx-sdk-google/sdk/go/google/iap"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own settings"})

    		ubx.Resource(iap.Settings, "ci-runner", iap.SettingsConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Settings } from "jsr:@ubx/sdk-google/google/iap/settings";

    export default stack("payments", () => {
      intent({ summary: "payments own settings" });

      resource(Settings, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.iap import Settings, SettingsConfig

    def describe():
        ubx.intent("payments own settings")

        ubx.resource(Settings, "ci-runner", SettingsConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
