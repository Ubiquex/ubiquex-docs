`aws_lambda_resource_policy` attaches a real resource-based access policy directly to an `aws_lambda_function`, a lower-level, full-document alternative to granting individual permissions one at a time through `aws_lambda_permission`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	lambda "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/lambda"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resource policy"})

    		ubx.Resource(lambda.ResourcePolicy, "example", lambda.ResourcePolicyConfig{
    			PolicyDocument: "example",
    			ResourceArn:    "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { ResourcePolicy } from "@ubx/sdk-aws/aws/lambda/resource_policy";

    export default stack("example", () => {
      intent({ summary: "example own resource policy" });

      resource(ResourcePolicy, "example", {
        policyDocument: "example",
        resourceArn: "arn:aws:iam::123456789012:policy/example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.lambda_.resource_policy import ResourcePolicy, ResourcePolicyConfig

    def describe():
        ubx.intent("example own resource policy")

        ubx.resource(ResourcePolicy, "example", ResourcePolicyConfig(
            policy_document="example",
            resource_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
