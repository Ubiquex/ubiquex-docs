`aws_lambda_resource_policy` attaches a real resource-based access policy directly to an `aws_lambda_function`, a lower-level, full-document alternative to granting individual permissions one at a time through `aws_lambda_permission`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	lambda "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/lambda"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own resource policy"})

    		ubx.Resource(lambda.ResourcePolicy, "ci-runner", lambda.ResourcePolicyConfig{
    			PolicyDocument: "example",
    			ResourceArn:    "arn:aws:iam::123456789012:policy/ci",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ResourcePolicy } from "./local-sdk/aws/sdk/typescript/aws/lambda/resource_policy.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own resource policy" });

      resource(ResourcePolicy, "ci-runner", {
        policyDocument: "example",
        resourceArn: "arn:aws:iam::123456789012:policy/ci",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.lambda_ import ResourcePolicy, ResourcePolicyConfig

    def describe():
        ubx.intent("payments own resource policy")

        ubx.resource(ResourcePolicy, "ci-runner", ResourcePolicyConfig(
            policy_document="example",
            resource_arn="arn:aws:iam::123456789012:policy/ci",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
