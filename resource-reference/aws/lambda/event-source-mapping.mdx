---

`aws_lambda_event_source_mapping` creates a real subscription that polls a stream or queue, a Kinesis stream, a DynamoDB stream, an SQS queue, an MSK or self-managed Kafka topic, and invokes an `aws_lambda_function` with the batched records it reads.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	lambda "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/lambda"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own event source mapping"})

    		ubx.Resource(lambda.EventSourceMapping, "example", lambda.EventSourceMappingConfig{
    			FunctionName:                        "example-function-name",
    			AmazonManagedKafkaEventSourceConfig: map[string]any{"consumer_group_id": "example"},
    			BatchSize:                           1,
    			BisectBatchOnFunctionError:          true,
    			DestinationConfig:                   map[string]any{"on_failure": map[string]any{"destination": "example"}},
    			DocumentDbeventSourceConfig:         map[string]any{"collection_name": "example-collection-name"},
    			Enabled:                             true,
    			EventSourceArn:                      "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
