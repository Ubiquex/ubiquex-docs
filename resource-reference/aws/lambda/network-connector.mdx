`aws_lambda_network_connector` configures the real VPC networking, subnets and security groups, that an `aws_lambda_capacity_provider`'s underlying EC2 instances use to reach private resources under Lambda Managed Instances.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	lambda "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/lambda"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own network connector"})

    		ubx.Resource(lambda.NetworkConnector, "example", lambda.NetworkConnectorConfig{
    			Configuration: map[string]any{"vpc_egress_configuration": map[string]any{"associated_compute_resource_types": []string{"example"}, "subnet_ids": []string{"example"}}},
    			Name:          "example",
    			OperatorRole:  "example",
    			Tags:          []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.lambda_ import NetworkConnector, NetworkConnectorConfig
    from ubx.aws.lambda_.network_connector import NetworkConnector_Configuration, NetworkConnector_Configuration_VpcEgressConfiguration, NetworkConnector_Tags

    def describe():
        ubx.intent("example own network connector")

        ubx.resource(NetworkConnector, "example", NetworkConnectorConfig(
            configuration=NetworkConnector_Configuration(vpc_egress_configuration=NetworkConnector_Configuration_VpcEgressConfiguration(associated_compute_resource_types=["example"], subnet_ids=["example"])),
            name="example",
            operator_role="example",
            tags=[NetworkConnector_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="configuration" type="object" required>
  The network configuration for the connector. Specify a VpcEgressConfiguration to enable outbound traffic routing through your VPC. Required.
  <Expandable title="properties">
    <ResponseField name="vpc_egress_configuration" type="object" required>
      The VPC egress configuration for the network connector. Specifies the subnets, security groups, and network protocol for routing outbound traffic through your VPC. Required.
      <Expandable title="properties">
        <ResponseField name="associated_compute_resource_types" type="list(string)" required>
          The types of Lambda compute resources that can use this connector. Currently, only MicroVm is supported. Required.
        </ResponseField>
        <ResponseField name="network_protocol" type="string">
          The network protocol for the connector. Specify IPv4 for IPv4-only networking, or DualStack for both IPv4 and IPv6. Optional.
        </ResponseField>
        <ResponseField name="security_group_ids" type="list(string)">
          The IDs of the VPC security groups to attach to the ENIs. Specify 0 to 5 security groups. All security groups must be in the same VPC as the subnets. Optional.
        </ResponseField>
        <ResponseField name="subnet_ids" type="list(string)" required>
          The IDs of the VPC subnets where Lambda provisions elastic network interfaces (ENIs). Specify 1 to 16 subnets. All subnets must be in the same VPC. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="name" type="string">
  A unique name for the network connector within your account and Region. Must be 1 to 64 alphanumeric characters, hyphens, or underscores. Optional.
</ResponseField>

<ResponseField name="operator_role" type="string">
  The ARN of the IAM role that Lambda assumes to manage elastic network interfaces in your VPC. This role must have permissions for ec2:CreateNetworkInterface and related describe operations. Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  A list of tags to apply to the network connector. Use tags to categorize network connectors for cost allocation, access control, or operational management. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">

</ResponseField>

<ResponseField name="state" type="string">
  The current state of the network connector. Computed by AWS after `ship`.
</ResponseField>
