`aws_lambda_microvm_image` builds a real versioned image, a Dockerfile plus application artifacts uploaded to S3, that Lambda MicroVMs launch on demand as individually isolated, VM-level-sandboxed compute instances. This is a genuinely new AWS service: Lambda MicroVMs launched June 22, 2026, expanded from an initial five AWS Regions to ten as of August 2026, and is recent enough that not all of its behavior is covered by pre-2026 general knowledge.

## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	lambda "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/lambda"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own microvm image"})

    		ubx.Resource(lambda.MicrovmImage, "example", lambda.MicrovmImageConfig{
    			AdditionalOsCapabilities: []string{"example"},
    			BaseImageArn:             "arn:aws:iam::123456789012:policy/example",
    			BaseImageVersion:         "example",
    			BuildRoleArn:             "arn:aws:iam::123456789012:policy/example",
    			CodeArtifact:             map[string]any{"uri": "example"},
    			CpuConfigurations:        []map[string]any{{"architecture": "example"}},
    			Description:              "Managed by ubx.",
    			EgressNetworkConnectors:  []string{"example"},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.lambda_ import MicrovmImage, MicrovmImageConfig
    from ubx.aws.lambda_.microvm_image import MicrovmImage_CodeArtifact, MicrovmImage_CpuConfigurations

    def describe():
        ubx.intent("example own microvm image")

        ubx.resource(MicrovmImage, "example", MicrovmImageConfig(
            additional_os_capabilities=["example"],
            base_image_arn="arn:aws:iam::123456789012:policy/example",
            base_image_version="example",
            build_role_arn="arn:aws:iam::123456789012:policy/example",
            code_artifact=MicrovmImage_CodeArtifact(uri="example"),
            cpu_configurations=[MicrovmImage_CpuConfigurations(architecture="example")],
            description="Managed by ubx.",
            egress_network_connectors=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
