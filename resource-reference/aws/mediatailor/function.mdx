`aws_media_tailor_function` defines a real custom function, an HTTP-backed endpoint with configurable headers, that MediaTailor can invoke as part of a sequential executor chain to run custom logic during ad decisioning or playback configuration. This resource has only thin official documentation, verified live this session against AWS's own CloudFormation reference and field schema rather than assumed from general knowledge, since it did not turn up a dedicated conceptual guide page the way MediaTailor's older resources do.

    import ubx_sdk as ubx
    from ubx.aws.mediatailor import Function, FunctionConfig
    from ubx.aws.mediatailor.function import Function_CustomOutputConfiguration, Function_HttpRequestConfiguration, Function_SequentialExecutorConfiguration, Function_SequentialExecutorConfiguration_FunctionList, Function_Tags

    def describe():
        ubx.intent("example own function")

        ubx.resource(Function, "example", FunctionConfig(
            function_id="example",
            function_type="example",
            custom_output_configuration=Function_CustomOutputConfiguration(runtime="example"),
            description="Managed by ubx.",
            http_request_configuration=Function_HttpRequestConfiguration(method_type="example", request_timeout_milliseconds=1, runtime="example", url="example"),
            sequential_executor_configuration=Function_SequentialExecutorConfiguration(function_list=[Function_SequentialExecutorConfiguration_FunctionList(function_id="example")], runtime="example", timeout_milliseconds=1),
            tags=[Function_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
