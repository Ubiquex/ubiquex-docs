`aws_route53_resolver_resolver_query_logging_config_association` associates a VPC with an `aws_route53_resolver_resolver_query_logging_config`, the resource that actually turns query logging on for that specific VPC's DNS traffic.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	route53resolver "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/route53resolver"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resolver query logging config association"})

    		ubx.Resource(route53resolver.ResolverQueryLoggingConfigAssociation, "example", route53resolver.ResolverQueryLoggingConfigAssociationConfig{
    			ResolverQueryLogConfigId: "example",
    			ResourceId:               "example",
    		})
    	}))
    }
    ```
  </Tab>
