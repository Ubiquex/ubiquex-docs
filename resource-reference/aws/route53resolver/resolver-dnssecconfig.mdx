`aws_route53_resolver_resolver_dnssecconfig` configures whether Route 53 Resolver validates DNSSEC signatures for outbound queries a VPC sends through an `aws_route53_resolver_resolver_endpoint`, rejecting responses that fail validation.

## Example

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resolver dnssecconfig"})

    		ubx.Resource(route53resolver.ResolverDnssecconfig, "example", route53resolver.ResolverDnssecconfigConfig{
    			ResourceId: "example",
    		})
    	}))
    }
