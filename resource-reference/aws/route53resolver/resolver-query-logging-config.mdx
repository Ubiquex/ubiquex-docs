`aws_route53_resolver_resolver_query_logging_config` configures real logging of DNS queries resolved through an `aws_route53_resolver_resolver_endpoint` or a VPC's default resolver, to a CloudWatch Logs group, S3 bucket, or Kinesis Data Firehose.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	route53resolver "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/route53resolver"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resolver query logging config"})

    		ubx.Resource(route53resolver.ResolverQueryLoggingConfig, "example", route53resolver.ResolverQueryLoggingConfigConfig{
    			Name:           "example",
    			DestinationArn: "arn:aws:iam::123456789012:policy/example",
    			Tags:           []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
