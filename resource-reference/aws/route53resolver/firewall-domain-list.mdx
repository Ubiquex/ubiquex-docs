`aws_route53_resolver_firewall_domain_list` defines a real list of domains that an `aws_route53_resolver_firewall_rule_group`'s rules match outbound DNS queries against, inside a VPC, distinct from the newer Global Resolver's own `aws_route53_global_resolver_firewall_domain_list`.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own firewall domain list"})

    			Domains:       []string{"example"},
    			Tags:          []map[string]any{{"key": "example"}},
    		})
    	}))
    }
