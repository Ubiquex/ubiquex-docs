## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	route "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/route"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own table"})

    		ubx.Resource(route.Table, "ci-runner", route.TableConfig{
    			VpcId: "example",
    			Tags:  []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Table } from "./local-sdk/aws/sdk/typescript/aws/route/table.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own table" });

      resource(Table, "ci-runner", {
        vpcId: "example",
        tags: [{ key: "example" }],
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.route import Table, TableConfig

    def describe():
        ubx.intent("payments own table")

        ubx.resource(Table, "ci-runner", TableConfig(
            vpc_id="example",
            tags=[{"key": "example"}],
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Input properties

<ResponseField name="tags" type="list(object)">
  Any tags assigned to the route table. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a user-defined tag key associated with the route table, used to organize and identify the route table within AWS. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="vpc_id" type="string" required>
  The ID of the VPC. Required.
</ResponseField>

## Output properties

<ResponseField name="route_table_id" type="string">
  The unique identifier of the route table, assigned by AWS (e.g., rtb-xxxxxxxx). Computed by AWS after `ship`. **(AI-inferred)**
</ResponseField>
