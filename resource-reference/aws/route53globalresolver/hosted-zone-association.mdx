`aws_route53_global_resolver_hosted_zone_association` associates a real Route 53 private hosted zone with an `aws_route53_global_resolver_global_resolver`, so its records become resolvable through Global Resolver's own anycast endpoints for authorized external clients.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	route53globalresolver "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/route53globalresolver"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own global resolver hosted zone association"})

    		ubx.Resource(route53globalresolver.HostedZoneAssociation, "example", route53globalresolver.HostedZoneAssociationConfig{
    			HostedZoneId: "example",
    			Name:         "example",
    			ResourceArn:  "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
