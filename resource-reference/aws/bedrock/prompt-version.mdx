`aws_bedrock_prompt_version` snapshots an `aws_bedrock_prompt`'s content at a point in time, the real, versioned target that gets deployed and referenced once a prompt has been tested and is ready for production use.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	bedrock "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/bedrock"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own prompt version"})

    		ubx.Resource(bedrock.PromptVersion, "example", bedrock.PromptVersionConfig{
    			PromptArn:   "arn:aws:iam::123456789012:policy/example",
    			Description: "Managed by ubx.",
    			Tags:        "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.bedrock import PromptVersion, PromptVersionConfig

    def describe():
        ubx.intent("example own prompt version")

        ubx.resource(PromptVersion, "example", PromptVersionConfig(
            prompt_arn="arn:aws:iam::123456789012:policy/example",
            description="Managed by ubx.",
            tags="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
