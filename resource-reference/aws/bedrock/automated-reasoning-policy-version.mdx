`aws_bedrock_automated_reasoning_policy_version` publishes a real, immutable version of an `aws_bedrock_automated_reasoning_policy`'s formal logic rules, letting a guardrail pin to a specific, unchanging version rather than the policy's current, editable definition.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    		ubx.Resource(bedrock.AutomatedReasoningPolicyVersion, "example", bedrock.AutomatedReasoningPolicyVersionConfig{
    			PolicyArn:                 "arn:aws:iam::123456789012:policy/example",
    			LastUpdatedDefinitionHash: "example",
    			Tags:                      []map[string]any{{"key": "example"}},
    		})
    	}))
    }
