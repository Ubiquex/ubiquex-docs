`aws_bedrock_flow_alias` points at a specific `aws_bedrock_flow_version`, letting a caller roll back a flow or A/B test two versions without changing whatever's invoking the flow by its alias.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	bedrock "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/bedrock"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own flow alias"})

    		ubx.Resource(bedrock.FlowAlias, "example", bedrock.FlowAliasConfig{
    			FlowArn:                  "arn:aws:iam::123456789012:policy/example",
    			Name:                     "example",
    			RoutingConfiguration:     []map[string]any{{"flow_version": "example"}},
    			ConcurrencyConfiguration: map[string]any{"type": "example"},
    			Description:              "Managed by ubx.",
    			Tags:                     "example",
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    import ubx_sdk as ubx
    from ubx.aws.bedrock import FlowAlias, FlowAliasConfig
    from ubx.aws.bedrock.flow_alias import FlowAlias_RoutingConfiguration, FlowAlias_ConcurrencyConfiguration

    def describe():
        ubx.intent("example own flow alias")

        ubx.resource(FlowAlias, "example", FlowAliasConfig(
            flow_arn="arn:aws:iam::123456789012:policy/example",
            name="example",
            routing_configuration=[FlowAlias_RoutingConfiguration(flow_version="example")],
            concurrency_configuration=FlowAlias_ConcurrencyConfiguration(type="example"),
            description="Managed by ubx.",
            tags="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="concurrency_configuration" type="object">
  Optional.
  <Expandable title="properties">
    <ResponseField name="max_concurrency" type="number">

    <ResponseField name="type" type="string" required>
      The concurrency configuration type for a Bedrock flow alias, which can be set to SINGLE to restrict the alias to one active flow invocation at a time or PARALLEL to allow multiple concurrent invocations, controlling how the alias handles runtime requests. Required. **(AI-inferred)**

</ResponseField>

<ResponseField name="routing_configuration" type="list(object)" required>
