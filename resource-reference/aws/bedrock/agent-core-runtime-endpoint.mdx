`aws_bedrock_agent_core_runtime_endpoint` is a stable, versioned entry point for invoking an `aws_bedrock_agent_core_runtime`, letting a caller manage and route to different runtime configurations by ARN rather than a single, unversioned invocation target.

## Example

    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	bedrock "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/bedrock"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own agent core runtime endpoint"})

    		ubx.Resource(bedrock.AgentCoreRuntimeEndpoint, "ci-runner", bedrock.AgentCoreRuntimeEndpointConfig{
    			AgentRuntimeId:      "example",
    			Name:                "ci-runner",
    			AgentRuntimeVersion: "example",
    			Description:         "Managed by ubx.",
    			Tags:                "example",
    		})
    	}))
    }
    ```

## Input properties

</ResponseField>

<ResponseField name="agent_runtime_version" type="string">
