`aws_bedrock_agent_core_configuration_bundle` packages a real, reusable set of settings that can be applied together to an `aws_bedrock_agent_core_runtime`, instead of configuring the same group of options individually on every runtime that needs them.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	bedrockagentcore "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/bedrockagentcore"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own agent core configuration bundle"})

    		ubx.Resource(bedrockagentcore.ConfigurationBundle, "example", bedrockagentcore.ConfigurationBundleConfig{
    			BundleName:    "example-bundle-name",
    			Components:    "example",
    			BranchName:    "example-branch-name",
    			CommitMessage: "example",
    			CreatedBy:     map[string]any{"name": "example-name"},
    			Description:   "Managed by ubx.",
    			KmsKeyArn:     "arn:aws:iam::123456789012:policy/example",
    			Tags:          []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

## Input properties

<ResponseField name="branch_name" type="string">
  The branch name for version tracking. Optional.
</ResponseField>

<ResponseField name="bundle_name" type="string" required>
  The name for the configuration bundle. Names must be unique within your account. Required.
</ResponseField>

<ResponseField name="commit_message" type="string">
  A commit message describing the version of the configuration bundle. Optional.
</ResponseField>

<ResponseField name="components" type="any" required>
  A map of component identifiers to their configurations. Required.
</ResponseField>

<ResponseField name="created_by" type="object">
  The source that created a configuration bundle version. Optional.
  <Expandable title="properties">
    <ResponseField name="arn" type="string">
      The Amazon Resource Name (ARN) of the source, if applicable. Optional.
    </ResponseField>
    <ResponseField name="name" type="string" required>
      The name of the source (for example, user, optimization-job, or system). Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="description" type="string">
  The description for the configuration bundle. Optional.
</ResponseField>

<ResponseField name="kms_key_arn" type="string">
  The ARN of the KMS key used to encrypt component configurations. Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags to assign to the configuration bundle. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="bundle_arn" type="string">
  The Amazon Resource Name (ARN) of the configuration bundle. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="bundle_id" type="string">
  The unique identifier of the configuration bundle. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="created_at" type="string">
  The timestamp when the configuration bundle was created. Computed by AWS after `ship`.
</ResponseField>

    </ResponseField>
    <ResponseField name="commit_message" type="string">

    </ResponseField>
    <ResponseField name="created_by" type="object">
      The source that created a configuration bundle version. Optional.
      <Expandable title="properties">
        <ResponseField name="arn" type="string">
          The Amazon Resource Name (ARN) of the source, if applicable. Optional.
        </ResponseField>

        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="parent_version_ids" type="list(string)">
      A list of parent version identifiers. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="updated_at" type="string">

</ResponseField>

<ResponseField name="version_id" type="string">
  The version identifier of the configuration bundle. Computed by AWS after `ship`.
</ResponseField>
