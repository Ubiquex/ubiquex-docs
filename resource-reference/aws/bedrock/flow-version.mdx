title: "aws_bedrock_flow_version"
description: "`aws_bedrock_flow_version` snapshots an `aws_bedrock_flow`'s definition at a point in time..."
---

`aws_bedrock_flow_version` snapshots an `aws_bedrock_flow`'s definition at a point in time, the real, immutable target an `aws_bedrock_flow_alias` actually points to and invokes.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	bedrock "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/bedrock"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own flow version"})

    		ubx.Resource(bedrock.FlowVersion, "example", bedrock.FlowVersionConfig{
    			FlowArn:     "arn:aws:iam::123456789012:policy/example",
    			Description: "Managed by ubx.",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { FlowVersion } from "@ubx/sdk-aws/aws/bedrock/flow_version";

    export default stack("example", () => {
      intent({ summary: "example own flow version" });

      resource(FlowVersion, "example", {
        flowArn: "arn:aws:iam::123456789012:policy/example",
        description: "Managed by ubx.",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.bedrock import FlowVersion, FlowVersionConfig

    def describe():
        ubx.intent("example own flow version")

        ubx.resource(FlowVersion, "example", FlowVersionConfig(
            flow_arn="arn:aws:iam::123456789012:policy/example",
            description="Managed by ubx.",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
