`aws_bedrock_data_source` connects an `aws_bedrock_knowledge_base` to where its real content actually lives, S3, Confluence, SharePoint, Salesforce, or a web source, the ingestion point a knowledge base has nothing to retrieve from without.

## Example

    	bedrock "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/bedrock"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own data source"})

    		ubx.Resource(bedrock.DataSource, "example", bedrock.DataSourceConfig{

    			VectorIngestionConfiguration:      map[string]any{"chunking_configuration": map[string]any{"chunking_strategy": "example"}},
    		})
    	}))
    }
