`aws_bedrock_agent_core_payment_connector` integrates an `aws_bedrock_agent_core_payment_manager` with one real external payment provider, Coinbase or Stripe, for instance, referencing a credential stored in AgentCore Identity so API keys and wallet secrets are never passed directly through the Payments API.

## Example

    package main

    import (

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    		ubx.Resource(bedrock.AgentCorePaymentConnector, "example", bedrock.AgentCorePaymentConnectorConfig{
    			ConnectorName:                    "example-connector-name",
    			ConnectorType:                    "example",
    			PaymentManagerId:                 "example",
    			CredentialProviderConfigurations: []map[string]any{{"coinbase_cdp": map[string]any{"credential_provider_arn": "example"}}},
    			Description:                      "Managed by ubx.",
    			ProvisionMode:                    "example",
    		})
    	}))
    }

## Input properties

<ResponseField name="connector_name" type="string" required>
  The name of the payment connector Required.
</ResponseField>

<ResponseField name="connector_type" type="string" required>
  Required.
</ResponseField>

<ResponseField name="credential_provider_configurations" type="list(object)">
  The credential provider configurations for the connector. Required when ProvisionMode is MANUAL or not specified. Empty for QUICK_CREATE until provisioning completes. Optional.
  <Expandable title="properties">
    <ResponseField name="coinbase_cdp" type="object">
      Optional; if omitted, computed by AWS.
      <Expandable title="properties">
        <ResponseField name="credential_provider_arn" type="string">
          The ARN of the credential provider (e.g., an AWS Secrets Manager secret or IAM role) that supplies the API credentials used by the Bedrock agent to authenticate with Coinbase CDP for payment operations. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="stripe_privy" type="object">
      Optional; if omitted, computed by AWS.
      <Expandable title="properties">
        <ResponseField name="credential_provider_arn" type="string">
          Optional; if omitted, computed by AWS.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="description" type="string">
  A description of the payment connector Optional.
</ResponseField>

<ResponseField name="payment_manager_id" type="string" required>

<ResponseField name="provision_mode" type="string">
  Optional.
