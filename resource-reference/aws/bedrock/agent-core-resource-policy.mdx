`aws_bedrock_agent_core_resource_policy` attaches a real resource-based access policy to an `aws_bedrock_agent_core_policy_engine`, controlling which principals, including from other accounts, can manage or evaluate against it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	"encoding/json"

    	bedrockagentcore "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/bedrockagentcore"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own agent core resource policy"})

    		accessPolicy, _ := json.Marshal(map[string]any{
    			"Version": "2012-10-17",
    			"Statement": []map[string]any{{
    				"Effect":   "Allow",
    				"Action":   "*",
    				"Resource": "*",
    			}},
    		})

    		ubx.Resource(bedrockagentcore.ResourcePolicy, "example", bedrockagentcore.ResourcePolicyConfig{
    			Policy:      string(accessPolicy),
    			ResourceArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```

  <Tab title="Python">
    ```python
    import json
    import ubx_sdk as ubx
    from ubx.aws.bedrockagentcore.resource_policy import ResourcePolicy, ResourcePolicyConfig

    def describe():
        ubx.intent("example own agent core resource policy")

        access_policy = json.dumps({
            "Version": "2012-10-17",
            "Statement": [{
                "Effect": "Allow",
                "Action": "*",
                "Resource": "*",
            }],
        })

        ubx.resource(ResourcePolicy, "example", ResourcePolicyConfig(
            policy=access_policy,
            resource_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="policy" type="string" required>
  The resource policy to create or update. Required.
</ResponseField>

<ResponseField name="resource_arn" type="string" required>
  The Amazon Resource Name (ARN) of the resource for which to create or update the resource policy. Required.
</ResponseField>
