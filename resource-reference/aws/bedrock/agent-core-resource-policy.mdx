`aws_bedrock_agent_core_resource_policy` attaches a real resource-based access policy to an `aws_bedrock_agent_core_policy_engine`, controlling which principals, including from other accounts, can manage or evaluate against it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	"encoding/json"

    	bedrock "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/bedrock"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own agent core resource policy"})

    		accessPolicy, _ := json.Marshal(map[string]any{
    			"Version": "2012-10-17",
    			"Statement": []map[string]any{{
    				"Effect":   "Allow",
    				"Action":   "*",
    				"Resource": "*",
    			}},
    		})

    		ubx.Resource(bedrock.AgentCoreResourcePolicy, "ci-runner", bedrock.AgentCoreResourcePolicyConfig{
    			Policy:      string(accessPolicy),
    			ResourceArn: "arn:aws:iam::123456789012:policy/ci",
    		})
    	}))
    }
    ```

    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import json
    import ubx_sdk as ubx
    from ubx.aws.bedrock import AgentCoreResourcePolicy, AgentCoreResourcePolicyConfig

    def describe():
        ubx.intent("payments own agent core resource policy")

        access_policy = json.dumps({
            "Version": "2012-10-17",
            "Statement": [{
                "Effect": "Allow",
                "Action": "*",
                "Resource": "*",
            }],
        })

        ubx.resource(AgentCoreResourcePolicy, "ci-runner", AgentCoreResourcePolicyConfig(
            policy=access_policy,
            resource_arn="arn:aws:iam::123456789012:policy/ci",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)

## Properties
