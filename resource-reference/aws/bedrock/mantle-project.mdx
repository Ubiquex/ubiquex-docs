`aws_bedrock_mantle_project` creates a real project under Amazon Bedrock's Project Mantle, a distributed inference engine that serves the latest open-weight models through an OpenAI-compatible endpoint (bedrock-mantle) alongside Bedrock's own native bedrock-runtime API. This is a genuinely new capability, so behavior here is verified live rather than assumed from general knowledge.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	bedrock "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/bedrock"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own mantle project"})

    		ubx.Resource(bedrock.MantleProject, "example", bedrock.MantleProjectConfig{
    			Name: "example",
    			Tags: []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
