`aws_bedrock_flow` visually links multiple foundation models, prompts, and AWS services into one generative AI workflow in Amazon Bedrock Prompt Flows, reducing what would otherwise be custom orchestration code to a defined graph of connected nodes.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	bedrock "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/bedrock"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own flow"})

    		ubx.Resource(bedrock.Flow, "example", bedrock.FlowConfig{
    			ExecutionRoleArn:         "arn:aws:iam::123456789012:policy/example",
    			Name:                     "example",
    			CustomerEncryptionKeyArn: "arn:aws:iam::123456789012:policy/example",
    			Definition:               map[string]any{"connections": []map[string]any{{"name": "example-name"}}},
    			DefinitionS3Location:     map[string]any{"bucket": "example"},
    			DefinitionString:         "example",
    			DefinitionSubstitutions:  "example",
    			Description:              "Managed by ubx.",
    		})
    	}))
    }
    ```
