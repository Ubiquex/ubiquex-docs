`aws_bedrock_blueprint` defines a real, reusable extraction schema, which fields to pull and how to structure them, that an `aws_bedrock_data_automation_project` applies when processing unstructured documents, images, or video into structured output.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	bedrock "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/bedrock"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own blueprint"})

    		ubx.Resource(bedrock.Blueprint, "example", bedrock.BlueprintConfig{
    			BlueprintName:        "example-blueprint-name",
    			Schema:               "example",
    			Type:                 "example",
    			KmsEncryptionContext: "example",
    			KmsKeyId:             "example",
    			Tags:                 []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.bedrock import Blueprint, BlueprintConfig
    from ubx.aws.bedrock.blueprint import Blueprint_Tags

    def describe():
        ubx.intent("example own blueprint")

        ubx.resource(Blueprint, "example", BlueprintConfig(
            blueprint_name="example-blueprint-name",
            schema="example",
            type="example",
            kms_encryption_context="example",
            kms_key_id="example",
            tags=[Blueprint_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
