`aws_bedrock_prompt` stores a reusable prompt template in Amazon Bedrock Prompt Management, created, evaluated, and shared through the console or API rather than hardcoded into application code, so the same tested prompt can be reused across an application.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	bedrock "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/bedrock"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own prompt"})

    		ubx.Resource(bedrock.Prompt, "example", bedrock.PromptConfig{
    			Name:                     "example",
    			CustomerEncryptionKeyArn: "arn:aws:iam::123456789012:policy/example",
    			DefaultVariant:           "example",
    			Description:              "Managed by ubx.",
    			Tags:                     "example",
    			Variants:                 []map[string]any{{"name": "example-name"}},
    		})
    	}))
    }
    ```
