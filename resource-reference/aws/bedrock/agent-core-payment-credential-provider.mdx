`aws_bedrock_agent_core_payment_credential_provider` registers real, stored payment credentials that an AgentCore agent can retrieve at runtime to complete a transaction through an `aws_bedrock_agent_core_payment_connector` or `aws_bedrock_agent_core_payment_manager`, without the credentials being hardcoded into the agent's own logic.

## Example

    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    import (
    	bedrock "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/bedrock"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own agent core payment credential provider"})

    		ubx.Resource(bedrock.AgentCorePaymentCredentialProvider, "ci-runner", bedrock.AgentCorePaymentCredentialProviderConfig{
    			CredentialProviderVendor:   "example",
    			Name:                       "ci-runner",
    			ProviderConfigurationInput: map[string]any{"coinbase_cdp_configuration": map[string]any{"api_key_id": "example"}},
    			Tags:                       []map[string]any{{"key": "example"}},
    		})
    	}))
    }
