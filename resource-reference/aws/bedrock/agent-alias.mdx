`aws_bedrock_agent_alias` points at a specific prepared version of an `aws_bedrock_agent`, letting a caller roll back or run two versions side by side for testing without disrupting whatever's currently invoking the agent in production.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	bedrock "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/bedrock"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own agent alias"})

    		ubx.Resource(bedrock.AgentAlias, "example", bedrock.AgentAliasConfig{
    			AgentAliasName:       "example-agent-alias-name",
    			AgentId:              "example",
    			Description:          "Managed by ubx.",
    			RoutingConfiguration: []map[string]any{{"agent_version": "example"}},
    			Tags:                 "example",
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    import ubx_sdk as ubx
    from ubx.aws.bedrock.agent_alias import AgentAlias, AgentAliasConfig

    def describe():
        ubx.intent("example own agent alias")

        ubx.resource(AgentAlias, "example", AgentAliasConfig(
            agent_alias_name="example-agent-alias-name",
            agent_id="example",
            description="Managed by ubx.",
            routing_configuration=[{"agent_version": "example"}],
            tags="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="agent_alias_name" type="string" required>
  Name for a resource. Required.
</ResponseField>

<ResponseField name="agent_id" type="string" required>
  Identifier for a resource. Required.
</ResponseField>

<ResponseField name="description" type="string">
  Description of the Resource. Optional.
</ResponseField>

<ResponseField name="routing_configuration" type="list(object)">
  Routing configuration for an Agent alias. Optional.
  <Expandable title="properties">
    <ResponseField name="agent_version" type="string">
      Specifies the version of the Bedrock agent (such as 'DRAFT' or a numeric version) that this routing configuration entry directs alias traffic to. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tags" type="any">
  A map of tag keys and values Optional.
</ResponseField>

## Output properties

    <ResponseField name="routing_configuration" type="list(object)">
      The routing configuration at the time of this history event, listing the agent versions to which the alias directed traffic. Optional; if omitted, computed by AWS. **(AI-inferred)**
      <Expandable title="properties">
        <ResponseField name="agent_version" type="string">
          The version of the Bedrock agent (for example, 'DRAFT' or a numeric version) that the routing configuration entry associated with this alias history event directs traffic to. Optional; if omitted, computed by AWS. **(AI-inferred)**

    </ResponseField>
    <ResponseField name="start_date" type="string">

</ResponseField>
