`aws_wafv2_regex_pattern_set` defines a real set of regular expressions that an `aws_wafv2_rule_group` or `aws_wafv2_web_acl` rule can match a request part against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	wafv2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/wafv2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own regex pattern set"})

    		ubx.Resource(wafv2.RegexPatternSet, "example", wafv2.RegexPatternSetConfig{
    			RegularExpressionList: []string{"example"},
    			Scope:                 "example",
    			Name:                  "example",
    			Description:           "Managed by ubx.",
    			Tags:                  []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
