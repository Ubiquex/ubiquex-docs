`aws_pcaconnector_ad_service_principal_name` registers a real Kerberos service principal name in Active Directory on behalf of an `aws_pcaconnector_ad_connector`, letting the connector authenticate to the domain the way a domain-joined service would.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	pcaconnectorad "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/pcaconnectorad"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own ad service principal name"})

    		ubx.Resource(pcaconnectorad.ServicePrincipalName, "example", pcaconnectorad.ServicePrincipalNameConfig{
    			ConnectorArn:             "arn:aws:iam::123456789012:policy/example",
    			DirectoryRegistrationArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
