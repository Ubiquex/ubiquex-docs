`aws_redshift_serverless_snapshot` creates a real point-in-time snapshot of an `aws_redshift_serverless_namespace`'s data, which can later be restored into a new namespace.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	redshiftserverless "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/redshiftserverless"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own snapshot"})

    		ubx.Resource(redshiftserverless.Snapshot, "example", redshiftserverless.SnapshotConfig{
    			SnapshotName:    "example-snapshot-name",
    			NamespaceName:   "example-namespace-name",
    			RetentionPeriod: 1,
    			Tags:            []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.redshiftserverless import Snapshot, SnapshotConfig

        ubx.intent("example own snapshot")
