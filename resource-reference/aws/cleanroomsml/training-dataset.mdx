`aws_clean_rooms_ml_training_dataset` points AWS Clean Rooms ML at the real AWS Glue table, and its user, item, and timestamp columns, used to train a lookalike model, Clean Rooms ML reads metadata about the location rather than copying the underlying training data itself.

    import ubx_sdk as ubx
    from ubx.aws.cleanroomsml import TrainingDataset, TrainingDatasetConfig
    from ubx.aws.cleanroomsml.training_dataset import TrainingDataset_TrainingData, TrainingDataset_TrainingData_InputConfig, TrainingDataset_TrainingData_InputConfig_DataSource, TrainingDataset_TrainingData_InputConfig_DataSource_GlueDataSource, TrainingDataset_Tags

    def describe():
        ubx.intent("example own training dataset")

        ubx.resource(TrainingDataset, "example", TrainingDatasetConfig(
            name="example",
            role_arn="arn:aws:iam::123456789012:policy/example",
            training_data=[TrainingDataset_TrainingData(input_config=TrainingDataset_TrainingData_InputConfig(data_source=TrainingDataset_TrainingData_InputConfig_DataSource(glue_data_source=TrainingDataset_TrainingData_InputConfig_DataSource_GlueDataSource(catalog_id="example"))))],
            description="Managed by ubx.",
            tags=[TrainingDataset_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
