`aws_rekognition_project` creates a real Rekognition Custom Labels project, the container that groups the `aws_rekognition_dataset`s and trained model versions built to detect a customer's own custom labels rather than Rekognition's built-in label set.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	rekognition "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/rekognition"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own project"})

    		ubx.Resource(rekognition.Project, "ci-runner", rekognition.ProjectConfig{
    			ProjectName: "example-project-name",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.rekognition import Project, ProjectConfig

    def describe():
        ubx.intent("payments own project")

        ubx.resource(Project, "ci-runner", ProjectConfig(
            project_name="example-project-name",
            tags=[{"key": "example"}],
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
