`aws_rekognition_collection` creates a real container of indexed face vectors that Rekognition's face-search operations compare newly detected faces against, independent of any specific `aws_rekognition_project`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	rekognition "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/rekognition"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own collection"})

    		ubx.Resource(rekognition.Collection, "ci-runner", rekognition.CollectionConfig{
    			CollectionId: "example",
    			Tags:         []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
