`aws_rekognition_dataset` imports a real set of labeled training or test images into an `aws_rekognition_project`, the data Custom Labels actually trains and evaluates a model against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	rekognition "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/rekognition"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dataset"})

    		ubx.Resource(rekognition.Dataset, "example", rekognition.DatasetConfig{
    			DatasetType: "example",
    			ProjectArn:  "arn:aws:iam::123456789012:policy/example",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.rekognition import Dataset, DatasetConfig
    from ubx.aws.rekognition.dataset import Dataset_Tags

    def describe():
        ubx.intent("example own dataset")

        ubx.resource(Dataset, "example", DatasetConfig(
            dataset_type="example",
            project_arn="arn:aws:iam::123456789012:policy/example",
            tags=[Dataset_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="dataset_type" type="string" required>
  The type of the dataset. Specify TRAIN to create a training dataset. Specify TEST to create a test dataset. Required.
</ResponseField>

<ResponseField name="project_arn" type="string">
  The ARN of the project to which the dataset belongs. Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An array of key-value pairs to apply to this resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The tag key (e.g., 'Name' or 'Project') that labels the Rekognition dataset, used for organizing and identifying the dataset in AWS. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a tag key attached to the AWS Rekognition Dataset, used for organizing, identifying, and managing the dataset within AWS. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="dataset_arn" type="string">
  The ARN of the dataset. Computed by AWS after `ship`.
</ResponseField>
