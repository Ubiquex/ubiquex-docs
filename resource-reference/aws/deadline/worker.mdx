`aws_deadline_worker` is a real, individual compute host, running the Deadline Cloud Worker Agent, that actually picks up and executes tasks from an `aws_deadline_queue` through its `aws_deadline_fleet`, reporting status and performance metrics back to the service as it works.

## Example

    ```go

    import (
    	deadline "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/deadline"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own worker"})

    		ubx.Resource(deadline.Worker, "example", deadline.WorkerConfig{
    			FarmId:         "example",
    			FleetId:        "example",
    			HostProperties: map[string]any{"host_name": "example-host-name"},
    			Tags:           []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
