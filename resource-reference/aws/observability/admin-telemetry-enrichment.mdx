`aws_observability_admin_telemetry_enrichment` configures real, additional AWS context, resource tags or account metadata, automatically attached to telemetry data governed by an `aws_observability_admin_telemetry_rule`, without each individual signal needing to carry that context itself.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	observability "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/observability"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own admin telemetry enrichment"})

    		ubx.Resource(observability.AdminTelemetryEnrichment, "ci-runner", observability.AdminTelemetryEnrichmentConfig{
    			Scope: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    ```

  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.observability import AdminTelemetryEnrichment, AdminTelemetryEnrichmentConfig

    def describe():
        ubx.intent("payments own admin telemetry enrichment")

        ubx.resource(AdminTelemetryEnrichment, "ci-runner", AdminTelemetryEnrichmentConfig(
            scope="example",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Input properties

<ResponseField name="scope" type="string" required>
  Scope of the Telemetry Enrichment Required.
</ResponseField>

## Output properties

<ResponseField name="status" type="string">
  Current status of the resource tags for telemetry feature (Running, Stopped, or Impaired). Computed by AWS after `ship`.
</ResponseField>
