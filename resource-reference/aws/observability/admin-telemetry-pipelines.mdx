`aws_observability_admin_telemetry_pipelines` defines a real telemetry pipeline under CloudWatch Observability Admin, describing how collected telemetry data is processed and routed on its way to a destination.

## Example

    import (
    	observability "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/observability"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own admin telemetry pipelines"})

    		ubx.Resource(observability.AdminTelemetryPipelines, "ci-runner", observability.AdminTelemetryPipelinesConfig{
    			Configuration: map[string]any{"body": "example"},
    			Name:          "ci-runner",
    			Tags:          []map[string]any{{"key": "example"}},
    		})
    	}))
    }
