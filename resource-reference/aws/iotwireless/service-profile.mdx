`aws_io_twireless_service_profile` is a real, reusable template of LoRaWAN network server behavior, applied to an `aws_io_twireless_wireless_device` alongside its own `aws_io_twireless_device_profile`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iotwireless "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/iotwireless"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own service profile"})

    		ubx.Resource(iotwireless.ServiceProfile, "example", iotwireless.ServiceProfileConfig{
    			Name: "example",
    			Tags: []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
