`aws_proton_service_template` defines a real, reusable service template, the application-level resources deployed on top of an `aws_proton_environment_template`'s shared infrastructure. Like the rest of AWS Proton, this is being discontinued, with support ending October 7, 2026, roughly a month and a half from now.

## Example

    ```go
    package main

    import (
    	proton "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/proton"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own service template"})

    		ubx.Resource(proton.ServiceTemplate, "example", proton.ServiceTemplateConfig{
    			Name:                 "example",
    			Description:          "Managed by ubx.",
    			DisplayName:          "example-display-name",
    			EncryptionKey:        "example",
    			PipelineProvisioning: "example",
    			Tags:                 []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.proton import ServiceTemplate, ServiceTemplateConfig
    from ubx.aws.proton.service_template import ServiceTemplate_Tags

    def describe():
        ubx.intent("example own service template")

        ubx.resource(ServiceTemplate, "example", ServiceTemplateConfig(
            name="example",
            description="Managed by ubx.",
            display_name="example-display-name",
            encryption_key="example",
            pipeline_provisioning="example",
            tags=[ServiceTemplate_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="description" type="string">
  &lt;p&gt;A description of the service template.&lt;/p&gt; Optional.
</ResponseField>

<ResponseField name="display_name" type="string">
  &lt;p&gt;The name of the service template as displayed in the developer interface.&lt;/p&gt; Optional.
</ResponseField>

<ResponseField name="encryption_key" type="string">
  &lt;p&gt;A customer provided encryption key that's used to encrypt data.&lt;/p&gt; Optional.
</ResponseField>

<ResponseField name="name" type="string">
  The unique name of the Proton service template. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="pipeline_provisioning" type="string">
  Determines whether the service instance pipeline is provisioned by AWS Proton or by the customer, with valid values CUSTOMER and PROTON. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="tags" type="list(object)">
  &lt;p&gt;An optional list of metadata items that you can associate with the Proton service template. A tag is a key-value pair.&lt;/p&gt; &lt;p&gt;For more information, see &lt;a href="https://docs.aws.amazon.com/proton/latest/userguide/resources.html"&gt;Proton resources and tagging&lt;/a&gt; in the &lt;i&gt;Proton User Guide&lt;/i&gt;.&lt;/p&gt; Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">

    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>
