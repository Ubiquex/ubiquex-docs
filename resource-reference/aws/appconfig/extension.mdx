`aws_app_config_extension` defines custom or AWS-authored behavior, notify SNS or EventBridge, run a Lambda check, that AppConfig invokes at specific lifecycle points (before a hosted configuration is created, or when a deployment starts, completes, or rolls back).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	appconfig "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/appconfig"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own extension"})

    		ubx.Resource(appconfig.Extension, "example", appconfig.ExtensionConfig{
    			Actions:             "example",
    			Name:                "example",
    			Description:         "Managed by ubx.",
    			LatestVersionNumber: 1,
    			Parameters:          "example",
    			Tags:                []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.appconfig import Extension, ExtensionConfig
    from ubx.aws.appconfig.extension import Extension_Tags

    def describe():
        ubx.intent("example own extension")

        ubx.resource(Extension, "example", ExtensionConfig(
            actions="example",
            name="example",
            description="Managed by ubx.",
            latest_version_number=1,
            parameters="example",
            tags=[Extension_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
