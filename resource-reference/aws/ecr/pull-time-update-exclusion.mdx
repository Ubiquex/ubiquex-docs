`aws_ecr_pull_time_update_exclusion` excludes a real, specific pull request, most commonly from an automated scanner or CI system, from updating an `aws_ecr_repository` image's own last-pulled timestamp, keeping that timestamp meaningful for real usage-based lifecycle policies.

## Example

    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	ecr "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/ecr"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own pull time update exclusion"})

    		ubx.Resource(ecr.PullTimeUpdateExclusion, "example", ecr.PullTimeUpdateExclusionConfig{
    			PrincipalArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
