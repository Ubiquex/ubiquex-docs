`aws_cognito_user_pool_user_to_group_attachment` is the real, actual membership grant linking an `aws_cognito_user_pool_user` to an `aws_cognito_user_pool_group`. Neither existing on its own makes a user a group member, this attachment does.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    			UserPoolId: "example",
    			Username:   "example",
    		})
    	}))
    }
