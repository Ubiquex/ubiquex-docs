`aws_cognito_terms` configures real terms of use and privacy policy document links for an `aws_cognito_user_pool_client`, shown on managed login's sign-up page in the language matching the user's own locale, with a defined fallback when no localized document exists.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cognito "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/cognito"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own terms"})

    		ubx.Resource(cognito.Terms, "example", cognito.TermsConfig{
    			Enforcement: "example",
    			Links:       "example",
    			TermsName:   "example-terms-name",
    			TermsSource: "example",
    			UserPoolId:  "example",
    			ClientId:    "example",
    		})
    	}))
    }
    ```
