`aws_cases_case_rule` automatically runs whenever an `aws_cases_case` is created or updated, matching conditions like a field value, a priority, or an SLA breach, and taking a real action, assigning an SLA, creating a task, sending an alert, without an agent doing it by hand.

## Example

    ```go
    package main

    import (
    	cases "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/cases"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own case rule"})

    		ubx.Resource(cases.CaseRule, "example", cases.CaseRuleConfig{
    			Name:        "example",
    			Rule:        map[string]any{"hidden": map[string]any{"conditions": []map[string]any{{"equal_to": map[string]any{"operand_one": map[string]any{"field_id": "example"}}}}}},
    			Description: "Managed by ubx.",
    			DomainId:    "example",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
