## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	outposts "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/outposts"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_outposts_connection"})

    		ubx.Data(outposts.Connection, "example", outposts.ConnectionConfig{
    			ConnectionId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
