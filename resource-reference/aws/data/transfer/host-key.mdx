---
title: "aws_transfer_host_key"
description: "`data_aws_transfer_host_key` looks up a real, already-existing `aws_transfer_host_key`."
---

It imports a real SSH host key into an `aws_transfer_server`, letting an organization present its own, previously trusted host key fingerprint to clients instead of the one Transfer Family generates by default.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	transfer "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/transfer"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own host key"})

    		ubx.Resource(transfer.HostKey, "example", transfer.HostKeyConfig{
    			ServerId:    "example",
    			Description: "Managed by ubx.",
    			HostKeyBody: "example",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.transfer import HostKey, HostKeyConfig
    from ubx.aws.transfer.host_key import HostKey_Tags

    def describe():
        ubx.intent("example own host key")

        ubx.resource(HostKey, "example", HostKeyConfig(
            server_id="example",
            description="Managed by ubx.",
            host_key_body="example",
            tags=[HostKey_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
