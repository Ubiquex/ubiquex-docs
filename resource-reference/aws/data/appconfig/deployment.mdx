## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	appconfig "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/appconfig"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_appconfig_deployment"})

    		ubx.Data(appconfig.Deployment, "example", appconfig.DeploymentConfig{
    			ApplicationId:    "example",
    			DeploymentNumber: 1,
    			EnvironmentId:    "example",
    		})
    	}))
    }
    ```

    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.appconfig import Deployment, DeploymentConfig

    def describe():
        ubx.intent("look up aws_appconfig_deployment")

        ubx.data(Deployment, "example", DeploymentConfig(
            application_id="example",
            deployment_number=1,
            environment_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
