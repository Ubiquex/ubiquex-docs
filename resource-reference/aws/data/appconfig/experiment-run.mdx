## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	appconfig "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/appconfig"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_appconfig_experiment_run"})

    		ubx.Data(appconfig.ExperimentRun, "example", appconfig.ExperimentRunConfig{
    			ApplicationIdentifier:          "example",
    			ExperimentDefinitionIdentifier: "example",
    			Run:                            1,
    		})
    	}))
    }
    ```
