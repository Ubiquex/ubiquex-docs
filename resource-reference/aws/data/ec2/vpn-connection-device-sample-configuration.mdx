`aws_ec2_vpn_connection_device_sample_configuration` reads a real, sample configuration file for a specific customer gateway device connecting to an AWS Site-to-Site VPN.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ec2_vpn_connection_device_sample_configuration"})
