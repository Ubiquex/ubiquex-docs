`aws_ec2_hosts` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ec2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/ec2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ec2_hosts"})

    		ubx.Data(ec2.Hosts, "example", ec2.HostsConfig{
    			Filter:     []map[string]any{{"name": "example-name"}},
    			HostIds:    []string{"example"},
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>

    from ubx.aws.data.ec2 import Hosts, HostsConfig

    def describe():
