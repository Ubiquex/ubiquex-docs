`aws_ec2_addresses` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

    ```go
    package main

    import (
    	ec2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/ec2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ec2_addresses"})

    		ubx.Data(ec2.Addresses, "example", ec2.AddressesConfig{
    			AllocationIds: []string{"example"},
    			DryRun:        true,
    			Filters:       []map[string]any{{"name": "example-name"}},
    			PublicIps:     []string{"example"},
    		})
    	}))
    }
    ```
