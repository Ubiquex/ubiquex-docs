`aws_ec2_image_attribute` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    ```go
    package main

    import (
    	ec2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/ec2"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ec2_image_attribute"})

    		ubx.Data(ec2.ImageAttribute, "example", ec2.ImageAttributeConfig{
    			Attribute: "example",
    			ImageId:   "example",
    			DryRun:    true,
    		})
    	}))
    }
    ```
