`aws_ec2_instance_types` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ec2_instance_types"})

    		ubx.Data(ec2.InstanceTypes, "example", ec2.InstanceTypesConfig{
    			DryRun:                     true,
    			Filters:                    []map[string]any{{"name": "example-name"}},
    			IncludeUnsupportedInRegion: true,
    			MaxResults:                 1,
    		})
    	}))
    }
