## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	lambda "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/lambda"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_lambda_tags"})

    		ubx.Data(lambda.Tags, "example", lambda.TagsConfig{
    			Resource: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.lambda_ import Tags, TagsConfig

    def describe():
        ubx.intent("look up aws_lambda_tags")

        ubx.data(Tags, "example", TagsConfig(

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
