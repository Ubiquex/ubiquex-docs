---
title: "aws_iam_user_policy"
description: "`data_aws_iam_user_policy` looks up a real, already-existing `aws_iam_user_policy`."
---

It attaches a real inline IAM policy directly to an `aws_iam_user`, embedded in the user itself rather than existing as a separate, reusable `aws_iam_policy` that could also be attached elsewhere.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iam "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/iam"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iam_user_policy"})

    		ubx.Data(iam.UserPolicy, "example", iam.UserPolicyConfig{
    			PolicyName: "example-policy-name",
    			UserName:   "example-user-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { UserPolicy } from "@ubx/sdk-aws/aws/data/iam/user_policy";

    export default stack("example", () => {
      intent({ summary: "look up aws_iam_user_policy" });

      data(UserPolicy, "example", {
        policyName: "example-policy-name",
        userName: "example-user-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.iam import UserPolicy, UserPolicyConfig

    def describe():
        ubx.intent("look up aws_iam_user_policy")

        ubx.data(UserPolicy, "example", UserPolicyConfig(
            policy_name="example-policy-name",
            user_name="example-user-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

<ResponseField name="user_name" type="string" required>
