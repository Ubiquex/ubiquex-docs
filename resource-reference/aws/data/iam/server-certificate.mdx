---
title: "aws_iam_server_certificate"
description: "`data_aws_iam_server_certificate` looks up a real, already-existing `aws_iam_server_certificate`."
---

It uploads a real X.509 certificate and private key to IAM for use with a Classic Load Balancer or CloudFront, IAM's original, legacy certificate storage from before AWS Certificate Manager existed to handle this instead.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iam "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/iam"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iam_server_certificate"})

    		ubx.Data(iam.ServerCertificate, "example", iam.ServerCertificateConfig{
    			ServerCertificateName: "example-server-certificate-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { ServerCertificate } from "@ubx/sdk-aws/aws/data/iam/server_certificate";

    export default stack("example", () => {
      intent({ summary: "look up aws_iam_server_certificate" });

      data(ServerCertificate, "example", {
        serverCertificateName: "example-server-certificate-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.iam import ServerCertificate, ServerCertificateConfig

    def describe():
        ubx.intent("look up aws_iam_server_certificate")

        ubx.data(ServerCertificate, "example", ServerCertificateConfig(
            server_certificate_name="example-server-certificate-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
