---
title: "aws_ecs_task_definition"
description: "`data_aws_ecs_task_definition` looks up a real, already-existing `aws_ecs_task_definition`."
---

It is the real blueprint, container images, CPU and memory, networking mode, IAM roles, that an `aws_ecs_service` or a standalone task actually launches from.

## Example

  <Tab title="Go">

    package main

    import (
    	ecs "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/ecs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own task definition"})

    		ubx.Resource(ecs.TaskDefinition, "example", ecs.TaskDefinitionConfig{
    			ContainerDefinitions:  []map[string]any{{"name": "example-name"}},
    			Cpu:                   "example",
    			EnableFaultInjection:  true,
    			EphemeralStorage:      map[string]any{"size_in_gi_b": 1},
    			ExecutionRoleArn:      "arn:aws:iam::123456789012:policy/example",
    			Family:                "example",
    			InferenceAccelerators: []map[string]any{{"device_name": "example-device-name"}},
    			IpcMode:               "example",
    		})
    	}))
    }

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.ecs import TaskDefinition, TaskDefinitionConfig
    from ubx.aws.ecs.task_definition import TaskDefinition_ContainerDefinitions, TaskDefinition_EphemeralStorage, TaskDefinition_InferenceAccelerators

    def describe():
        ubx.intent("example own task definition")

        ubx.resource(TaskDefinition, "example", TaskDefinitionConfig(
            container_definitions=[TaskDefinition_ContainerDefinitions(name="example-name")],
            cpu="example",
            enable_fault_injection=True,
            ephemeral_storage=TaskDefinition_EphemeralStorage(size_in_gi_b=1),
            execution_role_arn="arn:aws:iam::123456789012:policy/example",
            family="example",
            inference_accelerators=[TaskDefinition_InferenceAccelerators(device_name="example-device-name")],
            ipc_mode="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
