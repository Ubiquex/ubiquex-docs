    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_appfabric_ingestion_destinations"})

    		ubx.Data(appfabric.IngestionDestinations, "example", appfabric.IngestionDestinationsConfig{
    			AppBundleIdentifier: "example",
    			IngestionIdentifier: "example",
    			MaxResults:          1,
    		})
    	}))
    }

## Lookup arguments

<ResponseField name="app_bundle_identifier" type="string" required>
  The real, unique identifier of the app bundle this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="ingestion_identifier" type="string" required>
  The real, unique identifier of the ingestion this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="max_results" type="number">
  The maximum number of results to return in a single page. **(AI-inferred)** Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="ingestion_destinations" type="list(object)">
  The real ingestion destinations included in this result. **(AI-inferred)** Returned by the lookup.
</ResponseField>

<ResponseField name="next_token" type="string">
  An opaque pagination token from a previous response, requesting the next page of results. **(AI-inferred)** Returned by the lookup.
</ResponseField>
