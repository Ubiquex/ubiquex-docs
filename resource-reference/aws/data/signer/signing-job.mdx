## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	signer "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/signer"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_signer_signing_job"})

    		ubx.Data(signer.SigningJob, "example", signer.SigningJobConfig{
    			JobId: "example",
    		})
    	}))
    }
    ```
