---
title: "aws_signer_signing_profile"
description: "`data_aws_signer_signing_profile` looks up a real, already-existing `aws_signer_signing_profile`."
---

It creates a real, reusable code-signing configuration, its signing platform and certificate validity, that AWS Signer uses to produce a verifiable, tamper-evident signature over deployment artifacts such as Lambda code or IoT firmware.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	signer "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/signer"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own signing profile"})

    		ubx.Resource(signer.SigningProfile, "example", signer.SigningProfileConfig{
    			PlatformId:              "example",
    			SignatureValidityPeriod: map[string]any{"type": "example"},
    			Tags:                    []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.signer import SigningProfile, SigningProfileConfig
    from ubx.aws.signer.signing_profile import SigningProfile_SignatureValidityPeriod, SigningProfile_Tags

    def describe():
        ubx.intent("example own signing profile")

        ubx.resource(SigningProfile, "example", SigningProfileConfig(
            platform_id="example",
            signature_validity_period=SigningProfile_SignatureValidityPeriod(type="example"),
            tags=[SigningProfile_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
