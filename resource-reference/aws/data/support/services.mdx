`aws_support_services` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	support "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/support"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_support_services"})

    		ubx.Data(support.Services, "example", support.ServicesConfig{
    			Language:        "example",
    			ServiceCodeList: []string{"example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.support import Services, ServicesConfig

    def describe():
        ubx.intent("look up aws_support_services")

        ubx.data(Services, "example", ServicesConfig(
            language="example",
            service_code_list=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
