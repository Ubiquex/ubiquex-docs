<Tabs>

    ```go
    package main

    import (
    	datazone "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/datazone"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_datazone_subscription"})

    		ubx.Data(datazone.Subscription, "example", datazone.SubscriptionConfig{
    			DomainIdentifier: "example",
    			Identifier:       "example",
    		})
    	}))
    }
    ```
