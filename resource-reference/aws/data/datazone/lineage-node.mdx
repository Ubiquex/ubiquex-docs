## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	datazone "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/datazone"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_datazone_lineage_node"})

    		ubx.Data(datazone.LineageNode, "example", datazone.LineageNodeConfig{
    			DomainIdentifier: "example",
    			Identifier:       "example",
    		})
    	}))
    }
    ```
