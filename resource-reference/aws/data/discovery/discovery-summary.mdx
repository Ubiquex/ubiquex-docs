## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	discovery "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/discovery"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_discovery_discovery_summary"})

    		ubx.Data(discovery.DiscoverySummary, "example", discovery.DiscoverySummaryConfig{})
    	}))
    }
    ```
