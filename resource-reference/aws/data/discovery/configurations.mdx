`aws_discovery_configurations` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	discovery "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/discovery"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_discovery_configurations"})

    		ubx.Data(discovery.Configurations, "example", discovery.ConfigurationsConfig{
    			ConfigurationIds: []string{"example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Configurations } from "@ubx/sdk-aws/aws/data/discovery/configurations";

    export default stack("example", () => {
      intent({ summary: "look up aws_discovery_configurations" });

      data(Configurations, "example", {
        configurationIds: ["example"],
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.discovery import Configurations, ConfigurationsConfig

    def describe():
        ubx.intent("look up aws_discovery_configurations")

        ubx.data(Configurations, "example", ConfigurationsConfig(
            configuration_ids=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
