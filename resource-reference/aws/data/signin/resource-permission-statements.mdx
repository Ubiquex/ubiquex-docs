## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	signin "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/signin"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_signin_resource_permission_statements"})

    		ubx.Data(signin.ResourcePermissionStatements, "example", signin.ResourcePermissionStatementsConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```
