## Example

<Tabs>

    ```go
    package main

    import (
    	schemas "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/schemas"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_schemas_schema_versions"})

    		ubx.Data(schemas.SchemaVersions, "example", schemas.SchemaVersionsConfig{
    			RegistryName: "example-registry-name",
    			SchemaName:   "example-schema-name",
    			Limit:        1,
    		})
    	}))
    }
    ```
