`aws_schemas_schemas` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	schemas "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/schemas"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_schemas_schemas"})

    		ubx.Data(schemas.Schemas, "example", schemas.SchemasConfig{
    			RegistryName:     "example-registry-name",
    			Limit:            1,
    			SchemaNamePrefix: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Schemas } from "@ubx/sdk-aws/aws/data/schemas/schemas";

    export default stack("example", () => {
      intent({ summary: "look up aws_schemas_schemas" });

      data(Schemas, "example", {
        registryName: "example-registry-name",
        limit: 1,
        schemaNamePrefix: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.schemas import Schemas, SchemasConfig

    def describe():
        ubx.intent("look up aws_schemas_schemas")

        ubx.data(Schemas, "example", SchemasConfig(
            registry_name="example-registry-name",
            limit=1,
            schema_name_prefix="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
