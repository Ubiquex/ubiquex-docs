## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	states "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/states"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_states_map_run"})

    		ubx.Data(states.MapRun, "example", states.MapRunConfig{
    			MapRunArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.aws.data.states import MapRun, MapRunConfig

    def describe():
        ubx.intent("look up aws_states_map_run")

        ubx.data(MapRun, "example", MapRunConfig(
            map_run_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
