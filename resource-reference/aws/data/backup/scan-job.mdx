## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	backup "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/backup"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_backup_scan_job"})

    		ubx.Data(backup.ScanJob, "example", backup.ScanJobConfig{
    			ScanJobId: "example",
    		})
    	}))
    }
    ```
