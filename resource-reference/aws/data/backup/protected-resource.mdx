## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	backup "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/backup"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_backup_protected_resource"})

    		ubx.Data(backup.ProtectedResource, "example", backup.ProtectedResourceConfig{
    			ResourceArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
