## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	backup "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/backup"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_backup_report_job"})

    		ubx.Data(backup.ReportJob, "example", backup.ReportJobConfig{
    			ReportJobId: "example",
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.backup import ReportJob, ReportJobConfig

    def describe():
        ubx.intent("look up aws_backup_report_job")

        ubx.data(ReportJob, "example", ReportJobConfig(
            report_job_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
