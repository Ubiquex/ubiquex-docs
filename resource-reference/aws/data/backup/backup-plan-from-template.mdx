## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	backup "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/backup"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_backup_backup_plan_from_template"})

    		ubx.Data(backup.BackupPlanFromTemplate, "example", backup.BackupPlanFromTemplateConfig{
    			BackupPlanTemplateId: "example",
    		})
    	}))
    }
    ```
