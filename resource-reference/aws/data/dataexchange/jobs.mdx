    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_dataexchange_jobs"})

    		ubx.Data(dataexchange.Jobs, "example", dataexchange.JobsConfig{

    			MaxResults: 1,
    			RevisionId: "example",
    		})
    	}))
    }
