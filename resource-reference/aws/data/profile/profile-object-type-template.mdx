`aws_profile_profile_object_type_template` looks up a real, predefined object type template (such as for Salesforce or Zendesk data) provided by Customer Profiles.

## Example

  <Tab title="Go">

    import (
    	profile "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/profile"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_profile_profile_object_type_template"})

    		ubx.Data(profile.ProfileObjectTypeTemplate, "example", profile.ProfileObjectTypeTemplateConfig{
    			TemplateId: "example",
    		})
    	}))
    }
