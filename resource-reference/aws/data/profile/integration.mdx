## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	profile "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/profile"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_profile_integration"})

    		ubx.Data(profile.Integration, "example", profile.IntegrationConfig{
    			DomainName: "example-domain-name",
    			Uri:        "example",
    		})
    	}))
    }
    ```
