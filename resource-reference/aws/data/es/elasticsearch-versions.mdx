## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	es "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/es"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_es_elasticsearch_versions"})

    		ubx.Data(es.ElasticsearchVersions, "example", es.ElasticsearchVersionsConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { ElasticsearchVersions } from "@ubx/sdk-aws/aws/data/es/elasticsearch_versions";

    export default stack("example", () => {
      intent({ summary: "look up aws_es_elasticsearch_versions" });

      data(ElasticsearchVersions, "example", {
        maxResults: 1,
      });
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.es import ElasticsearchVersions, ElasticsearchVersionsConfig

    def describe():
        ubx.intent("look up aws_es_elasticsearch_versions")

        ubx.data(ElasticsearchVersions, "example", ElasticsearchVersionsConfig(
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
