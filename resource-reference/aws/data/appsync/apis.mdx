## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	appsync "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/appsync"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_appsync_apis"})

    		ubx.Data(appsync.Apis, "example", appsync.ApisConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.data.appsync import Apis, ApisConfig

    def describe():
        ubx.intent("look up aws_appsync_apis")

        ubx.data(Apis, "example", ApisConfig(
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
