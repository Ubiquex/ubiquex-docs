`aws_appsync_graphql_apis` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	appsync "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/appsync"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_appsync_graphql_apis"})

    		ubx.Data(appsync.GraphqlApis, "example", appsync.GraphqlApisConfig{
    			ApiType:    "example",
    			MaxResults: 1,
    			Owner:      "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.appsync import GraphqlApis, GraphqlApisConfig

    def describe():
        ubx.intent("look up aws_appsync_graphql_apis")

        ubx.data(GraphqlApis, "example", GraphqlApisConfig(
            api_type="example",
            max_results=1,
            owner="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
