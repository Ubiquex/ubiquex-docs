## Example

    package main

    import (
    	appsync "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/appsync"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_appsync_api_association"})

    		ubx.Data(appsync.ApiAssociation, "example", appsync.ApiAssociationConfig{
    			DomainName: "example-domain-name",
    		})
    	}))
    }
