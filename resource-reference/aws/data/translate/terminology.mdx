## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	translate "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/translate"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_translate_terminology"})

    		ubx.Data(translate.Terminology, "example", translate.TerminologyConfig{
    			Name:                  "example",
    			TerminologyDataFormat: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk

    export default stack("example", () => {
      intent({ summary: "look up aws_translate_terminology" });

      data(Terminology, "example", {
        name: "example",
        terminologyDataFormat: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.translate import Terminology, TerminologyConfig

    def describe():
        ubx.intent("look up aws_translate_terminology")

        ubx.data(Terminology, "example", TerminologyConfig(
            name="example",
            terminology_data_format="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
