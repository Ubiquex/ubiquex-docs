---
title: "aws_translate_parallel_data"
description: "`data_aws_translate_parallel_data` looks up a real, already-existing `aws_translate_parallel_data`."
---

It imports a real set of source-and-target sentence pairs that Amazon Translate uses to customize its own translation output, favoring an organization's preferred terminology and phrasing over the engine's generic default translations.

## Example

    import (
    	translate "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/translate"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own parallel data"})

    		ubx.Resource(translate.ParallelData, "example", translate.ParallelDataConfig{

    			ParallelDataConfig: map[string]any{"format": "example", "s3_uri": "example"},

    			Tags:               []map[string]any{{"key": "example"}},
    		})
    	}))
    }

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.translate import ParallelData, ParallelDataConfig
    from ubx.aws.translate.parallel_data import ParallelData_ParallelDataConfig, ParallelData_EncryptionKey, ParallelData_Tags

    def describe():
        ubx.intent("example own parallel data")

        ubx.resource(ParallelData, "example", ParallelDataConfig(
            name="example",
            parallel_data_config=ParallelData_ParallelDataConfig(format="example", s3_uri="example"),
            description="Managed by ubx.",
            encryption_key=ParallelData_EncryptionKey(id="example", type="example"),
            tags=[ParallelData_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
