## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	translate "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/translate"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_translate_languages"})

    		ubx.Data(translate.Languages, "example", translate.LanguagesConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.data.translate import Languages, LanguagesConfig

    def describe():
        ubx.intent("look up aws_translate_languages")

        ubx.data(Languages, "example", LanguagesConfig(
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="max_results" type="number">
  The maximum number of results to return in a single page. **(AI-inferred)** Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="display_language_code" type="string">
  The real code identifying the display language. **(AI-inferred)** Returned by the lookup.
</ResponseField>

<ResponseField name="languages" type="list(object)">
  The real languages included in this result. **(AI-inferred)** Returned by the lookup.
</ResponseField>

<ResponseField name="next_token" type="string">
  An opaque pagination token from a previous response, requesting the next page of results. **(AI-inferred)** Returned by the lookup.
</ResponseField>
