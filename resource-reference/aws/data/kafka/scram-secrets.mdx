## Example

    package main

    import (
    	kafka "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/kafka"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_kafka_scram_secrets"})

    		ubx.Data(kafka.ScramSecrets, "example", kafka.ScramSecretsConfig{
    			ClusterArn: "arn:aws:iam::123456789012:policy/example",
    			MaxResults: 1,
    		})
    	}))
    }
