## Example

    package main

    import (

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_kafka_vpc_connections"})

    		ubx.Data(kafka.VpcConnections, "example", kafka.VpcConnectionsConfig{
    			MaxResults: 1,
    		})
    	}))
    }
