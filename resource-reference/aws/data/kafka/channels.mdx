    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_kafka_channels"})

    		ubx.Data(kafka.Channels, "example", kafka.ChannelsConfig{
    			ClusterArn:      "arn:aws:iam::123456789012:policy/example",
    			MaxResults:      1,
    			TopicNameFilter: "example",
    		})
    	}))
    }

## Lookup arguments

<ResponseField name="cluster_arn" type="string" required>
  The real Amazon Resource Name (ARN) of the cluster this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="max_results" type="number">
  The maximum number of results to return in a single page. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="topic_name_filter" type="string">
  Restricts results to this topic name. **(AI-inferred)** Optional lookup argument.
</ResponseField>
