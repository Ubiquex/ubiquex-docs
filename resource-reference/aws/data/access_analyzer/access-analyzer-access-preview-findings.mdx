`aws_access_analyzer_access_preview_findings` reads the real findings from an IAM Access Analyzer access preview, simulating a proposed policy change before applying it.

## Example

    ```go
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_access_analyzer_access_preview_findings"})

    		ubx.Data(access_analyzer.AccessAnalyzerAccessPreviewFindings, "example", access_analyzer.AccessAnalyzerAccessPreviewFindingsConfig{
    			AccessPreviewId: "example",
    			AnalyzerArn:     "arn:aws:iam::123456789012:policy/example",
    			Filter:          map[string]any{"key": map[string]any{"contains": []string{"example"}}},
    			MaxResults:      1,
    		})
    	}))
    }
    ```
