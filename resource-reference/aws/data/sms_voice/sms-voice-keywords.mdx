`aws_sms_voice_keywords` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

    ```go

    import (
    	sms_voice "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/sms_voice"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_sms_voice_keywords"})

    		ubx.Data(sms_voice.SmsVoiceKeywords, "example", sms_voice.SmsVoiceKeywordsConfig{
    			OriginationIdentity: "example",
    			Filters:             []map[string]any{{"name": "example-name"}},
    			MaxResults:          1,
    		})
    	}))
    }
    ```
