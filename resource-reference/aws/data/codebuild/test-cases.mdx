## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	codebuild "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/codebuild"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_codebuild_test_cases"})

    		ubx.Data(codebuild.TestCases, "example", codebuild.TestCasesConfig{
    			ReportArn:  "arn:aws:iam::123456789012:policy/example",
    			Filter:     map[string]any{"keyword": "example"},
    			MaxResults: 1,
    		})
    	}))
    }
    ```
