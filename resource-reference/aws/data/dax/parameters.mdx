`aws_dax_parameters` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dax "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/dax"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_dax_parameters"})

    		ubx.Data(dax.Parameters, "example", dax.ParametersConfig{
    			ParameterGroupName: "example-parameter-group-name",
    			MaxResults:         1,
    			Source:             "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.aws.data.dax import Parameters, ParametersConfig

    def describe():

        ubx.data(Parameters, "example", ParametersConfig(
            parameter_group_name="example-parameter-group-name",
            max_results=1,
            source="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
