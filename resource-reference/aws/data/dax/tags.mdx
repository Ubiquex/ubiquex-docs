## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dax "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/dax"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_dax_tags"})

    		ubx.Data(dax.Tags, "example", dax.TagsConfig{
    			ResourceName: "example-resource-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Tags } from "@ubx/sdk-aws/aws/data/dax/tags";

    export default stack("example", () => {
      intent({ summary: "look up aws_dax_tags" });

      data(Tags, "example", {
        resourceName: "example-resource-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.dax import Tags, TagsConfig

    def describe():
        ubx.intent("look up aws_dax_tags")

        ubx.data(Tags, "example", TagsConfig(
            resource_name="example-resource-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
