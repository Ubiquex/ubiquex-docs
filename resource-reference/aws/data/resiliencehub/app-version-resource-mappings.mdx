`aws_resiliencehub_app_version_resource_mappings` reads the real mappings from a Resilience Hub application version's own logical resources to real AWS resources.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	resiliencehub "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/resiliencehub"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_resiliencehub_app_version_resource_mappings"})

    		ubx.Data(resiliencehub.AppVersionResourceMappings, "example", resiliencehub.AppVersionResourceMappingsConfig{
    			AppArn:     "arn:aws:iam::123456789012:policy/example",
    			AppVersion: "example",
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>
