`aws_resiliencehub_app_input_sources` reads the real input sources (CloudFormation stacks, Terraform state) a Resilience Hub application's own resources were imported from.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	resiliencehub "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/resiliencehub"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_resiliencehub_app_input_sources"})

    		ubx.Data(resiliencehub.AppInputSources, "example", resiliencehub.AppInputSourcesConfig{
    			AppArn:     "arn:aws:iam::123456789012:policy/example",
    			AppVersion: "example",
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>
