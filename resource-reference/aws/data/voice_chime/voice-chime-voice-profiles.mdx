    package main

    import (

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_voice_chime_voice_profiles"})

    		ubx.Data(voice_chime.VoiceChimeVoiceProfiles, "example", voice_chime.VoiceChimeVoiceProfilesConfig{
    			VoiceProfileDomainId: "example",
    			MaxResults:           1,
    		})
    	}))
    }

## Lookup arguments

<ResponseField name="max_results" type="number">
  The maximum number of results to return in a single page. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="voice_profile_domain_id" type="string" required>
  The real, unique identifier of the voice profile domain this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="next_token" type="string">

</ResponseField>

<ResponseField name="voice_profiles" type="list(object)">
  The real Amazon Chime SDK Voice Profiles included in this result. **(AI-inferred)** Returned by the lookup.
