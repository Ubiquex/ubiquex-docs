## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	sns "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/sns"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_sns_subscriptions_by_topic"})

    		ubx.Data(sns.SubscriptionsByTopic, "example", sns.SubscriptionsByTopicConfig{
    			TopicArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
