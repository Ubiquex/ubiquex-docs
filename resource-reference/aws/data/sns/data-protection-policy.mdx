## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	sns "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/sns"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_sns_data_protection_policy"})

    		ubx.Data(sns.DataProtectionPolicy, "example", sns.DataProtectionPolicyConfig{
    			ResourceArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
