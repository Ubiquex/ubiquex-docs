## Example

    package main

    import (

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_email_identities"})

    		ubx.Data(email.Identities, "example", email.IdentitiesConfig{

    			MaxItems:     1,
    		})
    	}))
    }
