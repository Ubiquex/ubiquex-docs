    ```go
    package main

    import (
    	api_detective "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/api_detective"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_api_detective_investigation"})

    		ubx.Data(api_detective.ApiDetectiveInvestigation, "example", api_detective.ApiDetectiveInvestigationConfig{
    			GraphArn:        "arn:aws:iam::123456789012:policy/example",
    			InvestigationId: "example",
    		})
    	}))
    }
    ```
