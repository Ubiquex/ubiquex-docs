## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dms "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/dms"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_dms_conversion_configuration"})

    		ubx.Data(dms.ConversionConfiguration, "example", dms.ConversionConfigurationConfig{
    			MigrationProjectIdentifier: "example",
    		})
    	}))
    }
    ```
