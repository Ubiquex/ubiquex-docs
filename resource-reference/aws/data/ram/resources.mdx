`aws_ram_resources` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	ram "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/ram"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ram_resources"})

    		ubx.Data(ram.Resources, "example", ram.ResourcesConfig{
    			ResourceOwner:       "example",
    			MaxResults:          1,
    			Principal:           "example",
    			ResourceArns:        []string{"example"},
    			ResourceRegionScope: "example",
    			ResourceShareArns:   []string{"example"},
    			ResourceType:        "example",
    		})
    	}))
    }
    ```
