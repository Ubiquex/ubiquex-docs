`aws_pinpoint_email_domain_statistics_report` reads real, aggregated sending statistics for a domain tracked by the Pinpoint Email Deliverability Dashboard.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	pinpoint_email "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/pinpoint_email"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_pinpoint_email_domain_statistics_report"})

    		ubx.Data(pinpoint_email.PinpointEmailDomainStatisticsReport, "example", pinpoint_email.PinpointEmailDomainStatisticsReportConfig{
    			Domain:    "example",
    			EndDate:   "example",
    			StartDate: "example",
    		})
    	}))
    }
    ```
