---
title: "aws_greengrass_connector_definitions"
description: "`data_aws_greengrass_connector_definitions` looks up a real, already-existing `aws_greengrass_connector_definition`."
---

It deploys a real, pre-built AWS integration, connecting to a local ML model or a serial port, for instance, to an `aws_greengrass_group`'s core device without a caller writing that integration logic themselves. Part of Greengrass V1's maintenance-only lifecycle since June 2023.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	greengrass "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/greengrass"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_greengrass_connector_definitions"})

    		ubx.Data(greengrass.ConnectorDefinitions, "example", greengrass.ConnectorDefinitionsConfig{
    			MaxResults: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">

    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.greengrass import ConnectorDefinitions, ConnectorDefinitionsConfig

    def describe():
        ubx.intent("look up aws_greengrass_connector_definitions")

        ubx.data(ConnectorDefinitions, "example", ConnectorDefinitionsConfig(
            max_results="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
