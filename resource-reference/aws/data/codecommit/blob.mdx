## Example

  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	codecommit "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/codecommit"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_codecommit_blob"})

    		ubx.Data(codecommit.Blob, "example", codecommit.BlobConfig{
    			BlobId:         "example",
    			RepositoryName: "example-repository-name",
    		})
    	}))
    }
    ```

    ```python
    from ubx.aws.data.codecommit import Blob, BlobConfig

    def describe():
        ubx.intent("look up aws_codecommit_blob")

        ubx.data(Blob, "example", BlobConfig(
            blob_id="example",
            repository_name="example-repository-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
