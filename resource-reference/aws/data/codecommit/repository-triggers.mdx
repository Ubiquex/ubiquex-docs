    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	codecommit "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/codecommit"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_codecommit_repository_triggers"})

    		ubx.Data(codecommit.RepositoryTriggers, "example", codecommit.RepositoryTriggersConfig{
    			RepositoryName: "example-repository-name",
    		})
    	}))
    }

    from ubx.aws.data.codecommit import RepositoryTriggers, RepositoryTriggersConfig

    def describe():
        ubx.intent("look up aws_codecommit_repository_triggers")

        ubx.data(RepositoryTriggers, "example", RepositoryTriggersConfig(

        ))

    if __name__ == "__main__":
