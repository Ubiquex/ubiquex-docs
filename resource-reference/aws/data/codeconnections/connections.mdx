`aws_codeconnections_connections` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_codeconnections_connections"})

    		ubx.Data(codeconnections.Connections, "example", codeconnections.ConnectionsConfig{
    			HostArnFilter:      "example",
    			MaxResults:         1,
    			ProviderTypeFilter: "example",
    		})
    	}))
    }
