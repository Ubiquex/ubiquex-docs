## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	codedeploy "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/codedeploy"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_codedeploy_application_revision"})

    		ubx.Data(codedeploy.ApplicationRevision, "example", codedeploy.ApplicationRevisionConfig{
    			ApplicationName: "example-application-name",
    			Revision:        map[string]any{"app_spec_content": map[string]any{"content": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
