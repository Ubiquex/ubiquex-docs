## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	codedeploy "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/codedeploy"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_codedeploy_on_premises_instance"})

    		ubx.Data(codedeploy.OnPremisesInstance, "example", codedeploy.OnPremisesInstanceConfig{
    			InstanceName: "example-instance-name",
    		})
    	}))
    }
    ```
