`aws_wellarchitected_notifications` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

    ```go
    package main

    import (
    	wellarchitected "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/wellarchitected"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_wellarchitected_notifications"})

    		ubx.Data(wellarchitected.Notifications, "example", wellarchitected.NotificationsConfig{
    			MaxResults:  1,
    			ResourceArn: "arn:aws:iam::123456789012:policy/example",
    			WorkloadId:  "example",
    		})
    	}))
    }
    ```
