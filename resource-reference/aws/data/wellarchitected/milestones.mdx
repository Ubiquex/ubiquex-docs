  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	wellarchitected "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/wellarchitected"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_wellarchitected_milestones"})

    		ubx.Data(wellarchitected.Milestones, "example", wellarchitected.MilestonesConfig{
    			WorkloadId: "example",
    			MaxResults: 1,
    		})
    	}))
    }
    ```
