## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	wellarchitected "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/wellarchitected"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_wellarchitected_tags_for_resource"})

    		ubx.Data(wellarchitected.TagsForResource, "example", wellarchitected.TagsForResourceConfig{
    			WorkloadArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
