---
title: "aws_wellarchitected_workloads"
description: "`data_aws_wellarchitected_workloads` looks up a real, already-existing `aws_well_architected_workload`."
---

It registers a real workload, an application or system, in the AWS Well-Architected Tool, the resource that a Well-Architected Framework review is actually conducted and tracked against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	wellarchitected "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/wellarchitected"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_wellarchitected_workloads"})

    		ubx.Data(wellarchitected.Workloads, "example", wellarchitected.WorkloadsConfig{
    			MaxResults:         1,
    			WorkloadNamePrefix: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { Workloads } from "./local-sdk/aws/sdk/typescript/aws/data/wellarchitected/workloads.ts";

    export default stack("example", () => {
      intent({ summary: "look up aws_wellarchitected_workloads" });
    ```

  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.wellarchitected import Workloads, WorkloadsConfig

    def describe():
        ubx.intent("look up aws_wellarchitected_workloads")

        ubx.data(Workloads, "example", WorkloadsConfig(
            max_results=1,
            workload_name_prefix="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="max_results" type="number">
  &lt;p&gt;The maximum number of results to return for this request.&lt;/p&gt; Optional lookup argument.
</ResponseField>

<ResponseField name="workload_name_prefix" type="string">
  &lt;p&gt;An optional string added to the beginning of each workload name returned in the results.&lt;/p&gt; Optional lookup argument.
</ResponseField>
