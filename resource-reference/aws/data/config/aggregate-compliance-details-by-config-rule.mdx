`aws_config_aggregate_compliance_details_by_config_rule` reads real, per-resource compliance details of a Config rule, aggregated across accounts and Regions.

## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	config "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/config"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_config_aggregate_compliance_details_by_config_rule"})

    		ubx.Data(config.AggregateComplianceDetailsByConfigRule, "example", config.AggregateComplianceDetailsByConfigRuleConfig{
    			AccountId:                   "example",
    			AwsRegion:                   "example",
    			ConfigRuleName:              "example-config-rule-name",
    			ConfigurationAggregatorName: "example-configuration-aggregator-name",
    			ComplianceType:              "example",
    			Limit:                       1,
    		})
    	}))
    }
    ```
