`aws_config_aggregate_compliance_by_conformance_packs` reads real, aggregated compliance status of a Config conformance pack across every account and Region in a Config aggregator.

## Example

  <Tab title="Go">
    ```go
    package main

    import (

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_config_aggregate_compliance_by_conformance_packs"})

    		ubx.Data(config.AggregateComplianceByConformancePacks, "example", config.AggregateComplianceByConformancePacksConfig{
    			ConfigurationAggregatorName: "example-configuration-aggregator-name",
    			Filters:                     map[string]any{"account_id": "example"},
    			Limit:                       1,
    		})
    	}))
    }
    ```
