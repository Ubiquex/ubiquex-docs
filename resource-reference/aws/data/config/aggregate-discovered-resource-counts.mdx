`aws_config_aggregate_discovered_resource_counts` reads real, aggregated resource counts discovered by Config across every account and Region in a Config aggregator.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	config "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/config"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_config_aggregate_discovered_resource_counts"})

    		ubx.Data(config.AggregateDiscoveredResourceCounts, "example", config.AggregateDiscoveredResourceCountsConfig{
    			ConfigurationAggregatorName: "example-configuration-aggregator-name",
    			Filters:                     map[string]any{"account_id": "example"},
    			Limit:                       1,
    		})
    	}))
    }
    ```
