---
title: "aws_config_delivery_channel_status"
description: "`data_aws_config_delivery_channel_status` looks up a real, already-existing `aws_config_delivery_channel`."
---

It tells AWS Config where to actually deliver what an `aws_config_configuration_recorder` records, a configuration snapshot and history to an S3 bucket, and change notifications to an SNS topic.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	config "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/config"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_config_delivery_channel_status"})

    		ubx.Data(config.DeliveryChannelStatus, "example", config.DeliveryChannelStatusConfig{
    			DeliveryChannelNames: []string{"example"},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.data.config import DeliveryChannelStatus, DeliveryChannelStatusConfig

    def describe():
        ubx.intent("look up aws_config_delivery_channel_status")

        ubx.data(DeliveryChannelStatus, "example", DeliveryChannelStatusConfig(
            delivery_channel_names=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
