---
title: "aws_config_connector"
description: "`data_aws_config_connector` looks up a real, already-existing `aws_config_connector`."
---

It connects a real third-party cloud provider's own resources to AWS Config, extending Config's compliance evaluation and configuration history beyond AWS's own resources to a genuinely multi-cloud inventory.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	config "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/config"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own connector"})

    		ubx.Resource(config.Connector, "example", config.ConnectorConfig{
    			ConnectorConfiguration: map[string]any{"azure": map[string]any{"client_identifier": "example", "tenant_identifier": "example"}},
    			Tags:                   []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.config import Connector, ConnectorConfig
    from ubx.aws.config.connector import Connector_ConnectorConfiguration, Connector_ConnectorConfiguration_Azure, Connector_Tags

    def describe():
        ubx.intent("example own connector")

        ubx.resource(Connector, "example", ConnectorConfig(
            connector_configuration=Connector_ConnectorConfiguration(azure=Connector_ConnectorConfiguration_Azure(client_identifier="example", tenant_identifier="example")),
            tags=[Connector_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
