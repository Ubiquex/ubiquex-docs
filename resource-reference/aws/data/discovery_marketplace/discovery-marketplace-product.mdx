## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	discovery_marketplace "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/discovery_marketplace"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_discovery_marketplace_product"})

    		ubx.Data(discovery_marketplace.DiscoveryMarketplaceProduct, "example", discovery_marketplace.DiscoveryMarketplaceProductConfig{
    			ProductId: "example",
    		})
    	}))
    }
    ```
  </Tab>
