## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	discovery_marketplace "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/discovery_marketplace"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_discovery_marketplace_listing"})

    		ubx.Data(discovery_marketplace.DiscoveryMarketplaceListing, "example", discovery_marketplace.DiscoveryMarketplaceListingConfig{
    			ListingId: "example",
    		})
    	}))
    }
    ```
  </Tab>
