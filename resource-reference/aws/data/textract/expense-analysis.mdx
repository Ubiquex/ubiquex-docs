    package main

    import (
    	textract "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/textract"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_textract_expense_analysis"})

    		ubx.Data(textract.ExpenseAnalysis, "example", textract.ExpenseAnalysisConfig{
    			JobId:      "example",
    			MaxResults: 1,
    		})
    	}))
    }
