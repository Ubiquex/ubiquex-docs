    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cognito_idp_device"})

    		ubx.Data(cognito_idp.CognitoIdpDevice, "example", cognito_idp.CognitoIdpDeviceConfig{

    			AccessToken: "example",
    		})
    	}))
    }

## Lookup arguments

<ResponseField name="access_token" type="string">
  The real, opaque access token used to authenticate this request. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="device_key" type="string" required>
  The real, unique key of the device this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
