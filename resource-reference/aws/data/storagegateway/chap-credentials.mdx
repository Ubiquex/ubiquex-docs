title: "aws_storagegateway_chap_credentials"

---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	storagegateway "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/storagegateway"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_storagegateway_chap_credentials"})

    		ubx.Data(storagegateway.ChapCredentials, "example", storagegateway.ChapCredentialsConfig{
    			TargetArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.storagegateway import ChapCredentials, ChapCredentialsConfig

    def describe():
        ubx.intent("look up aws_storagegateway_chap_credentials")

        ubx.data(ChapCredentials, "example", ChapCredentialsConfig(
            target_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

</ResponseField>

## Result properties

<ResponseField name="chap_credentials" type="list(object)">
