## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	storagegateway "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/storagegateway"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_storagegateway_gateway_information"})

    		ubx.Data(storagegateway.GatewayInformation, "example", storagegateway.GatewayInformationConfig{
    			GatewayArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
