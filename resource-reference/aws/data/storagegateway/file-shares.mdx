## Example

    package main

    import (

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_storagegateway_file_shares"})

    		ubx.Data(storagegateway.FileShares, "example", storagegateway.FileSharesConfig{
    			GatewayArn: "arn:aws:iam::123456789012:policy/example",
    			Limit:      1,
    		})
    	}))
    }
