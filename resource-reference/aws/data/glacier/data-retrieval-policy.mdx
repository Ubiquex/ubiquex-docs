## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glacier "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/glacier"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_glacier_data_retrieval_policy"})

    		ubx.Data(glacier.DataRetrievalPolicy, "example", glacier.DataRetrievalPolicyConfig{
    			AccountId: "example",
    		})
    	}))
    }
    ```
