## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glacier "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/glacier"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_glacier_tags_for_vault"})

    		ubx.Data(glacier.TagsForVault, "example", glacier.TagsForVaultConfig{
    			AccountId: "example",
    			VaultName: "example-vault-name",
    		})
    	}))
    }
    ```
