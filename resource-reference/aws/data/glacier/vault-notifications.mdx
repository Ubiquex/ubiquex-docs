## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glacier "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/glacier"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_glacier_vault_notifications"})

    		ubx.Data(glacier.VaultNotifications, "example", glacier.VaultNotificationsConfig{
    			AccountId: "example",
    			VaultName: "example-vault-name",
    		})
    	}))
    }
    ```
  </Tab>
