## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	resiliencehubv2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/resiliencehubv2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_resiliencehubv2_tags_for_resource"})

    		ubx.Data(resiliencehubv2.TagsForResource, "example", resiliencehubv2.TagsForResourceConfig{
    			ResourceArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.resiliencehubv2 import TagsForResource, TagsForResourceConfig

    def describe():
        ubx.intent("look up aws_resiliencehubv2_tags_for_resource")

        ubx.data(TagsForResource, "example", TagsForResourceConfig(
            resource_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
