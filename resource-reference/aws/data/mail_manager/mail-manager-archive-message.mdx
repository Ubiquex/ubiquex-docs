## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	mail_manager "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/mail_manager"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_mail_manager_archive_message"})

    		ubx.Data(mail_manager.MailManagerArchiveMessage, "example", mail_manager.MailManagerArchiveMessageConfig{
    			ArchivedMessageId: "example",
    		})
    	}))
    }
    ```
