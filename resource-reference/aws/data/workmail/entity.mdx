`aws_workmail_entity` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    	workmail "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/workmail"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_workmail_entity"})

    		ubx.Data(workmail.Entity, "example", workmail.EntityConfig{
    			Email:          "example",
    			OrganizationId: "example",
    		})
    	}))
    }
