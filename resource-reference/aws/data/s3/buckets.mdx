`aws_s3_buckets` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	s3 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/s3"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_s3_buckets"})

    		ubx.Data(s3.Buckets, "example", s3.BucketsConfig{
    			BucketRegion: "example",
    			MaxBuckets:   1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Buckets } from "@ubx/sdk-aws/aws/data/s3/buckets";

    export default stack("example", () => {
      intent({ summary: "look up aws_s3_buckets" });

      data(Buckets, "example", {
        bucketRegion: "example",
        maxBuckets: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.s3 import Buckets, BucketsConfig

    def describe():
        ubx.intent("look up aws_s3_buckets")

        ubx.data(Buckets, "example", BucketsConfig(
            bucket_region="example",
            max_buckets=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
