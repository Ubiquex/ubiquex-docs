## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	kinesis "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/kinesis"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_kinesis_tags_for_stream"})

    		ubx.Data(kinesis.TagsForStream, "example", kinesis.TagsForStreamConfig{
    			ExclusiveStartTagKey: "example",
    			Limit:                1,
    			StreamArn:            "arn:aws:iam::123456789012:policy/example",
    			StreamId:             "example",
    			StreamName:           "example-stream-name",
    		})
    	}))
    }
    ```
