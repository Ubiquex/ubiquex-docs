`aws_dynamodb_tables` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dynamodb "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/dynamodb"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_dynamodb_tables"})

    		ubx.Data(dynamodb.Tables, "example", dynamodb.TablesConfig{
    			ExclusiveStartTableName: "example-exclusive-start-table-name",
    			Limit:                   1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

    export default stack("example", () => {
      intent({ summary: "look up aws_dynamodb_tables" });

      data(Tables, "example", {

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.dynamodb import Tables, TablesConfig

    def describe():
        ubx.intent("look up aws_dynamodb_tables")

        ubx.data(Tables, "example", TablesConfig(
            exclusive_start_table_name="example-exclusive-start-table-name",
            limit=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
