## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dynamodb "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/dynamodb"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_dynamodb_time_to_live"})

    		ubx.Data(dynamodb.TimeToLive, "example", dynamodb.TimeToLiveConfig{
    			TableName: "example-table-name",
    		})
    	}))
    }
    ```
  </Tab>
