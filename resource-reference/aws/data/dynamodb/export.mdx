`aws_dynamodb_export` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dynamodb "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/dynamodb"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_dynamodb_export"})

    		ubx.Data(dynamodb.Export, "example", dynamodb.ExportConfig{
    			ExportArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Export } from "@ubx/sdk-aws/aws/data/dynamodb/export";

    export default stack("example", () => {
      intent({ summary: "look up aws_dynamodb_export" });

      data(Export, "example", {
        exportArn: "arn:aws:iam::123456789012:policy/example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.dynamodb import Export, ExportConfig

    def describe():
        ubx.intent("look up aws_dynamodb_export")

        ubx.data(Export, "example", ExportConfig(
            export_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
