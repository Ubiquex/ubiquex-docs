---
title: "aws_dynamodb_global_table_settings"
description: "`data_aws_dynamodb_global_table_settings` looks up a real, already-existing `aws_dynamo_db_global_table`."
---

It replicates an `aws_dynamo_db_table`'s own data across multiple AWS Regions with active-active, multi-Region writes, real low-latency global access and Regional failover that a single-Region table doesn't provide.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	dynamodb "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/dynamodb"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_dynamodb_global_table_settings"})

    		ubx.Data(dynamodb.GlobalTableSettings, "example", dynamodb.GlobalTableSettingsConfig{
    			GlobalTableName: "example-global-table-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { GlobalTableSettings } from "./local-sdk/aws/sdk/typescript/aws/data/dynamodb/global_table_settings.ts";

    export default stack("example", () => {
      intent({ summary: "look up aws_dynamodb_global_table_settings" });

      data(GlobalTableSettings, "example", {
        globalTableName: "example-global-table-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.dynamodb import GlobalTableSettings, GlobalTableSettingsConfig

    def describe():
        ubx.intent("look up aws_dynamodb_global_table_settings")

        ubx.data(GlobalTableSettings, "example", GlobalTableSettingsConfig(
            global_table_name="example-global-table-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
