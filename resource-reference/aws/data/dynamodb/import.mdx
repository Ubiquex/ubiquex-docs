## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	dynamodb "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/dynamodb"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_dynamodb_import"})

    		ubx.Data(dynamodb.Import, "example", dynamodb.ImportConfig{
    			ImportArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { Import } from "./local-sdk/aws/sdk/typescript/aws/data/dynamodb/import.ts";

    export default stack("example", () => {
      intent({ summary: "look up aws_dynamodb_import" });

      data(Import, "example", {
        importArn: "arn:aws:iam::123456789012:policy/example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.dynamodb import Import, ImportConfig

    def describe():
        ubx.intent("look up aws_dynamodb_import")

        ubx.data(Import, "example", ImportConfig(
            import_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
