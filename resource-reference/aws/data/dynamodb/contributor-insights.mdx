## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dynamodb "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/dynamodb"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_dynamodb_contributor_insights"})

    		ubx.Data(dynamodb.ContributorInsights, "example", dynamodb.ContributorInsightsConfig{
    			TableName: "example-table-name",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
