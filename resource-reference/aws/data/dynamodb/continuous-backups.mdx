## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	dynamodb "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/dynamodb"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_dynamodb_continuous_backups"})

    		ubx.Data(dynamodb.ContinuousBackups, "example", dynamodb.ContinuousBackupsConfig{
    			TableName: "example-table-name",
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.dynamodb import ContinuousBackups, ContinuousBackupsConfig

    def describe():
        ubx.intent("look up aws_dynamodb_continuous_backups")

        ubx.data(ContinuousBackups, "example", ContinuousBackupsConfig(
            table_name="example-table-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
