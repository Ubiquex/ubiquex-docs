`aws_elasticmapreduce_on_cluster_app_uipresigned_url` reads a real, generated, presigned URL for accessing an EMR cluster's own on-cluster application UI (such as YARN or Spark History Server).

    	elasticmapreduce "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/elasticmapreduce"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_elasticmapreduce_on_cluster_app_uipresigned_url"})

    		ubx.Data(elasticmapreduce.OnClusterAppUipresignedUrl, "example", elasticmapreduce.OnClusterAppUipresignedUrlConfig{
    			ClusterId:          "example",

    			ExecutionRoleArn:   "arn:aws:iam::123456789012:policy/example",
    			OnClusterAppUitype: "example",
