`aws_elasticmapreduce_persistent_app_uipresigned_url` reads a real, generated, presigned URL for accessing an EMR Persistent Application UI, which survives after the originating cluster is terminated.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	elasticmapreduce "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/elasticmapreduce"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_elasticmapreduce_persistent_app_uipresigned_url"})

    		ubx.Data(elasticmapreduce.PersistentAppUipresignedUrl, "example", elasticmapreduce.PersistentAppUipresignedUrlConfig{
    			PersistentAppUiid:   "example",
    			ApplicationId:       "example",
    			AuthProxyCall:       true,
    			ExecutionRoleArn:    "arn:aws:iam::123456789012:policy/example",
    			PersistentAppUitype: "example",
    		})
    	}))
    }
    ```
