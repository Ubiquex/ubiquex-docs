## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	monitoring "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/monitoring"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_monitoring_dashboards"})

    		ubx.Data(monitoring.Dashboards, "example", monitoring.DashboardsConfig{
    			DashboardNamePrefix: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.aws.data.monitoring import Dashboards, DashboardsConfig

    def describe():
        ubx.intent("look up aws_monitoring_dashboards")

        ubx.data(Dashboards, "example", DashboardsConfig(
            dashboard_name_prefix="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
