`aws_monitoring_dataset` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	monitoring "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/monitoring"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_monitoring_dataset"})

    		ubx.Data(monitoring.Dataset, "example", monitoring.DatasetConfig{
    			DatasetIdentifier: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Dataset } from "@ubx/sdk-aws/aws/data/monitoring/dataset";

    export default stack("example", () => {
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.monitoring import Dataset, DatasetConfig

    def describe():
        ubx.intent("look up aws_monitoring_dataset")

        ubx.data(Dataset, "example", DatasetConfig(
            dataset_identifier="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
