`aws_iotsitewise_asset_models` looks up a real IoT SiteWise asset model, the template defining the properties and hierarchy every asset of that type shares.

## Example

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iotsitewise_asset_models"})

    		ubx.Data(iotsitewise.AssetModels, "example", iotsitewise.AssetModelsConfig{
    			AssetModelTypes:   []string{"example"},

    		})
    	}))
    }
