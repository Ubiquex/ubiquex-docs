## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iotsitewise "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/iotsitewise"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iotsitewise_dashboards"})

    		ubx.Data(iotsitewise.Dashboards, "example", iotsitewise.DashboardsConfig{
    			ProjectId:  "example",
    			MaxResults: 1,
    		})
    	}))
    }
    ```
