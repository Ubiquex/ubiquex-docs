    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iotsitewise_query"})

    		ubx.Data(iotsitewise.Query, "example", iotsitewise.QueryConfig{
    			QueryId:       "example",
    			WorkspaceName: "example-workspace-name",
    		})
    	}))
    }
