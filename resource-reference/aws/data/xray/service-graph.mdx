`aws_xray_service_graph` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    import (
    	xray "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/xray"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_xray_service_graph"})

    		ubx.Data(xray.ServiceGraph, "example", xray.ServiceGraphConfig{
    			EndTime:   "example",
    			StartTime: "example",
    			GroupArn:  "arn:aws:iam::123456789012:policy/example",
    			GroupName: "example-group-name",
    		})
    	}))
    }
