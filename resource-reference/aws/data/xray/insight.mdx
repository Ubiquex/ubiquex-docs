`aws_xray_insight` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	xray "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/xray"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_xray_insight"})

    		ubx.Data(xray.Insight, "example", xray.InsightConfig{
    			InsightId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Insight } from "@ubx/sdk-aws/aws/data/xray/insight";

    export default stack("example", () => {
      intent({ summary: "look up aws_xray_insight" });

      data(Insight, "example", {
        insightId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.xray import Insight, InsightConfig

    def describe():
        ubx.intent("look up aws_xray_insight")

        ubx.data(Insight, "example", InsightConfig(
            insight_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
