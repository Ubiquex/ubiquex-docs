## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	xray "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/xray"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_xray_insight"})

    		ubx.Data(xray.Insight, "example", xray.InsightConfig{
    			InsightId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { Insight } from "./local-sdk/aws/sdk/typescript/aws/data/xray/insight.ts";

    export default stack("example", () => {
      intent({ summary: "look up aws_xray_insight" });

      data(Insight, "example", {
        insightId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.xray import Insight, InsightConfig

    def describe():
        ubx.intent("look up aws_xray_insight")

        ubx.data(Insight, "example", InsightConfig(
            insight_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
