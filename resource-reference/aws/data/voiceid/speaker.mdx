    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	voiceid "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/voiceid"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_voiceid_speaker"})

    		ubx.Data(voiceid.Speaker, "example", voiceid.SpeakerConfig{
    			DomainId:  "example",
    			SpeakerId: "example",
    		})
    	}))
    }
