`aws_securityhub_standards` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	securityhub "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/securityhub"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_securityhub_standards"})

    		ubx.Data(securityhub.Standards, "example", securityhub.StandardsConfig{
    			MaxResults: 1,
    			Providers:  []string{"example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.aws.data.securityhub import Standards, StandardsConfig

    def describe():

        ubx.data(Standards, "example", StandardsConfig(

            providers=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
