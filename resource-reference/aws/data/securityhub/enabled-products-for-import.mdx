`aws_securityhub_enabled_products_for_import` reads which real third-party security product integrations are currently enabled to import findings into Security Hub.

## Example

  <Tab title="Go">

    package main

    import (
    	securityhub "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/securityhub"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_securityhub_enabled_products_for_import"})

    		ubx.Data(securityhub.EnabledProductsForImport, "example", securityhub.EnabledProductsForImportConfig{
    			MaxResults: 1,
    		})
    	}))
    }
