---
title: "aws_managedblockchain_accessors"
description: "`data_aws_managedblockchain_accessors` looks up a real, already-existing `aws_managed_blockchain_accessor`."
---

It creates a real token-based access credential for Amazon Managed Blockchain's Ethereum node JSON-RPC APIs, an alternative to signing every request with SigV4 that instead authenticates with a bearer token.

## Example

    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	managedblockchain "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/managedblockchain"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_managedblockchain_accessors"})

    		ubx.Data(managedblockchain.Accessors, "example", managedblockchain.AccessorsConfig{
    			MaxResults:  1,
    			NetworkType: "example",
    		})
    	}))
    }

    from ubx.aws.data.managedblockchain import Accessors, AccessorsConfig

    def describe():

        ubx.data(Accessors, "example", AccessorsConfig(

        ))
