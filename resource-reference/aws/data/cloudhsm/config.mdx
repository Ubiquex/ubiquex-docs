## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudhsm "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudhsm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudhsm_config"})

    		ubx.Data(cloudhsm.Config, "example", cloudhsm.ConfigConfig{
    			ClientArn:     "arn:aws:iam::123456789012:policy/example",
    			ClientVersion: "example",
    			HapgList:      []string{"example"},
    		})
    	}))
    }
    ```

    def describe():
        ubx.intent("look up aws_cloudhsm_config")

        ubx.data(Config, "example", ConfigConfig(
            client_arn="arn:aws:iam::123456789012:policy/example",
            client_version="example",
            hapg_list=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="client_arn" type="string" required>
  The real Amazon Resource Name (ARN) of the client this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="client_version" type="string" required>
  The real version of the client. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="hapg_list" type="list(string)" required>
  The real hapgs included in this result. **(AI-inferred)** Required lookup argument.
</ResponseField>
