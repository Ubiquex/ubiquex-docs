`aws_cloudhsm_config` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudhsm "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudhsm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudhsm_config"})

    		ubx.Data(cloudhsm.Config, "example", cloudhsm.ConfigConfig{
    			ClientArn:     "arn:aws:iam::123456789012:policy/example",
    			ClientVersion: "example",
    			HapgList:      []string{"example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.cloudhsm import Config, ConfigConfig

    def describe():
        ubx.intent("look up aws_cloudhsm_config")

        ubx.data(Config, "example", ConfigConfig(
            client_arn="arn:aws:iam::123456789012:policy/example",
            client_version="example",
            hapg_list=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
