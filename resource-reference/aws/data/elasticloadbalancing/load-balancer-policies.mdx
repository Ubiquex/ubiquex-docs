---
title: "aws_elasticloadbalancing_load_balancer_policies"
description: "`data_aws_elasticloadbalancing_load_balancer_policies` looks up a real, already-existing `aws_elastic_load_balancing_load_balancer`."
---

It creates a Classic Load Balancer, ELB's original, legacy load balancing resource, layer 4 and basic layer 7 routing, superseded for new deployments by `aws_elastic_load_balancing_v2_load_balancer`'s own Application and Network Load Balancer types.

## Example

  <Tab title="Go">

    package main

    import (
    	elasticloadbalancing "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/elasticloadbalancing"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_elasticloadbalancing_load_balancer_policies"})

    		ubx.Data(elasticloadbalancing.LoadBalancerPolicies, "example", elasticloadbalancing.LoadBalancerPoliciesConfig{
    			LoadBalancerName: "example-load-balancer-name",
    			PolicyNames:      []string{"example"},
    		})
    	}))
    }

    import ubx_sdk as ubx
    from ubx.aws.data.elasticloadbalancing import LoadBalancerPolicies, LoadBalancerPoliciesConfig

    def describe():
        ubx.intent("look up aws_elasticloadbalancing_load_balancer_policies")

        ubx.data(LoadBalancerPolicies, "example", LoadBalancerPoliciesConfig(
            load_balancer_name="example-load-balancer-name",
            policy_names=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
