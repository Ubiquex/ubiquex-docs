## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	elasticloadbalancing "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/elasticloadbalancing"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_elasticloadbalancing_tags"})

    		ubx.Data(elasticloadbalancing.Tags, "example", elasticloadbalancing.TagsConfig{
    			LoadBalancerNames: []string{"example"},
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.aws.data.elasticloadbalancing import Tags, TagsConfig

    def describe():
        ubx.intent("look up aws_elasticloadbalancing_tags")

        ubx.data(Tags, "example", TagsConfig(
            load_balancer_names=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
