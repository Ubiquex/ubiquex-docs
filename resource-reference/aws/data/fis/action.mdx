`aws_fis_action` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	fis "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/fis"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_fis_action"})

    		ubx.Data(fis.Action, "example", fis.ActionConfig{
    			Id: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Action } from "@ubx/sdk-aws/aws/data/fis/action";

    export default stack("example", () => {
      intent({ summary: "look up aws_fis_action" });

      data(Action, "example", {
        id: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.fis import Action, ActionConfig

    def describe():
        ubx.intent("look up aws_fis_action")

        ubx.data(Action, "example", ActionConfig(
            id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
