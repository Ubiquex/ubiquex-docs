    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    		})
    	}))
    }

<ResponseField name="max_results" type="number">

</ResponseField>

<ResponseField name="workspace_id" type="string">
  The real, unique identifier of the workspace this lookup is scoped to. **(AI-inferred)** Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="grafana_versions" type="list(string)">

</ResponseField>
