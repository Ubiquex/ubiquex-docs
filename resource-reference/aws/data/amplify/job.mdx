## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	amplify "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/amplify"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_amplify_job"})

    		ubx.Data(amplify.Job, "example", amplify.JobConfig{
    			AppId:      "example",
    			BranchName: "example-branch-name",
    			JobId:      "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { Job } from "./local-sdk/aws/sdk/typescript/aws/data/amplify/job.ts";

    export default stack("example", () => {
      intent({ summary: "look up aws_amplify_job" });

      data(Job, "example", {
        appId: "example",
        branchName: "example-branch-name",
        jobId: "example",
      });
    });
    ```

  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.amplify import Job, JobConfig

    def describe():
        ubx.intent("look up aws_amplify_job")

        ubx.data(Job, "example", JobConfig(
            app_id="example",
            branch_name="example-branch-name",
            job_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="app_id" type="string" required>
  The real, unique identifier of the app this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="branch_name" type="string" required>
  The real name of the Amplify branch this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="job_id" type="string" required>
  The real, unique identifier of the job this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
