---
title: "aws_amplify_apps"
description: "`data_aws_amplify_apps` looks up a real, already-existing `aws_amplify_app`."
---

It creates the top-level Amplify Hosting application, the Git-connected project that Amplify's managed CI/CD pipeline builds and deploys from. Branches, domains, and build settings all attach to this app rather than standing alone.

## Example

  <Tab title="Go">

    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	amplify "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/amplify"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_amplify_apps"})

    		ubx.Data(amplify.Apps, "example", amplify.AppsConfig{
    			MaxResults: 1,
    		})
    	}))
    }

    import ubx_sdk as ubx
    from ubx.aws.data.amplify import Apps, AppsConfig

    def describe():
        ubx.intent("look up aws_amplify_apps")

        ubx.data(Apps, "example", AppsConfig(
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
