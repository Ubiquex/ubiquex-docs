## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	personalize "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/personalize"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_personalize_algorithm"})

    		ubx.Data(personalize.Algorithm, "example", personalize.AlgorithmConfig{
    			AlgorithmArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
