`aws_lakeformation_temporary_glue_partition_credentials` reads real, temporary, permission-scoped credentials for accessing a Glue table partition through Lake Formation.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	lakeformation "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/lakeformation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_lakeformation_temporary_glue_partition_credentials"})

    		ubx.Data(lakeformation.TemporaryGluePartitionCredentials, "example", lakeformation.TemporaryGluePartitionCredentialsConfig{
    			Partition:                map[string]any{"values": []string{"example"}},
    			TableArn:                 "arn:aws:iam::123456789012:policy/example",
    			AuditContext:             map[string]any{"additional_audit_context": "example"},
    			DurationSeconds:          7200,
    			Permissions:              []string{"example"},
    			SupportedPermissionTypes: []string{"example"},
    		})
    	}))
    }
    ```
