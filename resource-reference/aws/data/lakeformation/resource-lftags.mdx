## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	lakeformation "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/lakeformation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_lakeformation_resource_lftags"})

    		ubx.Data(lakeformation.ResourceLftags, "example", lakeformation.ResourceLftagsConfig{
    			Resource:           map[string]any{"catalog": map[string]any{"id": "example"}},
    			CatalogId:          "example",
    			ShowAssignedLftags: true,
    		})
    	}))
    }
    ```
