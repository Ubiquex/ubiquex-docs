## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	comprehend "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/comprehend"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_comprehend_endpoints"})

    		ubx.Data(comprehend.Endpoints, "example", comprehend.EndpointsConfig{
    			Filter:     map[string]any{"creation_time_after": "example"},
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.comprehend import Endpoints, EndpointsConfig
    from ubx.aws.data.comprehend.endpoints import Endpoints_Filter

    def describe():
        ubx.intent("look up aws_comprehend_endpoints")

        ubx.data(Endpoints, "example", EndpointsConfig(
            filter=Endpoints_Filter(creation_time_after="example"),
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
