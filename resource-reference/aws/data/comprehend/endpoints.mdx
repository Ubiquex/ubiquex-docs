## Example

    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	comprehend "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/comprehend"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_comprehend_endpoints"})

    		ubx.Data(comprehend.Endpoints, "example", comprehend.EndpointsConfig{
    			Filter:     map[string]any{"creation_time_after": "example"},
    			MaxResults: 1,
    		})
    	}))
    }
    ```
