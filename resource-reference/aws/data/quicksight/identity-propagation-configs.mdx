`aws_quicksight_identity_propagation_configs` reads the real configuration controlling how a QuickSight user's own identity is propagated to an underlying data source.

## Example

  <Tab title="Go">

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_quicksight_identity_propagation_configs"})

    		ubx.Data(quicksight.IdentityPropagationConfigs, "example", quicksight.IdentityPropagationConfigsConfig{
    			AwsAccountId: "example",
    			MaxResults:   1,
    		})
    	}))
    }
