## Example

    ```go
    package main

    import (
    	quicksight "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/quicksight"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_quicksight_flow_metadata"})

    		ubx.Data(quicksight.FlowMetadata, "example", quicksight.FlowMetadataConfig{
    			AwsAccountId: "example",
    			FlowId:       "example",
    		})
    	}))
    }
    ```
