`aws_messaging_chime_messaging_session_endpoint` reads the real WebSocket endpoint an app instance user connects to for real-time Chime SDK Messaging delivery.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_messaging_chime_messaging_session_endpoint"})

    }
