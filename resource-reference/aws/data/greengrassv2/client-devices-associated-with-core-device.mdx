`aws_greengrassv2_client_devices_associated_with_core_device` reads the real client devices (such as Bluetooth or Zigbee sensors) associated with a Greengrass V2 core device.

## Example

  <Tab title="Go">

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_greengrassv2_client_devices_associated_with_core_device"})

    		ubx.Data(greengrassv2.ClientDevicesAssociatedWithCoreDevice, "example", greengrassv2.ClientDevicesAssociatedWithCoreDeviceConfig{
    			CoreDeviceThingName: "example-core-device-thing-name",
    			MaxResults:          1,
    		})
    	}))
    }
