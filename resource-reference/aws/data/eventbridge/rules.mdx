## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	eventbridge "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/eventbridge"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_eventbridge_rules"})

    		ubx.Data(eventbridge.Rules, "example", eventbridge.RulesConfig{
    			EventBusName: "example-event-bus-name",
    			Limit:        1,
    			NamePrefix:   "example",
    		})
    	}))
    }
    ```
