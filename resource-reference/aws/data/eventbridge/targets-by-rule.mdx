## Example

    package main

    import (
    	eventbridge "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/eventbridge"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_eventbridge_targets_by_rule"})

    		ubx.Data(eventbridge.TargetsByRule, "example", eventbridge.TargetsByRuleConfig{
    			Rule:         "example",
    			EventBusName: "example-event-bus-name",
    			Limit:        1,
    		})
    	}))
    }
