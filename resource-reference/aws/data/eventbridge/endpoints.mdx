## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	eventbridge "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/eventbridge"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_eventbridge_endpoints"})

    		ubx.Data(eventbridge.Endpoints, "example", eventbridge.EndpointsConfig{
    			HomeRegion: "example",
    			MaxResults: 1,
    			NamePrefix: "example",
    		})
    	}))
    }
    ```
