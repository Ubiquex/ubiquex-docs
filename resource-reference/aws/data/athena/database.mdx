## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	athena "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/athena"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_athena_database"})

    		ubx.Data(athena.Database, "example", athena.DatabaseConfig{
    			CatalogName:  "example-catalog-name",
    			DatabaseName: "example-database-name",
    			WorkGroup:    "example",
    		})
    	}))
    }
    ```

            database_name="example-database-name",
            work_group="example",
