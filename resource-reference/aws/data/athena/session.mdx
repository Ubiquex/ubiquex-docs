## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	athena "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/athena"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_athena_session"})

    		ubx.Data(athena.Session, "example", athena.SessionConfig{
    			SessionId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

      data(Session, "example", {
        sessionId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.athena import Session, SessionConfig

    def describe():
        ubx.intent("look up aws_athena_session")

        ubx.data(Session, "example", SessionConfig(
            session_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
