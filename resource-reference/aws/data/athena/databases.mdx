## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	athena "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/athena"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_athena_databases"})

    		ubx.Data(athena.Databases, "example", athena.DatabasesConfig{
    			CatalogName: "example-catalog-name",
    			MaxResults:  1,
    			WorkGroup:   "example",
    		})
    	}))
    }
    ```

</ResponseField>

<ResponseField name="max_results" type="number">

</ResponseField>

<ResponseField name="work_group" type="string">
  The real Amazon Athena workgroup this lookup is scoped to. **(AI-inferred)** Optional lookup argument.
</ResponseField>
