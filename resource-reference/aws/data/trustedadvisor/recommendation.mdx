    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	trustedadvisor "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/trustedadvisor"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_trustedadvisor_recommendation"})

    		ubx.Data(trustedadvisor.Recommendation, "example", trustedadvisor.RecommendationConfig{
    			RecommendationIdentifier: "example",
    			Language:                 "example",
    		})
    	}))
    }
