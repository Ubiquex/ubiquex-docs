`aws_mq_users` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	mq "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/mq"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_mq_users"})

    		ubx.Data(mq.Users, "example", mq.UsersConfig{
    			BrokerId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Users } from "@ubx/sdk-aws/aws/data/mq/users";

    export default stack("example", () => {
      intent({ summary: "look up aws_mq_users" });

      data(Users, "example", {
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.mq import Users, UsersConfig

    def describe():
        ubx.intent("look up aws_mq_users")

        ubx.data(Users, "example", UsersConfig(
            broker_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
