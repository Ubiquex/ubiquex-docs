`aws_secretsmanager_random_password` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    	secretsmanager "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/secretsmanager"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_secretsmanager_random_password"})

    		ubx.Data(secretsmanager.RandomPassword, "example", secretsmanager.RandomPasswordConfig{
    			ExcludeCharacters:       "example",

    			ExcludeUppercase:        true,

    		})
    	}))
    }
