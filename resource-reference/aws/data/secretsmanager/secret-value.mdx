    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_secretsmanager_secret_value"})

    		ubx.Data(secretsmanager.SecretValue, "example", secretsmanager.SecretValueConfig{
    			SecretId:     "example",
    			VersionStage: "example",
    		})
    	}))
    }
