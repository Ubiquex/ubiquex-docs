## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	rekognition "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/rekognition"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_rekognition_segment_detection"})

    		ubx.Data(rekognition.SegmentDetection, "example", rekognition.SegmentDetectionConfig{
    			JobId:      "example",
    			MaxResults: 1,
    		})
    	}))
    }
    ```
