## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	securityagent "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/securityagent"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_securityagent_artifacts"})

    		ubx.Data(securityagent.Artifacts, "example", securityagent.ArtifactsConfig{
    			AgentSpaceId: "example",
    			MaxResults:   1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { Artifacts } from "./local-sdk/aws/sdk/typescript/aws/data/securityagent/artifacts.ts";

    export default stack("example", () => {
      intent({ summary: "look up aws_securityagent_artifacts" });

      data(Artifacts, "example", {
        agentSpaceId: "example",
        maxResults: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.securityagent import Artifacts, ArtifactsConfig

    def describe():
        ubx.intent("look up aws_securityagent_artifacts")

        ubx.data(Artifacts, "example", ArtifactsConfig(
            agent_space_id="example",
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Lookup arguments
