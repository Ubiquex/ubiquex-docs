## Example

    package main

    import (
    	apigateway "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/apigateway"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_apigateway_tags"})

    		ubx.Data(apigateway.Tags, "example", apigateway.TagsConfig{
    			ResourceArn: "arn:aws:iam::123456789012:policy/example",
    			Limit:       1,
    			Position:    "example",
    		})
    	}))
    }
