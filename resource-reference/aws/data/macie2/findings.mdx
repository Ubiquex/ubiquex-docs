---
title: "aws_macie2_findings"
description: "`data_aws_macie2_findings` looks up a real, already-existing `aws_macie_findings_filter`."
---

It defines real custom criteria, severity, finding type, affected resource, that either narrows which Macie findings a query returns or automatically archives findings matching those criteria, so routine noise doesn't have to be triaged by hand.

## Example

    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	macie2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/macie2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_macie2_findings"})

    		ubx.Data(macie2.Findings, "example", macie2.FindingsConfig{
    			FindingIds:   []string{"example"},
    			SortCriteria: map[string]any{"attribute_name": "example-attribute-name"},
    		})
    	}))
    }
    ```

    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.macie2 import Findings, FindingsConfig

    def describe():
        ubx.intent("look up aws_macie2_findings")

        ubx.data(Findings, "example", FindingsConfig(
            finding_ids=["example"],
            sort_criteria={"attribute_name": "example-attribute-name"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
