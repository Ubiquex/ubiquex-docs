---
title: "aws_macie2_bucket_statistics"
description: "`aws_macie2_bucket_statistics` reads real, aggregated statistics (object count, size, encryption status) about the S3 buckets Macie monitors."
---

    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    		})
    	}))
    }
