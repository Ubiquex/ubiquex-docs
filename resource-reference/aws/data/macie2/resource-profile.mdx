    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    import (
    	macie2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/macie2"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_macie2_resource_profile"})

    		ubx.Data(macie2.ResourceProfile, "example", macie2.ResourceProfileConfig{
    			ResourceArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
