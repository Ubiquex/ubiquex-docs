    from ubx.aws.data.databrew import Projects, ProjectsConfig

    def describe():
        ubx.intent("look up aws_databrew_projects")

        ubx.data(Projects, "example", ProjectsConfig(
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
