`aws_databrew_projects` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	databrew "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/databrew"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_databrew_projects"})

    		ubx.Data(databrew.Projects, "example", databrew.ProjectsConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Projects } from "@ubx/sdk-aws/aws/data/databrew/projects";

    export default stack("example", () => {
      intent({ summary: "look up aws_databrew_projects" });

      data(Projects, "example", {
        maxResults: 1,

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.databrew import Projects, ProjectsConfig

    def describe():
        ubx.intent("look up aws_databrew_projects")

        ubx.data(Projects, "example", ProjectsConfig(
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
