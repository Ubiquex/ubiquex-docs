## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	databrew "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/databrew"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_databrew_jobs"})

    		ubx.Data(databrew.Jobs, "example", databrew.JobsConfig{
    			DatasetName: "example-dataset-name",
    			MaxResults:  1,
    			ProjectName: "example-project-name",
    		})
    	}))
    }
    ```
