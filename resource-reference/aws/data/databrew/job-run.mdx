## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	databrew "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/databrew"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_databrew_job_run"})

    		ubx.Data(databrew.JobRun, "example", databrew.JobRunConfig{
    			Name:  "example",
    			RunId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    ```

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.databrew import JobRun, JobRunConfig

    def describe():
        ubx.intent("look up aws_databrew_job_run")

        ubx.data(JobRun, "example", JobRunConfig(
            name="example",
            run_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
