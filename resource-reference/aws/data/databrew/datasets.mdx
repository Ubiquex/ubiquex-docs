## Example

    package main

    import (
    	databrew "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/databrew"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_databrew_datasets"})

    		ubx.Data(databrew.Datasets, "example", databrew.DatasetsConfig{
    			MaxResults: 1,
    		})
    	}))
    }
