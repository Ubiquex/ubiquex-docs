## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	cloudtrail "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/cloudtrail"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudtrail_channels"})

    		ubx.Data(cloudtrail.Channels, "example", cloudtrail.ChannelsConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```
