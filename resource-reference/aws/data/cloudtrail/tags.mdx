`aws_cloudtrail_tags` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudtrail "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudtrail"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudtrail_tags"})

    		ubx.Data(cloudtrail.Tags, "example", cloudtrail.TagsConfig{
    			ResourceIdList: []string{"example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.cloudtrail import Tags, TagsConfig

    def describe():
        ubx.intent("look up aws_cloudtrail_tags")

        ubx.data(Tags, "example", TagsConfig(
            resource_id_list=["example"],

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
