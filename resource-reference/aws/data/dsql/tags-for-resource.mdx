## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	dsql "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/dsql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_dsql_tags_for_resource"})

    		ubx.Data(dsql.TagsForResource, "example", dsql.TagsForResourceConfig{
    			ResourceArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { TagsForResource } from "./local-sdk/aws/sdk/typescript/aws/data/dsql/tags_for_resource.ts";

    export default stack("example", () => {
      intent({ summary: "look up aws_dsql_tags_for_resource" });

      data(TagsForResource, "example", {
        resourceArn: "arn:aws:iam::123456789012:policy/example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.dsql import TagsForResource, TagsForResourceConfig

    def describe():
        ubx.intent("look up aws_dsql_tags_for_resource")

        ubx.data(TagsForResource, "example", TagsForResourceConfig(
            resource_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
