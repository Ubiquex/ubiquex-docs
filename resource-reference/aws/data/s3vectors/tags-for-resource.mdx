## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	s3vectors "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/s3vectors"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_s3vectors_tags_for_resource"})

    		ubx.Data(s3vectors.TagsForResource, "example", s3vectors.TagsForResourceConfig{
    			ResourceArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
