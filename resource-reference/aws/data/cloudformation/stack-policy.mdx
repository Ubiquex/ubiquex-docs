## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudformation "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/cloudformation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudformation_stack_policy"})

    		ubx.Data(cloudformation.StackPolicy, "example", cloudformation.StackPolicyConfig{
    			StackName: "example-stack-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.aws.data.cloudformation import StackPolicy, StackPolicyConfig

        ubx.run("example", describe)
