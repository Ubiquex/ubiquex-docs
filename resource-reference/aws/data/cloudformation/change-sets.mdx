---
title: "aws_cloudformation_change_sets"
description: "`data_aws_cloudformation_change_sets` looks up a real, already-existing `aws_cloud_formation_change_set`."
---

It computes a real, previewable set of changes that updating an `aws_cloud_formation_stack` to a new template or parameters would make, without actually applying them, so the changes can be reviewed before anything is modified.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudformation "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/cloudformation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudformation_change_sets"})

    		ubx.Data(cloudformation.ChangeSets, "example", cloudformation.ChangeSetsConfig{
    			StackName: "example-stack-name",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.data.cloudformation import ChangeSets, ChangeSetsConfig

    def describe():
        ubx.intent("look up aws_cloudformation_change_sets")

            stack_name="example-stack-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
