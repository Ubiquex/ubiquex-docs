`aws_cloudformation_stack_resources` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    	cloudformation "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudformation"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudformation_stack_resources"})

    		ubx.Data(cloudformation.StackResources, "example", cloudformation.StackResourcesConfig{
    			LogicalResourceId:  "example",
    			PhysicalResourceId: "example",
    			StackName:          "example-stack-name",
    		})
    	}))
    }
