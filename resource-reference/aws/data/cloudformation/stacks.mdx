`aws_cloudformation_stacks` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudformation "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudformation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudformation_stacks"})

    		ubx.Data(cloudformation.Stacks, "example", cloudformation.StacksConfig{
    			StackStatusFilter: []string{"example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Stacks } from "@ubx/sdk-aws/aws/data/cloudformation/stacks";

    export default stack("example", () => {
      intent({ summary: "look up aws_cloudformation_stacks" });

      data(Stacks, "example", {
        stackStatusFilter: ["example"],
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.cloudformation import Stacks, StacksConfig

    def describe():
        ubx.intent("look up aws_cloudformation_stacks")

        ubx.data(Stacks, "example", StacksConfig(
            stack_status_filter=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
