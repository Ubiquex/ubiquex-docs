    package main

    import (
    	cloudformation "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudformation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudformation_type"})

    		ubx.Data(cloudformation.Type, "example", cloudformation.TypeConfig{
    			VersionId: "example",
    		})
    	}))
    }
