---
title: "aws_cloudformation_generated_template"
description: "`data_aws_cloudformation_generated_template` looks up a real, already-existing `aws_cloud_formation_generated_template`."
---

It produces a real CloudFormation template generated FROM a set of existing, already-deployed resources, CloudFormation's own IaC-generator feature for bringing unmanaged infrastructure under an `aws_cloud_formation_stack` without hand-writing the template first.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudformation "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/cloudformation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudformation_generated_template"})

    		ubx.Data(cloudformation.GeneratedTemplate, "example", cloudformation.GeneratedTemplateConfig{
    			GeneratedTemplateName: "example-generated-template-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { GeneratedTemplate } from "@ubx/sdk-aws/aws/data/cloudformation/generated_template";

    export default stack("example", () => {
      intent({ summary: "look up aws_cloudformation_generated_template" });

      data(GeneratedTemplate, "example", {
        generatedTemplateName: "example-generated-template-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.cloudformation import GeneratedTemplate, GeneratedTemplateConfig

    def describe():
        ubx.intent("look up aws_cloudformation_generated_template")

        ubx.data(GeneratedTemplate, "example", GeneratedTemplateConfig(
            generated_template_name="example-generated-template-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
