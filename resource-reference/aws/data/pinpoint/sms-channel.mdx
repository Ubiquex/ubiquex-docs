## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	pinpoint "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/pinpoint"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_pinpoint_sms_channel"})

    		ubx.Data(pinpoint.SmsChannel, "example", pinpoint.SmsChannelConfig{
    			ApplicationId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  </Tab>
  <Tab title="Python">

    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.pinpoint import SmsChannel, SmsChannelConfig

    def describe():

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
