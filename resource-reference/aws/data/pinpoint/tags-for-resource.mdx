## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	pinpoint "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/pinpoint"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_pinpoint_tags_for_resource"})

    		ubx.Data(pinpoint.TagsForResource, "example", pinpoint.TagsForResourceConfig{
    			ResourceArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.aws.data.pinpoint import TagsForResource, TagsForResourceConfig

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
