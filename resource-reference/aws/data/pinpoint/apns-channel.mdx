## Example

    import (
    	pinpoint "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/pinpoint"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_pinpoint_apns_channel"})

    		ubx.Data(pinpoint.ApnsChannel, "example", pinpoint.ApnsChannelConfig{
    			ApplicationId: "example",
    		})
    	}))
    }
