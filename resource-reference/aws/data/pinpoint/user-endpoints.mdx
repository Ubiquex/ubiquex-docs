## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	pinpoint "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/pinpoint"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_pinpoint_user_endpoints"})

    		ubx.Data(pinpoint.UserEndpoints, "example", pinpoint.UserEndpointsConfig{
    			ApplicationId: "example",
    			UserId:        "example",
    		})
    	}))
    }
    ```
