  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	pinpoint "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/pinpoint"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_pinpoint_endpoint"})

    		ubx.Data(pinpoint.Endpoint, "example", pinpoint.EndpointConfig{
    			ApplicationId: "example",
    			EndpointId:    "example",
    		})
    	}))
    }
    ```

    from ubx.aws.data.pinpoint import Endpoint, EndpointConfig

    def describe():
        ubx.intent("look up aws_pinpoint_endpoint")

            application_id="example",
            endpoint_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
