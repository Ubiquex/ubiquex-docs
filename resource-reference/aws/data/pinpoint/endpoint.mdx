`aws_pinpoint_endpoint` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	pinpoint "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/pinpoint"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_pinpoint_endpoint"})

    		ubx.Data(pinpoint.Endpoint, "example", pinpoint.EndpointConfig{
    			ApplicationId: "example",
    			EndpointId:    "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Endpoint } from "@ubx/sdk-aws/aws/data/pinpoint/endpoint";

    export default stack("example", () => {
      intent({ summary: "look up aws_pinpoint_endpoint" });

      data(Endpoint, "example", {
        applicationId: "example",
        endpointId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.pinpoint import Endpoint, EndpointConfig

    def describe():
        ubx.intent("look up aws_pinpoint_endpoint")

        ubx.data(Endpoint, "example", EndpointConfig(
            application_id="example",
            endpoint_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
