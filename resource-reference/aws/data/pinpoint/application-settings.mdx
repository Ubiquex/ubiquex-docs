---
title: "aws_pinpoint_application_settings"
description: "`data_aws_pinpoint_application_settings` looks up a real, already-existing `aws_pinpoint_application_settings`."
---

It configures real account-level defaults, quiet hours, campaign hook, limits, for an `aws_pinpoint_app`, subject to the same October 30, 2026 Pinpoint retirement as the rest of its engagement surface.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	pinpoint "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/pinpoint"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own application settings"})

    		ubx.Resource(pinpoint.ApplicationSettings, "example", pinpoint.ApplicationSettingsConfig{
    			ApplicationId:            "example",
    			CampaignHook:             map[string]any{"lambda_function_name": "example-lambda-function-name"},
    			CloudWatchMetricsEnabled: true,
    			Limits:                   map[string]any{"daily": 1},
    			QuietTime:                map[string]any{"end": "example", "start": "example"},
    		})
    	}))
    }
    ```

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.pinpoint import ApplicationSettings, ApplicationSettingsConfig
    from ubx.aws.pinpoint.application_settings import ApplicationSettings_CampaignHook, ApplicationSettings_Limits, ApplicationSettings_QuietTime

    def describe():
        ubx.intent("example own application settings")

        ubx.resource(ApplicationSettings, "example", ApplicationSettingsConfig(
            application_id="example",
            campaign_hook=ApplicationSettings_CampaignHook(lambda_function_name="example-lambda-function-name"),
            cloud_watch_metrics_enabled=True,
            limits=ApplicationSettings_Limits(daily=1),
            quiet_time=ApplicationSettings_QuietTime(end="example", start="example"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
