## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	wafv2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/wafv2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_wafv2_mobile_sdk_release"})

    		ubx.Data(wafv2.MobileSdkRelease, "example", wafv2.MobileSdkReleaseConfig{
    			Platform:       "example",
    			ReleaseVersion: "example",
    		})
    	}))
    }
    ```
