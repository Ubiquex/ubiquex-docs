## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	lambda_microvms "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/lambda_microvms"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_lambda_microvms_tags"})

    		ubx.Data(lambda_microvms.LambdaMicrovmsTags, "example", lambda_microvms.LambdaMicrovmsTagsConfig{
    			Resource: "example",
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">

    # ubx sdk gen --only aws --lang py --out ./local-sdk

        ubx.intent("look up aws_lambda_microvms_tags")

        ubx.data(LambdaMicrovmsTags, "example", LambdaMicrovmsTagsConfig(

    if __name__ == "__main__":
        ubx.run("example", describe)
