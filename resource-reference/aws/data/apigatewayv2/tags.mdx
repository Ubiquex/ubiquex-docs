## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	apigatewayv2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/apigatewayv2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_apigatewayv2_tags"})

    		ubx.Data(apigatewayv2.Tags, "example", apigatewayv2.TagsConfig{
    			ResourceArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```

    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.apigatewayv2 import Tags, TagsConfig

    def describe():
        ubx.intent("look up aws_apigatewayv2_tags")

        ubx.data(Tags, "example", TagsConfig(
            resource_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
