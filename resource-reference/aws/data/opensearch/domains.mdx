## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	opensearch "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/opensearch"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_opensearch_domains"})

    		ubx.Data(opensearch.Domains, "example", opensearch.DomainsConfig{
    			DomainNames: []string{"example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { Domains } from "./local-sdk/aws/sdk/typescript/aws/data/opensearch/domains.ts";

    export default stack("example", () => {
      intent({ summary: "look up aws_opensearch_domains" });

      data(Domains, "example", {
        domainNames: ["example"],
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.opensearch import Domains, DomainsConfig

    def describe():
        ubx.intent("look up aws_opensearch_domains")

        ubx.data(Domains, "example", DomainsConfig(
            domain_names=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
