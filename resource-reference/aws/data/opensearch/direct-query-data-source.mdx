`aws_opensearch_direct_query_data_source` looks up a real direct query data source, letting OpenSearch query data in S3 or CloudWatch without ingesting it first.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	opensearch "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/opensearch"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_opensearch_direct_query_data_source"})

    		ubx.Data(opensearch.DirectQueryDataSource, "example", opensearch.DirectQueryDataSourceConfig{
    			DataSourceName: "example-data-source-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { DirectQueryDataSource } from "@ubx/sdk-aws/aws/data/opensearch/direct_query_data_source";

    export default stack("example", () => {
      intent({ summary: "look up aws_opensearch_direct_query_data_source" });

      data(DirectQueryDataSource, "example", {
        dataSourceName: "example-data-source-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.opensearch import DirectQueryDataSource, DirectQueryDataSourceConfig

    def describe():
        ubx.intent("look up aws_opensearch_direct_query_data_source")

        ubx.data(DirectQueryDataSource, "example", DirectQueryDataSourceConfig(
            data_source_name="example-data-source-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
