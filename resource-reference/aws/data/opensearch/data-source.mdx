`aws_opensearch_data_source` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	opensearch "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/opensearch"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_opensearch_data_source"})

    		ubx.Data(opensearch.DataSource, "example", opensearch.DataSourceConfig{
    			DomainName: "example-domain-name",
    			Name:       "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { DataSource } from "@ubx/sdk-aws/aws/data/opensearch/data_source";

    export default stack("example", () => {
      intent({ summary: "look up aws_opensearch_data_source" });

      data(DataSource, "example", {
        domainName: "example-domain-name",
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.opensearch import DataSource, DataSourceConfig

    def describe():
        ubx.intent("look up aws_opensearch_data_source")

        ubx.data(DataSource, "example", DataSourceConfig(
            domain_name="example-domain-name",
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
