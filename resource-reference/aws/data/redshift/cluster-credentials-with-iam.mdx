`aws_redshift_cluster_credentials_with_iam` reads real, temporary database credentials for a Redshift cluster, generated from the caller's own IAM identity.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	redshift "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/redshift"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_redshift_cluster_credentials_with_iam"})

    		ubx.Data(redshift.ClusterCredentialsWithIam, "example", redshift.ClusterCredentialsWithIamConfig{
    			ClusterIdentifier: "example",
    			CustomDomainName:  "example-custom-domain-name",
    			DbName:            "example-db-name",
    			DurationSeconds:   7200,
    		})
    	}))
    }
    ```
