    package main

    import (

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_redshift_recommendations"})

    		ubx.Data(redshift.Recommendations, "example", redshift.RecommendationsConfig{
    			ClusterIdentifier: "example",
    			MaxRecords:        1,
    			NamespaceArn:      "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }

<ResponseField name="max_records" type="number">
  The maximum number of records to return. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="namespace_arn" type="string">
  The real Amazon Resource Name (ARN) of the namespace this lookup is scoped to. **(AI-inferred)** Optional lookup argument.
</ResponseField>
