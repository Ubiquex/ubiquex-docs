## Example

    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iot_streams"})

    		ubx.Data(iot.Streams, "example", iot.StreamsConfig{
    			AscendingOrder: true,
    			MaxResults:     1,
    		})
    	}))
    }

## Lookup arguments

<ResponseField name="ascending_order" type="bool">
  Whether results are returned in ascending order. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="max_results" type="number">
  The maximum number of results to return in a single page. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="next_token" type="string">

</ResponseField>

<ResponseField name="streams" type="list(object)">
