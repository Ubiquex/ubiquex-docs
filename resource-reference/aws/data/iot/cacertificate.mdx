## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iot_cacertificate"})

    		ubx.Data(iot.Cacertificate, "example", iot.CacertificateConfig{
    			CertificateId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { Cacertificate } from "./local-sdk/aws/sdk/typescript/aws/data/iot/cacertificate.ts";

    export default stack("example", () => {
      intent({ summary: "look up aws_iot_cacertificate" });

      data(Cacertificate, "example", {
        certificateId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.iot import Cacertificate, CacertificateConfig

    def describe():
        ubx.intent("look up aws_iot_cacertificate")

        ubx.data(Cacertificate, "example", CacertificateConfig(
            certificate_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
