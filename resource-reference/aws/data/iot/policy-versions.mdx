## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iot_policy_versions"})

    		ubx.Data(iot.PolicyVersions, "example", iot.PolicyVersionsConfig{
    			PolicyName: "example-policy-name",
    		})
    	}))
    }
    ```
  </Tab>
