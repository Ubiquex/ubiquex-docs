`aws_iot_custom_metrics` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iot_custom_metrics"})

    		ubx.Data(iot.CustomMetrics, "example", iot.CustomMetricsConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.iot import CustomMetrics, CustomMetricsConfig

    def describe():
        ubx.intent("look up aws_iot_custom_metrics")

        ubx.data(CustomMetrics, "example", CustomMetricsConfig(
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
