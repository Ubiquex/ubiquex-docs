## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iot_packages"})

    		ubx.Data(iot.Packages, "example", iot.PackagesConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.data.iot import Packages, PackagesConfig

    def describe():
        ubx.intent("look up aws_iot_packages")

        ubx.data(Packages, "example", PackagesConfig(
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
