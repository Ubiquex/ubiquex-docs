`aws_iot_metric_values` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

    ```go
    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iot_metric_values"})

    		ubx.Data(iot.MetricValues, "example", iot.MetricValuesConfig{
    			EndTime:                "example",
    			MetricName:             "example-metric-name",
    			StartTime:              "example",
    			ThingName:              "example-thing-name",
    			DimensionName:          "example-dimension-name",
    			DimensionValueOperator: "example",
    			MaxResults:             1,
    		})
    	}))
    }
    ```
