    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iot_metric_values"})

    		ubx.Data(iot.MetricValues, "example", iot.MetricValuesConfig{
    			EndTime:                "example",
    			MetricName:             "example-metric-name",
    			StartTime:              "example",
    			ThingName:              "example-thing-name",
    			DimensionName:          "example-dimension-name",
    			DimensionValueOperator: "example",
    			MaxResults:             1,
    		})
    	}))
    }
