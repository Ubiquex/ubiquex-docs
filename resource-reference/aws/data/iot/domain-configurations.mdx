    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iot_domain_configurations"})

    			PageSize:    1,
    			ServiceType: "example",
    		})
    	}))
    }
