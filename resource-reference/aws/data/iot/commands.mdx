`aws_iot_commands` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iot_commands"})

    		ubx.Data(iot.Commands, "example", iot.CommandsConfig{
    			CommandParameterName: "example-command-parameter-name",
    			MaxResults:           1,
    			Namespace:            "example",
    			SortOrder:            "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.aws.data.iot import Commands, CommandsConfig

    def describe():
        ubx.intent("look up aws_iot_commands")

        ubx.data(Commands, "example", CommandsConfig(
            command_parameter_name="example-command-parameter-name",
            max_results=1,
            namespace="example",
            sort_order="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
