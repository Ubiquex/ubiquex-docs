## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iot_things"})

    		ubx.Data(iot.Things, "example", iot.ThingsConfig{
    			AttributeName:           "example-attribute-name",
    			AttributeValue:          "example",
    			MaxResults:              1,
    			ThingTypeName:           "example-thing-type-name",
    			UsePrefixAttributeValue: true,
    		})
    	}))
    }
    ```
