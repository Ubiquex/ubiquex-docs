## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iot_dimensions"})

    		ubx.Data(iot.Dimensions, "example", iot.DimensionsConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.data.iot import Dimensions, DimensionsConfig

    def describe():
        ubx.intent("look up aws_iot_dimensions")

        ubx.data(Dimensions, "example", DimensionsConfig(

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="max_results" type="number">

</ResponseField>

## Result properties

<ResponseField name="dimension_names" type="list(string)">

</ResponseField>

<ResponseField name="next_token" type="string">
