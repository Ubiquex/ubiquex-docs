`aws_iot_certificate` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_iot_certificate"})

    		ubx.Data(iot.Certificate, "example", iot.CertificateConfig{
    			CertificateId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Certificate } from "@ubx/sdk-aws/aws/data/iot/certificate";

    export default stack("example", () => {
      intent({ summary: "look up aws_iot_certificate" });

      data(Certificate, "example", {
        certificateId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.iot import Certificate, CertificateConfig

    def describe():
        ubx.intent("look up aws_iot_certificate")

        ubx.data(Certificate, "example", CertificateConfig(
            certificate_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
