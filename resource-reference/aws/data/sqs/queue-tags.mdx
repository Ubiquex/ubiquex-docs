---
title: "aws_sqs_queue_tags"
description: "`data_aws_sqs_queue_tags` looks up a real, already-existing `aws_sqs_queue`: Amazon SQS is a fully managed message queue."
---

`aws_sqs_queue` creates and configures one, standard or FIFO, with a real, typed binding in every SDK language.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	sqs "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/sqs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_sqs_queue_tags"})

    		ubx.Data(sqs.QueueTags, "example", sqs.QueueTagsConfig{
    			QueueUrl: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { QueueTags } from "./local-sdk/aws/sdk/typescript/aws/data/sqs/queue_tags.ts";

    export default stack("example", () => {
      intent({ summary: "look up aws_sqs_queue_tags" });

      data(QueueTags, "example", {
        queueUrl: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.sqs import QueueTags, QueueTagsConfig

    def describe():
        ubx.intent("look up aws_sqs_queue_tags")

        ubx.data(QueueTags, "example", QueueTagsConfig(
            queue_url="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
