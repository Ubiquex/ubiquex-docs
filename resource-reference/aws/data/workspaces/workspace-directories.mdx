`aws_workspaces_workspace_directories` looks up a real WorkSpaces directory, the Active Directory or Identity Center registration WorkSpaces are provisioned against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	workspaces "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/workspaces"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_workspaces_workspace_directories"})

    		ubx.Data(workspaces.WorkspaceDirectories, "example", workspaces.WorkspaceDirectoriesConfig{
    			DirectoryIds:            []string{"example"},
    			Filters:                 []map[string]any{{"name": "example-name"}},
    			Limit:                   1,
    			WorkspaceDirectoryNames: []string{"example"},
    		})
    	}))
    }
    ```
