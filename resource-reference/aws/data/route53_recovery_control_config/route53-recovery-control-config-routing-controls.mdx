`aws_route53_recovery_control_config_routing_controls` looks up a real Recovery Control routing control, an on/off switch used to redirect traffic during a failover.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_route53_recovery_control_config_routing_controls"})

    		ubx.Data(route53_recovery_control_config.Route53RecoveryControlConfigRoutingControls, "example", route53_recovery_control_config.Route53RecoveryControlConfigRoutingControlsConfig{
    			ControlPanelArn: "arn:aws:iam::123456789012:policy/example",
    			MaxResults:      1,
    		})
    	}))
