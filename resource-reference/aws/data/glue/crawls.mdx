## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_glue_crawls"})

    		ubx.Data(glue.Crawls, "example", glue.CrawlsConfig{
    			CrawlerName: "example-crawler-name",
    			Filters:     []map[string]any{{"field_name": "example-field-name"}},
    			MaxResults:  1,
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.data.glue import Crawls, CrawlsConfig

    def describe():
        ubx.intent("look up aws_glue_crawls")

        ubx.data(Crawls, "example", CrawlsConfig(
            crawler_name="example-crawler-name",
            filters=[{"field_name": "example-field-name"}],
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="crawler_name" type="string" required>
  The real name of the crawler this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="filters" type="list(object)">
  Real filter criteria restricting which results are returned. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="max_results" type="number">
  The maximum number of results to return in a single page. **(AI-inferred)** Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="crawls" type="list(object)">
  The real AWS Glue crawler runs included in this result. **(AI-inferred)** Returned by the lookup.
</ResponseField>

<ResponseField name="next_token" type="string">
  An opaque pagination token from a previous response, requesting the next page of results. **(AI-inferred)** Returned by the lookup.
</ResponseField>
