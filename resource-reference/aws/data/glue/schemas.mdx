---
title: "aws_glue_schemas"
description: "`data_aws_glue_schemas` looks up a real, already-existing `aws_glue_schema`."
---

It is a real, versioned schema definition, Avro, JSON, or Protobuf, registered so producers and consumers of streaming data, Kinesis or MSK, agree on and validate against the same actual message structure.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_glue_schemas"})

    		ubx.Data(glue.Schemas, "example", glue.SchemasConfig{
    			MaxResults: 1,
    			RegistryId: map[string]any{"registry_arn": "example"},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.glue import Schemas, SchemasConfig
    from ubx.aws.data.glue.schemas import Schemas_RegistryId

        ubx.intent("look up aws_glue_schemas")

            max_results=1,

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
