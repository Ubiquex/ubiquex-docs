`aws_glue_schemas` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_glue_schemas"})

    		ubx.Data(glue.Schemas, "example", glue.SchemasConfig{
    			MaxResults: 1,
    			RegistryId: map[string]any{"registry_arn": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Schemas } from "@ubx/sdk-aws/aws/data/glue/schemas";

    export default stack("example", () => {
      intent({ summary: "look up aws_glue_schemas" });

      data(Schemas, "example", {
        maxResults: 1,
        registryId: { registryArn: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.glue import Schemas, SchemasConfig

    def describe():
        ubx.intent("look up aws_glue_schemas")

        ubx.data(Schemas, "example", SchemasConfig(
            max_results=1,
            registry_id={"registry_arn": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
