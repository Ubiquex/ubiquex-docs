## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_glue_mltask_run"})

    		ubx.Data(glue.MltaskRun, "example", glue.MltaskRunConfig{
    			TaskRunId:   "example",
    			TransformId: "example",
    		})
    	}))
    }
    ```
  </Tab>
