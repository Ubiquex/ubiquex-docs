---
title: "aws_glue_schema_version"
description: "`data_aws_glue_schema_version` looks up a real, already-existing `aws_glue_schema_version`."
---

It publishes a real, immutable version of an `aws_glue_schema`'s definition, letting a Kafka or Kinesis producer/consumer pin to a specific, compatible schema version rather than the schema's current, evolving definition.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own schema version"})

    		ubx.Resource(glue.SchemaVersion, "example", glue.SchemaVersionConfig{
    			Schema:           map[string]any{"registry_name": "example-registry-name"},
    			SchemaDefinition: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import { intent, resource, stack } from "@ubx/sdk";

    import ubx_sdk as ubx
    from ubx.aws.glue import SchemaVersion, SchemaVersionConfig
    from ubx.aws.glue.schema_version import SchemaVersion_Schema

    def describe():
        ubx.intent("example own schema version")

        ubx.resource(SchemaVersion, "example", SchemaVersionConfig(
            schema=SchemaVersion_Schema(registry_name="example-registry-name"),
            schema_definition="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
