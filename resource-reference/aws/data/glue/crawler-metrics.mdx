---
title: "aws_glue_crawler_metrics"
description: "`data_aws_glue_crawler_metrics` looks up a real, already-existing `aws_glue_crawler`."
---

It scans a real data source, S3, a JDBC database, DynamoDB, and automatically infers and populates an `aws_glue_table`'s own schema in the Glue Data Catalog, rather than a caller writing that schema by hand.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_glue_crawler_metrics"})

    		ubx.Data(glue.CrawlerMetrics, "example", glue.CrawlerMetricsConfig{
    			CrawlerNameList: []string{"example"},
    			MaxResults:      1,
    		})
    	}))
    }
    ```

  <Tab title="Python">

    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.glue import CrawlerMetrics, CrawlerMetricsConfig

    def describe():
        ubx.intent("look up aws_glue_crawler_metrics")

        ubx.data(CrawlerMetrics, "example", CrawlerMetricsConfig(
            crawler_name_list=["example"],
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
