`aws_glue_catalogs` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_glue_catalogs"})

    		ubx.Data(glue.Catalogs, "example", glue.CatalogsConfig{
    			HasDatabases:    true,
    			IncludeRoot:     true,
    			MaxResults:      1,
    			ParentCatalogId: "example",
    			Recursive:       true,
    		})
    	}))
    }
