`aws_glue_plan` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/glue"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_glue_plan"})

    		ubx.Data(glue.Plan, "example", glue.PlanConfig{
    			Mapping:                  []map[string]any{{"source_path": "example"}},
    			Source:                   map[string]any{"database_name": "example-database-name"},

    			Language:                 "example",
    			Location:                 map[string]any{"dynamo_db": []map[string]any{{"name": "example-name"}}},
    			Sinks:                    []map[string]any{{"database_name": "example-database-name"}},
    		})
    	}))
    }
