## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	route53 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/route53"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_route53_tags_for_resource"})

    		ubx.Data(route53.TagsForResource, "example", route53.TagsForResourceConfig{
    			ResourceId:   "example",
    			ResourceType: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    import ubx_sdk as ubx
    ```
