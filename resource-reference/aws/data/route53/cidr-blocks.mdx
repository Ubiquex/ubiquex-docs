## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	route53 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/route53"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_route53_cidr_blocks"})

    		ubx.Data(route53.CidrBlocks, "example", route53.CidrBlocksConfig{
    			CollectionId: "example",
    			LocationName: "example-location-name",
    			MaxResults:   1,
    		})
    	}))
    }
    ```
