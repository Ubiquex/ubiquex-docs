    package main

    	ubx.Main(ubx.Stack("example", func() {

## Lookup arguments

<ResponseField name="global_network_id" type="string" required>
  The real, unique identifier of the AWS Cloud WAN global network this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="route_analysis_id" type="string" required>
  The real, unique identifier of the route analysis this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
