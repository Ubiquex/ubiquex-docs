`aws_redshift_data_statement` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	redshift_data "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/redshift_data"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_redshift_data_statement"})

    		ubx.Data(redshift_data.RedshiftDataStatement, "example", redshift_data.RedshiftDataStatementConfig{
    			Id:              "example",
    			WaitTimeSeconds: 7200,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { RedshiftDataStatement } from "@ubx/sdk-aws/aws/data/redshift_data/redshift_data_statement";

    export default stack("example", () => {
      intent({ summary: "look up aws_redshift_data_statement" });

      data(RedshiftDataStatement, "example", {
    ```

  </Tab>

    ```python
    import ubx_sdk as ubx

        ubx.data(RedshiftDataStatement, "example", RedshiftDataStatementConfig(

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
