    package main

    import (
    	redshift_data "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/redshift_data"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_redshift_data_table"})

    		ubx.Data(redshift_data.RedshiftDataTable, "example", redshift_data.RedshiftDataTableConfig{
    			Database:          "example",
    			ClusterIdentifier: "example",
    			ConnectedDatabase: "example",
    			DbUser:            "example",
    			MaxResults:        1,
    			Schema:            "example",
    			SecretArn:         "arn:aws:iam::123456789012:policy/example",
    			Table:             "example",
    		})
    	}))
    }
