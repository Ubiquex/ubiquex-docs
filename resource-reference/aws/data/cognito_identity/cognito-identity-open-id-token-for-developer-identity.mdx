`aws_cognito_identity_open_id_token_for_developer_identity` reads a real, generated OpenID Connect token for a Cognito identity pool developer-authenticated identity.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cognito_identity "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/cognito_identity"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cognito_identity_open_id_token_for_developer_identity"})

    		ubx.Data(cognito_identity.CognitoIdentityOpenIdTokenForDeveloperIdentity, "example", cognito_identity.CognitoIdentityOpenIdTokenForDeveloperIdentityConfig{
    			IdentityPoolId: "example",
    			Logins:         map[string]string{"managed-by": "ubx"},
    			PrincipalTags:  map[string]string{"managed-by": "ubx"},
    			TokenDuration:  7200,
    		})
    	}))
    }
    ```
