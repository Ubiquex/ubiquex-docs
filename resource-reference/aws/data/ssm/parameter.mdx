`aws_ssm_parameter` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ssm "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/ssm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ssm_parameter"})

    		ubx.Data(ssm.Parameter, "example", ssm.ParameterConfig{
    			Name:           "example",
    			WithDecryption: true,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Parameter } from "@ubx/sdk-aws/aws/data/ssm/parameter";

    export default stack("example", () => {
      intent({ summary: "look up aws_ssm_parameter" });

      data(Parameter, "example", {
        name: "example",
        withDecryption: true,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.ssm import Parameter, ParameterConfig

    def describe():
        ubx.intent("look up aws_ssm_parameter")

        ubx.data(Parameter, "example", ParameterConfig(
            name="example",
            with_decryption=True,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
