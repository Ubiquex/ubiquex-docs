`aws_ssm_parameters` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ssm "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/ssm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ssm_parameters"})

    		ubx.Data(ssm.Parameters, "example", ssm.ParametersConfig{
    			Filters:          []map[string]any{{"key": "example"}},
    			MaxResults:       1,
    			ParameterFilters: []map[string]any{{"key": "example"}},
    			Shared:           true,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.ssm import Parameters, ParametersConfig

    def describe():
        ubx.intent("look up aws_ssm_parameters")

        ubx.data(Parameters, "example", ParametersConfig(
            filters=[{"key": "example"}],
            max_results=1,
            parameter_filters=[{"key": "example"}],
            shared=True,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
