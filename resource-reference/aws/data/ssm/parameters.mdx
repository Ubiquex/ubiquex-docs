---
title: "aws_ssm_parameters"
description: "`data_aws_ssm_parameters` looks up a real, already-existing `aws_ssm_parameter`."
---

It stores a real string, string list, or secure string value in Parameter Store, that applications and other AWS resources reference by name at runtime instead of hardcoding configuration or (for SecureString) secrets directly.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ssm "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/ssm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ssm_parameters"})

    		ubx.Data(ssm.Parameters, "example", ssm.ParametersConfig{
    			Filters:          []map[string]any{{"key": "example"}},
    			MaxResults:       1,
    			ParameterFilters: []map[string]any{{"key": "example"}},
    			Shared:           true,
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.data.ssm import Parameters, ParametersConfig
    from ubx.aws.data.ssm.parameters import Parameters_Filters, Parameters_ParameterFilters

    def describe():
        ubx.intent("look up aws_ssm_parameters")

        ubx.data(Parameters, "example", ParametersConfig(
            filters=[Parameters_Filters(key="example")],
            max_results=1,
            parameter_filters=[Parameters_ParameterFilters(key="example")],
            shared=True,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
