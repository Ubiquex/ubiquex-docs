---
title: "aws_ssm_document"
description: "`data_aws_ssm_document` looks up a real, already-existing `aws_ssm_document`."
---

It defines a real, reusable automation, command, or policy document, in JSON or YAML, that Systems Manager runs against target instances, either directly or through an `aws_ssm_association`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ssm "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/ssm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own document"})

    		ubx.Resource(ssm.Document, "example", ssm.DocumentConfig{
    			Content:        "example",
    			Name:           "example",
    			Attachments:    []map[string]any{{"name": "example-name"}},
    			DocumentFormat: "example",
    			DocumentType:   "example",
    			Requires:       []map[string]any{{"name": "example-name"}},
    			Tags:           []map[string]any{{"key": "example"}},
    			TargetType:     "example",
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.ssm import Document, DocumentConfig
    from ubx.aws.ssm.document import Document_Attachments, Document_Requires, Document_Tags

    def describe():
        ubx.intent("example own document")

        ubx.resource(Document, "example", DocumentConfig(
            content="example",
            name="example",
            attachments=[Document_Attachments(name="example-name")],
            document_format="example",
            document_type="example",
            requires=[Document_Requires(name="example-name")],
            tags=[Document_Tags(key="example")],
            target_type="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
