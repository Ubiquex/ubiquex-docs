## Example

    package main

    import (

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ssm_available_patches"})

    		ubx.Data(ssm.AvailablePatches, "example", ssm.AvailablePatchesConfig{
    			Filters:    []map[string]any{{"key": "example"}},
    			MaxResults: 1,
    		})
    	}))
    }
