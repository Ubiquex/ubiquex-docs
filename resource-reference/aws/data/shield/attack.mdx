## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	shield "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/shield"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_shield_attack"})

    		ubx.Data(shield.Attack, "example", shield.AttackConfig{
    			AttackId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { Attack } from "./local-sdk/aws/sdk/typescript/aws/data/shield/attack.ts";

    export default stack("example", () => {
      intent({ summary: "look up aws_shield_attack" });

      data(Attack, "example", {
        attackId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.shield import Attack, AttackConfig

    def describe():
        ubx.intent("look up aws_shield_attack")

        ubx.data(Attack, "example", AttackConfig(
            attack_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
