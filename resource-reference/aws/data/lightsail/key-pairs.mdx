## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	lightsail "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/lightsail"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_lightsail_key_pairs"})

    		ubx.Data(lightsail.KeyPairs, "example", lightsail.KeyPairsConfig{
    			IncludeDefaultKeyPair: true,
    			PageToken:             "example",
    		})
    	}))
    }
    ```
