---
title: "aws_lightsail_static_ip"
description: "`data_aws_lightsail_static_ip` looks up a real, already-existing `aws_lightsail_static_ip`."
---

It allocates a real static public IP address that can be attached to a Lightsail instance, keeping the instance reachable at a fixed address across stops, starts, and replacements.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	lightsail "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/lightsail"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_lightsail_static_ip"})

    		ubx.Data(lightsail.StaticIp, "example", lightsail.StaticIpConfig{
    			StaticIpName: "example-static-ip-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.lightsail import StaticIp, StaticIpConfig

    def describe():
        ubx.intent("look up aws_lightsail_static_ip")

        ubx.data(StaticIp, "example", StaticIpConfig(
            static_ip_name="example-static-ip-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
