    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	rds "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/rds"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_rds_certificates"})

    		ubx.Data(rds.Certificates, "example", rds.CertificatesConfig{
    			CertificateIdentifier: "example",
    			Filters:               []map[string]any{{"name": "example-name"}},
    			MaxRecords:            1,
    		})
    	}))
    }

## Lookup arguments

<ResponseField name="certificate_identifier" type="string">
  The real, unique identifier of the certificate this lookup is scoped to. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="filters" type="list(object)">
  Real filter criteria restricting which results are returned. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="max_records" type="number">
  The maximum number of records to return. **(AI-inferred)** Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="certificates" type="list(object)">
  The real certificates included in this result. **(AI-inferred)** Returned by the lookup.
</ResponseField>

<ResponseField name="default_certificate_for_new_launches" type="string">

</ResponseField>
