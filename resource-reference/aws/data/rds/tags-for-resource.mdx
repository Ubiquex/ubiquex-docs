## Example

    ```go
    package main

    import (
    	rds "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/rds"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_rds_tags_for_resource"})

    		ubx.Data(rds.TagsForResource, "example", rds.TagsForResourceConfig{
    			ResourceName: "example-resource-name",
    			Filters:      []map[string]any{{"name": "example-name"}},
    		})
    	}))
    }
    ```
