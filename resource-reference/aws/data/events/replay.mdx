## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	events "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/events"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_events_replay"})

    		ubx.Data(events.Replay, "example", events.ReplayConfig{
    			ReplayName: "example-replay-name",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.data.events import Replay, ReplayConfig

    def describe():

        ubx.data(Replay, "example", ReplayConfig(
            replay_name="example-replay-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
