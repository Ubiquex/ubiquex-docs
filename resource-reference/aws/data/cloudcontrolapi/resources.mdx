    import (

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudcontrolapi_resources"})

    		ubx.Data(cloudcontrolapi.Resources, "example", cloudcontrolapi.ResourcesConfig{
    			TypeName:      "example-type-name",
    			MaxResults:    1,
    			ResourceModel: "example",
    			RoleArn:       "arn:aws:iam::123456789012:policy/example",
    			TypeVersionId: "example",
    		})
    	}))
    }
