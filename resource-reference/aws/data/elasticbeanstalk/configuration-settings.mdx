    ```go
    package main

    import (
    	elasticbeanstalk "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/elasticbeanstalk"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_elasticbeanstalk_configuration_settings"})

    		ubx.Data(elasticbeanstalk.ConfigurationSettings, "example", elasticbeanstalk.ConfigurationSettingsConfig{
    			ApplicationName: "example-application-name",
    			EnvironmentName: "example-environment-name",
    			TemplateName:    "example-template-name",
    		})
    	}))
    }
    ```
