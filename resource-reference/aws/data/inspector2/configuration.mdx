## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	inspector2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/inspector2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_inspector2_configuration"})

    		ubx.Data(inspector2.Configuration, "example", inspector2.ConfigurationConfig{
    			AccountId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.inspector2 import Configuration, ConfigurationConfig

    def describe():
        ubx.intent("look up aws_inspector2_configuration")

        ubx.data(Configuration, "example", ConfigurationConfig(
            account_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
