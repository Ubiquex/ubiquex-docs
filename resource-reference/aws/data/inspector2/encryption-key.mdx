title: "aws_inspector2_encryption_key"

---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	inspector2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/inspector2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_inspector2_encryption_key"})

    		ubx.Data(inspector2.EncryptionKey, "example", inspector2.EncryptionKeyConfig{
    			ResourceType: "example",
    			ScanType:     "example",
    		})
    	}))
    }
    ```
