`aws_thinclient_devices` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	thinclient "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/thinclient"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_thinclient_devices"})

    		ubx.Data(thinclient.Devices, "example", thinclient.DevicesConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Devices } from "@ubx/sdk-aws/aws/data/thinclient/devices";

    export default stack("example", () => {
      intent({ summary: "look up aws_thinclient_devices" });

      data(Devices, "example", {
        maxResults: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.thinclient import Devices, DevicesConfig

    def describe():
        ubx.intent("look up aws_thinclient_devices")

        ubx.data(Devices, "example", DevicesConfig(
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
