`aws_tagging_resources` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	tagging "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/tagging"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_tagging_resources"})

    		ubx.Data(tagging.Resources, "example", tagging.ResourcesConfig{
    			ExcludeCompliantResources: true,
    			IncludeComplianceDetails:  true,
    			ResourceArnlist:           []string{"example"},
    			ResourceTypeFilters:       []string{"example"},
    			ResourcesPerPage:          1,
    			TagFilters:                []map[string]any{{"key": "example"}},
    			TagsPerPage:               1,
    		})
    	}))
    }
    ```
