    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    import (

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudsearch_availability_options"})

    		ubx.Data(cloudsearch.AvailabilityOptions, "example", cloudsearch.AvailabilityOptionsConfig{

    		})
    	}))
    }
