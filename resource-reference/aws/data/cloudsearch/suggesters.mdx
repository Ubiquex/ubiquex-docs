    package main

    import (
    	cloudsearch "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudsearch"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudsearch_suggesters"})

    		ubx.Data(cloudsearch.Suggesters, "example", cloudsearch.SuggestersConfig{
    			DomainName:     "example-domain-name",
    			Deployed:       true,
    			SuggesterNames: []string{"example"},
    		})
    	}))
    }
