`aws_sts_web_identity_token` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

    ```go
    package main

    import (
    	sts "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/sts"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_sts_web_identity_token"})

    		ubx.Data(sts.WebIdentityToken, "example", sts.WebIdentityTokenConfig{
    			Audience:         []string{"example"},
    			SigningAlgorithm: "example",
    			DurationSeconds:  7200,
    			Tags:             []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
