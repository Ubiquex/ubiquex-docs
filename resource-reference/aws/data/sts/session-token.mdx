## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	sts "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/sts"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_sts_session_token"})

    		ubx.Data(sts.SessionToken, "example", sts.SessionTokenConfig{
    			DurationSeconds: 7200,
    			SerialNumber:    "example",
    			TokenCode:       "example",
    		})
    	}))
    }
    ```
