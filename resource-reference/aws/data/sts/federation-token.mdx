    from ubx.aws.data.sts import FederationToken, FederationTokenConfig

    def describe():
        ubx.intent("look up aws_sts_federation_token")

        access_policy = json.dumps({
            "Version": "2012-10-17",
            "Statement": [{
                "Effect": "Allow",
                "Action": "*",
                "Resource": "*",
            }],
        })

        ubx.data(FederationToken, "example", FederationTokenConfig(
            name="example",
            duration_seconds=7200,
            policy=access_policy,
            policy_arns=[{"arn": "example"}],
            tags=[{"key": "example"}],
        ))
