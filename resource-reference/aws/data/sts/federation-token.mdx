`aws_sts_federation_token` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	sts "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/sts"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_sts_federation_token"})

    		accessPolicy, _ := json.Marshal(map[string]any{
    			"Version": "2012-10-17",
    			"Statement": []map[string]any{{
    				"Effect":   "Allow",
    				"Action":   "*",
    				"Resource": "*",
    			}},
    		})

    		ubx.Data(sts.FederationToken, "example", sts.FederationTokenConfig{
    			Name:            "example",
    			DurationSeconds: 7200,
    			Policy:          string(accessPolicy),
    			PolicyArns:      []map[string]any{{"arn": "example"}},
    			Tags:            []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.data.sts import FederationToken, FederationTokenConfig

    def describe():
        ubx.intent("look up aws_sts_federation_token")

        access_policy = json.dumps({
            "Version": "2012-10-17",
            "Statement": [{
                "Effect": "Allow",
                "Action": "*",
                "Resource": "*",
            }],
        })

        ubx.data(FederationToken, "example", FederationTokenConfig(
            name="example",
            duration_seconds=7200,
            policy=access_policy,
            policy_arns=[{"arn": "example"}],
            tags=[{"key": "example"}],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
