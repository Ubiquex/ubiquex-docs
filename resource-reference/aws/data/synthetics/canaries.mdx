---
title: "aws_synthetics_canaries"
description: "`data_aws_synthetics_canaries` looks up a real, already-existing `aws_synthetics_canary`."
---

It creates a real, scheduled script, a Node.js or Python Selenium/Playwright-style script, that CloudWatch Synthetics runs on a recurring cadence against a live endpoint or UI flow, proactively detecting an outage or broken user journey before a real user hits it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	synthetics "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/synthetics"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_synthetics_canaries"})

    		ubx.Data(synthetics.Canaries, "example", synthetics.CanariesConfig{
    			MaxResults: 1,
    			Names:      []string{"example"},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.data.synthetics import Canaries, CanariesConfig

    def describe():
        ubx.intent("look up aws_synthetics_canaries")

        ubx.data(Canaries, "example", CanariesConfig(
            max_results=1,
            names=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

</ResponseField>
