## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	eks "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/eks"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_eks_insights"})

    		ubx.Data(eks.Insights, "example", eks.InsightsConfig{
    			ClusterName: "example-cluster-name",
    			Filter:      map[string]any{"categories": []string{"example"}},
    			MaxResults:  1,
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.eks import Insights, InsightsConfig
    from ubx.aws.data.eks.insights import Insights_Filter

    def describe():
        ubx.intent("look up aws_eks_insights")

        ubx.data(Insights, "example", InsightsConfig(
            cluster_name="example-cluster-name",
            filter=Insights_Filter(categories=["example"]),
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
