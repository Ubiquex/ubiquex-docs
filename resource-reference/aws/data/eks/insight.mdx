## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	eks "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/eks"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_eks_insight"})

    		ubx.Data(eks.Insight, "example", eks.InsightConfig{
    			ClusterName: "example-cluster-name",
    			Id:          "example",
    		})
    	}))
    }
    ```
