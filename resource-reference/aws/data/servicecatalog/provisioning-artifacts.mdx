`aws_servicecatalog_provisioning_artifacts` looks up one real, versioned provisioning artifact (the underlying CloudFormation template) of a Service Catalog product.

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
