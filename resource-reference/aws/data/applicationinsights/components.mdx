    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_applicationinsights_components"})

    		ubx.Data(applicationinsights.Components, "example", applicationinsights.ComponentsConfig{
    			ResourceGroupName: "example-resource-group-name",
    			AccountId:         "example",
    			MaxResults:        1,
    		})
    	}))
    }
