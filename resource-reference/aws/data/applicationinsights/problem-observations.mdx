    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_applicationinsights_problem_observations"})

    		ubx.Data(applicationinsights.ProblemObservations, "example", applicationinsights.ProblemObservationsConfig{
    			ProblemId: "example",
    			AccountId: "example",
    		})
    	}))
    }
