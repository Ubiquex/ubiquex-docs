    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	applicationinsights "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/applicationinsights"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_applicationinsights_problem"})

    		ubx.Data(applicationinsights.Problem, "example", applicationinsights.ProblemConfig{
    			ProblemId: "example",
    			AccountId: "example",
    		})
    	}))
    }
    ```

    def describe():
