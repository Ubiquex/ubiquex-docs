---
title: "aws_applicationinsights_observation"
description: "`aws_applicationinsights_observation` looks up one real observation (a detected anomaly) contributing to an Application Insights problem."
---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	applicationinsights "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/applicationinsights"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_applicationinsights_observation"})

    		ubx.Data(applicationinsights.Observation, "example", applicationinsights.ObservationConfig{
    			ObservationId: "example",
    			AccountId:     "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Observation } from "@ubx/sdk-aws/aws/data/applicationinsights/observation";

    export default stack("example", () => {
      intent({ summary: "look up aws_applicationinsights_observation" });

      data(Observation, "example", {
        observationId: "example",
        accountId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.applicationinsights import Observation, ObservationConfig

    def describe():
        ubx.intent("look up aws_applicationinsights_observation")

        ubx.data(Observation, "example", ObservationConfig(
            observation_id="example",
            account_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Lookup arguments

<ResponseField name="account_id" type="string">
  The real, unique identifier of the AWS account this lookup is scoped to. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="observation_id" type="string" required>
  The real, unique identifier of the observation this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
