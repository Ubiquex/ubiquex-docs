`aws_applicationinsights_component_configuration` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    	applicationinsights "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/applicationinsights"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_applicationinsights_component_configuration"})

    		ubx.Data(applicationinsights.ComponentConfiguration, "example", applicationinsights.ComponentConfigurationConfig{
    			ComponentName:     "example-component-name",
    			ResourceGroupName: "example-resource-group-name",
    			AccountId:         "example",
    		})
    	}))
    }
