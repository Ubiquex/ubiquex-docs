## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	medialive "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/medialive"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_medialive_schedule"})

    		ubx.Data(medialive.Schedule, "example", medialive.ScheduleConfig{
    			ChannelId:  "example",
    			MaxResults: 1,
    		})
    	}))
    }
    ```
