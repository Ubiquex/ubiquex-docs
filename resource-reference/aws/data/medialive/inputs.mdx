## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	medialive "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/medialive"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_medialive_inputs"})

    		ubx.Data(medialive.Inputs, "example", medialive.InputsConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx

    if __name__ == "__main__":
