## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	medialive "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/medialive"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_medialive_sdi_sources"})

    		ubx.Data(medialive.SdiSources, "example", medialive.SdiSourcesConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.data.medialive import SdiSources, SdiSourcesConfig

        ubx.data(SdiSources, "example", SdiSourcesConfig(
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
