## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	medialive "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/medialive"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_medialive_input_device"})

    		ubx.Data(medialive.InputDevice, "example", medialive.InputDeviceConfig{
    			InputDeviceId: "example",
    		})
    	}))
    }
    ```
