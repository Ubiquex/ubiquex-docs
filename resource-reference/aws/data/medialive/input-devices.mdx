    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_medialive_input_devices"})

    		ubx.Data(medialive.InputDevices, "example", medialive.InputDevicesConfig{
