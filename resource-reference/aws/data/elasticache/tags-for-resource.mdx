## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	elasticache "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/elasticache"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_elasticache_tags_for_resource"})

    		ubx.Data(elasticache.TagsForResource, "example", elasticache.TagsForResourceConfig{
    			ResourceName: "example-resource-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { TagsForResource } from "@ubx/sdk-aws/aws/data/elasticache/tags_for_resource";

    export default stack("example", () => {
      intent({ summary: "look up aws_elasticache_tags_for_resource" });

      data(TagsForResource, "example", {
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.elasticache import TagsForResource, TagsForResourceConfig

    def describe():
        ubx.intent("look up aws_elasticache_tags_for_resource")

        ubx.data(TagsForResource, "example", TagsForResourceConfig(
            resource_name="example-resource-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
