`aws_observabilityadmin_resource_telemetry_for_organization` reads real, discovered telemetry configuration for resources across an AWS Organization, via Observability Admin.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	observabilityadmin "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/observabilityadmin"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_observabilityadmin_resource_telemetry_for_organization"})

    		ubx.Data(observabilityadmin.ResourceTelemetryForOrganization, "example", observabilityadmin.ResourceTelemetryForOrganizationConfig{
    			AccountIdentifiers:          []string{"example"},
    			MaxResults:                  1,
    			ResourceIdentifierPrefix:    "example",
    			ResourceTags:                map[string]string{"managed-by": "ubx"},
    			ResourceTypes:               []string{"example"},
    			TelemetryConfigurationState: map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```
