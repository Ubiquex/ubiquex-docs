## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	swf "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/swf"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_swf_domain"})

    		ubx.Data(swf.Domain, "example", swf.DomainConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
