    package main

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_swf_workflow_type"})

    		ubx.Data(swf.WorkflowType, "example", swf.WorkflowTypeConfig{

    		})
    	}))
    }
