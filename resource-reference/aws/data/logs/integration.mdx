---
title: "aws_logs_integration"
description: "`data_aws_logs_integration` looks up a real, already-existing `aws_logs_integration`."
---

It creates a real, zero-ETL integration between CloudWatch Logs and Amazon OpenSearch Service, currently the only supported integration type, enabling curated vended-log dashboards powered by OpenSearch analytics without a separate ingestion pipeline. Only one integration is currently supported per account, and an existing integration cannot be modified in place.

## Example

    ```go
    package main

    import (
    	logs "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/logs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own integration"})

    		ubx.Resource(logs.Integration, "example", logs.IntegrationConfig{
    			IntegrationName: "example-integration-name",
    			IntegrationType: "example",
    			ResourceConfig:  map[string]any{"open_search_resource_config": map[string]any{"dashboard_viewer_principals": []string{"example"}, "data_source_role_arn": "example"}},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.logs import Integration, IntegrationConfig
    from ubx.aws.logs.integration import Integration_ResourceConfig, Integration_ResourceConfig_OpenSearchResourceConfig

    def describe():
        ubx.intent("example own integration")

        ubx.resource(Integration, "example", IntegrationConfig(
            integration_name="example-integration-name",
            integration_type="example",
            resource_config=Integration_ResourceConfig(open_search_resource_config=Integration_ResourceConfig_OpenSearchResourceConfig(dashboard_viewer_principals=["example"], data_source_role_arn="example")),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
