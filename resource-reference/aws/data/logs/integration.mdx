`aws_logs_integration` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	logs "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/logs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_logs_integration"})

    		ubx.Data(logs.Integration, "example", logs.IntegrationConfig{
    			IntegrationName: "example-integration-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Integration } from "@ubx/sdk-aws/aws/data/logs/integration";

    export default stack("example", () => {
      intent({ summary: "look up aws_logs_integration" });

      data(Integration, "example", {
        integrationName: "example-integration-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.logs import Integration, IntegrationConfig

    def describe():
        ubx.intent("look up aws_logs_integration")

        ubx.data(Integration, "example", IntegrationConfig(
            integration_name="example-integration-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
