## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	logs "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/logs"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_logs_data_protection_policy"})

    		ubx.Data(logs.DataProtectionPolicy, "example", logs.DataProtectionPolicyConfig{
    			LogGroupIdentifier: "example",
    		})
    	}))
    }
    ```
