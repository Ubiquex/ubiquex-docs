`aws_logs_configuration_templates` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

    ```go
    package main

    import (
    	logs "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/logs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_logs_configuration_templates"})

    		ubx.Data(logs.ConfigurationTemplates, "example", logs.ConfigurationTemplatesConfig{
    			DeliveryDestinationTypes: []string{"example"},
    			Limit:                    1,
    			LogTypes:                 []string{"example"},
    			ResourceTypes:            []string{"example"},
    			Service:                  "example",
    		})
    	}))
    }
    ```
