## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	logs "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/logs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_logs_lookup_tables"})

    		ubx.Data(logs.LookupTables, "example", logs.LookupTablesConfig{
    			LookupTableNamePrefix: "example",
    			MaxResults:            1,
    		})
    	}))
    }
    ```

    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.logs import LookupTables, LookupTablesConfig

    def describe():
        ubx.intent("look up aws_logs_lookup_tables")

        ubx.data(LookupTables, "example", LookupTablesConfig(
            lookup_table_name_prefix="example",
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="lookup_table_name_prefix" type="string">
  The real S3 key prefix used for the lookup table name. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="max_results" type="number">
  The maximum number of results to return in a single page. **(AI-inferred)** Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="lookup_tables" type="list(object)">
  The real lookup tables included in this result. **(AI-inferred)** Returned by the lookup.
</ResponseField>

<ResponseField name="next_token" type="string">
  &lt;p&gt;The token for the next set of items to return. The token expires after 24 hours.&lt;/p&gt; Returned by the lookup.
