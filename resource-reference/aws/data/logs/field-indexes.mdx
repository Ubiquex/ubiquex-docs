## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	logs "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/logs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_logs_field_indexes"})

    		ubx.Data(logs.FieldIndexes, "example", logs.FieldIndexesConfig{
    			LogGroupIdentifiers: []string{"example"},
    			IndexCategories:     []string{"example"},
    		})
    	}))
    }
    ```
