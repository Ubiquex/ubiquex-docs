    package main

    import (

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    			LoggingConfiguration:      map[string]any{"log_destination_configs": []map[string]any{{"log_destination": "example"}}},
    			EnableMonitoringDashboard: true,
    			FirewallName:              "example-firewall-name",
    		})
    	}))
    }

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.networkfirewall import LoggingConfiguration, LoggingConfigurationConfig
    from ubx.aws.networkfirewall.logging_configuration import LoggingConfiguration_LoggingConfiguration, LoggingConfiguration_LoggingConfiguration_LogDestinationConfigs

    def describe():
        ubx.intent("example own logging configuration")

        ubx.resource(LoggingConfiguration, "example", LoggingConfigurationConfig(
            firewall_arn="arn:aws:iam::123456789012:policy/example",
            logging_configuration=LoggingConfiguration_LoggingConfiguration(log_destination_configs=[LoggingConfiguration_LoggingConfiguration_LogDestinationConfigs(log_destination="example")]),
            enable_monitoring_dashboard=True,
            firewall_name="example-firewall-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
