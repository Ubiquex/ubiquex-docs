`aws_globalaccelerator_custom_routing_port_mappings_by_destination` reads the real port mappings of a Global Accelerator custom routing endpoint group, filtered to a specific destination address.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	globalaccelerator "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/globalaccelerator"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_globalaccelerator_custom_routing_port_mappings_by_destination"})

    		ubx.Data(globalaccelerator.CustomRoutingPortMappingsByDestination, "example", globalaccelerator.CustomRoutingPortMappingsByDestinationConfig{
    			DestinationAddress: "example",
    			EndpointId:         "example",
    			MaxResults:         1,
    		})
    	}))
    }
    ```
