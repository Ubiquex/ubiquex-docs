`aws_globalaccelerator_custom_routing_accelerator_attributes` reads the real, configured attributes (such as flow log settings) of a Global Accelerator custom routing accelerator.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	globalaccelerator "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/globalaccelerator"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_globalaccelerator_custom_routing_accelerator_attributes"})

    		ubx.Data(globalaccelerator.CustomRoutingAcceleratorAttributes, "example", globalaccelerator.CustomRoutingAcceleratorAttributesConfig{
    			AcceleratorArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
