    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_qbusiness_media"})

    		ubx.Data(qbusiness.Media, "example", qbusiness.MediaConfig{
    			ApplicationId:  "example",
    			ConversationId: "example",
    			MediaId:        "example",
    			MessageId:      "example",
    		})
    	}))
    }
