---
title: "aws_ivs_playback_key_pair"
description: "`data_aws_ivs_playback_key_pair` looks up a real, already-existing `aws_ivs_playback_key_pair`."
---

It signs real, private playback tokens for an `aws_ivs_channel` configured to require authorization, restricting viewing to clients that present a token signed by this specific key pair.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ivs "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/ivs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own playback key pair"})

    		ubx.Resource(ivs.PlaybackKeyPair, "example", ivs.PlaybackKeyPairConfig{
    			Name:              "example",
    			PublicKeyMaterial: "example",
    			Tags:              []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.ivs import PlaybackKeyPair, PlaybackKeyPairConfig
    from ubx.aws.ivs.playback_key_pair import PlaybackKeyPair_Tags

    def describe():
        ubx.intent("example own playback key pair")

        ubx.resource(PlaybackKeyPair, "example", PlaybackKeyPairConfig(
            name="example",
            public_key_material="example",
            tags=[PlaybackKeyPair_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
