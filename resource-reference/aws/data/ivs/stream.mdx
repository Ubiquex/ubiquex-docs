---
title: "aws_ivs_stream"
description: "`data_aws_ivs_stream` looks up a real, already-existing `aws_ivs_stream_key`."
---

It is the real, secret credential streaming software uses to authenticate and publish video to a specific `aws_ivs_channel`, distinct from the channel's own public ingest endpoint URL.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	ivs "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/ivs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ivs_stream"})

    		ubx.Data(ivs.Stream, "example", ivs.StreamConfig{
    			ChannelArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```typescript
    export default stack("example", () => {
    ```

  <Tab title="Python">

    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.ivs import Stream, StreamConfig

    def describe():
        ubx.intent("look up aws_ivs_stream")

        ubx.data(Stream, "example", StreamConfig(
            channel_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
