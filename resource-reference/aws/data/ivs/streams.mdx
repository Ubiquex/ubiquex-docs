`aws_ivs_streams` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ivs "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/ivs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ivs_streams"})

    		ubx.Data(ivs.Streams, "example", ivs.StreamsConfig{
    			FilterBy:   map[string]any{"health": "example"},
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

    export default stack("example", () => {

      data(Streams, "example", {

  </Tab>
  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.aws.data.ivs import Streams, StreamsConfig

    def describe():
        ubx.intent("look up aws_ivs_streams")

        ubx.data(Streams, "example", StreamsConfig(
            filter_by={"health": "example"},
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
