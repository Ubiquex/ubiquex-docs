## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	codeartifact "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/codeartifact"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_codeartifact_authorization_token"})

    		ubx.Data(codeartifact.AuthorizationToken, "example", codeartifact.AuthorizationTokenConfig{
    			Domain:          "example",
    			DomainOwner:     "example",
    			DurationSeconds: 7200,
    		})
    	}))
    }
    ```
