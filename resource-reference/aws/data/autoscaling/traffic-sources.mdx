`aws_autoscaling_traffic_sources` reads the real traffic sources (load balancers or VPC Lattice target groups) an Auto Scaling group registers its own instances with.

## Example

  <Tab title="Go">

    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    import (
    	autoscaling "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/autoscaling"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_autoscaling_traffic_sources"})

    		ubx.Data(autoscaling.TrafficSources, "example", autoscaling.TrafficSourcesConfig{
    			AutoScalingGroupName: "example-auto-scaling-group-name",
    			MaxRecords:           1,
    			TrafficSourceType:    "example",
    		})
    	}))
    }

<ResponseField name="traffic_source_type" type="string">
