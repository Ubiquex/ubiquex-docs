`aws_bedrock_agentcore_workload_access_token_for_user_id` reads a real, temporary workload identity access token for a Bedrock AgentCore agent, scoped to a specific user ID.

    ```go
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_bedrock_agentcore_workload_access_token_for_user_id"})

    		ubx.Data(bedrock_agentcore.BedrockAgentcoreWorkloadAccessTokenForUserId, "example", bedrock_agentcore.BedrockAgentcoreWorkloadAccessTokenForUserIdConfig{
    			UserId:       "example",
    			WorkloadName: "example-workload-name",
    		})
    	}))
    }
    ```
