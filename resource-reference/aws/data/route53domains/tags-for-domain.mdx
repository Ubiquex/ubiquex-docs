## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	route53domains "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/route53domains"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_route53domains_tags_for_domain"})

    		ubx.Data(route53domains.TagsForDomain, "example", route53domains.TagsForDomainConfig{
    			DomainName: "example-domain-name",
    		})
    	}))
    }
    ```
  </Tab>
