## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	route53domains "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/route53domains"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_route53domains_domain_detail"})

    		ubx.Data(route53domains.DomainDetail, "example", route53domains.DomainDetailConfig{
    			DomainName: "example-domain-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { DomainDetail } from "./local-sdk/aws/sdk/typescript/aws/data/route53domains/domain_detail.ts";

    export default stack("example", () => {
      intent({ summary: "look up aws_route53domains_domain_detail" });

      data(DomainDetail, "example", {
        domainName: "example-domain-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.route53domains import DomainDetail, DomainDetailConfig

    def describe():
        ubx.intent("look up aws_route53domains_domain_detail")

        ubx.data(DomainDetail, "example", DomainDetailConfig(
            domain_name="example-domain-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
