## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	mediastore "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/mediastore"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_mediastore_cors_policy"})

    		ubx.Data(mediastore.CorsPolicy, "example", mediastore.CorsPolicyConfig{
    			ContainerName: "example-container-name",
    		})
    	}))
    }
    ```
