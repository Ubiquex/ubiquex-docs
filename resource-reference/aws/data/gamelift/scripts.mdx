`aws_gamelift_scripts` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	gamelift "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/gamelift"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_gamelift_scripts"})

    		ubx.Data(gamelift.Scripts, "example", gamelift.ScriptsConfig{
    			Limit: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Scripts } from "@ubx/sdk-aws/aws/data/gamelift/scripts";

    export default stack("example", () => {
      intent({ summary: "look up aws_gamelift_scripts" });

      data(Scripts, "example", {
        limit: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.gamelift import Scripts, ScriptsConfig

    def describe():
        ubx.intent("look up aws_gamelift_scripts")

        ubx.data(Scripts, "example", ScriptsConfig(
            limit=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
