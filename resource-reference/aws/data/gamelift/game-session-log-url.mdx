    package main

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_gamelift_game_session_log_url"})

    		ubx.Data(gamelift.GameSessionLogUrl, "example", gamelift.GameSessionLogUrlConfig{
    			GameSessionId: "example",
    		})
    	}))
    }
