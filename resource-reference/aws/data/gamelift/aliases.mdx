## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	gamelift "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/gamelift"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_gamelift_aliases"})

    		ubx.Data(gamelift.Aliases, "example", gamelift.AliasesConfig{
    			Name:                "example",
    			Limit:               1,
    			RoutingStrategyType: "example",
    		})
    	}))
    }
    ```
