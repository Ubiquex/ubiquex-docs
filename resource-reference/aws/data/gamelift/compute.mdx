## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	gamelift "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/gamelift"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_gamelift_compute"})

    		ubx.Data(gamelift.Compute, "example", gamelift.ComputeConfig{
    			ComputeName: "example-compute-name",
    			FleetId:     "example",
    		})
    	}))
    }
    ```

    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.gamelift import Compute, ComputeConfig

    def describe():
        ubx.intent("look up aws_gamelift_compute")

        ubx.data(Compute, "example", ComputeConfig(
            compute_name="example-compute-name",
            fleet_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="compute_name" type="string" required>
  The real name of the compute resource this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="fleet_id" type="string" required>
  The real, unique identifier of the fleet this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
