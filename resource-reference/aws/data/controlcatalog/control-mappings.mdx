`aws_controlcatalog_control_mappings` reads the real mappings between an AWS Control Catalog control and the compliance frameworks or standards it implements.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	controlcatalog "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/controlcatalog"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_controlcatalog_control_mappings"})

    		ubx.Data(controlcatalog.ControlMappings, "example", controlcatalog.ControlMappingsConfig{
    			Filter:     map[string]any{"common_control_arns": []string{"example"}},
    			MaxResults: 1,
    		})
    	}))
    }
    ```
