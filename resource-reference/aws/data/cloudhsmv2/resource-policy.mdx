    ```go
    package main

    import (
    	cloudhsmv2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/cloudhsmv2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudhsmv2_resource_policy"})

    		ubx.Data(cloudhsmv2.ResourcePolicy, "example", cloudhsmv2.ResourcePolicyConfig{
    			ResourceArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
