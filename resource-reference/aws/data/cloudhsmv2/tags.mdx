## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudhsmv2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/cloudhsmv2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudhsmv2_tags"})

    		ubx.Data(cloudhsmv2.Tags, "example", cloudhsmv2.TagsConfig{
    			ResourceId: "example",
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>
