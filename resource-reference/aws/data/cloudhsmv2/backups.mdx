`aws_cloudhsmv2_backups` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudhsmv2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudhsmv2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudhsmv2_backups"})

    		ubx.Data(cloudhsmv2.Backups, "example", cloudhsmv2.BackupsConfig{
    			Filters:       map[string]string{"managed-by": "ubx"},
    			MaxResults:    1,
    			Shared:        true,
    			SortAscending: true,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.aws.data.cloudhsmv2 import Backups, BackupsConfig

    def describe():
        ubx.intent("look up aws_cloudhsmv2_backups")

        ubx.data(Backups, "example", BackupsConfig(
            filters={"managed-by": "ubx"},
            max_results=1,
            shared=True,

    if __name__ == "__main__":
