`aws_cloudfront_distribution_tenants_by_customization` reads the real CloudFront distribution tenants matching a given customization (such as a specific certificate).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudfront "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudfront"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudfront_distribution_tenants_by_customization"})

    		ubx.Data(cloudfront.DistributionTenantsByCustomization, "example", cloudfront.DistributionTenantsByCustomizationConfig{
    			CertificateArn: "arn:aws:iam::123456789012:policy/example",
    			Marker:         "example",
    			MaxItems:       1,
    			WebAclarn:      "example",
    		})
    	}))
    }
    ```
