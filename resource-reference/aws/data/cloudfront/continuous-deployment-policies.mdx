`aws_cloudfront_continuous_deployment_policies` looks up a real CloudFront continuous deployment policy, used to stage and gradually shift traffic to a new distribution configuration.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	cloudfront "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudfront"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudfront_continuous_deployment_policies"})

    		ubx.Data(cloudfront.ContinuousDeploymentPolicies, "example", cloudfront.ContinuousDeploymentPoliciesConfig{
    			Marker:   "example",
    			MaxItems: 1,
    		})
    	}))
    }
    ```
