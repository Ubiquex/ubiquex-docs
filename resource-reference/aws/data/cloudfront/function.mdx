## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudfront "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudfront"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudfront_function"})

    		ubx.Data(cloudfront.Function, "example", cloudfront.FunctionConfig{
    			Name:  "example",
    			Stage: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Function } from "@ubx/sdk-aws/aws/data/cloudfront/function";

    export default stack("example", () => {
      intent({ summary: "look up aws_cloudfront_function" });

      data(Function, "example", {
        name: "example",
        stage: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.cloudfront import Function, FunctionConfig

    def describe():
        ubx.intent("look up aws_cloudfront_function")

        ubx.data(Function, "example", FunctionConfig(
            name="example",
            stage="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
