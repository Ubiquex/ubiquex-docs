    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudfront_connection_function"})

    		ubx.Data(cloudfront.ConnectionFunction, "example", cloudfront.ConnectionFunctionConfig{

    			Stage:      "example",
    		})
    	}))
    }
