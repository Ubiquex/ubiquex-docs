`aws_cloudfront_cache_policies` looks up a real CloudFront cache policy, controlling what CloudFront includes in its own cache key and how long content is cached.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudfront "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudfront"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudfront_cache_policies"})

    		ubx.Data(cloudfront.CachePolicies, "example", cloudfront.CachePoliciesConfig{
    			Marker:   "example",
    			MaxItems: 1,
    			Type:     "example",
    		})
    	}))
    }
    ```
  </Tab>
