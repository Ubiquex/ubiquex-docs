## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudfront "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/cloudfront"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudfront_connection_groups"})

    		ubx.Data(cloudfront.ConnectionGroups, "example", cloudfront.ConnectionGroupsConfig{
    			AssociationFilter: map[string]any{"anycast_ip_list_id": "example"},
    			Marker:            "example",
    			MaxItems:          1,
    		})
    	}))
    }
    ```
