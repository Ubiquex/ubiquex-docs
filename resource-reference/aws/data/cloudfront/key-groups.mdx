    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudfront_key_groups"})

    		ubx.Data(cloudfront.KeyGroups, "example", cloudfront.KeyGroupsConfig{

    			MaxItems: 1,
    		})
    	}))
    }
