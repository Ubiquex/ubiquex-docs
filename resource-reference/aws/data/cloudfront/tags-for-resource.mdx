## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudfront "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudfront"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudfront_tags_for_resource"})

    		ubx.Data(cloudfront.TagsForResource, "example", cloudfront.TagsForResourceConfig{
    			Resource: "example",
    		})
    	}))
    }
    ```
  </Tab>
