## Example

    package main

    import (
    	cloudfront "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cloudfront"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cloudfront_distributions_by_connection_mode"})

    		ubx.Data(cloudfront.DistributionsByConnectionMode, "example", cloudfront.DistributionsByConnectionModeConfig{
    			ConnectionMode: "example",
    			Marker:         "example",
    			MaxItems:       1,
    		})
    	}))
    }
