    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	cur "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/cur"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_cur_tags_for_resource"})

    		ubx.Data(cur.TagsForResource, "example", cur.TagsForResourceConfig{
    			ReportName: "example-report-name",
    		})
    	}))
    }
    ```
