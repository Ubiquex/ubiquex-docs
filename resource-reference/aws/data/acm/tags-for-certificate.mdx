## Example

  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	acm "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/acm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_acm_tags_for_certificate"})

    		ubx.Data(acm.TagsForCertificate, "example", acm.TagsForCertificateConfig{
    			CertificateArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
