## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	directconnect "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/directconnect"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_directconnect_tags"})

    		ubx.Data(directconnect.Tags, "example", directconnect.TagsConfig{
    			ResourceArns: []string{"example"},
    		})
    	}))
    }
    ```

    export default stack("example", () => {

        resourceArns: ["example"],

  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.data.directconnect import Tags, TagsConfig

    def describe():
        ubx.intent("look up aws_directconnect_tags")

        ubx.data(Tags, "example", TagsConfig(
            resource_arns=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
