## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	codepipeline "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/codepipeline"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_codepipeline_job_details"})

    		ubx.Data(codepipeline.JobDetails, "example", codepipeline.JobDetailsConfig{
    			JobId: "example",
    		})
    	}))
    }
    ```
  </Tab>
