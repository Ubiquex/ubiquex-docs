title: "aws_ds_certificate"

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ds "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/ds"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ds_certificate"})

    		ubx.Data(ds.Certificate, "example", ds.CertificateConfig{
    			CertificateId: "example",
    			DirectoryId:   "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.ds import Certificate, CertificateConfig

    def describe():
        ubx.intent("look up aws_ds_certificate")

        ubx.data(Certificate, "example", CertificateConfig(
            certificate_id="example",
            directory_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>

## Lookup arguments

<ResponseField name="certificate_id" type="string" required>
  The real, unique identifier of the certificate this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="directory_id" type="string" required>
  The real, unique identifier of the directory this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
