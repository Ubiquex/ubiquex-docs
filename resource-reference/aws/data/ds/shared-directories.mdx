## Example

    package main

    import (
    	ds "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/data/ds"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ds_shared_directories"})

    		ubx.Data(ds.SharedDirectories, "example", ds.SharedDirectoriesConfig{
    			OwnerDirectoryId:   "example",
    			Limit:              1,
    			SharedDirectoryIds: []string{"example"},
    		})
    	}))
    }

## Lookup arguments

<ResponseField name="limit" type="number">
  The maximum number of results to return. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="owner_directory_id" type="string" required>
  The real, unique identifier of the owner directory this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="shared_directory_ids" type="list(string)">
  &lt;p&gt;A list of directory identifiers.&lt;/p&gt; Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="next_token" type="string">
  An opaque pagination token from a previous response, requesting the next page of results. **(AI-inferred)** Returned by the lookup.
</ResponseField>

<ResponseField name="shared_directories" type="list(object)">
  The real directories shared with this account. **(AI-inferred)** Returned by the lookup.
