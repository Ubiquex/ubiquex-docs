`aws_ds_settings` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ds "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/ds"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ds_settings"})

    		ubx.Data(ds.Settings, "example", ds.SettingsConfig{
    			DirectoryId: "example",
    			Status:      "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Settings } from "@ubx/sdk-aws/aws/data/ds/settings";

    export default stack("example", () => {
      intent({ summary: "look up aws_ds_settings" });

      data(Settings, "example", {
        directoryId: "example",
        status: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.ds import Settings, SettingsConfig

    def describe():
        ubx.intent("look up aws_ds_settings")

        ubx.data(Settings, "example", SettingsConfig(
            directory_id="example",
            status="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
