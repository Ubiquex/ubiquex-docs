## Example

    import (

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ds_ldapssettings"})

    		ubx.Data(ds.Ldapssettings, "example", ds.LdapssettingsConfig{
    			DirectoryId: "example",
    			Limit:       1,
    			Type:        "example",
    		})
    	}))
    }
