## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	ds "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/ds"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_ds_snapshot_limits"})

    		ubx.Data(ds.SnapshotLimits, "example", ds.SnapshotLimitsConfig{
    			DirectoryId: "example",
    		})
    	}))
    }
    ```
  </Tab>
