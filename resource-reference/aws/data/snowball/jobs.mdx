`aws_snowball_jobs` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against AWS's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	snowball "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/snowball"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_snowball_jobs"})

    		ubx.Data(snowball.Jobs, "example", snowball.JobsConfig{
    			MaxResults: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Jobs } from "@ubx/sdk-aws/aws/data/snowball/jobs";

    export default stack("example", () => {
      intent({ summary: "look up aws_snowball_jobs" });

      data(Jobs, "example", {
        maxResults: 1,
      });

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.data.snowball import Jobs, JobsConfig

    def describe():
        ubx.intent("look up aws_snowball_jobs")

        ubx.data(Jobs, "example", JobsConfig(
            max_results=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
