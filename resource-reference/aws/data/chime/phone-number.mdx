## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	chime "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/data/chime"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up aws_chime_phone_number"})

    		ubx.Data(chime.PhoneNumber, "example", chime.PhoneNumberConfig{
    			PhoneNumberId: "example",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx

    def describe():

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
