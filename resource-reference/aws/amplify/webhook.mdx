`aws_amplify_webhook` creates a real, callable URL that triggers a new build of an `aws_amplify_branch` when invoked, letting an external system kick off a deployment without needing a full source-provider Git integration.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	amplify "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/amplify"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own webhook"})

    		ubx.Resource(amplify.Webhook, "example", amplify.WebhookConfig{
    			BranchName:  "example-branch-name",
    			AppId:       "example",
    			Description: "Managed by ubx.",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Webhook } from "@ubx/sdk-aws/aws/amplify/webhook";

    export default stack("example", () => {

      resource(Webhook, "example", {
        branchName: "example-branch-name",
        appId: "example",
        description: "Managed by ubx.",
        tags: [{ key: "example" }],
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.amplify.webhook import Webhook, WebhookConfig

    def describe():
        ubx.intent("example own webhook")

        ubx.resource(Webhook, "example", WebhookConfig(
            branch_name="example-branch-name",
            app_id="example",
            description="Managed by ubx.",
            tags=[{"key": "example"}],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="app_id" type="string">
  The unique ID for an Amplify app. Optional.
</ResponseField>

<ResponseField name="branch_name" type="string" required>
  The name for a branch that is part of an Amplify app. Required.
</ResponseField>

<ResponseField name="description" type="string">
  The description for a webhook. Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags for the webhook. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  The Amazon Resource Name (ARN) for the webhook. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="webhook_id" type="string">
  The unique ID for a webhook. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="webhook_url" type="string">
  The URL of the webhook. Computed by AWS after `ship`.
</ResponseField>
