`aws_code_pipeline_custom_action_type` defines a caller's own real action an `aws_code_pipeline_pipeline` stage can invoke, beyond CodePipeline's built-in source, build, test, and deploy actions, for integrating a tool AWS doesn't provide a native action for.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	codepipeline "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/codepipeline"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own custom action type"})

    		ubx.Resource(codepipeline.CustomActionType, "example", codepipeline.CustomActionTypeConfig{
    			Category:                "example",
    			InputArtifactDetails:    map[string]any{"maximum_count": 1, "minimum_count": 1},
    			OutputArtifactDetails:   map[string]any{"maximum_count": 1, "minimum_count": 1},
    			Provider:                "example",
    			Version:                 "example",
    			ConfigurationProperties: []map[string]any{{"name": "example-name"}},
    			Settings:                map[string]any{"entity_url_template": "example"},
    			Tags:                    []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.codepipeline import CustomActionType, CustomActionTypeConfig
    from ubx.aws.codepipeline.custom_action_type import CustomActionType_InputArtifactDetails, CustomActionType_ConfigurationProperties, CustomActionType_Settings, CustomActionType_Tags

    def describe():
        ubx.intent("example own custom action type")

        ubx.resource(CustomActionType, "example", CustomActionTypeConfig(
            category="example",
            input_artifact_details=CustomActionType_InputArtifactDetails(maximum_count=1, minimum_count=1),
            output_artifact_details=CustomActionType_InputArtifactDetails(maximum_count=1, minimum_count=1),
            provider="example",
            version="example",
            configuration_properties=[CustomActionType_ConfigurationProperties(name="example-name")],
            settings=CustomActionType_Settings(entity_url_template="example"),
            tags=[CustomActionType_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="category" type="string" required>
  The category of the custom action, such as a build action or a test action. Required.
</ResponseField>

<ResponseField name="configuration_properties" type="list(object)">
  The configuration properties for the custom action. Optional.
  <Expandable title="properties">
    <ResponseField name="description" type="string">
      A short, user-facing description of the configuration property that is displayed to users in the CodePipeline console. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="key" type="bool">
      Indicates whether the configuration property is a key, which is used to uniquely identify the action instance within AWS CodePipeline. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="name" type="string">
      The unique, human-readable identifier for a configuration property of the custom action type, used as the property key when specifying action configuration values in a pipeline. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="queryable" type="bool">
      Indicates whether the configuration property can be used as a filter (queryable) when listing or viewing pipeline executions in the AWS CodePipeline console. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="required" type="bool">
      Indicates whether the configuration property is required for the custom action type to be used in a pipeline. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="secret" type="bool">
      Specifies whether the configuration property is a secret (such as a password) and should be hidden from view in the CodePipeline console. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="type" type="string">
      Specifies the data type of the configuration property that the custom action accepts, such as 'String', 'Number', or 'Boolean', which determines how the property value is handled by the pipeline. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="input_artifact_details" type="object" required>

    <ResponseField name="minimum_count" type="number" required>
      The minimum number of artifacts allowed for the action type. Required.

  </Expandable>
</ResponseField>

<ResponseField name="output_artifact_details" type="object" required>
  Returns information about the details of an artifact. Required.
  <Expandable title="properties">
    <ResponseField name="maximum_count" type="number" required>
      The maximum number of artifacts allowed for the action type. Required.
    </ResponseField>
    <ResponseField name="minimum_count" type="number" required>
      The minimum number of artifacts allowed for the action type. Required.
    </ResponseField>
  </Expandable>

<ResponseField name="provider" type="string" required>
  The provider of the service used in the custom action, such as AWS CodeDeploy. Required.
</ResponseField>

<ResponseField name="settings" type="object">
  Settings is a property of the AWS::CodePipeline::CustomActionType resource that provides URLs that users can access to view information about the CodePipeline custom action. Optional.
  <Expandable title="properties">
    <ResponseField name="entity_url_template" type="string">
      The URL returned to the AWS CodePipeline console that provides a deep link to the resources of the external system, such as the configuration page for an AWS CodeDeploy deployment group. This link is provided as part of the action display in the pipeline. Optional.
    </ResponseField>
    <ResponseField name="execution_url_template" type="string">
      The URL returned to the AWS CodePipeline console that contains a link to the top-level landing page for the external system, such as the console page for AWS CodeDeploy. This link is shown on the pipeline view page in the AWS CodePipeline console and provides a link to the execution entity of the external action. Optional.
    </ResponseField>
    <ResponseField name="revision_url_template" type="string">
      The URL returned to the AWS CodePipeline console that contains a link to the page where customers can update or change the configuration of the external action. Optional.
    </ResponseField>
    <ResponseField name="third_party_configuration_url" type="string">
      The URL of a sign-up page where users can sign up for an external service and perform initial configuration of the action provided by that service. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Any tags assigned to the custom action. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The tag key (such as 'Environment') used to label and categorize this AWS CodePipeline custom action type for management and identification purposes. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="version" type="string" required>
  The version identifier of the custom action. Required.
</ResponseField>
