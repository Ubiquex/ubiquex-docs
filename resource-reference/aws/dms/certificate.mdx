title: "aws_dms_certificate"

`aws_dms_certificate` imports a real SSL certificate into AWS DMS, used by an `aws_dms_endpoint` to encrypt its connection to a source or target database that requires certificate-based authentication.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	dms "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/dms"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own certificate"})

    		ubx.Resource(dms.Certificate, "ci-runner", dms.CertificateConfig{
    			CertificateIdentifier: "example",
    			CertificatePem:        "example",
    			CertificateWallet:     "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Certificate } from "./local-sdk/aws/sdk/typescript/aws/dms/certificate.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own certificate" });

      resource(Certificate, "ci-runner", {
        certificateIdentifier: "example",
        certificatePem: "example",
        certificateWallet: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.dms import Certificate, CertificateConfig

    def describe():
        ubx.intent("payments own certificate")

        ubx.resource(Certificate, "ci-runner", CertificateConfig(
            certificate_identifier="example",
            certificate_pem="example",
            certificate_wallet="example",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
