`aws_eks_pod_identity_association` maps a real IAM role to a Kubernetes service account in an `aws_eks_cluster`, giving pods using that service account automatically rotated, temporary AWS credentials without the OIDC-provider setup IAM roles for service accounts requires.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	"encoding/json"

    	eks "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/eks"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own pod identity association"})

    		accessPolicy, _ := json.Marshal(map[string]any{
    			"Version": "2012-10-17",
    			"Statement": []map[string]any{{
    				"Effect":   "Allow",
    				"Action":   "*",
    				"Resource": "*",
    			}},
    		})

    		ubx.Resource(eks.PodIdentityAssociation, "example", eks.PodIdentityAssociationConfig{
    			ClusterName:        "example-cluster-name",
    			Namespace:          "example",
    			RoleArn:            "arn:aws:iam::123456789012:policy/example",
    			ServiceAccount:     "example",
    			DisableSessionTags: true,
    			Policy:             string(accessPolicy),
    			Tags:               []map[string]any{{"key": "example"}},
    			TargetRoleArn:      "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```

  <Tab title="Python">
    ```python
    import json
    import ubx_sdk as ubx
    from ubx.aws.eks import PodIdentityAssociation, PodIdentityAssociationConfig

    def describe():
        ubx.intent("example own pod identity association")

        access_policy = json.dumps({
            "Version": "2012-10-17",
            "Statement": [{
                "Effect": "Allow",
                "Action": "*",
                "Resource": "*",
            }],
        })

        ubx.resource(PodIdentityAssociation, "example", PodIdentityAssociationConfig(
            cluster_name="example-cluster-name",
            namespace="example",
            role_arn="arn:aws:iam::123456789012:policy/example",
            service_account="example",
            disable_session_tags=True,
            policy=access_policy,
            tags=[{"key": "example"}],
            target_role_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="cluster_name" type="string" required>
  The cluster that the pod identity association is created for. Required.
</ResponseField>

<ResponseField name="disable_session_tags" type="bool">
  The Disable Session Tags of the pod identity association. Optional.
</ResponseField>

<ResponseField name="namespace" type="string" required>
  The Kubernetes namespace that the pod identity association is created for. Required.
</ResponseField>

<ResponseField name="policy" type="string">
  The policy of the pod identity association. Optional.
</ResponseField>

<ResponseField name="role_arn" type="string" required>
  The IAM role ARN that the pod identity association is created for. Required.
</ResponseField>

<ResponseField name="service_account" type="string" required>
  The Kubernetes service account that the pod identity association is created for. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An array of key-value pairs to apply to this resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The tag key used to label the Pod Identity Association, enabling cost allocation, access control, and resource identification. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a tag key-value pair attached to the EKS Pod Identity Association, used for labeling, organizing, and managing the association with custom metadata. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="target_role_arn" type="string">
  The Target Role Arn of the pod identity association. Optional.
</ResponseField>
