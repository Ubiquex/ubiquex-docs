title: "aws_eks_cluster"

---

`aws_eks_cluster` runs the real, managed Kubernetes control plane, API server, etcd, scheduler, that every `aws_eks_nodegroup` and Fargate profile in an EKS deployment runs its actual workloads against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	eks "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/eks"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own cluster"})

    		ubx.Resource(eks.Cluster, "example", eks.ClusterConfig{
    			ResourcesVpcConfig:           map[string]any{"subnet_ids": []string{"example"}},
    			RoleArn:                      "arn:aws:iam::123456789012:policy/example",
    			Name:                         "example",
    			AccessConfig:                 map[string]any{"authentication_mode": "example"},
    			ActiveCertificateAuthorityId: "example",
    			BootstrapSelfManagedAddons:   true,
    			ComputeConfig:                map[string]any{"enabled": true},
    			ControlPlaneScalingConfig:    map[string]any{"tier": "example"},
    		})
    	}))
    }
    ```
