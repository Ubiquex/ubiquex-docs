`aws_clean_rooms_id_namespace_association` associates an AWS Entity Resolution ID namespace with an `aws_clean_rooms_id_mapping_table`, the real link that makes a namespace's own resolved identities usable for matching inside a specific Clean Rooms collaboration.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own id namespace association"})

    		ubx.Resource(cleanrooms.IdNamespaceAssociation, "example", cleanrooms.IdNamespaceAssociationConfig{
