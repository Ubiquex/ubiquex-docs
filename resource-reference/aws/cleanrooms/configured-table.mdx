`aws_clean_rooms_configured_table` links a real S3, Athena, Glue, or Snowflake data source into AWS Clean Rooms with an attached analysis rule restricting what SQL queries can run against it, so a collaboration member's raw data is queryable under real, enforced constraints rather than exposed directly.

    import ubx_sdk as ubx
    from ubx.aws.cleanrooms import ConfiguredTable, ConfiguredTableConfig
    from ubx.aws.cleanrooms.configured_table import ConfiguredTable_TableReference, ConfiguredTable_TableReference_Athena, ConfiguredTable_AnalysisRules, ConfiguredTable_AnalysisRules_Policy, ConfiguredTable_AnalysisRules_Policy_V1, ConfiguredTable_AnalysisRules_Policy_V1_Aggregation, ConfiguredTable_Tags

    def describe():
        ubx.intent("example own configured table")

        ubx.resource(ConfiguredTable, "example", ConfiguredTableConfig(
            allowed_columns=["example"],
            analysis_method="example",
            name="example",
            table_reference=ConfiguredTable_TableReference(athena=ConfiguredTable_TableReference_Athena(catalog_name="example-catalog-name")),
            analysis_rules=[ConfiguredTable_AnalysisRules(policy=ConfiguredTable_AnalysisRules_Policy(v1=ConfiguredTable_AnalysisRules_Policy_V1(aggregation=ConfiguredTable_AnalysisRules_Policy_V1_Aggregation(additional_analyses="example"))))],
            description="Managed by ubx.",
            selected_analysis_methods=["example"],
            tags=[ConfiguredTable_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
