`aws_dynamo_db_backup` creates a real, on-demand backup of an `aws_dynamo_db_table`'s data at a specific point in time, stored independently of the table's own continuous point-in-time recovery window until explicitly deleted.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dynamodb "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/dynamodb"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own db backup"})

    		ubx.Resource(dynamodb.Backup, "example", dynamodb.BackupConfig{
    			BackupName: "example-backup-name",
    			TableName:  "example-table-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";

    import ubx_sdk as ubx
    from ubx.aws.dynamodb.backup import Backup, BackupConfig

    def describe():
        ubx.intent("example own db backup")

        ubx.resource(Backup, "example", BackupConfig(
            backup_name="example-backup-name",
            table_name="example-table-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="backup_name" type="string" required>
  The name for the backup. Required.
</ResponseField>

<ResponseField name="table_name" type="string" required>
  The name of the table to back up. Required.
</ResponseField>

## Output properties

<ResponseField name="backup_arn" type="string">
  The ARN associated with the backup. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="backup_creation_date_time" type="string">
  The time at which the backup was created. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="backup_id" type="string">
  The identifier portion of the backup ARN (server-generated). Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="backup_size_bytes" type="number">
  The size of the backup in bytes. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="backup_status" type="string">
  The current state of the backup. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="backup_type" type="string">
  The type of backup (USER, SYSTEM, or AWS_BACKUP). Computed by AWS after `ship`.
</ResponseField>
