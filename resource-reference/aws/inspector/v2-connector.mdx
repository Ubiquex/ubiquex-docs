`aws_inspector_v2_connector` is the real, account-level connection Amazon Inspector's Code Security capability uses to reach a specific source-code provider account, the resource an `aws_inspector_v2_code_security_integration` is actually built on top of.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	inspector "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/inspector"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v2 connector"})

    		ubx.Resource(inspector.V2Connector, "example", inspector.V2ConnectorConfig{
    			Name:        "example",
    			Provider:    "example",
    			Description: "Managed by ubx.",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
