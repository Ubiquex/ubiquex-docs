`aws_inspector_resource_group` was the real, tag-based filter defining exactly which EC2 instances belonged to an `aws_inspector_assessment_target` in Amazon Inspector Classic. Reaches full end of support May 20, 2026, alongside the rest of Inspector Classic.

## Example

    package main

    import (
    	inspector "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/inspector"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resource group"})

    		ubx.Resource(inspector.ResourceGroup, "example", inspector.ResourceGroupConfig{
    			ResourceGroupTags: []map[string]any{{"key": "example"}},
    		})
    	}))
