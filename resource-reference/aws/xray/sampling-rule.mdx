`aws_xray_sampling_rule` defines a real rule controlling what fraction of requests matching specific criteria, service name, HTTP method, URL path, X-Ray actually traces, balancing visibility against the cost and overhead of tracing every single request.

## Example

    ```go
    import (

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own sampling rule"})

    		ubx.Resource(xray.SamplingRule, "example", xray.SamplingRuleConfig{
    			RuleName:           "example-rule-name",
    			SamplingRule:       map[string]any{"fixed_rate": 1},
    			SamplingRuleRecord: map[string]any{"created_at": "example"},
    			SamplingRuleUpdate: map[string]any{"attributes": "example"},
    			Tags:               []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
