`aws_app_sync_channel_namespace` defines a namespace for AWS AppSync Events, the WebSocket-powered real-time PubSub API AppSync supports separately from GraphQL subscriptions, scoping which channels a client can publish to and subscribe against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	appsync "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/appsync"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own channel namespace"})

    		ubx.Resource(appsync.ChannelNamespace, "example", appsync.ChannelNamespaceConfig{
    			ApiId:              "example",
    			Name:               "example",
    			CodeHandlers:       "example",
    			CodeS3Location:     "example",
    			HandlerConfigs:     map[string]any{"on_publish": map[string]any{"behavior": "example", "integration": map[string]any{"data_source_name": "example-data-source-name"}}},
    			PublishAuthModes:   []map[string]any{{"auth_type": "example"}},
    			SubscribeAuthModes: []map[string]any{{"auth_type": "example"}},
    			Tags:               []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
