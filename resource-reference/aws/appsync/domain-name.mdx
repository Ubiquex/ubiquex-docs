`aws_app_sync_domain_name` registers a custom domain for an `aws_app_sync_api`, so a GraphQL endpoint is reachable at an organization's own domain rather than only at its default AppSync-issued URL.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	appsync "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/appsync"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own sync domain name"})

    		ubx.Resource(appsync.DomainName, "example", appsync.DomainNameConfig{
    			CertificateArn: "arn:aws:iam::123456789012:policy/example",
    			DomainName:     "example-domain-name",
    			Description:    "Managed by ubx.",
    			Tags:           []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    import ubx_sdk as ubx
    from ubx.aws.appsync.domain_name import DomainName, DomainNameConfig

    def describe():
        ubx.intent("example own sync domain name")

        ubx.resource(DomainName, "example", DomainNameConfig(
            certificate_arn="arn:aws:iam::123456789012:policy/example",
            domain_name="example-domain-name",
            description="Managed by ubx.",
            tags=[{"key": "example"}],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="certificate_arn" type="string" required>
  The Amazon Resource Name (ARN) of an AWS Certificate Manager certificate to associate with the AppSync custom domain name. Required. **(AI-inferred)**
</ResponseField>

<ResponseField name="description" type="string">
  An optional user-provided free-text description that identifies the purpose or context of the AppSync custom domain name in AWS management tools and APIs. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="domain_name" type="string" required>
  The fully qualified domain name (FQDN), such as api.example.com, that is associated with the AppSync API as its custom domain endpoint. Required. **(AI-inferred)**
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An arbitrary set of tags (key-value pairs) for this Domain Name. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key portion of a user-defined tag attached to the AWS::AppSync::DomainName resource, used to assign metadata for filtering, access control, or cost allocation to the AppSync custom domain. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="app_sync_domain_name" type="string">
  The actual AppSync domain name (e.g., `xxxxxxxxxx.appsync-api.us-east-1.amazonaws.com`) automatically generated by AppSync for this custom domain, read-only. Computed by AWS after `ship`. **(AI-inferred)**
</ResponseField>

<ResponseField name="domain_name_arn" type="string">
  The Amazon Resource Name (ARN) for the Domain Name. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="hosted_zone_id" type="string">
  The Route 53 hosted zone ID associated with the AppSync domain name, used to configure DNS records that route traffic to the AppSync endpoint. Computed by AWS after `ship`. **(AI-inferred)**
</ResponseField>
