## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	appsync "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/appsync"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own api"})

    		ubx.Resource(appsync.Api, "ci-runner", appsync.ApiConfig{
    			Name:         "ci-runner",
    			OwnerContact: "example",
    			Tags:         map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```
  </Tab>
