`aws_ec2_transit_gateway_route_table_propagation` enables a real transit gateway attachment's routes to be automatically propagated into a specific route table on an `aws_ec2_transit_gateway`, rather than requiring each route to be added by hand.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ec2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/ec2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own transit gateway route table propagation"})

    		ubx.Resource(ec2.TransitGatewayRouteTablePropagation, "example", ec2.TransitGatewayRouteTablePropagationConfig{
    			TransitGatewayAttachmentId: "example",
    			TransitGatewayRouteTableId: "example",
    		})
    	}))
    }
    ```
  </Tab>
