`aws_ec2_route_server_association` attaches an `aws_ec2_route_server` to a real VPC, the resource that actually turns the route server's BGP-learned routes into candidates for that VPC's own route tables.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ec2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/ec2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own route server association"})

    		ubx.Resource(ec2.RouteServerAssociation, "example", ec2.RouteServerAssociationConfig{
    			RouteServerId: "example",
    			VpcId:         "example",
    		})
    	}))
    }
    ```
  </Tab>
