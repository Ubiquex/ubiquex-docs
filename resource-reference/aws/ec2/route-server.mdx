`aws_ec2_route_server` creates a real, BGP-based dynamic routing resource for a VPC, automatically updating VPC and internet gateway route tables with preferred routes rather than requiring routes to be maintained by hand, giving fault-tolerant, software-defined-network-style routing inside a VPC.

## Example

    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	ec2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/ec2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own route server"})

    		ubx.Resource(ec2.RouteServer, "ci-runner", ec2.RouteServerConfig{
    			AmazonSideAsn:           1,
    			PersistRoutes:           "example",
    			PersistRoutesDuration:   7200,
    			SnsNotificationsEnabled: true,
    			Tags:                    []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

## Input properties

<ResponseField name="amazon_side_asn" type="number" required>
  The Amazon-side ASN of the Route Server. Required.
</ResponseField>

<ResponseField name="persist_routes" type="string">
  Whether to enable persistent routes Optional.
</ResponseField>

<ResponseField name="persist_routes_duration" type="number">
  The duration of persistent routes in minutes Optional.
</ResponseField>

<ResponseField name="sns_notifications_enabled" type="bool">
  Whether to enable SNS notifications Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An array of key-value pairs to apply to this resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  The Amazon Resource Name (ARN) of the Route Server. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="id" type="string">
  The ID of the Route Server. Computed by AWS after `ship`.
</ResponseField>
