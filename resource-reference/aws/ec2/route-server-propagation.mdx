`aws_ec2_route_server_propagation` enables a real `aws_ec2_route_server`'s learned BGP routes to be propagated into a specific VPC route table, the resource that actually applies the route server's routing decisions to real traffic.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ec2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/ec2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own route server propagation"})

    		ubx.Resource(ec2.RouteServerPropagation, "example", ec2.RouteServerPropagationConfig{
    			RouteServerId: "example",
    			RouteTableId:  "example",
    		})
    	}))
    }
    ```
  </Tab>
