`aws_ec2_network_insights_access_scope_analysis` runs a real, point-in-time check of an `aws_ec2_network_insights_access_scope`, the access-scope counterpart to what an `aws_ec2_network_insights_analysis` does for a single `aws_ec2_network_insights_path`.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own network insights access scope analysis"})

    			NetworkInsightsAccessScopeId: "example",

    		})
    	}))
    }
