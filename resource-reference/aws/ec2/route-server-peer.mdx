`aws_ec2_route_server_peer` establishes a real BGP peering session between an `aws_ec2_route_server` and an external or on-premises device, exchanging routes over that session.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ec2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/ec2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own route server peer"})

    		ubx.Resource(ec2.RouteServerPeer, "example", ec2.RouteServerPeerConfig{
    			BgpOptions:            map[string]any{"peer_asn": 1},
    			PeerAddress:           "example",
    			RouteServerEndpointId: "example",
    			Tags:                  []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.ec2 import RouteServerPeer, RouteServerPeerConfig
    from ubx.aws.ec2.route_server_peer import RouteServerPeer_BgpOptions, RouteServerPeer_Tags

    def describe():
        ubx.intent("example own route server peer")

        ubx.resource(RouteServerPeer, "example", RouteServerPeerConfig(
            bgp_options=RouteServerPeer_BgpOptions(peer_asn=1),
            peer_address="example",
            route_server_endpoint_id="example",
            tags=[RouteServerPeer_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="bgp_options" type="object" required>
  BGP Options Required.
  <Expandable title="properties">
    <ResponseField name="peer_asn" type="number">
      BGP ASN of the Route Server Peer Optional.
    </ResponseField>
    <ResponseField name="peer_liveness_detection" type="string">
      BGP Liveness Detection Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="peer_address" type="string" required>
  IP address of the Route Server Peer Required.
</ResponseField>

<ResponseField name="route_server_endpoint_id" type="string" required>
  Route Server Endpoint ID Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An array of key-value pairs to apply to this resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>
