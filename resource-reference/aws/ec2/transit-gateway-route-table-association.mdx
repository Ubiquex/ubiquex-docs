`aws_ec2_transit_gateway_route_table_association` associates a real transit gateway attachment with a specific route table on an `aws_ec2_transit_gateway`, controlling which routes that attachment's own traffic is evaluated against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ec2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/ec2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own transit gateway route table association"})

    		ubx.Resource(ec2.TransitGatewayRouteTableAssociation, "example", ec2.TransitGatewayRouteTableAssociationConfig{
    			TransitGatewayAttachmentId: "example",
    			TransitGatewayRouteTableId: "example",
    		})
    	}))
    }
    ```
