`aws_ec2_prefix_list` is a real, named, reusable set of CIDR blocks a caller manages once and references from security group rules and route tables, so updating the underlying IP ranges doesn't mean editing every rule that uses them individually.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ec2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/ec2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own prefix list"})

    		ubx.Resource(ec2.PrefixList, "example", ec2.PrefixListConfig{
    			AddressFamily:  "example",
    			PrefixListName: "example-prefix-list-name",
    			Entries:        []map[string]any{{"cidr": "example"}},
    			MaxEntries:     1,
    			Tags:           []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.ec2 import PrefixList, PrefixListConfig
    from ubx.aws.ec2.prefix_list import PrefixList_Entries, PrefixList_Tags

    def describe():
        ubx.intent("example own prefix list")

        ubx.resource(PrefixList, "example", PrefixListConfig(
            address_family="example",
            prefix_list_name="example-prefix-list-name",
            entries=[PrefixList_Entries(cidr="example")],
            max_entries=1,
            tags=[PrefixList_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="address_family" type="string" required>
  Ip Version of Prefix List. Required.
</ResponseField>

<ResponseField name="entries" type="list(object)">
  Entries of Prefix List. Optional.
  <Expandable title="properties">
    <ResponseField name="cidr" type="string">
      The CIDR block for the prefix list entry, defining the IP address range that this entry permits. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="description" type="string">
      Provides an optional human-readable description for the CIDR entry in an EC2 prefix list. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="max_entries" type="number">
  Max Entries of Prefix List. Optional.
</ResponseField>

<ResponseField name="prefix_list_name" type="string" required>
  Name of Prefix List. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">

    <ResponseField name="key" type="string">
