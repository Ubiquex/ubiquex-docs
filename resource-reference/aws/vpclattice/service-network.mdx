`aws_vpc_lattice_service_network` creates a real, logical boundary that `aws_vpc_lattice_service`s are published into and VPCs are associated with, the resource that actually lets services and consuming VPCs across separate accounts and VPCs discover and reach each other.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	vpclattice "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/vpclattice"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own service network"})

    		ubx.Resource(vpclattice.ServiceNetwork, "example", vpclattice.ServiceNetworkConfig{
    			Name:          "example",
    			AuthType:      "example",
    			SharingConfig: map[string]any{"enabled": true},
    			Tags:          []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.vpclattice import ServiceNetwork, ServiceNetworkConfig
    from ubx.aws.vpclattice.service_network import ServiceNetwork_SharingConfig, ServiceNetwork_Tags

    def describe():
        ubx.intent("example own service network")

        ubx.resource(ServiceNetwork, "example", ServiceNetworkConfig(
            name="example",
            auth_type="example",
            sharing_config=ServiceNetwork_SharingConfig(enabled=True),
            tags=[ServiceNetwork_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="auth_type" type="string">
  Determines the authentication method for the service network, accepting either NONE to allow open access or AWS_IAM to require IAM authentication for client requests. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="name" type="string">
  Specifies the name of the VPC Lattice service network, which must be unique within the account; if omitted, AWS CloudFormation generates a unique name. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="sharing_config" type="object">
  Defines the sharing configuration for the service network, with an 'enabled' toggle that allows the network to be shared with other AWS accounts through AWS Resource Access Manager. Optional. **(AI-inferred)**
