`aws_vpc_lattice_service_network_service_association` associates a real `aws_vpc_lattice_service` with an `aws_vpc_lattice_service_network`, the resource that actually publishes the service so consumers on the network can reach it.

## Example

    import (
    	vpclattice "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/vpclattice"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own service network service association"})

    		ubx.Resource(vpclattice.ServiceNetworkServiceAssociation, "example", vpclattice.ServiceNetworkServiceAssociationConfig{
    			ServiceIdentifier:        "example",
    			ServiceNetworkIdentifier: "example",
    			Tags:                     []map[string]any{{"key": "example"}},
    		})
    	}))
    }

    import ubx_sdk as ubx
    from ubx.aws.vpclattice import ServiceNetworkServiceAssociation, ServiceNetworkServiceAssociationConfig

    def describe():
        ubx.intent("example own service network service association")
