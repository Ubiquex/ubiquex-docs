`aws_vpc_lattice_domain_verification` verifies a real, custom domain name's ownership so it can be associated with an `aws_vpc_lattice_service`'s listener instead of only Lattice's own generated DNS name.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	vpclattice "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/vpclattice"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own domain verification"})

    		ubx.Resource(vpclattice.DomainVerification, "example", vpclattice.DomainVerificationConfig{
    			DomainName: "example-domain-name",
    			Tags:       []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.aws.vpclattice import DomainVerification, DomainVerificationConfig

        ubx.intent("example own domain verification")

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
