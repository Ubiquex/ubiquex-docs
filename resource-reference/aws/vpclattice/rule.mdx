`aws_vpc_lattice_rule` defines a real routing rule on an `aws_vpc_lattice_listener`, matching request path or header conditions and directing matching traffic to a specific `aws_vpc_lattice_target_group`.

## Example

    ```go
    import (
    	vpclattice "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/vpclattice"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own rule"})

    		ubx.Resource(vpclattice.Rule, "example", vpclattice.RuleConfig{
    			Action:             map[string]any{"fixed_response": map[string]any{"status_code": 1}},
    			Match:              map[string]any{"http_match": map[string]any{"header_matches": []map[string]any{{"name": "example-name"}}}},
    			Priority:           1,
    			Name:               "example",
    			ListenerIdentifier: "example",
    			ServiceIdentifier:  "example",
    			Tags:               []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.vpclattice import Rule, RuleConfig
    from ubx.aws.vpclattice.rule import Rule_Action, Rule_Action_FixedResponse, Rule_Match, Rule_Match_HttpMatch, Rule_Match_HttpMatch_HeaderMatches, Rule_Tags

    def describe():
        ubx.intent("example own rule")

        ubx.resource(Rule, "example", RuleConfig(
            action=Rule_Action(fixed_response=Rule_Action_FixedResponse(status_code=1)),
            match=Rule_Match(http_match=Rule_Match_HttpMatch(header_matches=[Rule_Match_HttpMatch_HeaderMatches(name="example-name")])),
            priority=1,
            name="example",
            listener_identifier="example",
            service_identifier="example",
            tags=[Rule_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
