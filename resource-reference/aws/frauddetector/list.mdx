`aws_fraud_detector_list` defines a real, named list of reference values, known-bad email domains or device IDs, that an `aws_fraud_detector_detector`'s rules can check an event's own fields against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	frauddetector "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/frauddetector"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own list"})

    		ubx.Resource(frauddetector.List, "example", frauddetector.ListConfig{
    			Name:         "example",
    			Description:  "Managed by ubx.",
    			Elements:     []string{"example"},
    			Tags:         []map[string]any{{"key": "example"}},
    			VariableType: "example",
    		})
    	}))
    }
    ```
