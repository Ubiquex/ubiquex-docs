Closed to new customers along with the rest of Fraud Detector since November 7, 2025.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	frauddetector "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/frauddetector"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own label"})

    		ubx.Resource(frauddetector.Label, "example", frauddetector.LabelConfig{
    			Name:        "example",
    			Description: "Managed by ubx.",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.frauddetector import Label, LabelConfig
    from ubx.aws.frauddetector.label import Label_Tags

    def describe():
        ubx.intent("example own label")

            name="example",
            description="Managed by ubx.",
            tags=[Label_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="description" type="string">
  The label description. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the label. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags associated with this label. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key of a tag attached to an AWS Fraud Detector label, used for resource organization, filtering, and cost allocation. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value component of a tag attached to an AWS Fraud Detector label, used to store arbitrary metadata for identifying and organizing the label resource. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>
