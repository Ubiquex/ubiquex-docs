`aws_fraud_detector_outcome` defined a real possible result, approve, review, block, that an `aws_fraud_detector_detector`'s own business rules could return for an evaluated event. Closed to new customers along with the rest of Fraud Detector since November 7, 2025.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	frauddetector "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/frauddetector"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own outcome"})

    		ubx.Resource(frauddetector.Outcome, "example", frauddetector.OutcomeConfig{
    			Name:        "example",
    			Description: "Managed by ubx.",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    import ubx_sdk as ubx
    from ubx.aws.frauddetector import Outcome, OutcomeConfig
    from ubx.aws.frauddetector.outcome import Outcome_Tags

    def describe():
        ubx.intent("example own outcome")

        ubx.resource(Outcome, "example", OutcomeConfig(
            name="example",
            description="Managed by ubx.",
            tags=[Outcome_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="description" type="string">
  The outcome description. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the outcome. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags associated with this outcome. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key of a tag assigned to the AWS Fraud Detector outcome, used to organize and identify the outcome resource. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>
