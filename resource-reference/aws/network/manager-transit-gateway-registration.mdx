`aws_network_manager_transit_gateway_registration` registers a real, existing EC2 Transit Gateway with an `aws_network_manager_global_network`, so Network Manager can represent and monitor it alongside an organization's other network infrastructure.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	network "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/network"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own manager transit gateway registration"})

    		ubx.Resource(network.ManagerTransitGatewayRegistration, "example", network.ManagerTransitGatewayRegistrationConfig{
    			GlobalNetworkId:   "example",
    			TransitGatewayArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
