`aws_network_manager_link` defines a real network connection, its bandwidth and provider, between two `aws_network_manager_site`s within an `aws_network_manager_global_network`.

## Example

    	network "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/network"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own manager link"})

    			Bandwidth:       map[string]any{"download_speed": 1},
    			GlobalNetworkId: "example",
    			SiteId:          "example",
    			Description:     "Managed by ubx.",
    			Provider:        "example",
    			Tags:            []map[string]any{{"key": "example"}},

    		})
    	}))
    }
