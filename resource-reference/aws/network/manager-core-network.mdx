`aws_network_manager_core_network` creates a real, centrally managed global network backbone, the core resource behind AWS Cloud WAN, that VPCs, Direct Connect gateways, and Site-to-Site VPNs attach to and route through.

## Example

    	network "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/network"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own manager core network"})

    		ubx.Resource(network.ManagerCoreNetwork, "example", network.ManagerCoreNetworkConfig{
    			GlobalNetworkId: "example",

    		})
    	}))
    }

</ResponseField>

<ResponseField name="global_network_id" type="string" required>
  The ID of the global network that your core network is a part of. Required.
</ResponseField>

<ResponseField name="policy_document" type="any">
  Live policy document for the core network, you must provide PolicyDocument in Json Format Optional.
</ResponseField>
