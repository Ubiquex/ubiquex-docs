`aws_network_manager_link_association` associates a real `aws_network_manager_link` with a specific device in an `aws_network_manager_global_network`, the resource that actually attaches that network connection to a device in Network Manager's own topology.

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own manager link association"})

    		ubx.Resource(network.ManagerLinkAssociation, "ci-runner", network.ManagerLinkAssociationConfig{

    			GlobalNetworkId: "example",
    			LinkId:          "example",
    		})
    	}))
    }
