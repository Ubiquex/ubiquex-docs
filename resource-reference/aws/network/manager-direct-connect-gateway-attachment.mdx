`aws_network_manager_direct_connect_gateway_attachment` attaches a real Direct Connect gateway to an `aws_network_manager_core_network`, extending Cloud WAN's routing to on-premises networks reachable over Direct Connect.

## Example

<Tabs>

    ```go
    import (
    	network "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/network"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own manager direct connect gateway attachment"})

    		ubx.Resource(network.ManagerDirectConnectGatewayAttachment, "example", network.ManagerDirectConnectGatewayAttachmentConfig{
    			CoreNetworkId:                      "example",
    			DirectConnectGatewayArn:            "arn:aws:iam::123456789012:policy/example",
    			EdgeLocations:                      []string{"example"},
    			ProposedNetworkFunctionGroupChange: map[string]any{"attachment_policy_rule_number": 1},
    			ProposedSegmentChange:              map[string]any{"attachment_policy_rule_number": 1},
    			RoutingPolicyLabel:                 "example",
    			Tags:                               []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

## Input properties

<ResponseField name="core_network_id" type="string" required>
  The ID of a core network for the Direct Connect Gateway attachment. Required.
</ResponseField>

<ResponseField name="direct_connect_gateway_arn" type="string" required>
  The ARN of the Direct Connect Gateway. Required.
</ResponseField>

<ResponseField name="edge_locations" type="list(string)" required>
  The Regions where the edges are located. Required.
</ResponseField>

<ResponseField name="proposed_network_function_group_change" type="object">
  The attachment to move from one network function group to another. Optional.
  <Expandable title="properties">
    <ResponseField name="attachment_policy_rule_number" type="number">
      The rule number in the policy document that applies to this change. Optional.
    </ResponseField>
    <ResponseField name="network_function_group_name" type="string">
      The name of the network function group to change. Optional.
    </ResponseField>
    <ResponseField name="tags" type="list(object)">
      The key-value tags that changed for the network function group. Optional.
      <Expandable title="properties">
        <ResponseField name="key" type="string">
          The key of a tag applied to the proposed network function group change for an AWS Network Manager Direct Connect Gateway attachment, used to add custom metadata to the change proposal. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="value" type="string">
          The value of a tag associated with the proposed network function group change for this Direct Connect gateway attachment, used to label or categorize the change. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="proposed_segment_change" type="object">
  The attachment to move from one segment to another. Optional.
  <Expandable title="properties">
    <ResponseField name="attachment_policy_rule_number" type="number">
      The rule number in the policy document that applies to this change. Optional.
    </ResponseField>
    <ResponseField name="segment_name" type="string">
      The name of the segment to change. Optional.
    </ResponseField>
    <ResponseField name="tags" type="list(object)">
      The key-value tags that changed for the segment. Optional.
      <Expandable title="properties">
        <ResponseField name="key" type="string">
          The key of a tag to apply to the proposed segment change for the Direct Connect Gateway attachment. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="value" type="string">
          The value portion of a tag attached to the proposed segment change, used to tag the segment update for this Direct Connect Gateway attachment in Network Manager. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="routing_policy_label" type="string">
  Routing policy label Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags for the attachment. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key of a tag attached to the Direct Connect Gateway attachment, used to categorize the attachment for cost allocation and management within AWS Network Manager. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a user-defined tag applied to the AWS Network Manager Direct Connect Gateway Attachment, used for organizing, identifying, and managing the attachment within AWS. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>
