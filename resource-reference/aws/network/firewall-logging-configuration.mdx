`aws_network_firewall_logging_configuration` configures real log destinations, CloudWatch Logs, S3, or Kinesis Data Firehose, for an `aws_network_firewall_firewall`'s own alert and flow logs.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	network "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/network"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own firewall logging configuration"})

    		ubx.Resource(network.FirewallLoggingConfiguration, "example", network.FirewallLoggingConfigurationConfig{
    			FirewallArn:               "arn:aws:iam::123456789012:policy/example",
    			LoggingConfiguration:      map[string]any{"log_destination_configs": []map[string]any{{"log_destination": "example"}}},
    			EnableMonitoringDashboard: true,
    			FirewallName:              "example-firewall-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";

      resource(FirewallLoggingConfiguration, "example", {
        firewallArn: "arn:aws:iam::123456789012:policy/example",
        loggingConfiguration: {
          logDestinationConfigs: [{ logDestination: "example" }],
        },
        enableMonitoringDashboard: true,
        firewallName: "example-firewall-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.network import FirewallLoggingConfiguration, FirewallLoggingConfigurationConfig

    def describe():
        ubx.intent("example own firewall logging configuration")

        ubx.resource(FirewallLoggingConfiguration, "example", FirewallLoggingConfigurationConfig(
            firewall_arn="arn:aws:iam::123456789012:policy/example",
            logging_configuration={"log_destination_configs": [{"log_destination": "example"}]},
            enable_monitoring_dashboard=True,
            firewall_name="example-firewall-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="enable_monitoring_dashboard" type="bool">
  Optional.
</ResponseField>

<ResponseField name="firewall_arn" type="string" required>
  A resource ARN. Required.
</ResponseField>

<ResponseField name="firewall_name" type="string">
  The name of the Network Firewall firewall resource that this logging configuration applies to, used as an alternative to the firewall ARN. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="logging_configuration" type="object" required>
  Specifies the logging destinations and configuration for the network firewall, including the list of log destination configs that define where firewall logs are sent (e.g., to CloudWatch Logs, S3, or Kinesis Data Firehose). Required. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="log_destination_configs" type="list(object)" required>
      Specifies the destinations and log types (alert or flow) to which Network Firewall sends its logs, such as Amazon CloudWatch Logs, Amazon S3, or Amazon Kinesis Data Firehose. Required. **(AI-inferred)**
      <Expandable title="properties">
        <ResponseField name="log_destination" type="any">
          Specifies the actual destination details as key-value pairs, such as the S3 bucket name, CloudWatch Logs log group, or Kinesis Data Firehose delivery stream, that correspond to the chosen LogDestinationType in the Network Firewall logging configuration. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="log_destination_type" type="string">
          Defines the kind of service that receives the firewall logs, such as Amazon S3, CloudWatch Logs, or Kinesis Data Firehose. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="log_type" type="string">
          The type of Network Firewall log to send to this destination, either ALERT for firewall alert logs or FLOW for firewall flow logs. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
