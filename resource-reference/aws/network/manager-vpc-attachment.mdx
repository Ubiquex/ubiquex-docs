`aws_network_manager_vpc_attachment` attaches a real VPC to an `aws_network_manager_core_network`, the mechanism that actually brings a VPC's subnets into Cloud WAN's centrally managed routing.

## Example

  <Tab title="Go">

    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    import (
    	network "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/network"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own manager vpc attachment"})

    		ubx.Resource(network.ManagerVpcAttachment, "ci-runner", network.ManagerVpcAttachmentConfig{
    			CoreNetworkId:                      "example",
    			SubnetArns:                         []string{"example"},
    			VpcArn:                             "arn:aws:iam::123456789012:policy/ci",
    			Options:                            map[string]any{"appliance_mode_support": true},
    			ProposedNetworkFunctionGroupChange: map[string]any{"attachment_policy_rule_number": 1},

    			RoutingPolicyLabel:                 "example",
    			Tags:                               []map[string]any{{"key": "example"}},
    		})
    	}))

## Input properties

<ResponseField name="core_network_id" type="string" required>
  The ID of a core network for the VPC attachment. Required.
</ResponseField>

<ResponseField name="options" type="object">
  Vpc options of the attachment. Optional.
  <Expandable title="properties">
    <ResponseField name="appliance_mode_support" type="bool">
      Indicates whether to enable ApplianceModeSupport Support for Vpc Attachment. Valid Values: true | false Optional.
    </ResponseField>
    <ResponseField name="dns_support" type="bool">
      Indicates whether to enable private DNS Support for Vpc Attachment. Valid Values: true | false Optional.
    </ResponseField>
    <ResponseField name="ipv6_support" type="bool">
      Indicates whether to enable Ipv6 Support for Vpc Attachment. Valid Values: enable | disable Optional.
    </ResponseField>
    <ResponseField name="security_group_referencing_support" type="bool">
      Indicates whether to enable Security Group Referencing Support for Vpc Attachment. Valid Values: true | false Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="proposed_network_function_group_change" type="object">
  The attachment to move from one network function group to another. Optional.
  <Expandable title="properties">
    <ResponseField name="attachment_policy_rule_number" type="number">
      The rule number in the policy document that applies to this change. Optional.
    </ResponseField>
    <ResponseField name="network_function_group_name" type="string">
      The name of the network function group to change. Optional.
    </ResponseField>
    <ResponseField name="tags" type="list(object)">
      The key-value tags that changed for the network function group. Optional.
      <Expandable title="properties">
        <ResponseField name="key" type="string">
          The key of a tag applied to the proposed network function group change for this VPC attachment. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="value" type="string">
          The value part of a tag attached to the proposed network function group change for the VPC attachment, used to label or categorize the network function group proposal with metadata such as environment or purpose. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="proposed_segment_change" type="object">
  The attachment to move from one segment to another. Optional.
  <Expandable title="properties">
    <ResponseField name="attachment_policy_rule_number" type="number">
      The rule number in the policy document that applies to this change. Optional.
    </ResponseField>
    <ResponseField name="segment_name" type="string">
      The name of the segment to change. Optional.
    </ResponseField>
    <ResponseField name="tags" type="list(object)">
      The key-value tags that changed for the segment. Optional.
      <Expandable title="properties">
        <ResponseField name="key" type="string">
          The key of a tag to be applied to the VPC attachment as part of the proposed segment change. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="value" type="string">
          The value of a tag key-value pair attached to the proposed segment change for the VPC attachment, used to provide metadata about the change. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="routing_policy_label" type="string">
  Routing policy label Optional.
</ResponseField>

<ResponseField name="subnet_arns" type="list(string)" required>
  Subnet Arn list Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags for the attachment. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a tag (key-value pair) attached to the AWS Network Manager VPC attachment, used to organize and identify the attachment within AWS. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="vpc_arn" type="string" required>
  The ARN of the VPC. Required.
</ResponseField>
