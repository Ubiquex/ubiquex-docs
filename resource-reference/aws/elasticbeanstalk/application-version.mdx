`aws_elastic_beanstalk_application_version` is one real, specific, immutable build of an `aws_elastic_beanstalk_application`'s own code, stored in S3, that an `aws_elastic_beanstalk_environment` actually deploys.

## Example

<Tabs>
  <Tab title="Go">

    package main

    import (
    	elasticbeanstalk "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/elasticbeanstalk"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own application version"})

    		ubx.Resource(elasticbeanstalk.ApplicationVersion, "example", elasticbeanstalk.ApplicationVersionConfig{
    			ApplicationName: "example-application-name",
    			SourceBundle:    map[string]any{"s3_bucket": "example", "s3_key": "example"},
    			Description:     "Managed by ubx.",
    		})
    	}))
    }

</ResponseField>

  A description of this application version. Optional.
</ResponseField>
