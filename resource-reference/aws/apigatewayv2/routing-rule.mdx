`aws_api_gateway_v2_routing_rule` defines additional request-matching logic for an `aws_api_gateway_v2_api`'s routes beyond the basic method-and-path match, used to route the same path differently based on request content.

## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	apigatewayv2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/apigatewayv2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own routing rule"})

    		ubx.Resource(apigatewayv2.RoutingRule, "example", apigatewayv2.RoutingRuleConfig{
    			Actions:       []map[string]any{{"invoke_api": map[string]any{"api_id": "example"}}},
    			Conditions:    []map[string]any{{"match_base_paths": map[string]any{"any_of": []string{"example"}}}},
    			DomainNameArn: "arn:aws:iam::123456789012:policy/example",
    			Priority:      1,
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.apigatewayv2 import RoutingRule, RoutingRuleConfig
    from ubx.aws.apigatewayv2.routing_rule import RoutingRule_Actions, RoutingRule_Actions_InvokeApi, RoutingRule_Conditions, RoutingRule_Conditions_MatchBasePaths

    def describe():
        ubx.intent("example own routing rule")

        ubx.resource(RoutingRule, "example", RoutingRuleConfig(
            actions=[RoutingRule_Actions(invoke_api=RoutingRule_Actions_InvokeApi(api_id="example"))],
            conditions=[RoutingRule_Conditions(match_base_paths=RoutingRule_Conditions_MatchBasePaths(any_of=["example"]))],
            domain_name_arn="arn:aws:iam::123456789012:policy/example",
            priority=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
