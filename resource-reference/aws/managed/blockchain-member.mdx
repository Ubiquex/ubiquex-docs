`aws_managed_blockchain_member` creates a real member identity within a Hyperledger Fabric network on Amazon Managed Blockchain, the account-level participant that peer nodes, `aws_managed_blockchain_node`, are created under.

## Example

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own blockchain member"})

    		ubx.Resource(managed.BlockchainMember, "example", managed.BlockchainMemberConfig{
    			MemberConfiguration:  map[string]any{"name": "example-name"},

    			NetworkId:            "example",
    		})
    	}))
    }
