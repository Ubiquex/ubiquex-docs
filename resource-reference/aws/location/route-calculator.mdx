`aws_location_route_calculator` creates a real route calculator resource in Amazon Location Service, used to estimate travel time and distance and to generate turn-by-turn directions between locations.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	location "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/location"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own route calculator"})

    		ubx.Resource(location.RouteCalculator, "example", location.RouteCalculatorConfig{
    			CalculatorName: "example-calculator-name",
    			DataSource:     "example",
    			Description:    "Managed by ubx.",
    			PricingPlan:    "example",
    			Tags:           []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
