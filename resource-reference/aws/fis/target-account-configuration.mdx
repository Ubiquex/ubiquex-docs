---
title: "aws_fis_target_account_configuration"
description: "`aws_fis_target_account_configuration` registers a real member account..."
---

`aws_fis_target_account_configuration` registers a real member account, and the IAM role FIS should assume in it, as an eligible target for an `aws_fis_experiment_template` run from a different, orchestrating account.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	fis "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/fis"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own target account configuration"})

    		ubx.Resource(fis.TargetAccountConfiguration, "example", fis.TargetAccountConfigurationConfig{
    			AccountId:            "example",
    			ExperimentTemplateId: "example",
    			RoleArn:              "arn:aws:iam::123456789012:policy/example",
    			Description:          "Managed by ubx.",
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    import ubx_sdk as ubx
    from ubx.aws.fis import TargetAccountConfiguration, TargetAccountConfigurationConfig

    def describe():
        ubx.intent("example own target account configuration")

        ubx.resource(TargetAccountConfiguration, "example", TargetAccountConfigurationConfig(
            account_id="example",
            experiment_template_id="example",
            role_arn="arn:aws:iam::123456789012:policy/example",
            description="Managed by ubx.",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="account_id" type="string" required>
  The AWS account ID of the target account. Required.
</ResponseField>

<ResponseField name="description" type="string">
  The description of the target account. Optional.
</ResponseField>

<ResponseField name="experiment_template_id" type="string" required>
  The ID of the experiment template. Required.
</ResponseField>

<ResponseField name="role_arn" type="string" required>
  The Amazon Resource Name (ARN) of an IAM role for the target account. Required.
</ResponseField>
