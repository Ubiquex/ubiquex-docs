`aws_transfer_workflow` defines a real sequence of post-upload processing steps, tagging, copying, running a Lambda function, that Transfer Family automatically runs against a file after it's fully uploaded to an `aws_transfer_server`.

## Example

    ```go
    package main

    import (
    	transfer "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/transfer"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own workflow"})

    		ubx.Resource(transfer.Workflow, "example", transfer.WorkflowConfig{
    			Steps:            []map[string]any{{"copy_step_details": map[string]any{"name": "example-name"}}},
    			Description:      "Managed by ubx.",
    			OnExceptionSteps: []map[string]any{{"copy_step_details": map[string]any{"name": "example-name"}}},
    			Tags:             []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.transfer import Workflow, WorkflowConfig
    from ubx.aws.transfer.workflow import Workflow_OnExceptionSteps, Workflow_OnExceptionSteps_CopyStepDetails, Workflow_OnExceptionSteps_TagStepDetails_Tags

    def describe():
        ubx.intent("example own workflow")

        ubx.resource(Workflow, "example", WorkflowConfig(
            steps=[Workflow_OnExceptionSteps(copy_step_details=Workflow_OnExceptionSteps_CopyStepDetails(name="example-name"))],
            description="Managed by ubx.",
            on_exception_steps=[Workflow_OnExceptionSteps(copy_step_details=Workflow_OnExceptionSteps_CopyStepDetails(name="example-name"))],
            tags=[Workflow_OnExceptionSteps_TagStepDetails_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
