`aws_media_convert_job_template` saves a real, reusable set of MediaConvert transcoding job settings, output format, codec, bitrate, and destination queue, so new jobs can be created from the template instead of re-specifying every setting.

## Example

    ```go
    package main

    import (
    	mediaconvert "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/mediaconvert"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own job template"})

    		ubx.Resource(mediaconvert.JobTemplate, "example", mediaconvert.JobTemplateConfig{
    			SettingsJson:         "example",
    			Name:                 "example",
    			AccelerationSettings: map[string]any{"mode": "example"},
    			Category:             "example",
    			Description:          "Managed by ubx.",
    			HopDestinations:      []map[string]any{{"priority": 1}},
    			Priority:             1,
    			Queue:                "example",
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.mediaconvert import JobTemplate, JobTemplateConfig
    from ubx.aws.mediaconvert.job_template import JobTemplate_AccelerationSettings, JobTemplate_HopDestinations

    def describe():
        ubx.intent("example own job template")

        ubx.resource(JobTemplate, "example", JobTemplateConfig(
            settings_json="example",
            name="example",
            acceleration_settings=JobTemplate_AccelerationSettings(mode="example"),
            category="example",
            description="Managed by ubx.",
            hop_destinations=[JobTemplate_HopDestinations(priority=1)],
            priority=1,
            queue="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="acceleration_settings" type="object">
  Configures accelerated transcoding behavior for jobs using this template, with the Mode property set to DISABLED, ENABLED, or PREFERRED to control whether faster GPU-based transcoding is used. Optional. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="mode" type="string" required>
      Indicates whether AWS Elemental MediaConvert job acceleration is enabled for jobs that use this template, with allowed values DISABLED, ENABLED, or PREFERRED. Required. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="category" type="string">
  Assigns a category to the job template for organizational grouping and management. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="description" type="string">
  A user-defined description that helps identify the purpose or contents of the AWS Elemental MediaConvert job template. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="hop_destinations" type="list(object)">
  Specifies the list of alternate MediaConvert queues (as hop destinations with associated priorities) to which a job is re-routed when it cannot be processed in the current queue, as defined in the job template. Optional. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="priority" type="number">
      This number sets the priority of this hop destination, controlling the order in which the job attempts to hop to the associated queue when higher-priority queues are unavailable. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="queue" type="string">
      The name of the MediaConvert queue to which the job will be sent when this hop destination is triggered as part of the job template's queue-hopping configuration. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="wait_minutes" type="number">
      Specifies the number of minutes to wait before transitioning the job to the next queue in the hop destination list. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="name" type="string">
  Specifies a user-defined name for the MediaConvert job template; if omitted, CloudFormation generates a unique name. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="priority" type="number">
  Specifies the default priority (0-50) for jobs created from this template, where higher values indicate a higher priority in the MediaConvert job queue. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="queue" type="string">
  The Amazon Resource Name (ARN) of the MediaConvert queue that jobs created from this template are submitted to, falling back to the account's default queue when not specified. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="settings_json" type="any" required>
  The JSON-formatted job settings that define the inputs, outputs, and encoding configurations for the MediaConvert job template. Required. **(AI-inferred)**
</ResponseField>

<ResponseField name="status_update_interval" type="string">
  Specifies how often (in seconds) MediaConvert sends job status updates to CloudWatch Events and your SNS topic, with valid values of 10, 12, 15, 20, 30, 60, 120, 180, 240, 300, 360, 420, 480, 540, or 600, defaulting to 600 if not set. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="tags" type="any">
  Optional.
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  The Amazon Resource Name (ARN) that uniquely identifies this AWS Elemental MediaConvert job template. Computed by AWS after `ship`. **(AI-inferred)**
</ResponseField>

<ResponseField name="id" type="string">
  The name of the MediaConvert job template, which is the resource's unique identifier within the AWS account. Computed by AWS after `ship`. **(AI-inferred)**
</ResponseField>
