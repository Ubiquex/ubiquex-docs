`aws_security_hub_connector_v2` connects a real external, non-AWS security tool's findings into Security Hub's next-generation (HubV2) model, launched at re:Invent in December 2025, the V2 counterpart to `aws_security_hub_connector`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	security "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/security"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own hub connector v2"})

    		ubx.Resource(security.HubConnectorV2, "example", security.HubConnectorV2Config{
    			Name:        "example",
    			Provider:    map[string]any{"azure": map[string]any{"awsconfig_connector_arn": "example"}},
    			Description: "Managed by ubx.",
    			KmsKeyArn:   "arn:aws:iam::123456789012:policy/example",
    			Tags:        "example",
    		})
    	}))
    }
    ```
  </Tab>
