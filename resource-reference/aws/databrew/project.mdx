`aws_data_brew_project` is DataBrew's real, interactive workspace, a grid view of an `aws_data_brew_dataset` where a caller explores value distributions and builds up an `aws_data_brew_recipe` of point-and-click transformations without writing code.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	databrew "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/databrew"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own project"})

    		ubx.Resource(databrew.Project, "example", databrew.ProjectConfig{
    			DatasetName: "example-dataset-name",
    			Name:        "example",
    			RecipeName:  "example-recipe-name",
    			RoleArn:     "arn:aws:iam::123456789012:policy/example",
    			Sample:      map[string]any{"type": "example"},
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.databrew import Project, ProjectConfig
    from ubx.aws.databrew.project import Project_Sample, Project_Tags

    def describe():
        ubx.intent("example own project")

        ubx.resource(Project, "example", ProjectConfig(
            dataset_name="example-dataset-name",
            name="example",
            recipe_name="example-recipe-name",
            role_arn="arn:aws:iam::123456789012:policy/example",
            sample=Project_Sample(type="example"),
            tags=[Project_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
