`aws_data_brew_dataset` connects AWS Glue DataBrew to a real source of raw data, an S3 location among the supported sources, the actual input an `aws_data_brew_project` explores and an `aws_data_brew_job` transforms.

## Example

    ```go
    	databrew "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/databrew"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dataset"})

    		ubx.Resource(databrew.Dataset, "example", databrew.DatasetConfig{
    			Input:         map[string]any{"data_catalog_input_definition": map[string]any{"catalog_id": "example"}},
    			Name:          "example",
    			Format:        "example",
    			FormatOptions: map[string]any{"csv": map[string]any{"delimiter": "example"}},
    			PathOptions:   map[string]any{"files_limit": map[string]any{"max_files": 1}},
    			Source:        "example",
    			Tags:          []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
