`aws_data_brew_job` actually runs an `aws_data_brew_recipe` against an `aws_data_brew_dataset` at production scale, terabytes of data, no infrastructure to provision, writing the cleaned, transformed output to S3.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	databrew "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/databrew"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own job"})

    		ubx.Resource(databrew.Job, "example", databrew.JobConfig{
    			Name:               "example",
    			RoleArn:            "arn:aws:iam::123456789012:policy/example",
    			Type:               "example",
    			DataCatalogOutputs: []map[string]any{{"catalog_id": "example"}},
    			DatabaseOutputs:    []map[string]any{{"database_options": map[string]any{"table_name": "example-table-name"}}},
    			DatasetName:        "example-dataset-name",
    			EncryptionKeyArn:   "arn:aws:iam::123456789012:policy/example",
    			EncryptionMode:     "example",
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.databrew import Job, JobConfig
    from ubx.aws.databrew.job import Job_DataCatalogOutputs, Job_DatabaseOutputs, Job_DataCatalogOutputs_DatabaseOptions

    def describe():
        ubx.intent("example own job")

            name="example",
            role_arn="arn:aws:iam::123456789012:policy/example",
            type="example",
            data_catalog_outputs=[Job_DataCatalogOutputs(catalog_id="example")],

            dataset_name="example-dataset-name",

    if __name__ == "__main__":
        ubx.run("example", describe)
