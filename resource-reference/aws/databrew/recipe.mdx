`aws_data_brew_recipe` is the real, saved sequence of transformations, chosen from DataBrew's 250-plus built-in options, built inside an `aws_data_brew_project` and then reusable, unchanged or updated, against other datasets rather than rebuilt from scratch each time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	databrew "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/databrew"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own recipe"})

    		ubx.Resource(databrew.Recipe, "example", databrew.RecipeConfig{
    			Name:        "example",
    			Steps:       []map[string]any{{"action": map[string]any{"operation": "example"}}},
    			Description: "Managed by ubx.",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.databrew import Recipe, RecipeConfig
    from ubx.aws.databrew.recipe import Recipe_Steps, Recipe_Steps_Action, Recipe_Tags

    def describe():
        ubx.intent("example own recipe")

        ubx.resource(Recipe, "example", RecipeConfig(
            name="example",
            steps=[Recipe_Steps(action=Recipe_Steps_Action(operation="example"))],
            description="Managed by ubx.",
            tags=[Recipe_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
