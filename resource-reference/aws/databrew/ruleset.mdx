`aws_data_brew_ruleset` defines real data-quality rules, checks for nulls, valid ranges, expected formats, that get validated against an `aws_data_brew_dataset`, catching bad data before it flows further downstream rather than only after a job has already run.

## Example

    ```go
    package main

    import (
    	databrew "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/databrew"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own ruleset"})

    		ubx.Resource(databrew.Ruleset, "example", databrew.RulesetConfig{
    			Name:        "example",
    			Rules:       []map[string]any{{"name": "example-name"}},
    			TargetArn:   "arn:aws:iam::123456789012:policy/example",
    			Description: "Managed by ubx.",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
