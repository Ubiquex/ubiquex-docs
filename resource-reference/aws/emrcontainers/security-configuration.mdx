`aws_emrcontainers_security_configuration` is a real, reusable security setup, including AWS Lake Formation integration, that gets applied to an `aws_emrcontainers_virtual_cluster` when it's created, rather than every virtual cluster's security being configured individually.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	emrcontainers "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/emrcontainers"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own security configuration"})

    		ubx.Resource(emrcontainers.SecurityConfiguration, "example", emrcontainers.SecurityConfigurationConfig{
    			SecurityConfigurationData: map[string]any{"authentication_configuration": map[string]any{"iamconfiguration": map[string]any{"system_role": "example"}}},
    			Name:                      "example",
    			ContainerProvider:         map[string]any{"id": "example"},
    			Tags:                      []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
