`aws_emrcontainers_virtual_cluster` registers a real Amazon EKS namespace as an EMR on EKS virtual cluster, the actual target Spark and Hive jobs submitted through EMR on EKS run against, using the underlying Kubernetes cluster's own compute rather than dedicated EMR instances.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	emrcontainers "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/emrcontainers"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own virtual cluster"})

    		ubx.Resource(emrcontainers.VirtualCluster, "example", emrcontainers.VirtualClusterConfig{
    			ContainerProvider:       map[string]any{"id": "example"},
    			Name:                    "example",
    			SecurityConfigurationId: "example",
    			SessionEnabled:          true,
    			Tags:                    []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
