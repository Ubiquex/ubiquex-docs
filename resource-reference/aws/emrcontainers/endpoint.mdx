`aws_emrcontainers_endpoint` is a real, managed gateway connecting an `aws_emr_studio` to an `aws_emrcontainers_virtual_cluster`, so Studio's notebooks can actually communicate with and submit interactive work to that virtual cluster.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	emrcontainers "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/emrcontainers"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own endpoint"})

    		ubx.Resource(emrcontainers.Endpoint, "example", emrcontainers.EndpointConfig{
    			ExecutionRoleArn:            "arn:aws:iam::123456789012:policy/example",
    			ReleaseLabel:                "example",
    			Type:                        "example",
    			VirtualClusterId:            "example",
    			Name:                        "example",
    			ConfigurationOverrides:      map[string]any{"application_configuration": []map[string]any{{"classification": "example"}}},
    			SessionIdleTimeoutInMinutes: 1,
    			Tags:                        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.emrcontainers import Endpoint, EndpointConfig
    from ubx.aws.emrcontainers.endpoint import Endpoint_ConfigurationOverrides, Endpoint_ConfigurationOverrides_ApplicationConfiguration, Endpoint_Tags

    def describe():
        ubx.intent("example own endpoint")

        ubx.resource(Endpoint, "example", EndpointConfig(
            execution_role_arn="arn:aws:iam::123456789012:policy/example",
            release_label="example",
            type="example",
            virtual_cluster_id="example",
            name="example",
            configuration_overrides=Endpoint_ConfigurationOverrides(application_configuration=[Endpoint_ConfigurationOverrides_ApplicationConfiguration(classification="example")]),
            session_idle_timeout_in_minutes=1,
            tags=[Endpoint_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
