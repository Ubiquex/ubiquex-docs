`aws_neptune_dbparameter_group` defines a real set of instance-level engine parameters that can be applied to an `aws_neptune_dbinstance` instead of hand-tuning the instance individually.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	neptune "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/neptune"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dbparameter group"})

    		ubx.Resource(neptune.DbparameterGroup, "example", neptune.DbparameterGroupConfig{
    			Description: "Managed by ubx.",
    			Family:      "example",
    			Parameters:  "example",
    			Name:        "example",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.neptune import DbparameterGroup, DbparameterGroupConfig
    from ubx.aws.neptune.dbparameter_group import DbparameterGroup_Tags

    def describe():
        ubx.intent("example own dbparameter group")

        ubx.resource(DbparameterGroup, "example", DbparameterGroupConfig(
            description="Managed by ubx.",
            family="example",
            parameters="example",
            name="example",
            tags=[DbparameterGroup_Tags(key="example")],
        ))

## Properties

<ResponseField name="description" type="string" required>
  Provides the customer-specified description for this DB parameter group. Required.
</ResponseField>

<ResponseField name="family" type="string" required>
  Must be `neptune1` for engine versions prior to 1.2.0.0, or `neptune1.2` for engine version `1.2.0.0` and higher. Required.
</ResponseField>

<ResponseField name="name" type="string">
  Provides the name of the DB parameter group. Optional.
</ResponseField>

<ResponseField name="parameters" type="any" required>
  The parameters to set for this DB parameter group. The parameters are expressed as a JSON object consisting of key-value pairs. Changes to dynamic parameters are applied immediately. During an update, if you have static parameters (whether they were changed or not), it triggers AWS CloudFormation to reboot the associated DB instance without failover. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An optional array of key-value pairs to apply to this DB parameter group. Optional.
