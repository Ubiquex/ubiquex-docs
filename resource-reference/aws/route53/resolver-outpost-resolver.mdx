`aws_route53_resolver_outpost_resolver` creates a real DNS resolver running on an AWS Outposts rack, extending Route 53 Resolver's own hybrid DNS resolution to workloads on that on-premises Outpost.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	route53 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/route53"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resolver outpost resolver"})

    		ubx.Resource(route53.ResolverOutpostResolver, "example", route53.ResolverOutpostResolverConfig{
    			Name:                  "example",
    			OutpostArn:            "arn:aws:iam::123456789012:policy/example",
    			PreferredInstanceType: "example",
    			InstanceCount:         1,
    			Tags:                  []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

## Input properties

<ResponseField name="instance_count" type="number">
  The number of OutpostResolvers. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  The OutpostResolver name. Required.
</ResponseField>

<ResponseField name="outpost_arn" type="string" required>
  The Outpost ARN. Required.
</ResponseField>

<ResponseField name="preferred_instance_type" type="string" required>
  The OutpostResolver instance type. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An array of key-value pairs to apply to this resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key of a tag assigned to the Route 53 Resolver Outpost resolver, used to organize, identify, and manage the resolver within AWS. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>
