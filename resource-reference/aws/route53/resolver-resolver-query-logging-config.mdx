`aws_route53_resolver_resolver_query_logging_config` configures real logging of DNS queries resolved through an `aws_route53_resolver_resolver_endpoint` or a VPC's default resolver, to a CloudWatch Logs group, S3 bucket, or Kinesis Data Firehose.

## Example

    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	route53 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/route53"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own resolver resolver query logging config"})

    		ubx.Resource(route53.ResolverResolverQueryLoggingConfig, "ci-runner", route53.ResolverResolverQueryLoggingConfigConfig{
    			Name:           "ci-runner",
    			DestinationArn: "arn:aws:iam::123456789012:policy/ci",
    			Tags:           []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
