`aws_route53_dnssec` enables DNSSEC signing for an `aws_route53_hosted_zone`, the resource that actually turns on cryptographic signing of that zone's DNS responses so resolvers can verify they haven't been tampered with.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	route53 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/route53"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dnssec"})

    		ubx.Resource(route53.Dnssec, "example", route53.DnssecConfig{
    			HostedZoneId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
