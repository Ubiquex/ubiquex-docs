`aws_route53_resolver_resolver_query_logging_config_association` associates a VPC with an `aws_route53_resolver_resolver_query_logging_config`, the resource that actually turns query logging on for that specific VPC's DNS traffic.

## Example

    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	route53 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/route53"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own resolver resolver query logging config association"})

    		ubx.Resource(route53.ResolverResolverQueryLoggingConfigAssociation, "ci-runner", route53.ResolverResolverQueryLoggingConfigAssociationConfig{
    			ResolverQueryLogConfigId: "example",
    			ResourceId:               "example",
    		})
    	}))
    }
    ```
