`aws_route53_global_resolver_hosted_zone_association` associates a real Route 53 private hosted zone with an `aws_route53_global_resolver_global_resolver`, so its records become resolvable through Global Resolver's own anycast endpoints for authorized external clients.

## Example

  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	route53 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/route53"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own global resolver hosted zone association"})

    		ubx.Resource(route53.GlobalResolverHostedZoneAssociation, "ci-runner", route53.GlobalResolverHostedZoneAssociationConfig{
    			HostedZoneId: "example",
    			Name:         "ci-runner",
    			ResourceArn:  "arn:aws:iam::123456789012:policy/ci",
    		})
    	}))
    }
    ```
