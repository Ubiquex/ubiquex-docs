`aws_route53_record_set_group` creates a real batch of `aws_route53_record_set`-equivalent DNS records within an `aws_route53_hosted_zone` in a single resource, useful when a template needs to manage many records together.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	route53 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/route53"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own record set group"})

    		ubx.Resource(route53.RecordSetGroup, "example", route53.RecordSetGroupConfig{
    			Comment:        "example",
    			HostedZoneId:   "example",
    			HostedZoneName: "example-hosted-zone-name",
    			RecordSets:     []map[string]any{{"name": "example-name"}},
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    import ubx_sdk as ubx
    from ubx.aws.route53 import RecordSetGroup, RecordSetGroupConfig

    def describe():
        ubx.intent("example own record set group")

        ubx.resource(RecordSetGroup, "example", RecordSetGroupConfig(
            comment="example",
            hosted_zone_id="example",
            hosted_zone_name="example-hosted-zone-name",
            record_sets=[{"name": "example-name"}],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
