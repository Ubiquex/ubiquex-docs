`aws_route53_resolver_resolver_dnssecconfig` configures whether Route 53 Resolver validates DNSSEC signatures for outbound queries a VPC sends through an `aws_route53_resolver_resolver_endpoint`, rejecting responses that fail validation.

## Example

    ```go
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own resolver resolver dnssecconfig"})

    		ubx.Resource(route53.ResolverResolverDnssecconfig, "ci-runner", route53.ResolverResolverDnssecconfigConfig{
    			ResourceId: "example",
    		})
    	}))
    }
    ```
