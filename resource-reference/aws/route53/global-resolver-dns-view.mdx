`aws_route53_global_resolver_dns_view` defines a real DNS view, which of an account's private hosted zones and rules apply, that an `aws_route53_global_resolver_global_resolver` resolves queries against for a given set of clients.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own global resolver dns view"})

    		ubx.Resource(route53.GlobalResolverDnsView, "example", route53.GlobalResolverDnsViewConfig{

    			Description:           "Managed by ubx.",
    			DnssecValidation:      "example",
    			EdnsClientSubnet:      "example",

    		})
    	}))
