`aws_app_test_test_case` defines the smallest real unit of a mainframe modernization test in AWS Mainframe Modernization Application Testing, typically one independent piece of business logic that modifies data, captured against the real mainframe and later replayed and compared against a modernized target.

## Example

<Tabs>

    ```go
    package main

    import (
    	apptest "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/apptest"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own test test case"})

    		ubx.Resource(apptest.TestCase, "example", apptest.TestCaseConfig{
    			Name:        "example",
    			Steps:       []map[string]any{{"name": "example-name"}},
    			Description: "Managed by ubx.",
    			Tags:        "example",
    		})
    	}))
    }
    ```
