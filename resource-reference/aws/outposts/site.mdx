`aws_outposts_site` registers a real physical location, its address and operating hours, where an AWS Outposts rack or server will be installed, extending AWS infrastructure and APIs into that on-premises facility.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	outposts "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/outposts"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own site"})

    		ubx.Resource(outposts.Site, "example", outposts.SiteConfig{
    			Name:                   "example",
    			Description:            "Managed by ubx.",
    			Notes:                  "example",
    			OperatingAddress:       map[string]any{"address_line1": "example"},
    			RackPhysicalProperties: map[string]any{"fiber_optic_cable_type": "example"},
    			ShippingAddress:        map[string]any{"address_line1": "example"},
    			Tags:                   []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
