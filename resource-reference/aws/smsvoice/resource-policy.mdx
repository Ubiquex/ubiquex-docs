`aws_smsvoice_resource_policy` attaches a real resource-based access policy to an `aws_smsvoice_phone_number` or pool, controlling which principals, including from other accounts, can send through it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	smsvoice "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/smsvoice"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resource policy"})

    		ubx.Resource(smsvoice.ResourcePolicy, "example", smsvoice.ResourcePolicyConfig{
    			PolicyDocument: "example",
    			ResourceArn:    "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.aws.smsvoice import ResourcePolicy, ResourcePolicyConfig

    def describe():

            policy_document="example",
            resource_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Properties

<ResponseField name="policy_document" type="any" required>
  The JSON formatted resource-based policy to attach. Required.
</ResponseField>

<ResponseField name="resource_arn" type="string" required>
  The Amazon Resource Name (ARN) of the AWS End User Messaging SMS and Voice resource to attach the resource-based policy to. Required.
</ResponseField>
