`aws_translate_parallel_data` imports a real set of source-and-target sentence pairs that Amazon Translate uses to customize its own translation output, favoring an organization's preferred terminology and phrasing over the engine's generic default translations.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	translate "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/translate"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own parallel data"})

    		ubx.Resource(translate.ParallelData, "example", translate.ParallelDataConfig{
    			Name:               "example",
    			ParallelDataConfig: map[string]any{"format": "example"},
    			Description:        "Managed by ubx.",
    			EncryptionKey:      map[string]any{"id": "example"},
    			Tags:               []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx

    def describe():

    if __name__ == "__main__":
        ubx.run("example", describe)
