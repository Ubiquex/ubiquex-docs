`aws_app_flow_connector` registers a real custom connector, backed by a Lambda function, that an `aws_app_flow_flow` can use as a source or destination beyond AppFlow's own built-in SaaS connectors.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	appflow "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/appflow"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own flow connector"})

    		ubx.Resource(appflow.Connector, "example", appflow.ConnectorConfig{
    			ConnectorProvisioningConfig: map[string]any{"lambda": map[string]any{"lambda_arn": "example"}},
    			ConnectorProvisioningType:   "example",
    			ConnectorLabel:              "example",
    			Description:                 "Managed by ubx.",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Connector } from "@ubx/sdk-aws/aws/appflow/connector";

    export default stack("example", () => {
      intent({ summary: "example own flow connector" });

      resource(Connector, "example", {
        connectorProvisioningConfig: { lambda: { lambdaArn: "example" } },
        connectorProvisioningType: "example",
        connectorLabel: "example",
        description: "Managed by ubx.",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.appflow.connector import Connector, ConnectorConfig

    def describe():
        ubx.intent("example own flow connector")

        ubx.resource(Connector, "example", ConnectorConfig(
            connector_provisioning_config={"lambda": {"lambda_arn": "example"}},
            connector_provisioning_type="example",
            connector_label="example",
            description="Managed by ubx.",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="connector_label" type="string">
  The name of the connector. The name is unique for each ConnectorRegistration in your AWS account. Optional.
</ResponseField>

<ResponseField name="connector_provisioning_config" type="object" required>
  Contains information about the configuration of the connector being registered. Required.
  <Expandable title="properties">
    <ResponseField name="lambda" type="object">
      Contains information about the configuration of the lambda which is being registered as the connector. Optional.
      <Expandable title="properties">
        <ResponseField name="lambda_arn" type="string" required>
          Lambda ARN of the connector being registered. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="connector_provisioning_type" type="string" required>
  The provisioning type of the connector. Currently the only supported value is LAMBDA. Required.
</ResponseField>

<ResponseField name="description" type="string">
  A description about the connector that's being registered. Optional.
</ResponseField>

## Output properties

<ResponseField name="connector_arn" type="string">
  The arn of the connector. The arn is unique for each ConnectorRegistration in your AWS account. Computed by AWS after `ship`.
</ResponseField>
