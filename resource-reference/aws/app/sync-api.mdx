`aws_app_sync_api` creates a GraphQL API in AWS AppSync, a managed, serverless endpoint that resolves queries against one or more real backend data sources, with built-in subscriptions for pushing real-time updates to connected clients.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	appsync "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/appsync"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own sync api"})

    		ubx.Resource(appsync.Api, "example", appsync.ApiConfig{
    			Name:         "example",
    			EventConfig:  map[string]any{"auth_providers": []map[string]any{{"auth_type": "example"}}},
    			OwnerContact: "example",
    			Tags:         []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
