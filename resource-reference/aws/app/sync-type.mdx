`aws_app_sync_type` defines a real, individual GraphQL type, in SDL or JSON format, within an `aws_app_sync_api`, a more granular alternative to defining an API's entire schema at once.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	appsync "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/appsync"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own sync type"})

    		ubx.Resource(appsync.Type, "example", appsync.TypeConfig{
    			ApiId:      "example",
    			Definition: "example",
    			Format:     "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Type } from "@ubx/sdk-aws/aws/appsync/type";

    export default stack("example", () => {
      intent({ summary: "example own sync type" });

      resource(Type, "example", {
        apiId: "example",
        definition: "example",

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.appsync.type import Type, TypeConfig

    def describe():
        ubx.intent("example own sync type")

        ubx.resource(Type, "example", TypeConfig(
            api_id="example",
            definition="example",
            format="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="api_id" type="string" required>
  The API ID. Required.
</ResponseField>

<ResponseField name="definition" type="string" required>
  The type definition, in GraphQL Schema Definition Language (SDL) format. Required.
</ResponseField>

<ResponseField name="format" type="string" required>
  The type format: SDL or JSON. Required.
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  The Amazon Resource Name (ARN) of the type. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="name" type="string">
  The type name. Computed by AWS after `ship`.
</ResponseField>
