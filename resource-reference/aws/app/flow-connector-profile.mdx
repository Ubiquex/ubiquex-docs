`aws_app_flow_connector_profile` stores the authenticated connection details, credentials and endpoint, for one SaaS source or destination (Salesforce, Zendesk, Slack, and similar), that an `aws_app_flow_flow` then draws on to move data without re-authenticating per flow.

## Example

    ```go
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own flow connector profile"})

    		ubx.Resource(appflow.ConnectorProfile, "example", appflow.ConnectorProfileConfig{
    			ConnectionMode:         "example",
    			ConnectorProfileName:   "example-connector-profile-name",
    			ConnectorType:          "example",
    			ConnectorLabel:         "example",
    			ConnectorProfileConfig: map[string]any{"connector_profile_credentials": map[string]any{"amplitude": map[string]any{"api_key": "example"}}},
    			Kmsarn:                 "example",
    		})
    	}))
    }
    ```
