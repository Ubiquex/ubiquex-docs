`aws_app_stream_app_block` packages the application files and setup script needed to stream one application, built by an `aws_app_stream_app_block_builder` and uploaded to S3, the app-block model that lets an application be added to a stack without baking it into a full image.

## Example

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.appstream import AppBlock, AppBlockConfig
    from ubx.aws.appstream.app_block import AppBlock_PostSetupScriptDetails_ScriptS3Location, AppBlock_PostSetupScriptDetails, AppBlock_Tags

    def describe():
        ubx.intent("example own app block")

        ubx.resource(AppBlock, "example", AppBlockConfig(
            name="example",
            source_s3_location=AppBlock_PostSetupScriptDetails_ScriptS3Location(s3_bucket="example"),
            description="Managed by ubx.",
            display_name="example-display-name",
            packaging_type="example",
            post_setup_script_details=AppBlock_PostSetupScriptDetails(executable_path="example", script_s3_location=AppBlock_PostSetupScriptDetails_ScriptS3Location(s3_bucket="example"), timeout_in_seconds=1),
            setup_script_details=AppBlock_PostSetupScriptDetails(executable_path="example", script_s3_location=AppBlock_PostSetupScriptDetails_ScriptS3Location(s3_bucket="example"), timeout_in_seconds=1),
            tags=[AppBlock_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
