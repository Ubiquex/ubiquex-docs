`aws_app_config_application` is the top-level container in AWS AppConfig, the logical grouping that an `aws_app_config_environment` and `aws_app_config_configuration_profile` both attach to. AppConfig itself lets you change application behavior, feature flags, allow lists, throttling limits, in production without redeploying code.

    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	app "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/app"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own config application"})

    		ubx.Resource(app.ConfigApplication, "ci-runner", app.ConfigApplicationConfig{
    			Name:        "ci-runner",
    			Description: "Managed by ubx.",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }

<ResponseField name="tags" type="list(object)">
