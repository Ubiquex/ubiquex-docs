`aws_certificate_manager_acme_domain_validation` pre-approves a real domain and provisions the required CNAME record for an `aws_certificate_manager_acme_endpoint`, so an ACME client can later request a certificate for that domain without a separate manual validation step each time. ACM's ACME support was announced in July 2026, roughly a month ago.

    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own manager acme domain validation"})

    		ubx.Resource(certificate.ManagerAcmeDomainValidation, "example", certificate.ManagerAcmeDomainValidationConfig{
    			AcmeEndpointArn:      "arn:aws:iam::123456789012:policy/example",
    			DomainName:           "example-domain-name",
    			PrevalidationOptions: map[string]any{"dns_prevalidation": map[string]any{"domain_scope": map[string]any{"exact_domain": "example"}}},
    			Tags:                 []map[string]any{{"key": "example"}},
    		})
    	}))
    }

## Input properties

<ResponseField name="acme_endpoint_arn" type="string" required>
  The ARN of the ACME endpoint this domain validation is associated with. Required.
</ResponseField>

<ResponseField name="domain_name" type="string" required>
  The domain name to validate. Required.
</ResponseField>

<ResponseField name="prevalidation_options" type="object" required>
  Prevalidation method configuration. Currently only DNS-based prevalidation is supported. Required.
  <Expandable title="properties">
    <ResponseField name="dns_prevalidation" type="object" required>
      DNS-based prevalidation options for the domain validation. Required.
      <Expandable title="properties">
        <ResponseField name="domain_scope" type="object">
          Controls which certificate types are authorized to be issued for the domain via the ACME endpoint. Optional.
          <Expandable title="properties">
            <ResponseField name="exact_domain" type="string">
              Whether certificates may be issued for the exact domain. Optional.
            </ResponseField>
            <ResponseField name="subdomains" type="string">
              Whether certificates may be issued for subdomains of the domain. Optional.
            </ResponseField>
            <ResponseField name="wildcards" type="string">
              Whether wildcard certificates may be issued for the domain. Optional.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="hosted_zone_id" type="string">
          The Route 53 hosted zone ID for automatic DNS record management. When provided, the service creates the validation DNS record on the customer's behalf. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags associated with the domain validation. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  The Amazon Resource Name (ARN) of the domain validation. Computed by AWS after `ship`.
</ResponseField>
