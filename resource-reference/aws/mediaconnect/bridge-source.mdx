    ```python
    import ubx_sdk as ubx
    from ubx.aws.mediaconnect import BridgeSource, BridgeSourceConfig
    from ubx.aws.mediaconnect.bridge_source import BridgeSource_FlowSource, BridgeSource_NetworkSource

    def describe():
        ubx.intent("example own bridge source")

        ubx.resource(BridgeSource, "example", BridgeSourceConfig(
            bridge_arn="arn:aws:iam::123456789012:policy/example",
            name="example",
            flow_source=BridgeSource_FlowSource(flow_arn="example"),
            network_source=BridgeSource_NetworkSource(multicast_ip="example", network_name="example-network-name", port=1, protocol="example"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="bridge_arn" type="string" required>
  The Amazon Resource Number (ARN) of the bridge. Required.
</ResponseField>

<ResponseField name="flow_source" type="object">
  The source of the bridge. A flow source originates in MediaConnect as an existing cloud flow. Optional.
  <Expandable title="properties">
    <ResponseField name="flow_arn" type="string" required>
      The ARN of the cloud flow used as a source of this bridge. Required.
    </ResponseField>
    <ResponseField name="flow_vpc_interface_attachment" type="object">
      The settings for attaching a VPC interface to an resource. Optional.
      <Expandable title="properties">
        <ResponseField name="vpc_interface_name" type="string">
          The name of the VPC interface to use for this resource. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the source. Required.
</ResponseField>

<ResponseField name="network_source" type="object">
  The source of the bridge. A network source originates at your premises. Optional.
  <Expandable title="properties">
    <ResponseField name="multicast_ip" type="string" required>
      The network source multicast IP. Required.
    </ResponseField>
    <ResponseField name="multicast_source_settings" type="object">
      The settings related to the multicast source. Optional.
      <Expandable title="properties">
        <ResponseField name="multicast_source_ip" type="string">
          The IP address of the source for source-specific multicast (SSM). Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="network_name" type="string" required>
      The network source's gateway network name. Required.
    </ResponseField>
    <ResponseField name="port" type="number" required>
      The network source port. Required.
    </ResponseField>
    <ResponseField name="protocol" type="string" required>
      Specifies the transport protocol (RTP or RTP-FEC) used by the network source to ingest the media stream into the bridge. Required. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>
