---

`aws_evs_environment` deploys a real VMware Cloud Foundation stack, SDDC Manager, NSX Manager, vCenter Server, and an initial set of ESXi hosts, directly on EC2 bare-metal instances inside a caller's own VPC, letting existing VMware-based workloads migrate to AWS unmodified rather than needing conversion to a different hypervisor.

    import ubx_sdk as ubx
    from ubx.aws.evs import Environment, EnvironmentConfig
    from ubx.aws.evs.environment import Environment_ConnectivityInfo, Environment_Hosts, Environment_InitialVlans, Environment_InitialVlans_EdgeVtep

    def describe():
        ubx.intent("example own environment")

        ubx.resource(Environment, "example", EnvironmentConfig(
            service_access_subnet_id="example",
            terms_accepted=True,
            vcf_version="example",
            vpc_id="example",
            connectivity_info=Environment_ConnectivityInfo(private_route_server_peerings=["example"]),
            environment_name="example-environment-name",
            hosts=[Environment_Hosts(dedicated_host_id="example")],
            initial_vlans=Environment_InitialVlans(edge_vtep=Environment_InitialVlans_EdgeVtep(cidr="example"), expansion_vlan1=Environment_InitialVlans_EdgeVtep(cidr="example"), expansion_vlan2=Environment_InitialVlans_EdgeVtep(cidr="example"), hcx=Environment_InitialVlans_EdgeVtep(cidr="example"), nsx_up_link=Environment_InitialVlans_EdgeVtep(cidr="example"), vm_management=Environment_InitialVlans_EdgeVtep(cidr="example"), vmk_management=Environment_InitialVlans_EdgeVtep(cidr="example"), vmotion=Environment_InitialVlans_EdgeVtep(cidr="example"), vsan=Environment_InitialVlans_EdgeVtep(cidr="example"), vtep=Environment_InitialVlans_EdgeVtep(cidr="example")),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
