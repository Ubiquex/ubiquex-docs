`aws_api_gateway_method` defines one HTTP method (GET, POST, and similar) on an `aws_api_gateway_resource`, together with its own authorization, request validation, and backend integration, the real unit a REST API caller actually invokes.

## Example

  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    import (
    	api "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/api"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own gateway method"})

    		ubx.Resource(api.GatewayMethod, "ci-runner", api.GatewayMethodConfig{
    			HttpMethod:          "example",
    			ResourceId:          "example",
    			RestApiId:           "example",
    			ApiKeyRequired:      true,
    			AuthorizationScopes: []string{"example"},
    			AuthorizationType:   "example",
    			AuthorizerId:        "example",
    			Integration:         map[string]any{"type": "example"},
    		})
    	}))
    }
    ```
