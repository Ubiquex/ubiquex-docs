`aws_api_gateway_v2_routing_rule` defines additional request-matching logic for an `aws_api_gateway_v2_api`'s routes beyond the basic method-and-path match, used to route the same path differently based on request content.

## Example

    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { GatewayV2RoutingRule } from "./local-sdk/aws/sdk/typescript/aws/api/gateway_v2_routing_rule.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own gateway v2 routing rule" });

      resource(GatewayV2RoutingRule, "ci-runner", {
        actions: [{ invokeApi: { apiId: "example" } }],
        conditions: [{ matchBasePaths: { anyOf: ["example"] } }],
        domainNameArn: "arn:aws:iam::123456789012:policy/ci",
        priority: 1,
      });
    });
