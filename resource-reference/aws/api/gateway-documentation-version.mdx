`aws_api_gateway_documentation_version` snapshots an `aws_api_gateway_rest_api`'s documentation parts at a point in time, kept versioned separately from the API's own deployments so docs and API behavior can each move independently.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	api "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/api"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own gateway documentation version"})

    		ubx.Resource(api.GatewayDocumentationVersion, "ci-runner", api.GatewayDocumentationVersionConfig{
    			DocumentationVersion: "example",
    			RestApiId:            "example",
    			Description:          "Managed by ubx.",
    		})
    	}))
    }
    ```
