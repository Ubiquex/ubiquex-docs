`aws_application_signals_discovery` enables real, automatic discovery of an account's services, APIs, and their dependencies for CloudWatch Application Signals, the mechanism that populates its Services and Application Map views from OpenTelemetry-instrumented traces without each service being registered by hand.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	applicationsignals "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/applicationsignals"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own discovery"})

    		ubx.Resource(applicationsignals.Discovery, "example", applicationsignals.DiscoveryConfig{
    			AccountId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Discovery } from "@ubx/sdk-aws/aws/applicationsignals/discovery";

    export default stack("example", () => {
      intent({ summary: "example own discovery" });

      resource(Discovery, "example", {
        accountId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.applicationsignals import Discovery, DiscoveryConfig

    def describe():
        ubx.intent("example own discovery")

        ubx.resource(Discovery, "example", DiscoveryConfig(
            account_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
