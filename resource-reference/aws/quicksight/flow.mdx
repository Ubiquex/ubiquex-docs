`aws_quick_sight_flow` defines a real Quick Flow, an automated, multi-step workflow within Amazon Quick, part of the AI-workspace capabilities QuickSight evolved into after its October 2025 rename to Amazon Quick Suite (shortened to Amazon Quick in 2026).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	quicksight "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/quicksight"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own flow"})

    		ubx.Resource(quicksight.Flow, "example", quicksight.FlowConfig{
    			AwsAccountId:   "example",
    			FlowDefinition: "example",
    			Name:           "example",
    			Description:    "Managed by ubx.",
    			Permissions:    []map[string]any{{"actions": []string{"example"}}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.quicksight import Flow, FlowConfig
    from ubx.aws.quicksight.flow import Flow_Permissions

    def describe():
        ubx.intent("example own flow")

        ubx.resource(Flow, "example", FlowConfig(
            aws_account_id="example",
            flow_definition="example",
            name="example",
            description="Managed by ubx.",
            permissions=[Flow_Permissions(actions=["example"])],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
