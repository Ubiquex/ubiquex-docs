`aws_quick_sight_limits_profile` defines a real set of usage limits, embedding session counts, API rate limits, that can be applied to control consumption of QuickSight/Amazon Quick resources within an account or by a specific group of users.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	quicksight "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/quicksight"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own sight limits profile"})

    		ubx.Resource(quicksight.LimitsProfile, "example", quicksight.LimitsProfileConfig{
    			AccountId:      "example",
    			ProfileName:    "example-profile-name",
    			ResourceLimits: "example",
    			Description:    "Managed by ubx.",
    		})
    	}))
    }
    ```
