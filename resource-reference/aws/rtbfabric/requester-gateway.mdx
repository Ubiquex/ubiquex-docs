`aws_rtbfabric_requester_gateway` creates a real gateway within AWS RTB Fabric that originates real-time bidding requests, the side of the fabric an ad exchange's own bid-request traffic enters through.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	rtbfabric "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/rtbfabric"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own requester gateway"})

    		ubx.Resource(rtbfabric.RequesterGateway, "example", rtbfabric.RequesterGatewayConfig{
    			SecurityGroupIds: []string{"example"},
    			SubnetIds:        []string{"example"},
    			VpcId:            "example",
    			Description:      "Managed by ubx.",
    			Tags:             []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
