`aws_rtbfabric_outbound_external_link` connects a real `aws_rtbfabric_requester_gateway` out to an external bidder or exchange, the outbound counterpart to `aws_rtbfabric_inbound_external_link` within AWS RTB Fabric.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	rtbfabric "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/rtbfabric"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own outbound external link"})

    		ubx.Resource(rtbfabric.OutboundExternalLink, "example", rtbfabric.OutboundExternalLinkConfig{
    			GatewayId:       "example",
    			LinkLogSettings: map[string]any{"application_logs": map[string]any{"link_application_log_sampling": map[string]any{"error_log": 1}}},
    			PublicEndpoint:  "example",
    			LinkAttributes:  map[string]any{"customer_provided_id": "example"},
    			Tags:            []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
