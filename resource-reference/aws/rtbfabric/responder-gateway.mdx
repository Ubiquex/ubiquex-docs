`aws_rtbfabric_responder_gateway` creates a real gateway within AWS RTB Fabric that receives real-time bidding requests and returns bid responses, the side of the fabric a bidder's own infrastructure sits behind.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	rtbfabric "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/rtbfabric"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own responder gateway"})

    		ubx.Resource(rtbfabric.ResponderGateway, "example", rtbfabric.ResponderGatewayConfig{
    			Port:              1,
    			Protocol:          "example",
    			SecurityGroupIds:  []any{"example"},
    			SubnetIds:         []string{"example"},
    			VpcId:             "example",
    			AcmCertificateArn: "arn:aws:iam::123456789012:policy/example",
    			Description:       "Managed by ubx.",
    			DomainName:        "example-domain-name",
    		})
    	}))
    }
    ```
