---

`aws_synthetics_canary` creates a real, scheduled script, a Node.js or Python Selenium/Playwright-style script, that CloudWatch Synthetics runs on a recurring cadence against a live endpoint or UI flow, proactively detecting an outage or broken user journey before a real user hits it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	synthetics "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/synthetics"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own canary"})

    		ubx.Resource(synthetics.Canary, "example", synthetics.CanaryConfig{
    			ArtifactS3Location:                    "example",
    			ExecutionRoleArn:                      "arn:aws:iam::123456789012:policy/example",
    			Name:                                  "example",
    			RuntimeVersion:                        "example",
    			Schedule:                              map[string]any{"expression": "example"},
    			ArtifactConfig:                        map[string]any{"s3_encryption": map[string]any{"encryption_mode": "example"}},
    			BrowserConfigs:                        []map[string]any{{"browser_type": "example"}},
    			DeleteLambdaResourcesOnCanaryDeletion: true,
    		})
    	}))
    }
    ```
