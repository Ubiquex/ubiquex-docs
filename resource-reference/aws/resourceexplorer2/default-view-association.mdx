`aws_resource_explorer2_default_view_association` sets a real `aws_resource_explorer2_view` as the default for its AWS Region, so a search that doesn't explicitly name a view uses this one automatically.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	resourceexplorer2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/resourceexplorer2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own default view association"})

    		ubx.Resource(resourceexplorer2.DefaultViewAssociation, "example", resourceexplorer2.DefaultViewAssociationConfig{
    			ViewArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    import ubx_sdk as ubx
    from ubx.aws.resourceexplorer2 import DefaultViewAssociation, DefaultViewAssociationConfig

    def describe():
        ubx.intent("example own default view association")

        ubx.resource(DefaultViewAssociation, "example", DefaultViewAssociationConfig(
            view_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="view_arn" type="string" required>
  The ARN of the Resource Explorer view that will be set as the default view for the account. Required. **(AI-inferred)**
</ResponseField>

## Output properties

<ResponseField name="associated_aws_principal" type="string">
  The AWS principal that the default view is associated with, used as the unique identifier for this resource. Computed by AWS after `ship`.
</ResponseField>
