`aws_network_firewall_tlsinspection_configuration` defines real certificate and scope settings that let an `aws_network_firewall_firewall_policy` decrypt, inspect, and re-encrypt TLS traffic rather than only inspecting it in encrypted form.

## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	networkfirewall "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/networkfirewall"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own firewall tlsinspection configuration"})

    		ubx.Resource(networkfirewall.TlsinspectionConfiguration, "example", networkfirewall.TlsinspectionConfigurationConfig{
    			TlsinspectionConfiguration:     map[string]any{"server_certificate_configurations": []map[string]any{{"certificate_authority_arn": "example"}}},
    			TlsinspectionConfigurationName: "example-tlsinspection-configuration-name",
    			Description:                    "Managed by ubx.",
    			Tags:                           []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
