`aws_network_firewall_vpc_endpoint_association` attaches a real additional subnet-scoped firewall endpoint to an existing `aws_network_firewall_firewall`, letting one firewall protect multiple VPCs or provide multiple endpoints within a single VPC's Availability Zone, a capability AWS Network Firewall added in May 2025.

## Example

    ```go
    package main

    import (
    	networkfirewall "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/networkfirewall"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own vpc endpoint association"})

    		ubx.Resource(networkfirewall.VpcEndpointAssociation, "example", networkfirewall.VpcEndpointAssociationConfig{
    			FirewallArn:   "arn:aws:iam::123456789012:policy/example",
    			SubnetMapping: map[string]any{"subnet_id": "example"},
    			VpcId:         "example",
    			Description:   "Managed by ubx.",
    			Tags:          []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.networkfirewall import VpcEndpointAssociation, VpcEndpointAssociationConfig
    from ubx.aws.networkfirewall.vpc_endpoint_association import VpcEndpointAssociation_SubnetMapping, VpcEndpointAssociation_Tags

    def describe():
        ubx.intent("example own vpc endpoint association")

        ubx.resource(VpcEndpointAssociation, "example", VpcEndpointAssociationConfig(
            firewall_arn="arn:aws:iam::123456789012:policy/example",
            subnet_mapping=VpcEndpointAssociation_SubnetMapping(subnet_id="example"),
            vpc_id="example",
            description="Managed by ubx.",
            tags=[VpcEndpointAssociation_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
