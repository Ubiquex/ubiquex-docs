`aws_work_spaces_web_user_access_logging_settings` configures real logging of user access events, sign-ins and session starts, for an `aws_work_spaces_web_portal`, delivered to CloudWatch Logs.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	workspacesweb "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/workspacesweb"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own user access logging settings"})

    		ubx.Resource(workspacesweb.UserAccessLoggingSettings, "example", workspacesweb.UserAccessLoggingSettingsConfig{
    			KinesisStreamArn: "arn:aws:iam::123456789012:policy/example",
    			Tags:             []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
