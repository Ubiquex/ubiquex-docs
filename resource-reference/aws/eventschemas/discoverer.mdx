`aws_event_schemas_discoverer` automatically infers real event schemas from an `aws_events_event_bus`'s own live traffic and adds them to a registry, so a caller doesn't have to write every event schema by hand from a sample payload.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	eventschemas "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/eventschemas"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own discoverer"})

    		ubx.Resource(eventschemas.Discoverer, "example", eventschemas.DiscovererConfig{
    			SourceArn:    "arn:aws:iam::123456789012:policy/example",
    			CrossAccount: true,
    			Description:  "Managed by ubx.",
    			Tags:         []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.eventschemas import Discoverer, DiscovererConfig
    from ubx.aws.eventschemas.discoverer import Discoverer_Tags

    def describe():
        ubx.intent("example own discoverer")

        ubx.resource(Discoverer, "example", DiscovererConfig(
            source_arn="arn:aws:iam::123456789012:policy/example",
            cross_account=True,
            description="Managed by ubx.",
            tags=[Discoverer_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="cross_account" type="bool">
  Defines whether event schemas from other accounts are discovered. Default is True. Optional.
</ResponseField>

<ResponseField name="description" type="string">
  A description for the discoverer. Optional.
</ResponseField>

<ResponseField name="source_arn" type="string" required>
  The ARN of the event bus. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags associated with the resource. Optional.

  </Expandable>
</ResponseField>
