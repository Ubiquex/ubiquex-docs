`aws_event_schemas_schema` is one real, versioned schema definition, OpenAPI or JSONSchema, inside an `aws_event_schemas_registry`, describing the actual structure of one type of event so code bindings can be generated against it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	eventschemas "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/eventschemas"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own schema"})

    		ubx.Resource(eventschemas.Schema, "example", eventschemas.SchemaConfig{
    			Content:      "example",
    			RegistryName: "example-registry-name",
    			Type:         "example",
    			Description:  "Managed by ubx.",
    			SchemaName:   "example-schema-name",
    			Tags:         []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.eventschemas import Schema, SchemaConfig
    from ubx.aws.eventschemas.schema import Schema_Tags

    def describe():
        ubx.intent("example own schema")

        ubx.resource(Schema, "example", SchemaConfig(
            content="example",
            registry_name="example-registry-name",
            type="example",
            description="Managed by ubx.",
            schema_name="example-schema-name",
            tags=[Schema_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="content" type="string" required>
  The source of the schema definition. Required.
</ResponseField>

<ResponseField name="description" type="string">
  A description of the schema. Optional.
</ResponseField>

<ResponseField name="registry_name" type="string" required>
  The name of the schema registry. Required.
</ResponseField>

<ResponseField name="schema_name" type="string">
  The name of the schema. Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags associated with the resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key of a user-defined tag attached to an EventBridge Schema, used to organize and categorize the schema within AWS. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a tag key attached to the EventBridge Schema, used to store arbitrary metadata for the schema resource. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="type" type="string" required>
  The type of schema. Valid types include OpenApi3 and JSONSchemaDraft4. Required.
</ResponseField>
