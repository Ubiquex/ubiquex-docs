`aws_network_manager_global_network` creates a real top-level container that groups an organization's `aws_network_manager_device`s, `aws_network_manager_site`s, and `aws_network_manager_link`s into a single network Network Manager can visualize and monitor.

## Example

    import (
    	networkmanager "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/networkmanager"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own global network"})

    		ubx.Resource(networkmanager.GlobalNetwork, "example", networkmanager.GlobalNetworkConfig{
    			CreatedAt:   "example",
    			Description: "Managed by ubx.",
    			State:       "example",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
