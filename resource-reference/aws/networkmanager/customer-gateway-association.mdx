`aws_network_manager_customer_gateway_association` associates a real `aws_customer_gateway` with a device and site in an `aws_network_manager_global_network`, so Network Manager can represent that on-premises VPN endpoint in its own topology.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	networkmanager "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/networkmanager"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own customer gateway association"})

    		ubx.Resource(networkmanager.CustomerGatewayAssociation, "example", networkmanager.CustomerGatewayAssociationConfig{
    			CustomerGatewayArn: "arn:aws:iam::123456789012:policy/example",
    			DeviceId:           "example",
    			GlobalNetworkId:    "example",
    			LinkId:             "example",
    		})
    	}))
    }
    ```
  </Tab>
