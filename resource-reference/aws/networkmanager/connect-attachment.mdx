`aws_network_manager_connect_attachment` creates a real GRE- or no-encapsulation-based Connect attachment on an `aws_network_manager_core_network`, providing the transport that `aws_network_manager_connect_peer`s establish BGP sessions over.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	networkmanager "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/networkmanager"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own manager connect attachment"})

    		ubx.Resource(networkmanager.ConnectAttachment, "example", networkmanager.ConnectAttachmentConfig{
    			CoreNetworkId:                      "example",
    			EdgeLocation:                       "example",
    			Options:                            map[string]any{"protocol": "example"},
    			TransportAttachmentId:              "example",
    			NetworkFunctionGroupName:           "example-networkmanager-function-group-name",
    			ProposedNetworkFunctionGroupChange: map[string]any{"attachment_policy_rule_number": 1},
    			ProposedSegmentChange:              map[string]any{"attachment_policy_rule_number": 1},
    			RoutingPolicyLabel:                 "example",
    		})
    	}))
    }
    ```

## Input properties

<ResponseField name="core_network_id" type="string" required>
  ID of the CoreNetwork that the attachment will be attached to. Required.
</ResponseField>

<ResponseField name="edge_location" type="string" required>
  Edge location of the attachment. Required.
</ResponseField>

<ResponseField name="network_function_group_name" type="string">
  The name of the network function group attachment. Optional.
</ResponseField>

<ResponseField name="options" type="object" required>
  Connect attachment options for protocol Required.
  <Expandable title="properties">
    <ResponseField name="protocol" type="string">
      Tunnel protocol for connect attachment Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="proposed_network_function_group_change" type="object">
  The attachment to move from one network function group to another. Optional.
  <Expandable title="properties">
    <ResponseField name="attachment_policy_rule_number" type="number">
      The rule number in the policy document that applies to this change. Optional.
    </ResponseField>
    <ResponseField name="network_function_group_name" type="string">
      The name of the network function group to change. Optional.
    </ResponseField>
    <ResponseField name="tags" type="list(object)">
      The key-value tags that changed for the network function group. Optional.
      <Expandable title="properties">
        <ResponseField name="key" type="string">
          The key of a tag applied to the proposed network function group change for the AWS Network Manager connect attachment. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="value" type="string">
          The value of a tag key-value pair in the list of tags to attach to the network function group when the proposed network function group change, defined on the AWS Network Manager Connect Attachment, is applied. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="proposed_segment_change" type="object">
  The attachment to move from one segment to another. Optional.
  <Expandable title="properties">
    <ResponseField name="attachment_policy_rule_number" type="number">
      The rule number in the policy document that applies to this change. Optional.
    </ResponseField>
    <ResponseField name="segment_name" type="string">
      The name of the segment to change. Optional.
    </ResponseField>
    <ResponseField name="tags" type="list(object)">
      The list of key-value tags that changed for the segment. Optional.
      <Expandable title="properties">
        <ResponseField name="key" type="string">
          The key of a tag in the list of tags associated with the proposed segment change for the Network Manager connect attachment, used to specify metadata for the segment association. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="value" type="string">
          The value of a key-value tag that will be applied to the Connect Attachment when the proposed segment change is executed. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="routing_policy_label" type="string">
  Routing policy label Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags for the attachment. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      The value component of a user-defined tag attached to the AWS Network Manager Connect Attachment, used for organizing and identifying the attachment through metadata. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="transport_attachment_id" type="string" required>
  Id of transport attachment Required.
</ResponseField>
