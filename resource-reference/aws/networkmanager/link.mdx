`aws_network_manager_link` defines a real network connection, its bandwidth and provider, between two `aws_network_manager_site`s within an `aws_network_manager_global_network`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	networkmanager "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/networkmanager"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own manager link"})

    		ubx.Resource(networkmanager.Link, "example", networkmanager.LinkConfig{
    			Bandwidth:       map[string]any{"download_speed": 1},
    			GlobalNetworkId: "example",
    			SiteId:          "example",
    			Description:     "Managed by ubx.",
    			Provider:        "example",
    			Tags:            []map[string]any{{"key": "example"}},
    			Type:            "example",
    		})
    	}))
    }
    ```
