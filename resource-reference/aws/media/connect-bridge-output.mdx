`aws_media_connect_bridge_output` defines a real destination that an `aws_media_connect_bridge` sends its content to, the bridge's own counterpart to a flow's `aws_media_connect_flow_output`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	media "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/media"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own connect bridge output"})

    		ubx.Resource(media.ConnectBridgeOutput, "ci-runner", media.ConnectBridgeOutputConfig{
    			BridgeArn:     "arn:aws:iam::123456789012:policy/ci",
    			Name:          "ci-runner",
    			NetworkOutput: map[string]any{"ip_address": "example"},
    		})
    	}))
    }
    ```
  </Tab>

## Properties

<ResponseField name="bridge_arn" type="string" required>
  The Amazon Resource Number (ARN) of the bridge. Required.
</ResponseField>

<ResponseField name="name" type="string" required>
  The network output name. Required.
</ResponseField>

<ResponseField name="network_output" type="object" required>
  The output of the bridge. A network output is delivered to your premises. Required.
  <Expandable title="properties">
    <ResponseField name="ip_address" type="string" required>
      The network output IP Address. Required.
    </ResponseField>
    <ResponseField name="network_name" type="string" required>
      The network output's gateway network name. Required.
    </ResponseField>
    <ResponseField name="port" type="number" required>
      The network output port. Required.
    </ResponseField>
    <ResponseField name="protocol" type="string" required>
      The network output protocol. Required.
    </ResponseField>
    <ResponseField name="ttl" type="number" required>
      The network output TTL. Required.
    </ResponseField>
  </Expandable>
</ResponseField>
