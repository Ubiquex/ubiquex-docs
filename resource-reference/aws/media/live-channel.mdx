`aws_media_live_channel` creates a real live video encoding channel, MediaLive's core resource, that takes one or more `aws_media_live_input`s and encodes them into one or more outputs.

## Example

    ```go
    	media "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/media"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own live channel"})

    		ubx.Resource(media.LiveChannel, "example", media.LiveChannelConfig{
    			Name:                  "example",
    			AnywhereSettings:      map[string]any{"channel_placement_group_id": "example"},
    			CdiInputSpecification: map[string]any{"resolution": "example"},
    			ChannelClass:          "example",
    			ChannelEngineVersion:  map[string]any{"version": "example"},
    			ChannelSecurityGroups: []string{"example"},
    			Destinations:          []map[string]any{{"id": "example"}},
    			DryRun:                true,
    		})
    	}))
    }
    ```
