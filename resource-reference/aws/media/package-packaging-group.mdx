`aws_media_package_packaging_group` creates a real top-level container for `aws_media_package_packaging_configuration`s, the v1 MediaPackage VOD counterpart to a live channel.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	mediapackage "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/mediapackage"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own packaging group"})

    		ubx.Resource(mediapackage.PackagingGroup, "example", mediapackage.PackagingGroupConfig{
    			Id:               "example",
    			Authorization:    map[string]any{"cdn_identifier_secret": "example", "secrets_role_arn": "example"},
    			EgressAccessLogs: map[string]any{"log_group_name": "example-log-group-name"},
    			Tags:             []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.mediapackage import PackagingGroup, PackagingGroupConfig
    from ubx.aws.mediapackage.packaging_group import PackagingGroup_Authorization, PackagingGroup_EgressAccessLogs, PackagingGroup_Tags

    def describe():
        ubx.intent("example own packaging group")

        ubx.resource(PackagingGroup, "example", PackagingGroupConfig(
            id="example",
            authorization=PackagingGroup_Authorization(cdn_identifier_secret="example", secrets_role_arn="example"),
            egress_access_logs=PackagingGroup_EgressAccessLogs(log_group_name="example-log-group-name"),
            tags=[PackagingGroup_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="authorization" type="object">
  Defines the CDN authorization configuration for the packaging group, specifying the identifier secret and the IAM role that MediaPackage uses to access AWS Secrets Manager for rotating credentials. Optional. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="cdn_identifier_secret" type="string" required>
      The Amazon Resource Name (ARN) for the secret in AWS Secrets Manager that is used for CDN authorization. Required.
    </ResponseField>
    <ResponseField name="secrets_role_arn" type="string" required>
      The Amazon Resource Name (ARN) for the IAM role that allows MediaPackage to communicate with AWS Secrets Manager. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="egress_access_logs" type="object">
  Specifies the CloudWatch Logs log group where egress access logs for this packaging group are sent. Optional. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="log_group_name" type="string">
      Sets a custom AWS CloudWatch log group name for egress logs. If a log group name isn't specified, the default name is used: /aws/MediaPackage/VodEgressAccessLogs. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="id" type="string" required>
  The ID of the PackagingGroup. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  A collection of tags associated with a resource Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      The value portion of a key-value tag assigned to the MediaPackage packaging group, used for resource metadata, organization, and cost allocation. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  The ARN of the PackagingGroup. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="domain_name" type="string">
  The fully qualified domain name for Assets in the PackagingGroup. Computed by AWS after `ship`.
</ResponseField>
