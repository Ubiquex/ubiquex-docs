`aws_media_package_v2_channel_policy` attaches a real resource-based access policy to an `aws_media_package_v2_channel`, controlling which principals can ingest into or manage that channel.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	mediapackagev2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/mediapackagev2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own package v2 channel policy"})

    		ubx.Resource(mediapackagev2.ChannelPolicy, "example", mediapackagev2.ChannelPolicyConfig{
    			ChannelGroupName: "example-channel-group-name",
    			ChannelName:      "example-channel-name",
    			Policy:           "example",
    		})
    	}))
    }
    ```
