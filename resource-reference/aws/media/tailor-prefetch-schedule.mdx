`aws_media_tailor_prefetch_schedule` tells AWS Elemental MediaTailor to fetch and prepare ad content ahead of a specific ad break (or, for a recurring schedule, every ad avail in a live event), reducing the latency of ad insertion at actual playback time.

## Input properties

<ResponseField name="consumption" type="object">
  Optional.
  <Expandable title="properties">
    <ResponseField name="avail_matching_criteria" type="list(object)">
      If you only want MediaTailor to insert prefetched ads into avails that match specific dynamic variables, set the avail matching criteria. Optional.
      <Expandable title="properties">
        <ResponseField name="dynamic_variable" type="string">
          Optional; if omitted, computed by AWS.
        </ResponseField>
        <ResponseField name="operator" type="string">
          Optional; if omitted, computed by AWS.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="end_time" type="string" required>
      The time when MediaTailor no longer considers the prefetched ads for use in an ad break, as an ISO 8601 date-time. Required.
    </ResponseField>
    <ResponseField name="start_time" type="string">
      The time when prefetched ads are considered for use in an ad break, as an ISO 8601 date-time. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="name" type="string" required>
  The name to assign to the prefetch schedule. Required.
</ResponseField>

<ResponseField name="playback_configuration_name" type="string" required>
  The name of the playback configuration. Required.
</ResponseField>

<ResponseField name="recurring_prefetch_configuration" type="object">
  Optional.
  <Expandable title="properties">
    <ResponseField name="end_time" type="string" required>
      The end time for the window that MediaTailor prefetches and inserts ads in a live event, as an ISO 8601 date-time. Required.
    </ResponseField>
    <ResponseField name="recurring_consumption" type="object" required>
      Required.
      <Expandable title="properties">
        <ResponseField name="avail_matching_criteria" type="list(object)">
          The configuration for the dynamic variables that determine which ad breaks that MediaTailor inserts prefetched ads in. Optional.
          <Expandable title="properties">
            <ResponseField name="dynamic_variable" type="string">
              Optional; if omitted, computed by AWS.
            </ResponseField>
            <ResponseField name="operator" type="string">
              Optional; if omitted, computed by AWS.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="retrieved_ad_expiration_seconds" type="number">
          The number of seconds that an ad is available for insertion after it was prefetched. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="recurring_retrieval" type="object" required>
      Required.
      <Expandable title="properties">
        <ResponseField name="delay_after_avail_end_seconds" type="number">
          The number of seconds that MediaTailor waits after an ad avail before prefetching ads for the next avail. Optional.
        </ResponseField>
        <ResponseField name="dynamic_variables" type="any">
          The dynamic variables to use for substitution during prefetch requests to the ADS. Optional.
        </ResponseField>
        <ResponseField name="traffic_shaping_retrieval_window" type="object">
          Optional.
          <Expandable title="properties">
            <ResponseField name="retrieval_window_duration_seconds" type="number">
              The amount of time, in seconds, that MediaTailor spreads prefetch requests to the ADS. Optional.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="traffic_shaping_tps_configuration" type="object">
          Optional.
          <Expandable title="properties">
            <ResponseField name="peak_concurrent_users" type="number">
              The expected peak number of concurrent viewers for your content. Optional.
            </ResponseField>
            <ResponseField name="peak_tps" type="number">
              The maximum number of transactions per second (TPS) that your ad decision server (ADS) can handle. Optional.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="traffic_shaping_type" type="string">
          Indicates the type of traffic shaping used to limit the number of requests to the ADS at one time. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="start_time" type="string">
      The start time for the window that MediaTailor prefetches and inserts ads in a live event, as an ISO 8601 date-time. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="retrieval" type="object">
  Optional.

    <ResponseField name="dynamic_variables" type="any">

    </ResponseField>
    <ResponseField name="traffic_shaping_tps_configuration" type="object">
      Optional.

          The expected peak number of concurrent viewers for your content. Optional.
        </ResponseField>
        <ResponseField name="peak_tps" type="number">

        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="traffic_shaping_type" type="string">
      Indicates the type of traffic shaping used to limit the number of requests to the ADS at one time. Optional.
    </ResponseField>

</ResponseField>

  The frequency that MediaTailor creates prefetch schedules. Optional.
</ResponseField>

<ResponseField name="stream_id" type="string">
  An optional stream identifier that MediaTailor uses to prefetch ads for multiple streams that use the same playback configuration. Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  The tags assigned to the prefetch schedule. Optional.
  <Expandable title="properties">
