`aws_vpc_lattice_service` creates a real, logical application service in VPC Lattice, with its own DNS name and `aws_vpc_lattice_listener`s, that consumers reach through an `aws_vpc_lattice_service_network` without VPC peering or Transit Gateway routes.

## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    import (
    	vpc "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/vpc"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own lattice service"})

    		ubx.Resource(vpc.LatticeService, "ci-runner", vpc.LatticeServiceConfig{
    			Name:               "ci-runner",
    			AuthType:           "example",
    			CertificateArn:     "arn:aws:iam::123456789012:policy/ci",
    			CustomDomainName:   "example-custom-domain-name",
    			IdleTimeoutSeconds: 7200,
    			Tags:               []map[string]any{{"key": "example"}},
    		})
    	}))
    }
