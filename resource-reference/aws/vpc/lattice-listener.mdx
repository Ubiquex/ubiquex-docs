`aws_vpc_lattice_listener` defines a real port and protocol, HTTP, HTTPS, or gRPC, that an `aws_vpc_lattice_service` accepts requests on, routing them according to its `aws_vpc_lattice_rule`s.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    			ServiceIdentifier: "example",

    	}))
