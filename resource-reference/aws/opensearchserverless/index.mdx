## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	opensearchserverless "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/opensearchserverless"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own index"})

    		ubx.Resource(opensearchserverless.Index, "example", opensearchserverless.IndexConfig{
    			CollectionEndpoint: "example",
    			IndexName:          "example-index-name",
    			Mappings:           map[string]any{"properties": "example"},
    			Settings:           map[string]any{"analysis": map[string]any{"analyzer": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.opensearchserverless import Index, IndexConfig
    from ubx.aws.opensearchserverless.index import Index_Mappings, Index_Settings, Index_Settings_Analysis

    def describe():
        ubx.intent("example own index")

        ubx.resource(Index, "example", IndexConfig(
            collection_endpoint="example",
            index_name="example-index-name",
            mappings=Index_Mappings(properties="example"),
            settings=Index_Settings(analysis=Index_Settings_Analysis(analyzer="example")),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="collection_endpoint" type="string" required>
  The endpoint for the collection. Required.
</ResponseField>

<ResponseField name="index_name" type="string" required>
  The name of the OpenSearch Serverless index. Required.
</ResponseField>

<ResponseField name="mappings" type="object">
  Index Mappings Optional.
  <Expandable title="properties">
    <ResponseField name="properties" type="any">
      Defines the fields within the mapping, including their types and configurations Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="settings" type="object">
  The settings object configures index-level settings such as the number of primary shards and replica shards for the OpenSearch Serverless index. Optional. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="analysis" type="object">
      Custom analysis configuration including analyzers, tokenizers, and filters Optional.
      <Expandable title="properties">
        <ResponseField name="analyzer" type="any">
          Custom analyzer definitions Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="index" type="object">
      The 'settings.index' object specifies index-level configuration for the OpenSearch Serverless index, including settings like the number of primary shards and replica shards. Optional. **(AI-inferred)**
      <Expandable title="properties">
        <ResponseField name="knn" type="bool">
          Enable/disable k-nearest neighbor search capability Optional.
        </ResponseField>
        <ResponseField name="knn_algo_param_ef_search" type="number">
          Size of the dynamic list for the nearest neighbors Optional.
        </ResponseField>
        <ResponseField name="refresh_interval" type="string">
          How often to perform refresh operation (e.g. '1s', '5s') Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="uuid" type="string">
  The unique identifier for the index. Computed by AWS after `ship`.
</ResponseField>
