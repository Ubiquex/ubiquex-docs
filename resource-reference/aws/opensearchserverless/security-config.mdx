`aws_open_search_serverless_security_config` defines a real SAML or federated authentication configuration for OpenSearch Serverless, letting users sign in to an `aws_open_search_serverless_collection` with an external identity provider rather than only IAM credentials.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	opensearchserverless "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/opensearchserverless"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own security config"})

    		ubx.Resource(opensearchserverless.SecurityConfig, "example", opensearchserverless.SecurityConfigConfig{
    			Name:                 "example",
    			Description:          "Managed by ubx.",
    			IamFederationOptions: map[string]any{"group_attribute": "example"},
    			SamlOptions:          map[string]any{"metadata": "example"},
    			Type:                 "example",
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.opensearchserverless import SecurityConfig, SecurityConfigConfig
    from ubx.aws.opensearchserverless.security_config import SecurityConfig_IamFederationOptions, SecurityConfig_SamlOptions

    def describe():
        ubx.intent("example own security config")

        ubx.resource(SecurityConfig, "example", SecurityConfigConfig(
            name="example",
            description="Managed by ubx.",
            iam_federation_options=SecurityConfig_IamFederationOptions(group_attribute="example"),
            saml_options=SecurityConfig_SamlOptions(metadata="example"),
            type="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="description" type="string">
  Security config description Optional.
</ResponseField>

<ResponseField name="iam_federation_options" type="object">
  Describe IAM federation options in form of key value map Optional.
  <Expandable title="properties">
    <ResponseField name="group_attribute" type="string">
      Group attribute for this IAM federation integration Optional.
    </ResponseField>
    <ResponseField name="user_attribute" type="string">
      User attribute for this IAM federation integration Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="name" type="string">
  The friendly name of the security config Optional.
</ResponseField>

<ResponseField name="saml_options" type="object">
  Describes saml options in form of key value map Optional.
  <Expandable title="properties">
    <ResponseField name="group_attribute" type="string">
      Group attribute for this saml integration Optional.
    </ResponseField>
    <ResponseField name="metadata" type="string" required>
      The XML saml provider metadata document that you want to use Required.
    </ResponseField>
    <ResponseField name="open_search_serverless_entity_id" type="string">
      Custom entity id attribute to override default entity id for this saml integration Optional.
    </ResponseField>
    <ResponseField name="session_timeout" type="number">
      Defines the session timeout in minutes Optional.
    </ResponseField>
    <ResponseField name="user_attribute" type="string">
      Custom attribute for this saml integration Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="type" type="string">
  Config type for security config Optional.
</ResponseField>

## Output properties

<ResponseField name="iam_identity_center_options" type="object">

  <Expandable title="properties">

    <ResponseField name="application_description" type="string">

    <ResponseField name="application_name" type="string">
      The name of the IAM Identity Center application used to integrate with OpenSearch Serverless Optional.
    </ResponseField>
    <ResponseField name="group_attribute" type="string">
      Group attribute for this IAM Identity Center integration Optional.
    </ResponseField>
    <ResponseField name="instance_arn" type="string" required>
      The ARN of the IAM Identity Center instance used to integrate with OpenSearch Serverless Required.

      User attribute for this IAM Identity Center integration Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="id" type="string">
  The identifier of the security config Computed by AWS after `ship`.
</ResponseField>
