`aws_open_search_serverless_access_policy` defines real data-access rules, which principals can read or write which indexes, for an `aws_open_search_serverless_collection`, OpenSearch Serverless's own fine-grained access-control layer distinct from broader IAM permissions.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	"encoding/json"

    	opensearchserverless "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/opensearchserverless"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own search serverless access policy"})

    		accessPolicy, _ := json.Marshal(map[string]any{
    			"Version": "2012-10-17",
    			"Statement": []map[string]any{{
    				"Effect":   "Allow",
    				"Action":   "*",
    				"Resource": "*",
    			}},
    		})

    		ubx.Resource(opensearchserverless.AccessPolicy, "example", opensearchserverless.AccessPolicyConfig{
    			Name:        "example",
    			Policy:      string(accessPolicy),
    			Type:        "example",
    			Description: "Managed by ubx.",
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
