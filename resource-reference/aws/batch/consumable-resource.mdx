`aws_batch_consumable_resource` defines a real constraint beyond vCPU, GPU, or memory, a third-party license token pool, a database connection limit, an API rate cap, that AWS Batch's resource-aware scheduling checks before starting a job, so a job doesn't launch, consume compute time, and then fail discovering the real resource it needed was unavailable.

    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	batch "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/batch"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own consumable resource"})

    		ubx.Resource(batch.ConsumableResource, "ci-runner", batch.ConsumableResourceConfig{
    			ResourceType:           "example",
    			TotalQuantity:          1,
    			ConsumableResourceName: "example-consumable-resource-name",
    			Tags:                   "example",
    		})
    	}))
    }
    ```
