`aws_game_lift_script` uploads real server-side logic for GameLift Realtime Servers, a lighter-weight alternative to `aws_game_lift_build`'s full custom executable for game logic that fits GameLift's own managed network layer.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	gamelift "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/gamelift"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own script"})

    		ubx.Resource(gamelift.Script, "example", gamelift.ScriptConfig{
    			StorageLocation: map[string]any{"bucket": "example", "key": "example", "role_arn": "example"},
    			Name:            "example",
    			NodeJsVersion:   "example",
    			Tags:            []map[string]any{{"key": "example"}},
    			Version:         "example",
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.gamelift import Script, ScriptConfig
    from ubx.aws.gamelift.script import Script_StorageLocation, Script_Tags

    def describe():
        ubx.intent("example own script")

        ubx.resource(Script, "example", ScriptConfig(
            storage_location=Script_StorageLocation(bucket="example", key="example", role_arn="example"),
            name="example",
            node_js_version="example",
            tags=[Script_Tags(key="example")],
            version="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="name" type="string">
  A descriptive label that is associated with a script. Script names do not need to be unique. Optional.
</ResponseField>

<ResponseField name="node_js_version" type="string">
  The Node.js version used for execution of the Realtime script. Optional.
</ResponseField>

<ResponseField name="storage_location" type="object" required>
  The Amazon S3 bucket and key that store the zipped script file, along with the IAM role that allows GameLift to access that location. Required. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="bucket" type="string" required>
      An Amazon S3 bucket identifier. This is the name of the S3 bucket. Required.
    </ResponseField>
    <ResponseField name="key" type="string" required>
      The name of the zip file that contains the script files. Required.
    </ResponseField>
    <ResponseField name="object_version" type="string">
      The version of the file, if object versioning is turned on for the bucket. Amazon GameLift uses this information when retrieving files from your S3 bucket. To retrieve a specific version of the file, provide an object version. To retrieve the latest version of the file, do not set this parameter. Optional.
    </ResponseField>
    <ResponseField name="role_arn" type="string" required>
      The Amazon Resource Name (ARN) for an IAM role that allows Amazon GameLift to access the S3 bucket. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An array of key-value pairs to apply to this resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key of a tag attached to the GameLift script, used to identify and categorize the resource. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The string value paired with a tag key on the GameLift script, used to label and categorize the resource for cost allocation and identity-based policies. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="version" type="string">
  The version that is associated with a script. Version strings do not need to be unique. Optional.
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  The Amazon Resource Name (ARN) that is assigned to a Amazon GameLift script resource and uniquely identifies it. ARNs are unique across all Regions. In a GameLift script ARN, the resource ID matches the Id value. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="creation_time" type="string">
  A time stamp indicating when this data object was created. Format is a number expressed in Unix time as milliseconds (for example "1469498468.057"). Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="id" type="string">
  A unique identifier for the Realtime script Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="size_on_disk" type="number">
  The file size of the uploaded Realtime script, expressed in bytes. When files are uploaded from an S3 location, this value remains at "0". Computed by AWS after `ship`.
</ResponseField>
