`aws_quick_sight_agent` creates a real, purpose-built AI agent within Amazon Quick, backed by an `aws_quick_sight_action_connector` for taking real actions and a knowledge source for grounding, the entity an end user or an embedded chat experience actually converses with.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	quicksight "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/quicksight"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own sight agent"})

    		ubx.Resource(quicksight.Agent, "example", quicksight.AgentConfig{
    			AgentId:           "example",
    			AwsAccountId:      "example",
    			Name:              "example",
    			ActionConnectors:  []string{"example"},
    			AgentLifecycle:    "example",
    			CustomPromptInput: map[string]any{"existing_prompt": map[string]any{"model_profile_id": "example"}},
    			Description:       "Managed by ubx.",
    			IconId:            "example",
    		})
    	}))
    }
    ```

## Input properties

<ResponseField name="action_connectors" type="list(string)">
  A list of ActionConnector ARNs (max 10) attached to the agent. Optional.
</ResponseField>

<ResponseField name="agent_id" type="string" required>
  The unique identifier for the agent. Required.
</ResponseField>

<ResponseField name="agent_lifecycle" type="string">
  The lifecycle stage of the agent. PREVIEW or PUBLISHED. Optional.
</ResponseField>

<ResponseField name="aws_account_id" type="string" required>
  The ID of the Amazon Web Services account where the agent is being created. Required.
</ResponseField>

<ResponseField name="custom_prompt_input" type="object">
  Custom prompt configuration. Specify either ExistingPrompt or NewPrompt. Optional.
  <Expandable title="properties">
    <ResponseField name="existing_prompt" type="object">
      Reference to an existing custom prompt profile. Optional.
      <Expandable title="properties">
        <ResponseField name="model_profile_id" type="string" required>
          The identifier of the model profile. Required.
        </ResponseField>
        <ResponseField name="qbs_aws_account_id" type="string" required>
          The QBS AWS account identifier. Required.
        </ResponseField>
        <ResponseField name="subscription_id" type="string" required>
          The subscription identifier. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="new_prompt" type="object">
      Parameters for creating a new custom prompt configuration. Optional.
      <Expandable title="properties">
        <ResponseField name="custom_instructions" type="string">
          Custom instructions for the agent behavior. Optional.
        </ResponseField>
        <ResponseField name="identity" type="string">
          The identity or persona of the agent. Optional.
        </ResponseField>
        <ResponseField name="output_style" type="string">
          The output style for the agent responses. Optional.
        </ResponseField>
        <ResponseField name="response_length" type="string">
          The desired response length for the agent. Optional.
        </ResponseField>
        <ResponseField name="tone" type="string">
          The tone used in agent responses. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="description" type="string">
  A description of the agent. Optional.
</ResponseField>

<ResponseField name="icon_id" type="string">
  The icon identifier for the agent. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  The display name of the agent. Required.
</ResponseField>

<ResponseField name="spaces" type="list(string)">
  A list of Space ARNs (max 10) attached to the agent. Optional.
</ResponseField>

<ResponseField name="starter_prompts" type="list(string)">
  A list of up to 3 starter prompts displayed to users. Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  A list of key-value pairs to associate with the agent resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="welcome_message" type="string">
  The welcome message displayed when a user opens the agent. Optional.
</ResponseField>

## Output properties

<ResponseField name="agent_status" type="string">
  The current status of the agent. One of ACTIVE, CREATING, UPDATING, or FAILED. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="arn" type="string">
  The Amazon Resource Name (ARN) of the agent. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="created_at" type="string">
  The date and time the agent was created. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="creator" type="string">
  The ARN of the user who created the agent. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="custom_prompt_interface" type="object">
  Read-only view of the resolved custom prompt interface for the agent. Computed by AWS after `ship`.
  <Expandable title="properties">
    <ResponseField name="custom_instructions" type="string">
      Custom instructions for the agent behavior. Optional.
    </ResponseField>
    <ResponseField name="identity" type="string">
      The identity or persona of the agent. Optional.
    </ResponseField>
    <ResponseField name="model_profile_id" type="string">
      The identifier of the model profile. Optional.
    </ResponseField>
    <ResponseField name="output_style" type="string">
      The output style for the agent responses. Optional.
    </ResponseField>
    <ResponseField name="prompt_summary" type="string">
      A summary of the resolved prompt. Optional.
    </ResponseField>
    <ResponseField name="qbs_aws_account_id" type="string">
      The QBS AWS account identifier. Optional.
    </ResponseField>
    <ResponseField name="response_length" type="string">
      The desired response length for the agent. Optional.
    </ResponseField>
    <ResponseField name="subscription_id" type="string">
      The subscription identifier. Optional.
    </ResponseField>
    <ResponseField name="tone" type="string">
      The tone used in agent responses. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="error_message" type="string">
  The error message if the agent is in FAILED status. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="updated_at" type="string">
  The date and time the agent was last updated. Computed by AWS after `ship`.
</ResponseField>
