`aws_quick_sight_limits_profile` defines a real set of usage limits, embedding session counts, API rate limits, that can be applied to control consumption of QuickSight/Amazon Quick resources within an account or by a specific group of users.

## Example

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own sight limits profile"})

    		ubx.Resource(quick.SightLimitsProfile, "ci-runner", quick.SightLimitsProfileConfig{
    			AccountId:      "example",
    			ProfileName:    "example-profile-name",
    			ResourceLimits: "example",
    			Description:    "Managed by ubx.",
    		})
    	}))
    }
