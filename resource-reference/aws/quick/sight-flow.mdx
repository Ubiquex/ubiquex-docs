`aws_quick_sight_flow` defines a real Quick Flow, an automated, multi-step workflow within Amazon Quick, part of the AI-workspace capabilities QuickSight evolved into after its October 2025 rename to Amazon Quick Suite (shortened to Amazon Quick in 2026).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	quick "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/quick"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own sight flow"})

    		ubx.Resource(quick.SightFlow, "example", quick.SightFlowConfig{
    			AwsAccountId:   "example",
    			FlowDefinition: "example",
    			Name:           "example",
    			Description:    "Managed by ubx.",
    			Permissions:    []map[string]any{{"actions": []string{"example"}}},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.quick import SightFlow, SightFlowConfig

    def describe():
        ubx.intent("example own sight flow")

        ubx.resource(SightFlow, "example", SightFlowConfig(
            aws_account_id="example",
            flow_definition="example",
            name="example",
            description="Managed by ubx.",
            permissions=[{"actions": ["example"]}],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
