`aws_quick_sight_knowledge_base` connects a real content source, SharePoint, OneDrive, and similar, into Amazon Quick's Quick Index component, so its agents can ground answers in that organization's own documents.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	quick "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/quick"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own sight knowledge base"})

    		ubx.Resource(quick.SightKnowledgeBase, "example", quick.SightKnowledgeBaseConfig{
    			AwsAccountId:               "example",
    			DataSourceArn:              "arn:aws:iam::123456789012:policy/example",
    			KnowledgeBaseConfiguration: map[string]any{"template_configuration": map[string]any{"template": "example"}},
    			KnowledgeBaseId:            "example",
    			Name:                       "example",
    			AccessControlConfiguration: map[string]any{"is_aclenabled": true},
    			Description:                "Managed by ubx.",
    			IsEmailNotificationOptedForIngestionFailures: true,
    		})
    	}))
    }
    ```
