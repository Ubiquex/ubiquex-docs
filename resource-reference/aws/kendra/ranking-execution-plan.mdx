`aws_kendra_ranking_execution_plan` provisions Kendra Intelligent Ranking, a real, standalone semantic re-ranking API a caller applies to results from their own separate search system, OpenSearch or Elasticsearch, for instance, independent of using a full `aws_kendra_index`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	kendra "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/kendra"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own ranking execution plan"})

    		ubx.Resource(kendra.RankingExecutionPlan, "example", kendra.RankingExecutionPlanConfig{
    			Name:          "example",
    			CapacityUnits: map[string]any{"rescore_capacity_units": 1},
    			Description:   "Managed by ubx.",
    			Tags:          []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
