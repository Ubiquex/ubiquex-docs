`aws_kendra_query_suggestions_block_list` blocks specific real terms from appearing in an `aws_kendra_index`'s own autocomplete query suggestions, without removing the underlying documents that contain them from search results entirely.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	kendra "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/kendra"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own query suggestions block list"})

    		ubx.Resource(kendra.QuerySuggestionsBlockList, "example", kendra.QuerySuggestionsBlockListConfig{
    			IndexId:      "example",
    			Name:         "example",
    			RoleArn:      "arn:aws:iam::123456789012:policy/example",
    			SourceS3Path: map[string]any{"bucket": "example"},
    			Description:  "Managed by ubx.",
    			Tags:         []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

## Input properties

</ResponseField>

<ResponseField name="index_id" type="string" required>

</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the block list. Required.
</ResponseField>

<ResponseField name="role_arn" type="string" required>
  The Amazon Resource Name (ARN) of an IAM role with permission to access the S3 bucket that contains the block list text file. Required.
</ResponseField>
