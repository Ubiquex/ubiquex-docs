`aws_kendra_index` is the real, core Amazon Kendra search index, built from one or more `aws_kendra_data_source`s, that a caller's own natural-language search queries actually run against.

## Example

<Tabs>

    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    import (
    	kendra "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/kendra"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own index"})

    		ubx.Resource(kendra.Index, "example", kendra.IndexConfig{
    			Edition:                           "example",
    			Name:                              "example",
    			RoleArn:                           "arn:aws:iam::123456789012:policy/example",
    			CapacityUnits:                     map[string]any{"query_capacity_units": 1},
    			Description:                       "Managed by ubx.",
    			DocumentMetadataConfigurations:    []map[string]any{{"name": "example-name"}},
    			ServerSideEncryptionConfiguration: map[string]any{"kms_key_id": "example"},
    			Tags:                              []map[string]any{{"key": "example"}},
    		})
    	}))
    ```
