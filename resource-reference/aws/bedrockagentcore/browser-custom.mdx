`aws_bedrock_agent_core_browser_custom` configures AgentCore's Browser tool with the caller's own network and access settings, giving an agent a real, isolated environment to interact with live websites at scale, beyond AgentCore's own default browser configuration.

## Example

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own browser custom"})

    import ubx_sdk as ubx
    from ubx.aws.bedrockagentcore import BrowserCustom, BrowserCustomConfig
    from ubx.aws.bedrockagentcore.browser_custom import BrowserCustom_NetworkConfiguration, BrowserCustom_BrowserSigning, BrowserCustom_Certificates, BrowserCustom_Certificates_CertificateLocation, BrowserCustom_EnterprisePolicies, BrowserCustom_EnterprisePolicies_Location, BrowserCustom_RecordingConfig

    def describe():
        ubx.intent("example own browser custom")

        ubx.resource(BrowserCustom, "example", BrowserCustomConfig(
            name="example",
            network_configuration=BrowserCustom_NetworkConfiguration(network_mode="example"),
            browser_signing=BrowserCustom_BrowserSigning(enabled=True),
            certificates=[BrowserCustom_Certificates(certificate_location=BrowserCustom_Certificates_CertificateLocation(secret_arn="example"))],
            description="Managed by ubx.",
            enterprise_policies=[BrowserCustom_EnterprisePolicies(location=BrowserCustom_EnterprisePolicies_Location(bucket="example"))],
            execution_role_arn="arn:aws:iam::123456789012:policy/example",
            recording_config=BrowserCustom_RecordingConfig(enabled=True),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
