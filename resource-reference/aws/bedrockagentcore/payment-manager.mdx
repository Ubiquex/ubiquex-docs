`aws_bedrock_agent_core_payment_manager` is the top-level AgentCore Payments resource for an AWS account, provisioning a real workload identity in AgentCore Identity and coordinating every payment operation run through it.

## Example

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own payment manager"})

    		ubx.Resource(bedrockagentcore.PaymentManager, "example", bedrockagentcore.PaymentManagerConfig{
    			AuthorizerType:          "example",
    			Name:                    "example",
    			RoleArn:                 "arn:aws:iam::123456789012:policy/example",

    			Tags:                    []map[string]any{{"key": "example"}},
    		})
    	}))
    }

    import ubx_sdk as ubx

    from ubx.aws.bedrockagentcore.payment_manager import PaymentManager_AuthorizerConfiguration, PaymentManager_AuthorizerConfiguration_CustomJwtauthorizer, PaymentManager_Tags
