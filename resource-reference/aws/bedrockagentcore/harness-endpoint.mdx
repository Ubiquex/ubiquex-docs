`aws_bedrock_agent_core_harness_endpoint` exposes a real, invokable endpoint for an `aws_bedrock_agent_core_harness`, AgentCore's fully managed agent runtime (general availability June 17, 2026) that provides the orchestration loop around a model, instructions, and tools so an agent can be called directly via InvokeHarnessCommand.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	bedrockagentcore "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/bedrockagentcore"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own harness endpoint"})

    		ubx.Resource(bedrockagentcore.HarnessEndpoint, "example", bedrockagentcore.HarnessEndpointConfig{
    			EndpointName:  "example-endpoint-name",
    			HarnessId:     "example",
    			Description:   "Managed by ubx.",
    			Tags:          []map[string]any{{"key": "example"}},
    			TargetVersion: "example",
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.bedrockagentcore import HarnessEndpoint, HarnessEndpointConfig
    from ubx.aws.bedrockagentcore.harness_endpoint import HarnessEndpoint_Tags

    def describe():
        ubx.intent("example own harness endpoint")

        ubx.resource(HarnessEndpoint, "example", HarnessEndpointConfig(
            endpoint_name="example-endpoint-name",
            harness_id="example",
            description="Managed by ubx.",
            tags=[HarnessEndpoint_Tags(key="example")],
            target_version="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="description" type="string">
  The description of the endpoint. Optional.
</ResponseField>

<ResponseField name="endpoint_name" type="string" required>
  The name of the endpoint. Must start with a letter and contain only alphanumeric characters and underscores. Required.
</ResponseField>

<ResponseField name="harness_id" type="string" required>
  The ID of the harness that the endpoint belongs to. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags to apply to the harness endpoint resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="target_version" type="string">
  The harness version that the endpoint points to and serves invocations from. Optional.
</ResponseField>
