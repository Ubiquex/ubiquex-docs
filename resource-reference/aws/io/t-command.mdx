`aws_io_t_command` defines a real, reusable, named remote command with its own custom payload schema, that AWS IoT Device Management Commands sends to a targeted device and tracks execution status for, without a caller managing MQTT topics, payload formats, or status tracking by hand. Reached general availability November 2024.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own t command"})

    		ubx.Resource(iot.Command, "example", iot.CommandConfig{
    			CommandId:           "example",
    			CreatedAt:           "example",
    			Deprecated:          true,
    			Description:         "Managed by ubx.",
    			DisplayName:         "example-display-name",
    			LastUpdatedAt:       "example",
    			MandatoryParameters: []map[string]any{{"name": "example-name"}},
    			Namespace:           "example",
    		})
    	}))
    }
    ```
