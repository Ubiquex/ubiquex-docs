`aws_io_tsite_wise_access_policy` grants a real IAM user, group, or IAM Identity Center identity permission to view or administer a specific `aws_io_tsite_wise_portal`, SiteWise Monitor's own access-control mechanism separate from IAM policies on the underlying SiteWise API.

    }

## Input properties

<ResponseField name="access_policy_identity" type="object" required>
  The identity for this access policy. Choose either an SSO user or group or an IAM user or role. Required.
  <Expandable title="properties">
    <ResponseField name="iam_role" type="object">
      Contains information for an IAM role identity in an access policy. Optional.
      <Expandable title="properties">
        <ResponseField name="arn" type="string">
          The ARN of the IAM role. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="iam_user" type="object">
      Contains information for an IAM user identity in an access policy. Optional.
      <Expandable title="properties">
        <ResponseField name="arn" type="string">
          The ARN of the IAM user. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="user" type="object">
      Contains information for a user identity in an access policy. Optional.
      <Expandable title="properties">
        <ResponseField name="id" type="string">
          The AWS SSO ID of the user. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="access_policy_permission" type="string" required>
  The permission level for this access policy. Valid values are ADMINISTRATOR or VIEWER. Required.
</ResponseField>

<ResponseField name="access_policy_resource" type="object" required>
  The AWS IoT SiteWise Monitor resource for this access policy. Choose either portal or project but not both. Required.
  <Expandable title="properties">
    <ResponseField name="portal" type="object">
      A portal resource. Optional.
      <Expandable title="properties">
        <ResponseField name="id" type="string">
          The ID of the portal. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="project" type="object">
      A project resource. Optional.
      <Expandable title="properties">
        <ResponseField name="id" type="string">
          The ID of the project. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
