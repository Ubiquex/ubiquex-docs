`aws_io_twireless_wireless_gateway` registers a real LoRaWAN gateway, the physical radio hardware that receives transmissions from an `aws_io_twireless_wireless_device` and forwards them to AWS IoT Wireless.

## Example

    import (
    	iotwireless "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/iotwireless"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own wireless gateway"})

    		ubx.Resource(iotwireless.WirelessGateway, "example", iotwireless.WirelessGatewayConfig{

    			Name:                 "example",
    			Description:          "Managed by ubx.",
    			LastUplinkReceivedAt: "example",
    			Tags:                 []map[string]any{{"key": "example"}},
    			ThingArn:             "arn:aws:iam::123456789012:policy/example",
    			ThingName:            "example-thing-name",
    		})
    	}))
    }

    from ubx.aws.iotwireless.wireless_gateway import WirelessGateway_LoRaWan, WirelessGateway_Tags

    def describe():

## Input properties

<ResponseField name="description" type="string">
  Description of Wireless Gateway. Optional.
</ResponseField>

<ResponseField name="last_uplink_received_at" type="string">
  The date and time when the most recent uplink was received. Optional.
</ResponseField>

<ResponseField name="lo_ra_wan" type="object" required>
  The LoRaWAN configuration for the wireless gateway, specifying the gateway EUI and service profile ID. Required. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="gateway_eui" type="string" required>
      The LoRaWAN gateway EUI is the unique 16-character hexadecimal identifier for the gateway, required to register it with AWS IoT Wireless. Required. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="rf_region" type="string" required>
      The LoRaWAN RF region for the gateway, which specifies the frequency band (e.g., US915, EU868) used for communication. Required. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="name" type="string">
  Name of Wireless Gateway. Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  A list of key-value pairs that contain metadata for the gateway. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key of a tag attached to an AWS IoT wireless gateway, used to organize, identify, and filter the gateway resource for management and cost tracking. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a tag attached to the AWS IoT Wireless Gateway, used for arbitrary metadata, identification, and cost allocation. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="thing_arn" type="string">
  Thing Arn. Passed into Update to associate a Thing with the Wireless Gateway. Optional.
</ResponseField>

<ResponseField name="thing_name" type="string">
  Thing Name. If there is a Thing created, this can be returned with a Get call. Optional.
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  Arn for Wireless Gateway. Returned upon successful create. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="id" type="string">
  Id for Wireless Gateway. Returned upon successful create. Computed by AWS after `ship`.
</ResponseField>
