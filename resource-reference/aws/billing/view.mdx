`aws_billing_view` defines a segment of AWS cost and usage data, either the account's own default primary view or a custom, filtered view built and shared, via AWS Resource Access Manager, with other accounts so they see only the cost data the view's filter actually exposes.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	billing "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/billing"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own view"})

    		ubx.Resource(billing.View, "example", billing.ViewConfig{
    			Name:                 "example",
    			SourceViews:          []string{"example"},
    			DataFilterExpression: map[string]any{"dimensions": map[string]any{"key": "example"}},
    			Description:          "Managed by ubx.",
    			Tags:                 []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
