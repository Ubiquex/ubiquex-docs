`aws_cloud_formation_module_default_version` designates which registered `aws_cloud_formation_module_version` CloudFormation uses by default whenever a template references that module without pinning to a specific version.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudformation "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/cloudformation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own formation module default version"})

    		ubx.Resource(cloudformation.ModuleDefaultVersion, "example", cloudformation.ModuleDefaultVersionConfig{
    			Arn:        "arn:aws:iam::123456789012:policy/example",
    			ModuleName: "example-module-name",
    			VersionId:  "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript

    import { ModuleDefaultVersion } from "@ubx/sdk-aws/aws/cloudformation/module_default_version";
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.cloudformation.module_default_version import ModuleDefaultVersion, ModuleDefaultVersionConfig

    def describe():
        ubx.intent("example own formation module default version")

        ubx.resource(ModuleDefaultVersion, "example", ModuleDefaultVersionConfig(
            arn="arn:aws:iam::123456789012:policy/example",
            module_name="example-module-name",
            version_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="arn" type="string">
  The Amazon Resource Name (ARN) of the module version to set as the default version. Optional.
</ResponseField>

<ResponseField name="module_name" type="string">
  The name of a module existing in the registry. Optional.
</ResponseField>

<ResponseField name="version_id" type="string">
  The ID of an existing version of the named module to set as the default. Optional.
</ResponseField>
