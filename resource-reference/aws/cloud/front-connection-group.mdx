`aws_cloud_front_connection_group` creates a real grouping of connections for a CloudFront distribution, part of CloudFront's multi-tenant distribution model, that lets multiple distributions or tenants share a common set of network-level connection settings rather than each provisioning its own.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudfront "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/cloudfront"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own front connection group"})

    		ubx.Resource(cloudfront.ConnectionGroup, "example", cloudfront.ConnectionGroupConfig{
    			Name:            "example",
    			AnycastIpListId: "example",
    			Enabled:         true,
    			Ipv6Enabled:     true,
    			Tags:            []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.cloudfront.connection_group import ConnectionGroup, ConnectionGroupConfig

    def describe():
        ubx.intent("example own front connection group")

        ubx.resource(ConnectionGroup, "example", ConnectionGroupConfig(
            name="example",
            anycast_ip_list_id="example",
            enabled=True,
            ipv6_enabled=True,
            tags=[{"key": "example"}],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="anycast_ip_list_id" type="string">
  The ID of the Anycast static IP list. Optional.
</ResponseField>

<ResponseField name="enabled" type="bool">
  Whether the connection group is enabled. Optional.
</ResponseField>

<ResponseField name="ipv6_enabled" type="bool">
  IPv6 is enabled for the connection group. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the connection group. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  A complex type that contains zero or more ``Tag`` elements. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>
