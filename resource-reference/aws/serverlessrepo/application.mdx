`aws_serverless_repo_application` publishes a real SAM application to the AWS Serverless Application Repository, packaging its template and code so other accounts can discover, review, and deploy it directly from the repository rather than copying source by hand.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	serverlessrepo "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/serverlessrepo"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own repo application"})

    		ubx.Resource(serverlessrepo.Application, "example", serverlessrepo.ApplicationConfig{
    			Author:          "example",
    			Description:     "Managed by ubx.",
    			Name:            "example",
    			HomePageUrl:     "example",
    			Labels:          []string{"example"},
    			LicenseBody:     "example",
    			ReadmeBody:      "example",
    			SemanticVersion: "example",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.serverlessrepo.application import Application, ApplicationConfig

    def describe():
        ubx.intent("example own repo application")

        ubx.resource(Application, "example", ApplicationConfig(
            author="example",
            description="Managed by ubx.",
            name="example",
            home_page_url="example",
            labels=["example"],
            license_body="example",
            readme_body="example",
            semantic_version="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="author" type="string" required>
  The name of the author publishing the app. Required.
</ResponseField>

<ResponseField name="description" type="string" required>
  The description of the application. Required.
</ResponseField>

<ResponseField name="home_page_url" type="string">
  A URL with more information about the application. Optional.
</ResponseField>

<ResponseField name="labels" type="list(string)">
  Labels to improve discovery of apps in search results. Optional.
</ResponseField>

<ResponseField name="license_body" type="string">
  A local text file that contains the license of the app. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the application. Required.
</ResponseField>

<ResponseField name="readme_body" type="string">
  A text readme file in Markdown language that contains a more detailed description of the application. Optional.
</ResponseField>

<ResponseField name="semantic_version" type="string">
  The semantic version of the application. Optional.
</ResponseField>

<ResponseField name="source_code_url" type="string">
  A link to a public repository for the source code of your application. Optional.
</ResponseField>

<ResponseField name="spdx_license_id" type="string">
  A valid identifier from https://spdx.org/licenses/. Optional.
</ResponseField>

<ResponseField name="template_body" type="string">
  The local raw packaged AWS SAM template file of your application. Optional.
</ResponseField>

## Output properties

<ResponseField name="application_id" type="string">
  The Amazon Resource Name (ARN) of the application. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="creation_time" type="string">
  The date and time this resource was created. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="is_verified_author" type="bool">
  Whether the author of this application has been verified. Computed by AWS after `ship`.
</ResponseField>
