`aws_athena_prepared_statement` saves a real, parameterized SQL statement under an `aws_athena_work_group`, so the same query shape can be executed repeatedly with different real parameter values substituted in.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	athena "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/athena"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own prepared statement"})

    		ubx.Resource(athena.PreparedStatement, "example", athena.PreparedStatementConfig{
    			QueryStatement: "example",
    			StatementName:  "example-statement-name",
    			WorkGroup:      "example",
    			Description:    "Managed by ubx.",
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    import ubx_sdk as ubx
    from ubx.aws.athena.prepared_statement import PreparedStatement, PreparedStatementConfig

    def describe():
        ubx.intent("example own prepared statement")

        ubx.resource(PreparedStatement, "example", PreparedStatementConfig(
            query_statement="example",
            statement_name="example-statement-name",
            work_group="example",
            description="Managed by ubx.",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="description" type="string">
  The description of the prepared statement. Optional.
</ResponseField>

<ResponseField name="query_statement" type="string" required>
  The query string for the prepared statement. Required.
</ResponseField>

<ResponseField name="statement_name" type="string" required>
  The name of the prepared statement. Required.
</ResponseField>

<ResponseField name="work_group" type="string" required>
  The name of the workgroup to which the prepared statement belongs. Required.
</ResponseField>
