`aws_athena_named_query` saves a real, named SQL query against an `aws_athena_work_group`, for reuse from the console or API instead of re-typing it for every ad hoc analysis.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	athena "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/athena"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own named query"})

    		ubx.Resource(athena.NamedQuery, "example", athena.NamedQueryConfig{
    			Database:    "example",
    			QueryString: "example",
    			Name:        "example",
    			Description: "Managed by ubx.",
    			WorkGroup:   "example",
    		})
    	}))
    }
    ```
  </Tab>

    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH

    from ubx.aws.athena import NamedQuery, NamedQueryConfig

    def describe():
        ubx.intent("example own named query")

        ubx.resource(NamedQuery, "example", NamedQueryConfig(
            database="example",
            query_string="example",
            name="example",
            description="Managed by ubx.",
            work_group="example",

    if __name__ == "__main__":

## Input properties

<ResponseField name="database" type="string" required>
  The database to which the query belongs. Required.
</ResponseField>

<ResponseField name="description" type="string">
  The query description. Optional.
</ResponseField>

<ResponseField name="name" type="string">
  The query name. Optional.
</ResponseField>

<ResponseField name="query_string" type="string" required>
  The contents of the query with all query statements. Required.
</ResponseField>

<ResponseField name="work_group" type="string">
  The name of the workgroup that contains the named query. Optional.
</ResponseField>

## Output properties

<ResponseField name="named_query_id" type="string">
  The unique ID of the query. Computed by AWS after `ship`.
</ResponseField>
