`aws_bedrock_mantle_project` creates a real project under Amazon Bedrock's Project Mantle, a distributed inference engine that serves the latest open-weight models through an OpenAI-compatible endpoint (bedrock-mantle) alongside Bedrock's own native bedrock-runtime API. This is a genuinely new capability, so behavior here is verified live rather than assumed from general knowledge.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	bedrockmantle "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/bedrockmantle"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own project"})

    		ubx.Resource(bedrockmantle.Project, "example", bedrockmantle.ProjectConfig{
    			Name: "example",
    			Tags: []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.aws.bedrockmantle import Project, ProjectConfig
    from ubx.aws.bedrockmantle.project import Project_Tags

    def describe():
        ubx.intent("example own project")

        ubx.resource(Project, "example", ProjectConfig(
            name="example",
            tags=[Project_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

</ResponseField>
