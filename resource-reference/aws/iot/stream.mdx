`aws_io_t_stream` delivers a real, large file, a firmware image, in ordered chunks over MQTT to a device running an `aws_io_t_job`, AWS IoT's own mechanism for file delivery to devices that can't make a normal HTTPS download request.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iot "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/iot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own stream"})

    		ubx.Resource(iot.Stream, "example", iot.StreamConfig{
    			Files:       []map[string]any{{"file_id": 1}},
    			RoleArn:     "arn:aws:iam::123456789012:policy/example",
    			StreamId:    "example",
    			Description: "Managed by ubx.",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.iot import Stream, StreamConfig
    from ubx.aws.iot.stream import Stream_Files, Stream_Tags

    def describe():
        ubx.intent("example own stream")

        ubx.resource(Stream, "example", StreamConfig(
            files=[Stream_Files(file_id=1)],
            role_arn="arn:aws:iam::123456789012:policy/example",
            stream_id="example",
            description="Managed by ubx.",
            tags=[Stream_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="description" type="string">
  The description of the stream. Optional.
</ResponseField>

<ResponseField name="files" type="list(object)" required>
  The files to stream. Required.
  <Expandable title="properties">
    <ResponseField name="file_id" type="number">
      The unique numeric identifier for this file within the AWS IoT Stream, used by devices to reference and load the specific file from the stream. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="s3_location" type="object">
      Specifies the Amazon S3 location (bucket, key, and optional version) of a file to be added to the AWS IoT stream, used to retrieve the file content for the stream. Optional; if omitted, computed by AWS. **(AI-inferred)**
      <Expandable title="properties">
        <ResponseField name="bucket" type="string">
          The name of the S3 bucket that contains the file referenced by this IoT stream file entry. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="key" type="string">
          Specifies the S3 object key (path) of the file to be added to the AWS IoT stream, locating the object within the bucket defined by the same s3_location block. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="version" type="string">
          The version ID of the S3 object that contains the file to be streamed. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="role_arn" type="string" required>
  An IAM role that allows the IoT service principal to access your S3 files. Required.
</ResponseField>

<ResponseField name="stream_id" type="string" required>
  The stream ID. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Metadata which can be used to manage streams. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>
