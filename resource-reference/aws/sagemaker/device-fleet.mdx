`aws_sage_maker_device_fleet` creates a real logical grouping of `aws_sage_maker_device`s that edge-deployed models are managed and monitored across together.

## Example

    package main

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own device fleet"})

    		ubx.Resource(sagemaker.DeviceFleet, "example", sagemaker.DeviceFleetConfig{
    			DeviceFleetName: "example-device-fleet-name",
    			OutputConfig:    map[string]any{"s3_output_location": "example"},
    			RoleArn:         "arn:aws:iam::123456789012:policy/example",
    			Description:     "Managed by ubx.",
    			Tags:            []map[string]any{{"key": "example"}},
    		})
    	}))
    }

    import ubx_sdk as ubx
    from ubx.aws.sagemaker import DeviceFleet, DeviceFleetConfig
