`aws_sage_maker_image` registers a real, reusable container image, and its `aws_sage_maker_app_image_config` variants, that can be selected as the kernel environment for an `aws_sage_maker_app` running in SageMaker Studio.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	sagemaker "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/sagemaker"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own maker image"})

    		ubx.Resource(sagemaker.Image, "example", sagemaker.ImageConfig{
    			ImageName:        "example-image-name",
    			ImageRoleArn:     "arn:aws:iam::123456789012:policy/example",
    			ImageDescription: "example",
    			ImageDisplayName: "example-image-display-name",
    			Tags:             []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.sagemaker.image import Image, ImageConfig

    def describe():
        ubx.intent("example own maker image")

        ubx.resource(Image, "example", ImageConfig(
            image_name="example-image-name",
            image_role_arn="arn:aws:iam::123456789012:policy/example",
            image_description="example",
            image_display_name="example-image-display-name",
            tags=[{"key": "example"}],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="image_description" type="string">
  A description of the image. Optional.
</ResponseField>

<ResponseField name="image_display_name" type="string">
  The display name of the image. Optional.
</ResponseField>

<ResponseField name="image_name" type="string" required>
  The name of the image. Required.
</ResponseField>

<ResponseField name="image_role_arn" type="string" required>
  The Amazon Resource Name (ARN) of an IAM role that enables Amazon SageMaker to perform tasks on behalf of the customer. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An array of key-value pairs to apply to this resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">

  </Expandable>
</ResponseField>

## Output properties
