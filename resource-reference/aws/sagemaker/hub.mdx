`aws_sage_maker_hub` creates a real private SageMaker JumpStart hub, a curated, organization-scoped catalog of foundation models and pretrained models that mirrors JumpStart's own public model hub but under an account's own governance and access controls.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	sagemaker "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/sagemaker"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own hub"})

    		ubx.Resource(sagemaker.Hub, "example", sagemaker.HubConfig{
    			HubDescription:    "example",
    			HubName:           "example-hub-name",
    			HubDisplayName:    "example-hub-display-name",
    			HubSearchKeywords: []string{"example"},
    			S3StorageConfig:   map[string]any{"s3_output_path": "example"},
    			Tags:              []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.sagemaker import Hub, HubConfig
    from ubx.aws.sagemaker.hub import Hub_S3StorageConfig, Hub_Tags

    def describe():
        ubx.intent("example own hub")

        ubx.resource(Hub, "example", HubConfig(
            hub_description="example",
            hub_name="example-hub-name",
            hub_display_name="example-hub-display-name",
            hub_search_keywords=["example"],
            s3_storage_config=Hub_S3StorageConfig(s3_output_path="example"),
            tags=[Hub_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="hub_description" type="string" required>
  A description of the hub. Required.
</ResponseField>

<ResponseField name="hub_display_name" type="string">
  The display name of the hub. Optional.
</ResponseField>

<ResponseField name="hub_name" type="string" required>
  The name of the hub. Required.
</ResponseField>

<ResponseField name="hub_search_keywords" type="list(string)">
  The searchable keywords for the hub. Optional.
</ResponseField>

<ResponseField name="s3_storage_config" type="object">
  The Amazon S3 storage configuration for the hub. Optional.
  <Expandable title="properties">
    <ResponseField name="s3_output_path" type="string">
      The Amazon S3 bucket prefix for hosting hub content. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags to associate with the hub. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key of a tag attached to the SageMaker Hub, used to identify and categorize the hub for resource management and cost allocation. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value component of a tag attached to the SageMaker Hub, which pairs with the corresponding key to organize, identify, and manage the hub resource within your AWS environment. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>
