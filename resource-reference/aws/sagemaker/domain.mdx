`aws_sage_maker_domain` creates a real SageMaker Studio domain, the top-level container, its VPC, authentication mode, and default settings, that `aws_sage_maker_user_profile`s, `aws_sage_maker_space`s, and their `aws_sage_maker_app`s all live inside.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own domain"})

    			AppSecurityGroupManagement: "example",
    			DefaultSpaceSettings:       map[string]any{"execution_role": "example"},
    			DomainSettings:             map[string]any{"docker_settings": map[string]any{"enable_docker_access": "example"}},
    			HomeEfsFileSystemCreation:  "example",
    		})
    	}))
    }

    import ubx_sdk as ubx
