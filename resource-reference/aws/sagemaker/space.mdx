`aws_sage_maker_space` creates a real, shared or private workspace within an `aws_sage_maker_domain` that an `aws_sage_maker_app` runs inside, an alternative to a per-user `aws_sage_maker_user_profile` for collaborative or shared Studio environments.

## Example

    ```go
    import (
    	sagemaker "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/sagemaker"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own maker space"})

    		ubx.Resource(sagemaker.Space, "example", sagemaker.SpaceConfig{
    			DomainId:             "example",
    			SpaceName:            "example-space-name",
    			OwnershipSettings:    map[string]any{"owner_user_profile_name": "example-owner-user-profile-name"},
    			SpaceDisplayName:     "example-space-display-name",
    			SpaceSettings:        map[string]any{"app_type": "example"},
    			SpaceSharingSettings: map[string]any{"sharing_type": "example"},
    			Tags:                 []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
