`aws_refactor_spaces_service` registers a real extracted microservice, and the routing rule that sends its share of an `aws_refactor_spaces_application`'s traffic to it, the mechanism Refactor Spaces uses to migrate traffic incrementally rather than all at once. Like the rest of Refactor Spaces, this closed to new customers November 7, 2025.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	refactorspaces "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/refactorspaces"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own spaces service"})

    		ubx.Resource(refactorspaces.Service, "example", refactorspaces.ServiceConfig{
    			ApplicationIdentifier: "example",
    			EndpointType:          "example",
    			EnvironmentIdentifier: "example",
    			Name:                  "example",
    			Description:           "Managed by ubx.",
    			LambdaEndpoint:        map[string]any{"arn": "example"},
    			Tags:                  []map[string]any{{"key": "example"}},
    			UrlEndpoint:           map[string]any{"url": "example"},
    		})
    	}))
    }
    ```
  </Tab>
