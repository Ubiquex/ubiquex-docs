`aws_notifications_notification_hub` designates a real AWS Region where AWS User Notifications stores, processes, and replicates an account's notification data, an account-level setting rather than a per-notification resource.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	notifications "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/notifications"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own notification hub"})

    		ubx.Resource(notifications.NotificationHub, "ci-runner", notifications.NotificationHubConfig{
    			Region: "example",
    		})
    	}))
    }
    ```
  </Tab>
