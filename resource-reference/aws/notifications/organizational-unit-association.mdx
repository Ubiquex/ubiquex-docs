`aws_notifications_organizational_unit_association` applies a real `aws_notifications_notification_configuration` across every account in an AWS Organizations organizational unit at once, rather than configuring User Notifications account by account.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	notifications "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/notifications"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own organizational unit association"})

    		ubx.Resource(notifications.OrganizationalUnitAssociation, "example", notifications.OrganizationalUnitAssociationConfig{
    			NotificationConfigurationArn: "arn:aws:iam::123456789012:policy/example",
    			OrganizationalUnitId:         "example",
    		})
    	}))
    }
    ```
  </Tab>
