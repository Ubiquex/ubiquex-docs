## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	opensearchserverless "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/opensearchserverless"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own search serverless collection index"})

    		ubx.Resource(opensearchserverless.CollectionIndex, "example", opensearchserverless.CollectionIndexConfig{
    			Id:          "example",
    			IndexName:   "example-index-name",
    			IndexSchema: "example",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.opensearchserverless.collection_index import CollectionIndex, CollectionIndexConfig

            id="example",
            index_name="example-index-name",
            index_schema="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
