`aws_open_search_service_application` creates a real OpenSearch Service Application, a unified analytics workspace that spans multiple `aws_open_search_service_domain`s and OpenSearch Serverless collections behind one access point, distinct from a domain's own role as a single managed cluster.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	open "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/open"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own search service application"})

    		ubx.Resource(open.SearchServiceApplication, "example", open.SearchServiceApplicationConfig{
    			Name:                     "example",
    			AppConfigs:               []map[string]any{{"key": "example"}},
    			DataSources:              []map[string]any{{"data_source_arn": "example"}},
    			Endpoint:                 "example",
    			IamIdentityCenterOptions: map[string]any{"enabled": true},
    			KmsKeyArn:                "arn:aws:iam::123456789012:policy/example",
    			Tags:                     []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

## Input properties

<ResponseField name="app_configs" type="list(object)">
  List of application configurations. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key name that identifies a specific configuration parameter within the app_configs list of an AWS OpenSearch Service application. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a single key-value pair within the AppConfigs list, which supplies custom settings for the OpenSearch Service application when it is provisioned. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="data_sources" type="list(object)">
  List of data sources. Optional.
  <Expandable title="properties">
    <ResponseField name="data_source_arn" type="any">
      Specifies the Amazon Resource Name (ARN) of an underlying AWS data source (for example, an S3 bucket or CloudWatch Logs log group) that this OpenSearch Service application uses for direct querying. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="data_source_description" type="string">
      Specifies a descriptive text for a data source that is associated with the Amazon OpenSearch Service application. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="endpoint" type="string">
  The endpoint for the application. Optional.
</ResponseField>

<ResponseField name="iam_identity_center_options" type="object">
  Options for configuring IAM Identity Center Optional.
  <Expandable title="properties">
    <ResponseField name="enabled" type="bool">
      Whether IAM Identity Center is enabled. Optional.
    </ResponseField>
    <ResponseField name="iam_identity_center_instance_arn" type="any">
      The ARN of the AWS IAM Identity Center instance used to authenticate users for this OpenSearch Service application. Optional. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="iam_role_for_identity_center_application_arn" type="string">
      The ARN of the IAM role for Identity Center application. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="kms_key_arn" type="string">
  The ARN of the KMS key used to encrypt the application. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the application. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An arbitrary set of tags (key-value pairs) for this application. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a user-defined tag key assigned to the OpenSearch Service application, used for metadata management, cost allocation, and access control. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  Amazon Resource Name (ARN) format. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="id" type="string">
  The identifier of the application. Computed by AWS after `ship`.
</ResponseField>
