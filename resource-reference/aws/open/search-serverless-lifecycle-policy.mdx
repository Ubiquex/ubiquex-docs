`aws_open_search_serverless_lifecycle_policy` defines a real retention rule for an `aws_open_search_serverless_collection`'s time-series indexes, automatically deleting indexes older than a configured age rather than requiring manual cleanup.

## Example

    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    import (

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own search serverless lifecycle policy"})

    		accessPolicy, _ := json.Marshal(map[string]any{
    			"Version": "2012-10-17",
    			"Statement": []map[string]any{{
    				"Effect":   "Allow",
    				"Action":   "*",
    				"Resource": "*",
    			}},
    		})

    		ubx.Resource(open.SearchServerlessLifecyclePolicy, "ci-runner", open.SearchServerlessLifecyclePolicyConfig{
    			Name:        "ci-runner",
    			Policy:      string(accessPolicy),
    			Type:        "example",
    			Description: "Managed by ubx.",
    		})
    	}))
    }
