`aws_entity_resolution_schema_mapping` defines which real input fields from an AWS Glue table an `aws_entity_resolution_matching_workflow` actually uses for matching, either a caller's own custom schema or one brought directly from an existing Glue input.

## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	entityresolution "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/entityresolution"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own schema mapping"})

    		ubx.Resource(entityresolution.SchemaMapping, "example", entityresolution.SchemaMappingConfig{
    			MappedInputFields: []map[string]any{{"field_name": "example-field-name"}},
    			SchemaName:        "example-schema-name",
    			Description:       "Managed by ubx.",
    			Tags:              []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.entityresolution import SchemaMapping, SchemaMappingConfig
    from ubx.aws.entityresolution.schema_mapping import SchemaMapping_MappedInputFields, SchemaMapping_Tags

    def describe():
        ubx.intent("example own schema mapping")

        ubx.resource(SchemaMapping, "example", SchemaMappingConfig(
            mapped_input_fields=[SchemaMapping_MappedInputFields(field_name="example-field-name")],
            schema_name="example-schema-name",
            description="Managed by ubx.",
            tags=[SchemaMapping_Tags(key="example")],
        ))

    if __name__ == "__main__":
