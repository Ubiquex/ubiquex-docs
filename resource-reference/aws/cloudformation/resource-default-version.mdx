`aws_cloud_formation_resource_default_version` designates which registered `aws_cloud_formation_resource_version` CloudFormation uses by default whenever a template references that custom resource type without pinning to a specific version.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudformation "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/cloudformation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resource default version"})

    		ubx.Resource(cloudformation.ResourceDefaultVersion, "example", cloudformation.ResourceDefaultVersionConfig{
    			TypeName:       "example-type-name",
    			TypeVersionArn: "arn:aws:iam::123456789012:policy/example",
    			VersionId:      "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { ResourceDefaultVersion } from "@ubx/sdk-aws/aws/cloudformation/resource_default_version";

    export default stack("example", () => {
      intent({ summary: "example own resource default version" });

      resource(ResourceDefaultVersion, "example", {
        typeName: "example-type-name",
        typeVersionArn: "arn:aws:iam::123456789012:policy/example",
        versionId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.cloudformation import ResourceDefaultVersion, ResourceDefaultVersionConfig

    def describe():
        ubx.intent("example own resource default version")

        ubx.resource(ResourceDefaultVersion, "example", ResourceDefaultVersionConfig(
            type_name="example-type-name",
            type_version_arn="arn:aws:iam::123456789012:policy/example",
            version_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="type_name" type="string">
  The name of the type being registered. We recommend that type names adhere to the following pattern: company_or_organization::service::type. Optional.
</ResponseField>

<ResponseField name="type_version_arn" type="string">
  The Amazon Resource Name (ARN) of the type version. Optional.
</ResponseField>

<ResponseField name="version_id" type="string">
  The ID of an existing version of the resource to set as the default. Optional.
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  The Amazon Resource Name (ARN) of the type. This is used to uniquely identify a ResourceDefaultVersion Computed by AWS after `ship`.
</ResponseField>
