`aws_cloud_formation_generated_template` produces a real CloudFormation template generated FROM a set of existing, already-deployed resources, CloudFormation's own IaC-generator feature for bringing unmanaged infrastructure under an `aws_cloud_formation_stack` without hand-writing the template first.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudformation "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/cloudformation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own formation generated template"})

    		ubx.Resource(cloudformation.GeneratedTemplate, "example", cloudformation.GeneratedTemplateConfig{
    			GeneratedTemplateName: "example-generated-template-name",
    			TemplateConfiguration: map[string]any{"deletion_policy": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { GeneratedTemplate } from "@ubx/sdk-aws/aws/cloudformation/generated_template";

    export default stack("example", () => {
      intent({ summary: "example own formation generated template" });

      resource(GeneratedTemplate, "example", {
        generatedTemplateName: "example-generated-template-name",
        templateConfiguration: { deletionPolicy: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.cloudformation.generated_template import GeneratedTemplate, GeneratedTemplateConfig

    def describe():
        ubx.intent("example own formation generated template")

        ubx.resource(GeneratedTemplate, "example", GeneratedTemplateConfig(
            generated_template_name="example-generated-template-name",
            template_configuration={"deletion_policy": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="generated_template_name" type="string" required>
  The name assigned to the generated template. Required.
</ResponseField>

<ResponseField name="template_configuration" type="object">
  The configuration details of the generated template. Optional.
  <Expandable title="properties">
    <ResponseField name="deletion_policy" type="string">
      The DeletionPolicy assigned to resources in the generated template. Optional.
    </ResponseField>
    <ResponseField name="update_replace_policy" type="string">
      The UpdateReplacePolicy assigned to resources in the generated template. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="creation_time" type="string">
  The time the generated template was created. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="generated_template_id" type="string">
  The Amazon Resource Name (ARN) of the generated template. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="last_updated_time" type="string">
  The time the generated template was last updated. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="progress" type="object">
  A summary of the progress of the template generation. Computed by AWS after `ship`.
  <Expandable title="properties">
    <ResponseField name="resources_failed" type="number">
      The number of resources that failed the template generation. Optional.
    </ResponseField>
    <ResponseField name="resources_pending" type="number">
      The number of resources that are still pending the template generation. Optional.
    </ResponseField>
    <ResponseField name="resources_processing" type="number">
      The number of resources that are in-process for the template generation. Optional.
    </ResponseField>
    <ResponseField name="resources_succeeded" type="number">
      The number of resources that succeeded the template generation. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="status" type="string">
  The status of the template generation. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="total_warnings" type="number">
  The number of warnings generated for this template. Computed by AWS after `ship`.
</ResponseField>
