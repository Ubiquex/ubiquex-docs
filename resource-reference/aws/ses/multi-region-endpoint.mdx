`aws_ses_multi_region_endpoint` creates a real SES Global Endpoint, automatically splitting sending traffic active-active across a primary and secondary Region and failing over entirely to the healthy Region if one becomes degraded. SES Global Endpoints launched in December 2024, roughly a year and a half ago.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ses "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/ses"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own multi region endpoint"})

    		ubx.Resource(ses.MultiRegionEndpoint, "example", ses.MultiRegionEndpointConfig{
    			Details:      map[string]any{"route_details": []map[string]any{{"region": "example"}}},
    			EndpointName: "example-endpoint-name",
    			Tags:         []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
