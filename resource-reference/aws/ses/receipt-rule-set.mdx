`aws_ses_receipt_rule_set` creates a real, named, ordered container of `aws_ses_receipt_rule`s under SES's original (pre-Mail-Manager) receiving pipeline; only one rule set can be active for an account at a time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	ses "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/ses"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own receipt rule set"})

    		ubx.Resource(ses.ReceiptRuleSet, "ci-runner", ses.ReceiptRuleSetConfig{
    			RuleSetName: "example-rule-set-name",
    		})
    	}))
    }
    ```
  </Tab>
