`aws_ses_contact_list` creates a real, managed list of email contacts and their topic subscription preferences, that SES's own bulk and marketing-email sending features send to and respect unsubscribes against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ses "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/ses"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own contact list"})

    		ubx.Resource(ses.ContactList, "example", ses.ContactListConfig{
    			ContactListName: "example-contact-list-name",
    			Description:     "Managed by ubx.",
    			Tags:            []map[string]any{{"key": "example"}},
    			Topics:          []map[string]any{{"default_subscription_status": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.ses import ContactList, ContactListConfig

    def describe():
        ubx.intent("example own contact list")

        ubx.resource(ContactList, "example", ContactListConfig(
            contact_list_name="example-contact-list-name",
            description="Managed by ubx.",
            tags=[{"key": "example"}],
            topics=[{"default_subscription_status": "example"}],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
