`aws_ses_mail_manager_archive` creates a real, immutable, searchable archive that an `aws_ses_mail_manager_rule_set` can route copies of email into, retained independently of the mailbox it was originally delivered to, for compliance and e-discovery use cases.

## Example

    ```go
    import (
    	ses "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/ses"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own mail manager archive"})

    		ubx.Resource(ses.MailManagerArchive, "example", ses.MailManagerArchiveConfig{
    			ArchiveName: "example-archive-name",
    			KmsKeyArn:   "arn:aws:iam::123456789012:policy/example",
    			Retention:   map[string]any{"retention_period": "example"},
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
