`aws_lightsail_distribution` creates a real content delivery network distribution, powered by CloudFront under the hood, that caches and accelerates delivery of content hosted on a Lightsail instance, bucket, or load balancer.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	lightsail "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/lightsail"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own distribution"})

    		ubx.Resource(lightsail.Distribution, "example", lightsail.DistributionConfig{
    			BundleId:              "example",
    			DefaultCacheBehavior:  map[string]any{"behavior": "example"},
    			DistributionName:      "example-distribution-name",
    			Origin:                map[string]any{"name": "example-name"},
    			CacheBehaviorSettings: map[string]any{"allowed_httpmethods": "example"},
    			CacheBehaviors:        []map[string]any{{"behavior": "example"}},
    			CertificateName:       "example-certificate-name",
    			IpAddressType:         "example",
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.lightsail import Distribution, DistributionConfig
    from ubx.aws.lightsail.distribution import Distribution_DefaultCacheBehavior, Distribution_Origin, Distribution_CacheBehaviorSettings, Distribution_CacheBehaviors

    def describe():
        ubx.intent("example own distribution")

            bundle_id="example",
            default_cache_behavior=Distribution_DefaultCacheBehavior(behavior="example"),
            distribution_name="example-distribution-name",

## Input properties

<ResponseField name="bundle_id" type="string" required>
  The bundle ID to use for the distribution. Required.
</ResponseField>

<ResponseField name="cache_behavior_settings" type="object">
  Describes the cache settings of an Amazon Lightsail content delivery network (CDN) distribution. Optional.
  <Expandable title="properties">
    <ResponseField name="allowed_httpmethods" type="string">
      The HTTP methods that are processed and forwarded to the distribution's origin. Optional.
    </ResponseField>
    <ResponseField name="cached_httpmethods" type="string">
      The HTTP method responses that are cached by your distribution. Optional.
    </ResponseField>
    <ResponseField name="default_ttl" type="number">
      The default amount of time that objects stay in the distribution's cache before the distribution forwards another request to the origin to determine whether the content has been updated. Optional.
    </ResponseField>
    <ResponseField name="forwarded_cookies" type="object">
      Describes whether an Amazon Lightsail content delivery network (CDN) distribution forwards cookies to the origin and, if so, which ones. Optional.
      <Expandable title="properties">
        <ResponseField name="cookies_allow_list" type="list(string)">
          The specific cookies to forward to your distribution's origin. Optional.
        </ResponseField>
        <ResponseField name="option" type="string">
          Specifies which cookies to forward to the distribution's origin for a cache behavior: all, none, or allow-list to forward only the cookies specified in the cookiesAllowList parameter. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="forwarded_headers" type="object">
      Describes the request headers that a Lightsail distribution bases caching on. Optional.
      <Expandable title="properties">
        <ResponseField name="headers_allow_list" type="list(string)">
          The specific headers to forward to your distribution's origin. Optional.
        </ResponseField>
        <ResponseField name="option" type="string">
          The headers that you want your distribution to forward to your origin and base caching on. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="forwarded_query_strings" type="object">
      Describes the query string parameters that an Amazon Lightsail content delivery network (CDN) distribution to bases caching on. Optional.
      <Expandable title="properties">
        <ResponseField name="option" type="bool">
          Indicates whether the distribution forwards and caches based on query strings. Optional.
        </ResponseField>
        <ResponseField name="query_strings_allow_list" type="list(string)">
          The specific query strings that the distribution forwards to the origin. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="maximum_ttl" type="number">
      The maximum amount of time that objects stay in the distribution's cache before the distribution forwards another request to the origin to determine whether the object has been updated. Optional.
    </ResponseField>
    <ResponseField name="minimum_ttl" type="number">
      The minimum amount of time that objects stay in the distribution's cache before the distribution forwards another request to the origin to determine whether the object has been updated. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="cache_behaviors" type="list(object)">
  An array of objects that describe the per-path cache behavior for the distribution. Optional.
  <Expandable title="properties">
    <ResponseField name="behavior" type="string">
      Specifies the cache behavior for the associated path pattern, with supported values of 'cache' or 'no-cache' indicating whether matching requests are cached. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="path" type="string">
      The path pattern (such as `/images/*`) that determines which requests this cache behavior applies to in the Lightsail distribution. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="certificate_name" type="string">
  The certificate attached to the Distribution. Optional.
</ResponseField>

<ResponseField name="default_cache_behavior" type="object" required>
  Describes the default cache behavior of an Amazon Lightsail content delivery network (CDN) distribution. Required.
  <Expandable title="properties">
    <ResponseField name="behavior" type="string">
      The cache behavior of the distribution. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="distribution_name" type="string" required>
  The name for the distribution. Required.
</ResponseField>

<ResponseField name="ip_address_type" type="string">
  The IP address type for the distribution. Optional.
</ResponseField>

<ResponseField name="is_enabled" type="bool">
  Indicates whether the distribution is enabled. Optional.
</ResponseField>

<ResponseField name="origin" type="object" required>
  Describes the origin resource of an Amazon Lightsail content delivery network (CDN) distribution. Required.
  <Expandable title="properties">
    <ResponseField name="name" type="string">
      The name of the origin resource. Optional.
    </ResponseField>
    <ResponseField name="protocol_policy" type="string">
      The protocol that your Amazon Lightsail distribution uses when establishing a connection with your origin to pull content. Optional.
    </ResponseField>
    <ResponseField name="region_name" type="string">
      The AWS Region name of the origin resource. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An array of key-value pairs to apply to this resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key of a tag to attach to the Lightsail distribution, used to identify and categorize the resource. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value part of a tag attached to the Lightsail distribution, used to store arbitrary metadata (e.g., environment, owner) for resource organization, cost allocation, or access control. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

</ResponseField>

<ResponseField name="status" type="string">
  The status of the distribution. Computed by AWS after `ship`.
</ResponseField>
