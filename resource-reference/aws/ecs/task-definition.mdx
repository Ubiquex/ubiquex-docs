`aws_ecs_task_definition` is the real blueprint, container images, CPU and memory, networking mode, IAM roles, that an `aws_ecs_service` or a standalone task actually launches from.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ecs "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/ecs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own task definition"})

    		ubx.Resource(ecs.TaskDefinition, "example", ecs.TaskDefinitionConfig{
    			ContainerDefinitions:  []map[string]any{{"name": "example-name"}},
    			Cpu:                   "example",
    			EnableFaultInjection:  true,
    			EphemeralStorage:      map[string]any{"size_in_gi_b": 1},
    			ExecutionRoleArn:      "arn:aws:iam::123456789012:policy/example",
    			Family:                "example",
    			InferenceAccelerators: []map[string]any{{"device_name": "example-device-name"}},
    			IpcMode:               "example",
    		})
    	}))
    }
    ```
