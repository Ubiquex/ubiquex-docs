`aws_logs_query_definition` saves a real, named CloudWatch Logs Insights query for reuse from the console or API, rather than re-typing it for every ad hoc investigation.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	logs "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/logs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own query definition"})

    		ubx.Resource(logs.QueryDefinition, "ci-runner", logs.QueryDefinitionConfig{
    			Name:          "ci-runner",
    			QueryString:   "example",
    			LogGroupNames: []string{"example"},
    			Parameters:    []map[string]any{{"name": "example-name"}},
    			QueryLanguage: "example",
    		})
    	}))
    }
    ```
  </Tab>

    from ubx.aws.logs import QueryDefinition, QueryDefinitionConfig

    def describe():

        ubx.resource(QueryDefinition, "ci-runner", QueryDefinitionConfig(
            name="ci-runner",
            query_string="example",
            log_group_names=["example"],

## Input properties

<ResponseField name="log_group_names" type="list(string)">
  Optionally define specific log groups as part of your query definition Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  A name for the saved query definition Required.
</ResponseField>

<ResponseField name="parameters" type="list(object)">
  Use this parameter to include specific query parameters as part of your query definition. Query parameters are supported only for Logs Insights QL queries. Query parameters allow you to use placeholder variables in your query string that are substituted with values at execution time. Use the &#123;&#123;parameterName&#125;&#125; syntax in your query string to reference a parameter. Optional.
  <Expandable title="properties">
    <ResponseField name="default_value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="description" type="string">
      An optional human-readable description of the query definition, used to identify its purpose or contents. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="name" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="query_language" type="string">
  Query language of the query string. Possible values are CWLI, SQL, PPL, with CWLI being the default. Optional.
</ResponseField>

<ResponseField name="query_string" type="string" required>
  The query string to use for this definition Required.
</ResponseField>

## Output properties

<ResponseField name="query_definition_id" type="string">
  Unique identifier of a query definition Computed by AWS after `ship`.
</ResponseField>
