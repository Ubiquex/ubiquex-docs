`aws_logs_log_stream` creates a real, individual sequence of log events within an `aws_logs_log_group`, typically one per source, an EC2 instance, a Lambda invocation, that log events actually get appended to.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	logs "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/logs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own log stream"})

    		ubx.Resource(logs.LogStream, "example", logs.LogStreamConfig{
    			LogGroupName:  "example-log-group-name",
    			LogStreamName: "example-log-stream-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";

    export default stack("example", () => {

      resource(LogStream, "example", {
        logGroupName: "example-log-group-name",

      });

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.logs.log_stream import LogStream, LogStreamConfig

    def describe():
        ubx.intent("example own log stream")

        ubx.resource(LogStream, "example", LogStreamConfig(
            log_group_name="example-log-group-name",
            log_stream_name="example-log-stream-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="log_group_name" type="string" required>
  The name of the log group where the log stream is created. Required.
</ResponseField>

<ResponseField name="log_stream_name" type="string">
  The name of the log stream. The name must be unique wihtin the log group. Optional.
</ResponseField>
