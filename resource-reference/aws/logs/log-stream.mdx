`aws_logs_log_stream` creates a real, individual sequence of log events within an `aws_logs_log_group`, typically one per source, an EC2 instance, a Lambda invocation, that log events actually get appended to.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	logs "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/logs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own log stream"})

    		ubx.Resource(logs.LogStream, "ci-runner", logs.LogStreamConfig{
    			LogGroupName:  "example-log-group-name",
    			LogStreamName: "example-log-stream-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { LogStream } from "./local-sdk/aws/sdk/typescript/aws/logs/log_stream.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own log stream" });

      resource(LogStream, "ci-runner", {
        logGroupName: "example-log-group-name",
        logStreamName: "example-log-stream-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.logs import LogStream, LogStreamConfig

    def describe():
        ubx.intent("payments own log stream")

        ubx.resource(LogStream, "ci-runner", LogStreamConfig(
            log_group_name="example-log-group-name",
            log_stream_name="example-log-stream-name",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Properties

<ResponseField name="log_group_name" type="string" required>
  The name of the log group where the log stream is created. Required.
</ResponseField>

<ResponseField name="log_stream_name" type="string">
  The name of the log stream. The name must be unique wihtin the log group. Optional.
</ResponseField>
