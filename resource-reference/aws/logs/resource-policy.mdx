`aws_logs_resource_policy` attaches a real resource-based access policy scoped to specific `aws_logs_log_group`s, controlling which other AWS services or accounts can write log events into them.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	logs "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/logs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resource policy"})

    		ubx.Resource(logs.ResourcePolicy, "example", logs.ResourcePolicyConfig{
    			PolicyDocument: "example",
    			PolicyName:     "example-policy-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { ResourcePolicy } from "@ubx/sdk-aws/aws/logs/resource_policy";

    export default stack("example", () => {

      resource(ResourcePolicy, "example", {
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.logs.resource_policy import ResourcePolicy, ResourcePolicyConfig

    def describe():
        ubx.intent("example own resource policy")

        ubx.resource(ResourcePolicy, "example", ResourcePolicyConfig(
            policy_document="example",
            policy_name="example-policy-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
