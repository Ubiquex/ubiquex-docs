`aws_macie_allow_list` defines a real list of text patterns or exact values, stored as an S3 object, that Amazon Macie should ignore when scanning for sensitive data, keeping known-benign content out of Macie's findings.

## Example

    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    import (
    	macie "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/macie"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own allow list"})

    		ubx.Resource(macie.AllowList, "ci-runner", macie.AllowListConfig{
    			Criteria:    "example",
    			Name:        "ci-runner",
    			Description: "Managed by ubx.",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
