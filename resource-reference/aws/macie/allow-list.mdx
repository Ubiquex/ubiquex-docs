`aws_macie_allow_list` defines a real list of text patterns or exact values, stored as an S3 object, that Amazon Macie should ignore when scanning for sensitive data, keeping known-benign content out of Macie's findings.

## Example

<Tabs>

    ```go
    package main

    import (
    	macie "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/macie"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own allow list"})

    		ubx.Resource(macie.AllowList, "example", macie.AllowListConfig{
    			Criteria:    "example",
    			Name:        "example",
    			Description: "Managed by ubx.",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
