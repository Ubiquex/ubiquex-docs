`aws_macie_custom_data_identifier` defines a real, custom sensitive-data pattern, a regular expression with optional keyword and proximity rules, that Macie checks for during a scan beyond its own built-in managed data identifiers.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	macie "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/macie"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own custom data identifier"})

    		ubx.Resource(macie.CustomDataIdentifier, "example", macie.CustomDataIdentifierConfig{
    			Name:                 "example",
    			Regex:                "example",
    			Description:          "Managed by ubx.",
    			IgnoreWords:          []string{"example"},
    			Keywords:             []string{"example"},
    			MaximumMatchDistance: 1,
    			Tags:                 []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

<ResponseField name="ignore_words" type="list(string)">

</ResponseField>

<ResponseField name="keywords" type="list(string)">
  Keywords to be matched against. Optional.
</ResponseField>

<ResponseField name="maximum_match_distance" type="number">
  Maximum match distance. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>

</ResponseField>
