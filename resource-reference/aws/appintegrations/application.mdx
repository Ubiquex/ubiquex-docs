`aws_app_integrations_application` configures a reusable connection to an external application inside Amazon AppIntegrations, most commonly consumed by Amazon Connect's own third-party integrations, the general application-level counterpart to a specific `aws_app_integrations_data_integration` or `aws_app_integrations_event_integration`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	appintegrations "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/appintegrations"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own application"})

    		ubx.Resource(appintegrations.Application, "example", appintegrations.ApplicationConfig{
    			ApplicationSourceConfig: map[string]any{"external_url_config": map[string]any{"access_url": "example"}},
    			Name:                    "example",
    			Namespace:               "example",
    			ApplicationConfig:       map[string]any{"contact_handling": map[string]any{"scope": "example"}},
    			ApplicationType:         "example",
    			Description:             "Managed by ubx.",
    			IframeConfig:            map[string]any{"allow": []string{"example"}},
    			InitializationTimeout:   1,
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.appintegrations import Application, ApplicationConfig
    from ubx.aws.appintegrations.application import Application_ApplicationSourceConfig, Application_ApplicationSourceConfig_ExternalUrlConfig, Application_ApplicationConfig, Application_ApplicationConfig_ContactHandling, Application_IframeConfig

    def describe():
        ubx.intent("example own application")

        ubx.resource(Application, "example", ApplicationConfig(
            application_source_config=Application_ApplicationSourceConfig(external_url_config=Application_ApplicationSourceConfig_ExternalUrlConfig(access_url="example")),
            name="example",
            namespace="example",
            application_config=Application_ApplicationConfig(contact_handling=Application_ApplicationConfig_ContactHandling(scope="example")),
            application_type="example",
            description="Managed by ubx.",
            iframe_config=Application_IframeConfig(allow=["example"]),
            initialization_timeout=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="application_config" type="object">
  Optional.
  <Expandable title="properties">
    <ResponseField name="contact_handling" type="object">
      Optional.
      <Expandable title="properties">
        <ResponseField name="scope" type="string" required>
          Specifies the operational scope of contact handling (the only supported value is DRIVE) for the AppIntegrations application, meaning the app acts as the primary interface during a contact. Required. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="application_source_config" type="object" required>
  Application source config Required.
  <Expandable title="properties">
    <ResponseField name="external_url_config" type="object" required>
      Configures the external URL for the AppIntegrations application, specifying the access URL and the list of approved origins. Required. **(AI-inferred)**
      <Expandable title="properties">
        <ResponseField name="access_url" type="string" required>
          The URL that users are redirected to when they access the application from the AppIntegrations UI. Required. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="approved_origins" type="list(string)">
          Specifies a list of allowed origins for the external URL of the AppIntegrations application, enabling CORS access from those domains. Optional. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="application_type" type="string">
  The type of application Optional.
</ResponseField>

<ResponseField name="description" type="string">
  The application description. Optional.
</ResponseField>

<ResponseField name="iframe_config" type="object">
  Iframe configuration for the application, specifying the AccessUrl to embed and the ApprovedOrigins list that controls which domains can host the iframe. Optional. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="allow" type="list(string)">
      Defines the list of allowed origins that are permitted to embed this AppIntegrations application in an iframe. Optional. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="sandbox" type="list(string)">
      Defines the sandbox permissions for the iframe used to display the application, where each string in the list corresponds to a token such as 'allow-scripts' or 'allow-same-origin' that controls what the iframe content can do. Optional. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="initialization_timeout" type="number">
  The initialization timeout in milliseconds. Required when IsService is true. Optional.
</ResponseField>

<ResponseField name="is_service" type="bool">
  Indicates if the application is a service Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the application. Required.
</ResponseField>

<ResponseField name="namespace" type="string" required>
  The namespace of the application. Required.
</ResponseField>

<ResponseField name="permissions" type="list(string)">
  The configuration of events or requests that the application has access to. Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  The tags (keys and values) associated with the application. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      The tag value for a key-value pair attached to the AWS AppIntegrations application, used to organize, identify, and manage the application resource within AWS. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>
