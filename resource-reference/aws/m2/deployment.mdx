`aws_m2_deployment` deploys a real, specific `aws_m2_application` version onto an `aws_m2_environment`, the resource that actually promotes application code from registered to running under AWS Mainframe Modernization.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	m2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/m2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own deployment"})

    		ubx.Resource(m2.Deployment, "example", m2.DeploymentConfig{
    			ApplicationId:      "example",
    			ApplicationVersion: 1,
    			EnvironmentId:      "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Deployment } from "@ubx/sdk-aws/aws/m2/deployment";

    export default stack("example", () => {
      intent({ summary: "example own deployment" });

      resource(Deployment, "example", {
        applicationId: "example",
        applicationVersion: 1,
        environmentId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.m2 import Deployment, DeploymentConfig

    def describe():
        ubx.intent("example own deployment")

        ubx.resource(Deployment, "example", DeploymentConfig(
            application_id="example",
            application_version=1,
            environment_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>

## Input properties

<ResponseField name="application_id" type="string" required>
  The application ID. Required.
</ResponseField>

<ResponseField name="application_version" type="number" required>
  The version number of the application to deploy Required.
</ResponseField>

<ResponseField name="environment_id" type="string" required>
  The environment ID. Required.
</ResponseField>

## Output properties

<ResponseField name="deployment_id" type="string">
  The deployment ID. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="status" type="string">
  The status of the deployment. Computed by AWS after `ship`.
</ResponseField>
