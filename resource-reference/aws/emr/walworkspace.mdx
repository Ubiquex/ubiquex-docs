`aws_emr_walworkspace` is a real, durable, managed storage layer, outliving any single `aws_emr_cluster`, that Apache HBase write-ahead logs get written into, so a caller can point a brand-new cluster at the same S3 root directory and WAL workspace after a cluster or Availability Zone failure and recover unflushed HBase data within minutes.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    			Tags:             []map[string]any{{"key": "example"}},
    			WalworkspaceName: "example-walworkspace-name",
    		})
    	}))
    }
