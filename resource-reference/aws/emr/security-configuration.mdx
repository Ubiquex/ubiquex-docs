`aws_emr_security_configuration` defines a real, reusable set of encryption and authentication settings, at-rest, in-transit, Kerberos, that an `aws_emr_cluster` references instead of specifying that configuration inline on every cluster that needs it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	emr "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/emr"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own security configuration"})

    		ubx.Resource(emr.SecurityConfiguration, "example", emr.SecurityConfigurationConfig{
    			SecurityConfiguration: "example",
    			Name:                  "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only aws --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { SecurityConfiguration } from "./local-sdk/aws/sdk/typescript/aws/emr/security_configuration.ts";

    export default stack("example", () => {
      intent({ summary: "example own security configuration" });

      resource(SecurityConfiguration, "example", {

    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.emr import SecurityConfiguration, SecurityConfigurationConfig

    def describe():

        ubx.resource(SecurityConfiguration, "example", SecurityConfigurationConfig(
            security_configuration="example",
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
