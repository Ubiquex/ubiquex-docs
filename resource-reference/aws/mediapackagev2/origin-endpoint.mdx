`aws_media_package_v2_origin_endpoint` defines a real packaging configuration on an `aws_media_package_v2_channel`, the endpoint players actually request the packaged output stream from.

## Example

<Tabs>
  <Tab title="Go">
    ```go

    import (
    	mediapackagev2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/mediapackagev2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own origin endpoint"})

    		ubx.Resource(mediapackagev2.OriginEndpoint, "example", mediapackagev2.OriginEndpointConfig{
    			ChannelGroupName:                "example-channel-group-name",
    			ChannelName:                     "example-channel-name",
    			ContainerType:                   "example",
    			OriginEndpointName:              "example-origin-endpoint-name",
    			DashManifests:                   []map[string]any{{"audio_timeline_pattern": "example"}},
    			Description:                     "Managed by ubx.",
    			ForceEndpointErrorConfiguration: map[string]any{"endpoint_error_conditions": []string{"example"}},
    			MssManifests:                    []map[string]any{{"filter_configuration": map[string]any{"clip_start_time": "example"}}},
    		})
    	}))
    }
    ```
