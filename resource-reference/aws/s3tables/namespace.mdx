`aws_s3_tables_namespace` creates a real, logical grouping of `aws_s3_tables_table`s within an `aws_s3_tables_table_bucket`, a SQL-style schema/namespace for organizing Iceberg tables rather than leaving them all flat in the bucket.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	s3tables "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/s3tables"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own namespace"})

    		ubx.Resource(s3tables.Namespace, "example", s3tables.NamespaceConfig{
    			Namespace:      "example",
    			TableBucketArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Namespace } from "@ubx/sdk-aws/aws/s3tables/namespace";

    export default stack("example", () => {
      intent({ summary: "example own namespace" });

      resource(Namespace, "example", {
        namespace: "example",
        tableBucketArn: "arn:aws:iam::123456789012:policy/example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.s3tables import Namespace, NamespaceConfig

    def describe():
        ubx.intent("example own namespace")

        ubx.resource(Namespace, "example", NamespaceConfig(
            namespace="example",
            table_bucket_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties
