`aws_s3_tables_table_bucket` creates a real S3 Tables bucket, a bucket type purpose-built to store Apache Iceberg tables rather than arbitrary objects, that `aws_s3_tables_table`s are created inside.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	s3tables "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/s3tables"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own table bucket"})

    		ubx.Resource(s3tables.TableBucket, "example", s3tables.TableBucketConfig{
    			TableBucketName:           "example-table-bucket-name",
    			EncryptionConfiguration:   map[string]any{"kmskey_arn": "example"},
    			MetricsConfiguration:      map[string]any{"status": "example"},
    			ReplicationConfiguration:  map[string]any{"role": "example", "rules": []map[string]any{{"destinations": []map[string]any{{"destination_table_bucket_arn": "example"}}}}},
    			StorageClassConfiguration: map[string]any{"storage_class": "example"},
    			Tags:                      []map[string]any{{"key": "example"}},
    			UnreferencedFileRemoval:   map[string]any{"noncurrent_days": 1},
    		})
    	}))
    }
    ```

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.s3tables import TableBucket, TableBucketConfig
    from ubx.aws.s3tables.table_bucket import TableBucket_EncryptionConfiguration, TableBucket_MetricsConfiguration, TableBucket_ReplicationConfiguration, TableBucket_ReplicationConfiguration_Rules, TableBucket_ReplicationConfiguration_Rules_Destinations, TableBucket_StorageClassConfiguration, TableBucket_Tags, TableBucket_UnreferencedFileRemoval

    def describe():
        ubx.intent("example own table bucket")

        ubx.resource(TableBucket, "example", TableBucketConfig(
            table_bucket_name="example-table-bucket-name",
            encryption_configuration=TableBucket_EncryptionConfiguration(kmskey_arn="example"),
            metrics_configuration=TableBucket_MetricsConfiguration(status="example"),
            replication_configuration=TableBucket_ReplicationConfiguration(role="example", rules=[TableBucket_ReplicationConfiguration_Rules(destinations=[TableBucket_ReplicationConfiguration_Rules_Destinations(destination_table_bucket_arn="example")])]),
            storage_class_configuration=TableBucket_StorageClassConfiguration(storage_class="example"),
            tags=[TableBucket_Tags(key="example")],
            unreferenced_file_removal=TableBucket_UnreferencedFileRemoval(noncurrent_days=1),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="encryption_configuration" type="object">
  Specifies encryption settings for the table bucket Optional.
  <Expandable title="properties">
    <ResponseField name="kmskey_arn" type="string">
      ARN of the KMS key to use for encryption Optional.
    </ResponseField>
    <ResponseField name="ssealgorithm" type="string">
      Server-side encryption algorithm Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="metrics_configuration" type="object">
  Settings governing the Metric configuration for the table bucket. Optional.
  <Expandable title="properties">
    <ResponseField name="status" type="string">
      Indicates whether Metrics are enabled. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="replication_configuration" type="object">
  Specifies replication configuration for the table bucket Optional.
  <Expandable title="properties">
    <ResponseField name="role" type="string" required>
      The ARN of the IAM role to use for replication Required.
    </ResponseField>
    <ResponseField name="rules" type="list(object)" required>
      List of replication rules Required.
      <Expandable title="properties">
        <ResponseField name="destinations" type="list(object)">
          Specifies the list of destination table buckets that this S3 Tables replication rule uses as targets for replicating the table bucket's data. Optional; if omitted, computed by AWS. **(AI-inferred)**
          <Expandable title="properties">
            <ResponseField name="destination_table_bucket_arn" type="string">
              The ARN of the destination table bucket to which replication rules send copies of table data. Optional; if omitted, computed by AWS. **(AI-inferred)**
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="storage_class_configuration" type="object">
  Specifies storage class settings for the table bucket Optional.
  <Expandable title="properties">
    <ResponseField name="storage_class" type="string">
      The storage class for the table bucket Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="table_bucket_name" type="string" required>
  A name for the table bucket. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  User tags (key-value pairs) to associate with the table bucket. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key of a tag attached to the S3 table bucket, used to categorize and identify the resource for management and cost allocation. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      Specifies the value of a tag key attached to the S3 Tables table bucket, used for resource organization, cost allocation, and tag-based IAM access control. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="unreferenced_file_removal" type="object">
  Settings governing the Unreferenced File Removal maintenance action. Unreferenced file removal identifies and deletes all objects that are not referenced by any table snapshots. Optional.
  <Expandable title="properties">
    <ResponseField name="noncurrent_days" type="number">
      S3 permanently deletes noncurrent objects after the number of days specified by the NoncurrentDays property. Optional.
    </ResponseField>
    <ResponseField name="status" type="string">
      Indicates whether the Unreferenced File Removal maintenance action is enabled. Optional.
    </ResponseField>
    <ResponseField name="unreferenced_days" type="number">
      For any object not referenced by your table and older than the UnreferencedDays property, S3 creates a delete marker and marks the object version as noncurrent. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="table_bucket_arn" type="string">
  The Amazon Resource Name (ARN) of the specified table bucket. Computed by AWS after `ship`.
</ResponseField>
