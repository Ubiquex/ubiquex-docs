`aws_fms_policy` defines a real, centrally managed security policy, WAF rules, security group settings, Shield Advanced protection, that AWS Firewall Manager automatically applies and keeps enforced across every account and resource in an AWS Organization matching the policy's scope.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	fms "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/fms"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own policy"})

    		ubx.Resource(fms.Policy, "example", fms.PolicyConfig{
    			ExcludeResourceTags:       true,
    			PolicyName:                "example-policy-name",
    			RemediationEnabled:        true,
    			SecurityServicePolicyData: map[string]any{"type": "example"},
    			DeleteAllPolicyResources:  true,
    			ExcludeMap:                map[string]any{"account": []string{"example"}},
    			IncludeMap:                map[string]any{"account": []string{"example"}},
    			PolicyDescription:         "example",
    		})
    	}))
    }
    ```
  </Tab>
