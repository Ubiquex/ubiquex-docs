`aws_fms_notification_channel` configures a real SNS topic that AWS Firewall Manager publishes notifications to about an `aws_fms_policy`'s own compliance status changes across an organization's accounts.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	fms "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/fms"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own notification channel"})

    		ubx.Resource(fms.NotificationChannel, "ci-runner", fms.NotificationChannelConfig{
    			SnsRoleName: "example-sns-role-name",
    			SnsTopicArn: "arn:aws:iam::123456789012:policy/ci",
    		})
    	}))
    }
    ```
  </Tab>
