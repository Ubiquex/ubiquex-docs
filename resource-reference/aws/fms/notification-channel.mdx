`aws_fms_notification_channel` configures a real SNS topic that AWS Firewall Manager publishes notifications to about an `aws_fms_policy`'s own compliance status changes across an organization's accounts.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	fms "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/fms"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own notification channel"})

    		ubx.Resource(fms.NotificationChannel, "example", fms.NotificationChannelConfig{
    			SnsRoleName: "example-sns-role-name",
    			SnsTopicArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.fms import NotificationChannel, NotificationChannelConfig

    def describe():
        ubx.intent("example own notification channel")

        ubx.resource(NotificationChannel, "example", NotificationChannelConfig(
            sns_role_name="example-sns-role-name",
            sns_topic_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
