`hashicorp/aws` provider schema, in every SDK language.

real enough to save and run exactly as shown.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	pinpoint "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/pinpoint"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own app"})

    		ubx.Resource(pinpoint.App, "ci-runner", pinpoint.AppConfig{
    			Name: "ci-runner",
    			Tags: map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```
