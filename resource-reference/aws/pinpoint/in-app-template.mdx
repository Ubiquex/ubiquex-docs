`aws_pinpoint_in_app_template` creates a real, reusable in-app message template, layout and content shown inside an application's own UI rather than as a push notification, subject to the same October 30, 2026 Pinpoint retirement.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	pinpoint "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/pinpoint"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own in app template"})

    		ubx.Resource(pinpoint.InAppTemplate, "example", pinpoint.InAppTemplateConfig{
    			TemplateName:        "example-template-name",
    			Content:             []map[string]any{{"background_color": "example"}},
    			CustomConfig:        "example",
    			Layout:              "example",
    			Tags:                "example",
    			TemplateDescription: "example",
    		})
    	}))
    }
    ```
