`aws_s3_access_grants_location` registers a real S3 bucket or prefix with an `aws_s3_access_grants_instance`, the scope that individual `aws_s3_access_grant`s are then issued against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	s3 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/s3"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own access grants location"})

    		ubx.Resource(s3.AccessGrantsLocation, "example", s3.AccessGrantsLocationConfig{
    			IamRoleArn:    "arn:aws:iam::123456789012:policy/example",
    			LocationScope: "example",
    			Tags:          []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
