`aws_s3_files_mount_target` creates a real, VPC-reachable network endpoint for an `aws_s3_files_file_system`, the resource that actually lets EC2, ECS, EKS, or Lambda mount that S3 data as a POSIX file system over the network.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	s3files "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/s3files"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own files mount target"})

    		ubx.Resource(s3files.MountTarget, "example", s3files.MountTargetConfig{
    			FileSystemId:   "example",
    			SubnetId:       "example",
    			IpAddressType:  "example",
    			Ipv4Address:    "example",
    			Ipv6Address:    "example",
    			SecurityGroups: []string{"example"},
    		})
    	}))
    }
    ```
