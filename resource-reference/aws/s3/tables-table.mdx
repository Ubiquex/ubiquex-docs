`aws_s3_tables_table` creates a real Apache Iceberg table within an `aws_s3_tables_table_bucket`, with S3 Tables handling the table's own compaction, snapshot management, and file optimization automatically in the background.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	s3 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/s3"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own tables table"})

    		ubx.Resource(s3.TablesTable, "example", s3.TablesTableConfig{
    			Namespace:                 "example",
    			OpenTableFormat:           "example",
    			TableBucketArn:            "arn:aws:iam::123456789012:policy/example",
    			TableName:                 "example-table-name",
    			Compaction:                map[string]any{"status": "example"},
    			IcebergMetadata:           map[string]any{"iceberg_partition_spec": map[string]any{"fields": []map[string]any{{"name": "example-name"}}}},
    			SnapshotManagement:        map[string]any{"max_snapshot_age_hours": 1},
    			StorageClassConfiguration: map[string]any{"storage_class": "example"},
    		})
    	}))
    }
    ```
  </Tab>
