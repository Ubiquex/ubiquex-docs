`aws_s3_outposts_bucket` creates a real S3 bucket physically located on an AWS Outposts rack, for workloads that must keep S3-compatible storage on premises rather than in an AWS Region.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	s3outposts "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/s3outposts"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own outposts bucket"})

    		ubx.Resource(s3outposts.Bucket, "example", s3outposts.BucketConfig{
    			BucketName:             "example-bucket-name",
    			OutpostId:              "example",
    			LifecycleConfiguration: map[string]any{"rules": []map[string]any{{"abort_incomplete_multipart_upload": map[string]any{"days_after_initiation": 1}}}},
    			Tags:                   []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
