`aws_s3_access_grants_instance` creates a real, account-level S3 Access Grants instance, the resource that an `aws_s3_access_grants_location` and individual `aws_s3_access_grant`s are registered under to vend temporary, prefix-scoped S3 credentials.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	s3 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/s3"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own access grants instance"})

    		ubx.Resource(s3.AccessGrantsInstance, "example", s3.AccessGrantsInstanceConfig{
    			IdentityCenterArn: "arn:aws:iam::123456789012:policy/example",
    			Tags:              []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
