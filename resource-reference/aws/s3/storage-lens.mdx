`aws_s3_storage_lens` creates a real S3 Storage Lens configuration, an organization- or account-wide dashboard of storage usage and activity metrics across every `aws_s3_bucket` in scope, including cost-optimization recommendations.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	s3 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/s3"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own storage lens"})

    		ubx.Resource(s3.StorageLens, "example", s3.StorageLensConfig{
    			Tags: []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { StorageLens } from "@ubx/sdk-aws/aws/s3/storage_lens";

    export default stack("example", () => {
      intent({ summary: "example own storage lens" });

      resource(StorageLens, "example", {
        tags: [{ key: "example" }],
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.s3 import StorageLens, StorageLensConfig
    from ubx.aws.s3.storage_lens import StorageLens_Tags

    def describe():
        ubx.intent("example own storage lens")

        ubx.resource(StorageLens, "example", StorageLensConfig(
            tags=[StorageLens_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
