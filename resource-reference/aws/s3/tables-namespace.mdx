`aws_s3_tables_namespace` creates a real, logical grouping of `aws_s3_tables_table`s within an `aws_s3_tables_table_bucket`, a SQL-style schema/namespace for organizing Iceberg tables rather than leaving them all flat in the bucket.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	s3 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/s3"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own tables namespace"})

    		ubx.Resource(s3.TablesNamespace, "example", s3.TablesNamespaceConfig{
    			Namespace:      "example",
    			TableBucketArn: "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    if __name__ == "__main__":
        ubx.run("example", describe)
