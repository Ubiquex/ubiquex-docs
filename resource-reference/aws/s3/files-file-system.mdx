`aws_s3_files_file_system` creates a real, POSIX-compliant file system scoped to an S3 bucket or prefix, built on Amazon EFS technology, so that bucket's data can be mounted and accessed with full file-system semantics from EC2, ECS, EKS, or Lambda without the data ever leaving S3. Amazon S3 Files reached general availability on April 7, 2026, roughly four months ago, so behavior here is verified live rather than assumed from general knowledge.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	s3 "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/s3"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own files file system"})

    		ubx.Resource(s3.FilesFileSystem, "example", s3.FilesFileSystemConfig{
    			Bucket:              "example",
    			RoleArn:             "arn:aws:iam::123456789012:policy/example",
    			AcceptBucketWarning: true,
    			ClientToken:         "example",
    			KmsKeyId:            "example",
    			Prefix:              "example",
    			Tags:                []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
