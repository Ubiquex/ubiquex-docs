`aws_s3_storage_lens_group` defines a real, filtered subset of objects, by prefix, tag, size, or age, within an `aws_s3_storage_lens` configuration, so its metrics can be broken out for just that subset instead of only whole-bucket totals.

## Example

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own storage lens group"})

    		ubx.Resource(s3.StorageLensGroup, "example", s3.StorageLensGroupConfig{
    			Filter: map[string]any{"and": map[string]any{"match_any_prefix": []string{"example"}}},
    			Name:   "example",
    			Tags:   []map[string]any{{"key": "example"}},
    		})
    	}))
    }
