`aws_s3_files_file_system_policy` attaches a real resource-based access policy to an `aws_s3_files_file_system`, controlling which principals can mount and access it as a file system.

## Example

<Tabs>

    package main

    import (
    	s3files "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/s3files"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own files file system policy"})

    		ubx.Resource(s3files.FileSystemPolicy, "example", s3files.FileSystemPolicyConfig{
    			FileSystemId: "example",
    			Policy:       "example",
    		})
    	}))
    }
