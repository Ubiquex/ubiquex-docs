`aws_ssm_resource_policy` attaches a real resource-based access policy to an `aws_ssm_parameter`, controlling which principals, including from other accounts, can read or write it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	ssm "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/ssm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own resource policy"})

    		ubx.Resource(ssm.ResourcePolicy, "ci-runner", ssm.ResourcePolicyConfig{
    			Policy:      "example",
    			ResourceArn: "arn:aws:iam::123456789012:policy/ci",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.ssm import ResourcePolicy, ResourcePolicyConfig

    def describe():
        ubx.intent("payments own resource policy")

        ubx.resource(ResourcePolicy, "ci-runner", ResourcePolicyConfig(
            policy="example",
            resource_arn="arn:aws:iam::123456789012:policy/ci",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)

## Input properties

<ResponseField name="policy" type="any" required>
  Actual policy statement. Required.
</ResponseField>

<ResponseField name="resource_arn" type="string" required>
  Arn of OpsItemGroup etc. Required.
</ResponseField>

## Output properties

<ResponseField name="policy_hash" type="string">
  A snapshot identifier for the policy over time. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="policy_id" type="string">
  An unique identifier within the policies of a resource. Computed by AWS after `ship`.
</ResponseField>
