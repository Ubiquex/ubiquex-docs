`aws_open_search_data_source` connects a real external data source, most commonly Amazon S3, to an `aws_open_search_service_domain`, so OpenSearch can query that data directly without first ingesting a full copy of it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	opensearch "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/opensearch"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own data source"})

    		ubx.Resource(opensearch.DataSource, "example", opensearch.DataSourceConfig{
    			DataSourceType: map[string]any{"s3_glue_data_catalog": map[string]any{"role_arn": "example"}},
    			DomainName:     "example-domain-name",
    			Name:           "example",
    			Description:    "Managed by ubx.",
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.opensearch import DataSource, DataSourceConfig
    from ubx.aws.opensearch.data_source import DataSource_DataSourceType, DataSource_DataSourceType_S3GlueDataCatalog

    def describe():
        ubx.intent("example own data source")

        ubx.resource(DataSource, "example", DataSourceConfig(
            data_source_type=DataSource_DataSourceType(s3_glue_data_catalog=DataSource_DataSourceType_S3GlueDataCatalog(role_arn="example")),
            domain_name="example-domain-name",
            name="example",
            description="Managed by ubx.",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="data_source_type" type="object" required>
  The type of data source. Required.
  <Expandable title="properties">
    <ResponseField name="s3_glue_data_catalog" type="object">
      Configuration for an S3 Glue Data Catalog data source. Optional.
      <Expandable title="properties">
        <ResponseField name="role_arn" type="string" required>
          The ARN of the IAM role that grants OpenSearch Service permission to access the Glue Data Catalog. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="description" type="string">
  A description of the data source. Optional.
</ResponseField>

<ResponseField name="domain_name" type="string" required>
  The name of the OpenSearch Service domain. Required.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the data source. Required.
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  The Amazon Resource Name (ARN) of the data source. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="status" type="string">
  The status of the data source. Computed by AWS after `ship`.
</ResponseField>
