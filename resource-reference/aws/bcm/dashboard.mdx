`aws_bcm_dashboard` combines multiple real cost data visualizations, Cost Explorer, Savings Plans, and Reserved Instance metrics among them, into one shareable dashboard, built and permissioned programmatically rather than reassembled by hand each time in the console.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	bcm "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/bcm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dashboard"})

    		ubx.Resource(bcm.Dashboard, "example", bcm.DashboardConfig{
    			Name:        "example",
    			Widgets:     []map[string]any{{"configs": []map[string]any{{"display_config": map[string]any{"graph": "example"}}}}},
    			Description: "Managed by ubx.",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    import ubx_sdk as ubx
    from ubx.aws.bcm.dashboard import Dashboard, DashboardConfig

    def describe():
        ubx.intent("example own dashboard")

        ubx.resource(Dashboard, "example", DashboardConfig(
            name="example",
            widgets=[{"configs": [{"display_config": {"graph": "example"}}]}],
            description="Managed by ubx.",
            tags=[{"key": "example"}],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
