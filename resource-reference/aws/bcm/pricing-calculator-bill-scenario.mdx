`aws_bcm_pricing_calculator_bill_scenario` models planned AWS usage and commitments, Savings Plans, Reserved Instances, inside the AWS Pricing Calculator, generating a real estimated bill from actual discount and benefit-sharing preferences rather than a rough, unofficial estimate.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	bcmpricingcalculator "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/bcmpricingcalculator"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own pricing calculator bill scenario"})

    		ubx.Resource(bcmpricingcalculator.BillScenario, "example", bcmpricingcalculator.BillScenarioConfig{
    			Name:                                  "example",
    			CostCategoryGroupSharingPreferenceArn: "arn:aws:iam::123456789012:policy/example",
    			ExpiresAt:                             "example",
    			GroupSharingPreference:                "example",
    			Tags:                                  []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
