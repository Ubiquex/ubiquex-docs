`aws_service_catalog_tag_option_association` attaches a real `aws_service_catalog_tag_option` to a specific product or `aws_service_catalog_portfolio`, the resource that actually makes that governed tag available to be applied there.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	servicecatalog "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/servicecatalog"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own tag option association"})

    		ubx.Resource(servicecatalog.TagOptionAssociation, "example", servicecatalog.TagOptionAssociationConfig{
    			ResourceId:  "example",
    			TagOptionId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    ```

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.servicecatalog import TagOptionAssociation, TagOptionAssociationConfig

    def describe():
        ubx.intent("example own tag option association")

            resource_id="example",
            tag_option_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
