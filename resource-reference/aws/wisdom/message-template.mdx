`aws_wisdom_message_template` defines a real, reusable response template, chat, email, or SMS, with variable placeholders, that a contact center agent or an `aws_wisdom_aiagent` can insert into a conversation instead of typing the same response by hand each time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	wisdom "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/wisdom"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own message template"})

    		ubx.Resource(wisdom.MessageTemplate, "example", wisdom.MessageTemplateConfig{
    			ChannelSubtype:        "example",
    			Content:               map[string]any{"email_message_template_content": map[string]any{"body": map[string]any{"html": map[string]any{"content": "example"}}}},
    			KnowledgeBaseArn:      "arn:aws:iam::123456789012:policy/example",
    			Name:                  "example",
    			DefaultAttributes:     map[string]any{"agent_attributes": map[string]any{"first_name": "example-first-name"}},
    			Description:           "Managed by ubx.",
    			GroupingConfiguration: map[string]any{"criteria": "example"},
    			Language:              "example",
    		})
    	}))
    }
    ```
