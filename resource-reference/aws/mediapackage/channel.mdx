`aws_media_package_channel` creates a real entry point that receives a live video stream from an encoder such as MediaLive and prepares it for packaging, the original MediaPackage (v1) channel resource, now superseded for new deployments by `aws_media_package_v2_channel`.

## Example

    ```go
    package main

    import (
    	mediapackage "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/mediapackage"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own package channel"})

    		ubx.Resource(mediapackage.Channel, "example", mediapackage.ChannelConfig{
    			Id:                "example",
    			Description:       "Managed by ubx.",
    			EgressAccessLogs:  map[string]any{"log_group_name": "example-log-group-name"},
    			IngressAccessLogs: map[string]any{"log_group_name": "example-log-group-name"},
    			Tags:              []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
