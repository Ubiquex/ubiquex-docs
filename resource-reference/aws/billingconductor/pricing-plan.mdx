`aws_billing_conductor_pricing_plan` bundles one or more `aws_billing_conductor_pricing_rule`s into the pricing an `aws_billing_conductor_billing_group`'s member accounts see, letting an organization define custom, marked-up, or discounted rates for chargeback and showback separate from AWS's own real bill.

## Example

    ```go
    package main

    import (
    	billingconductor "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/billingconductor"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own pricing plan"})

    		ubx.Resource(billingconductor.PricingPlan, "example", billingconductor.PricingPlanConfig{
    			Name:            "example",
    			Description:     "Managed by ubx.",
    			PricingRuleArns: []string{"example"},
    			Tags:            []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
