`aws_dev_ops_agent_trigger` defines a real event condition, an incident, a deployment failure, that automatically starts an `aws_dev_ops_agent_agent_space`'s own operator app working on it, rather than a human having to invoke the agent by hand.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	devopsagent "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/devopsagent"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own trigger"})

    		ubx.Resource(devopsagent.Trigger, "example", devopsagent.TriggerConfig{
    			Action:       "example",
    			AgentSpaceId: "example",
    			Condition:    map[string]any{"schedule": map[string]any{"expression": "example"}},
    			Type:         "example",
    			Status:       "example",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.devopsagent import Trigger, TriggerConfig
    from ubx.aws.devopsagent.trigger import Trigger_Condition, Trigger_Condition_Schedule

    def describe():
        ubx.intent("example own trigger")

        ubx.resource(Trigger, "example", TriggerConfig(
            action="example",
            agent_space_id="example",
            condition=Trigger_Condition(schedule=Trigger_Condition_Schedule(expression="example")),
            type="example",
            status="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="action" type="any" required>
  The action to perform when the trigger fires. A JSON object containing actionType and task. Required.
</ResponseField>

<ResponseField name="agent_space_id" type="string" required>
  The unique identifier of the parent Agent Space. Required.
</ResponseField>

<ResponseField name="condition" type="object" required>
  The condition that causes the trigger to fire. Required.
  <Expandable title="properties">
    <ResponseField name="schedule" type="object" required>
      Schedule configuration for a time-based trigger. Required.
      <Expandable title="properties">
        <ResponseField name="expression" type="string" required>
          A cron or rate expression that defines when the trigger fires. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="status" type="string">
  The status of the trigger. Active triggers fire on schedule; Inactive triggers are paused. Optional.
</ResponseField>

<ResponseField name="type" type="string" required>
  The type of trigger. Required.
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  The Amazon Resource Name (ARN) of the trigger. Nested under the parent Agent Space: arn:&lt;partition&gt;:aidevops:&lt;region&gt;:&lt;account-id&gt;:agentspace/&lt;agentspace-id&gt;/trigger/&lt;trigger-id&gt;. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="created_at" type="string">
  The timestamp when the trigger was created. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="trigger_id" type="string">
  The unique identifier of the trigger (assigned by the service on Create). Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="updated_at" type="string">
  The timestamp when the trigger was last updated. Computed by AWS after `ship`.
</ResponseField>
