`aws_cloud_front_response_headers_policy` adds or overrides real HTTP response headers, CORS, security headers among them, on every response an `aws_cloud_front_distribution` returns, without the origin itself needing to set them.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cloudfront "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/cloudfront"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own response headers policy"})

    		ubx.Resource(cloudfront.ResponseHeadersPolicy, "example", cloudfront.ResponseHeadersPolicyConfig{
    			ResponseHeadersPolicyConfig: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>
