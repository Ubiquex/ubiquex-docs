`aws_redshift_serverless_workgroup` creates a real compute configuration, its capacity in RPUs and network settings, that queries against an `aws_redshift_serverless_namespace` actually run through.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	redshift "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/redshift"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own serverless workgroup"})

    		ubx.Resource(redshift.ServerlessWorkgroup, "example", redshift.ServerlessWorkgroupConfig{
    			WorkgroupName:          "example-workgroup-name",
    			BaseCapacity:           1,
    			ConfigParameters:       []map[string]any{{"parameter_key": "example"}},
    			EnhancedVpcRouting:     true,
    			MaxCapacity:            1,
    			NamespaceName:          "example-namespace-name",
    			Port:                   1,
    			PricePerformanceTarget: map[string]any{"level": 1},
    		})
    	}))
    }
    ```
