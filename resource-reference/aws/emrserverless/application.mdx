`aws_emrserverless_application` runs real Spark or Hive jobs with automatically provisioned and scaled compute, no `aws_emr_cluster` to size, launch, or terminate, EMR's serverless alternative for workloads that don't need persistent cluster infrastructure.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	emrserverless "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/emrserverless"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own application"})

    		ubx.Resource(emrserverless.Application, "ci-runner", emrserverless.ApplicationConfig{
    			ReleaseLabel:                "example",
    			Type:                        "example",
    			Name:                        "ci-runner",
    			Architecture:                "example",
    			AutoStartConfiguration:      map[string]any{"enabled": true},
    			AutoStopConfiguration:       map[string]any{"enabled": true},
    			IdentityCenterConfiguration: map[string]any{"identity_center_instance_arn": "example"},
    			ImageConfiguration:          map[string]any{"image_uri": "example"},
    		})
    	}))
    }
    ```
