`aws_glue_registry` is a real, named collection of related `aws_glue_schema`s in the Glue Schema Registry, organizing schemas the way an `aws_glue_catalog` organizes tables.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own registry"})

    		ubx.Resource(glue.Registry, "example", glue.RegistryConfig{
    			Name:        "example",
    			Description: "Managed by ubx.",
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.glue import Registry, RegistryConfig
    from ubx.aws.glue.registry import Registry_Tags

    def describe():
        ubx.intent("example own registry")

        ubx.resource(Registry, "example", RegistryConfig(
            name="example",
            description="Managed by ubx.",
            tags=[Registry_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="description" type="string">
  A description of the registry. If description is not provided, there will not be any default value for this. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  Name of the registry to be created of max length of 255, and may only contain letters, numbers, hyphen, underscore, dollar sign, or hash mark. No whitespace. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  List of tags to tag the Registry Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The tag key part of a key-value pair used to tag an AWS Glue registry, enabling you to categorize and filter the resource. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value portion of a user-defined tag attached to this AWS Glue registry, used to categorize, identify, or manage the registry. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  Amazon Resource Name for the created Registry. Computed by AWS after `ship`.
</ResponseField>
