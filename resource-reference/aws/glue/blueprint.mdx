`aws_glue_blueprint` packages a real, reusable workflow template, generating a full set of `aws_glue_job`s, crawlers, and triggers from a small set of caller-provided parameters, rather than a caller assembling the same recurring pipeline shape by hand each time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own blueprint"})

    		ubx.Resource(glue.Blueprint, "ci-runner", glue.BlueprintConfig{
    			BlueprintLocation: "example",
    			Name:              "ci-runner",
    			Description:       "Managed by ubx.",
    			Tags:              []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    # ubx sdk gen --only aws --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/aws/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.aws.glue import Blueprint, BlueprintConfig

    def describe():
        ubx.intent("payments own blueprint")

        ubx.resource(Blueprint, "ci-runner", BlueprintConfig(
            blueprint_location="example",
            name="ci-runner",
            description="Managed by ubx.",
            tags=[{"key": "example"}],
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Input properties

<ResponseField name="blueprint_location" type="string" required>
  Specifies a path in Amazon S3 where the blueprint is published. Required.
</ResponseField>

<ResponseField name="description" type="string">
  A description of the blueprint. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the blueprint. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  The tags to be applied to this blueprint. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      A user-defined tag key attached to the AWS Glue Blueprint resource within the tags map, used for organizing and managing the blueprint. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value component of a tag (key-value pair) attached to the AWS Glue Blueprint, used for organizing, tracking, and allocating costs for the blueprint. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">

</ResponseField>

<ResponseField name="created_on" type="string">
  The date and time the blueprint was registered. Computed by AWS after `ship`.

<ResponseField name="last_modified_on" type="string">

</ResponseField>

<ResponseField name="status" type="string">

</ResponseField>
