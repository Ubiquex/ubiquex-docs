`aws_glue_integration_resource_property` sets a real, shared configuration property applied to every `aws_glue_integration` built from the same underlying resource, changing it affects every integration that resource is a part of rather than one integration individually.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own integration resource property"})

    		ubx.Resource(glue.IntegrationResourceProperty, "example", glue.IntegrationResourcePropertyConfig{
    			ResourceArn:                "arn:aws:iam::123456789012:policy/example",
    			SourceProcessingProperties: map[string]any{"role_arn": "example"},
    			Tags:                       []map[string]any{{"key": "example"}},
    			TargetProcessingProperties: map[string]any{"role_arn": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";

      resource(IntegrationResourceProperty, "example", {
        resourceArn: "arn:aws:iam::123456789012:policy/example",
        sourceProcessingProperties: { roleArn: "example" },
        tags: [{ key: "example" }],
        targetProcessingProperties: { roleArn: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.glue import IntegrationResourceProperty, IntegrationResourcePropertyConfig

    def describe():
        ubx.intent("example own integration resource property")

        ubx.resource(IntegrationResourceProperty, "example", IntegrationResourcePropertyConfig(
            resource_arn="arn:aws:iam::123456789012:policy/example",
            source_processing_properties={"role_arn": "example"},
            tags=[{"key": "example"}],
            target_processing_properties={"role_arn": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="resource_arn" type="string" required>
  The connection ARN of the source, or the database ARN of the target. Required.
</ResponseField>

<ResponseField name="source_processing_properties" type="object">
  The resource properties associated with the integration source. Optional.
  <Expandable title="properties">
    <ResponseField name="role_arn" type="string" required>
      The IAM role to access the Glue connection. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An array of key-value pairs to apply to this resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key of a tag assigned to the AWS Glue integration resource, enabling you to categorize and manage the integration according to your organizational conventions. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="target_processing_properties" type="object">
  The resource properties associated with the integration target. Optional.
  <Expandable title="properties">
    <ResponseField name="connection_name" type="string">
      The Glue network connection to configure the Glue job running in the customer VPC. Optional.
    </ResponseField>
    <ResponseField name="event_bus_arn" type="string">
      The ARN of an Eventbridge event bus to receive the integration status notification. Optional.
    </ResponseField>
    <ResponseField name="kms_arn" type="string">
      The ARN of the KMS key used for encryption. Optional.
    </ResponseField>
    <ResponseField name="role_arn" type="string" required>
      The IAM role to access the Glue database. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="resource_property_arn" type="string">
  The integration resource property ARN. Computed by AWS after `ship`.
</ResponseField>
