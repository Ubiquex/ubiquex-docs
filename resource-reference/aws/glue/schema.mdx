`aws_glue_schema` is a real, versioned schema definition, Avro, JSON, or Protobuf, registered so producers and consumers of streaming data, Kinesis or MSK, agree on and validate against the same actual message structure.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own schema"})

    		ubx.Resource(glue.Schema, "example", glue.SchemaConfig{
    			Compatibility:     "example",
    			DataFormat:        "example",
    			Name:              "example",
    			CheckpointVersion: map[string]any{"is_latest": true},
    			Description:       "Managed by ubx.",
    			Registry:          map[string]any{"name": "example-name"},
    			SchemaDefinition:  "example",
    			Tags:              []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.glue import Schema, SchemaConfig
    from ubx.aws.glue.schema import Schema_CheckpointVersion, Schema_Registry, Schema_Tags

    def describe():
        ubx.intent("example own schema")

        ubx.resource(Schema, "example", SchemaConfig(
            compatibility="example",
            data_format="example",
            name="example",
            checkpoint_version=Schema_CheckpointVersion(is_latest=True),
            description="Managed by ubx.",
            registry=Schema_Registry(name="example-name"),
            schema_definition="example",
            tags=[Schema_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="checkpoint_version" type="object">
  Specify checkpoint version for update. This is only required to update the Compatibility. Optional.
  <Expandable title="properties">
    <ResponseField name="is_latest" type="bool">
      Indicates if the latest version needs to be updated. Optional.
    </ResponseField>
    <ResponseField name="version_number" type="number">
      Indicates the version number in the schema to update. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="compatibility" type="string" required>
  Compatibility setting for the schema. Required.
</ResponseField>

<ResponseField name="data_format" type="string" required>
  Data format name to use for the schema. Accepted values: 'AVRO', 'JSON', 'PROTOBUF' Required.
</ResponseField>

<ResponseField name="description" type="string">
  A description of the schema. If description is not provided, there will not be any default value for this. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  Name of the schema. Required.
</ResponseField>

<ResponseField name="registry" type="object">
  Identifier for the registry which the schema is part of. Optional.
  <Expandable title="properties">
    <ResponseField name="arn" type="string">
      Amazon Resource Name for the Registry. Optional.
    </ResponseField>
    <ResponseField name="name" type="string">
      Name of the registry in which the schema will be created. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="schema_definition" type="string">
  Definition for the initial schema version in plain-text. Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  List of tags to tag the schema Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a tag attached to an AWS Glue schema, which allows you to assign custom metadata to the schema for cost tracking, access control, and operational organization. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  Amazon Resource Name for the Schema. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="initial_schema_version_id" type="string">
  Represents the version ID associated with the initial schema version. Computed by AWS after `ship`.
</ResponseField>
