`aws_glue_catalog` is a real, top-level metadata catalog, the container every `aws_glue_database` and `aws_glue_table` in one Glue Data Catalog deployment actually belongs to, queryable directly by Athena, Redshift Spectrum, and EMR without a caller copying the same metadata into each service separately.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own catalog"})

    		ubx.Resource(glue.Catalog, "example", glue.CatalogConfig{
    			Name:                             "example",
    			AllowFullTableExternalDataAccess: "example",
    			Description:                      "Managed by ubx.",
    			FederatedCatalog:                 map[string]any{"connection_name": "example-connection-name"},
    			OverwriteChildResourcePermissionsWithDefault: "example",
    			Parameters:            "example",
    			Tags:                  []map[string]any{{"key": "example"}},
    			TargetRedshiftCatalog: map[string]any{"catalog_arn": "example"},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.glue import Catalog, CatalogConfig
    from ubx.aws.glue.catalog import Catalog_FederatedCatalog, Catalog_Tags, Catalog_TargetRedshiftCatalog

    def describe():
        ubx.intent("example own catalog")

            name="example",

            description="Managed by ubx.",

            overwrite_child_resource_permissions_with_default="example",

            target_redshift_catalog=Catalog_TargetRedshiftCatalog(catalog_arn="example"),

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="allow_full_table_external_data_access" type="string">
  Allows third-party engines to access data in Amazon S3 locations that are registered with Lake Formation. Optional.
</ResponseField>

<ResponseField name="description" type="string">
  A description of the catalog. Optional.
</ResponseField>

<ResponseField name="federated_catalog" type="object">
  A FederatedCatalog structure that references an entity outside the Glue Data Catalog. Optional.
  <Expandable title="properties">
    <ResponseField name="connection_name" type="string">
      The name of the connection to an external data source. Optional.
    </ResponseField>
    <ResponseField name="identifier" type="string">
      A unique identifier for the federated catalog. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the catalog to create. Required.
</ResponseField>

<ResponseField name="overwrite_child_resource_permissions_with_default" type="string">
  Specifies whether to overwrite child resource permissions with the default permissions. Optional.
</ResponseField>

<ResponseField name="parameters" type="any">
  A map of key-value pairs that define parameters and properties of the catalog. Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An array of key-value pairs to apply to this resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a tag key in the Tags property of the AWS Glue Catalog, used to attach custom metadata to the Glue Data Catalog for identification and resource management. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="target_redshift_catalog" type="object">
  A structure that describes a target catalog for resource linking. Optional.
  <Expandable title="properties">
    <ResponseField name="catalog_arn" type="string" required>
      The Amazon Resource Name (ARN) of the catalog resource. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="catalog_id" type="string">
  The ID of the catalog. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="catalog_properties" type="object">
  A structure that specifies data lake access properties and other custom properties. Computed by AWS after `ship`.
  <Expandable title="properties">
    <ResponseField name="custom_properties" type="any">
      Additional key-value properties for the catalog. Optional.
    </ResponseField>
    <ResponseField name="data_lake_access_properties" type="object">
      Data lake access properties for the catalog. Optional.
      <Expandable title="properties">
        <ResponseField name="allow_full_table_external_data_access" type="string">
          Allows third-party engines to access data in Amazon S3 locations that are registered with Lake Formation. Optional.
        </ResponseField>
        <ResponseField name="catalog_type" type="string">
          Specifies a federated catalog type for the native catalog resource. Optional.
        </ResponseField>
        <ResponseField name="data_lake_access" type="bool">
          Turns on or off data lake access for Apache Spark applications that access Amazon Redshift databases in the Data Catalog from any non-Redshift engine. Optional.
        </ResponseField>
        <ResponseField name="data_transfer_role" type="string">
          A role that will be assumed by Glue for transferring data into/out of the staging bucket during a query. Optional.
        </ResponseField>
        <ResponseField name="kms_key" type="string">
          An encryption key that will be used for the staging bucket that will be created along with the catalog. Optional.
        </ResponseField>
        <ResponseField name="managed_workgroup_name" type="string">
          The name of the managed workgroup associated with the catalog. Optional.
        </ResponseField>
        <ResponseField name="managed_workgroup_status" type="string">
          The status of the managed workgroup. Optional.
        </ResponseField>
        <ResponseField name="redshift_database_name" type="string">
          The name of the Redshift database. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="create_database_default_permissions" type="list(object)">
  An array of PrincipalPermissions objects for default database permissions. Computed by AWS after `ship`.
  <Expandable title="properties">
    <ResponseField name="permissions" type="list(string)">
      The list of AWS Glue permissions (e.g., 'ALL' or 'CREATE_TABLE') granted by default to the specified principal on newly created databases in the Data Catalog. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="principal" type="object">
      Defines the principal (IAM user, role, or all users) that the Glue database's default permissions are granted to, via its Data Lake principal identifier. Optional; if omitted, computed by AWS. **(AI-inferred)**
      <Expandable title="properties">
        <ResponseField name="data_lake_principal_identifier" type="string">
          The unique identifier (such as an IAM role ARN, IAM user ARN, or AWS account ID) of the Data Lake principal that receives the default permissions for databases created in the Glue catalog. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="create_table_default_permissions" type="list(object)">
  An array of PrincipalPermissions objects for default table permissions. Computed by AWS after `ship`.
  <Expandable title="properties">
    <ResponseField name="permissions" type="list(string)">
      The list of permissions (e.g., ALL, SELECT, INSERT) to be granted to the principal in the default permissions for tables created in this Glue catalog database. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="principal" type="object">
      Specifies the principal (such as an IAM role, user, or AWS account) that receives the configured default permissions on newly created tables in the Glue data catalog. Optional; if omitted, computed by AWS. **(AI-inferred)**
      <Expandable title="properties">
        <ResponseField name="data_lake_principal_identifier" type="string">
          The ARN or identifier of the IAM role, user, or group that is granted default permissions on the Glue catalog table. Optional; if omitted, computed by AWS. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="create_time" type="number">
  The time at which the catalog was created. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="resource_arn" type="string">
  The Amazon Resource Name (ARN) of the catalog. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="update_time" type="number">
  The time at which the catalog was last updated. Computed by AWS after `ship`.
</ResponseField>
