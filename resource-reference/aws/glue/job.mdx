`aws_glue_job` defines a real, actual ETL script, Spark or Python, that transforms data read from a source into a target, run on demand, on a schedule via `aws_glue_trigger`, or as part of a larger workflow.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own job"})

    		ubx.Resource(glue.Job, "example", glue.JobConfig{
    			Command:           map[string]any{"name": "example-name"},
    			Role:              "example",
    			Name:              "example",
    			AllocatedCapacity: 1,
    			Connections:       map[string]any{"connections": []string{"example"}},
    			DefaultArguments:  "example",
    			Description:       "Managed by ubx.",
    			ExecutionClass:    "example",
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    import ubx_sdk as ubx
    from ubx.aws.glue.job import Job, JobConfig

    def describe():
        ubx.intent("example own job")

        ubx.resource(Job, "example", JobConfig(
            command={"name": "example-name"},
            role="example",
            name="example",
            allocated_capacity=1,
            connections={"connections": ["example"]},
            default_arguments="example",
            description="Managed by ubx.",
            execution_class="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
