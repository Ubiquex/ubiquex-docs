`aws_glue_schema_version_metadata` attaches a real, searchable key-value metadata pair to a specific `aws_glue_schema_version`, for tagging or filtering versions beyond the schema registry's own built-in fields.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	glue "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/glue"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own schema version metadata"})

    		ubx.Resource(glue.SchemaVersionMetadata, "example", glue.SchemaVersionMetadataConfig{
    			Key:             "example",
    			SchemaVersionId: "example",
    			Value:           "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

        ))

    if __name__ == "__main__":
