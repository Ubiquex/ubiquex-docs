`aws_s3_outposts_bucket_policy` attaches a real resource-based access policy to an `aws_s3_outposts_bucket`, controlling which principals on the local Outpost network can access it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	s3outposts "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/s3outposts"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own bucket policy"})

    		ubx.Resource(s3outposts.BucketPolicy, "example", s3outposts.BucketPolicyConfig{
    			Bucket:         "example",
    			PolicyDocument: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.s3outposts import BucketPolicy, BucketPolicyConfig

    def describe():
        ubx.intent("example own bucket policy")

        ubx.resource(BucketPolicy, "example", BucketPolicyConfig(
            bucket="example",
            policy_document="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="bucket" type="string" required>
  The Amazon Resource Name (ARN) of the specified bucket. Required.
</ResponseField>

<ResponseField name="policy_document" type="any" required>
  A policy document containing permissions to add to the specified bucket. Required.
</ResponseField>
