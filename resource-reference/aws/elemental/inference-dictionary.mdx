`aws_elemental_inference_dictionary` defines a real, custom vocabulary, team names, product names, brand names, that an `aws_elemental_inference_feed` applies to improve accuracy when its AI-generated live subtitles encounter domain-specific terms a general model wouldn't recognize correctly.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	elemental "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/elemental"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own inference dictionary"})

    		ubx.Resource(elemental.InferenceDictionary, "example", elemental.InferenceDictionaryConfig{
    			Language: "example",
    			Name:     "example",
    			Entries:  "example",
    			Tags:     "example",
    		})
    	}))
    }
    ```
