`aws_iam_oidcprovider` establishes real trust between IAM and an external OpenID Connect identity provider, GitHub Actions or an EKS cluster's own OIDC issuer among common ones, so an `aws_iam_role` can be assumed directly using that provider's tokens instead of a long-lived IAM credential.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iam "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/iam"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own oidcprovider"})

    		ubx.Resource(iam.Oidcprovider, "example", iam.OidcproviderConfig{
    			ClientIdList:   []string{"example"},
    			Tags:           []map[string]any{{"key": "example"}},
    			ThumbprintList: []string{"example"},
    			Url:            "example",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.iam import Oidcprovider, OidcproviderConfig
    from ubx.aws.iam.oidcprovider import Oidcprovider_Tags

    def describe():
        ubx.intent("example own oidcprovider")

        ubx.resource(Oidcprovider, "example", OidcproviderConfig(
            client_id_list=["example"],
            tags=[Oidcprovider_Tags(key="example")],
            thumbprint_list=["example"],
            url="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="client_id_list" type="list(string)">
  The list of client IDs (also known as audiences) that are allowed to use the OIDC provider to authenticate to AWS. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Defines one or more custom tag key-value pairs to attach to the IAM OIDC provider, which can be used for cost allocation, access control, and metadata management. Optional. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Specifies the key (name) of a tag assigned to the AWS IAM OIDC provider, allowing you to filter, control access, and track costs for the resource. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a tag attached to the IAM OIDC provider, used to manage, identify, and categorize the resource. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="thumbprint_list" type="list(string)">
  The list of server certificate thumbprints used to identify the OIDC identity provider's server certificate. Optional. **(AI-inferred)**
</ResponseField>

<ResponseField name="url" type="string">
  Specifies the URL of the OIDC identity provider (IdP), which must be a valid HTTPS URL that matches the issuer in the OIDC tokens issued by the provider. Optional. **(AI-inferred)**
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  Amazon Resource Name (ARN) of the OIDC provider Computed by AWS after `ship`.
</ResponseField>
