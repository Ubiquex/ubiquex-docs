`aws_iam_service_linked_role` creates a real IAM role predefined by an AWS service, with a trust policy and permissions the service itself owns and controls, that account owners cannot edit and generally cannot delete while the service still has resources depending on it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iam "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/iam"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own service linked role"})

    		ubx.Resource(iam.ServiceLinkedRole, "example", iam.ServiceLinkedRoleConfig{
    			AwsserviceName: "example-awsservice-name",
    			CustomSuffix:   "example",
    			Description:    "Managed by ubx.",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.aws.iam.service_linked_role import ServiceLinkedRole, ServiceLinkedRoleConfig

    def describe():
        ubx.intent("example own service linked role")

        ubx.resource(ServiceLinkedRole, "example", ServiceLinkedRoleConfig(
            awsservice_name="example-awsservice-name",
            custom_suffix="example",
            description="Managed by ubx.",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="awsservice_name" type="string">
  The service principal for the AWS service to which this role is attached. Optional.
</ResponseField>

<ResponseField name="custom_suffix" type="string">
  A string that you provide, which is combined with the service-provided prefix to form the complete role name. Optional.
</ResponseField>

<ResponseField name="description" type="string">
  The description of the role. Optional.
</ResponseField>

## Output properties

<ResponseField name="role_name" type="string">
  The name of the role. Computed by AWS after `ship`.
</ResponseField>
