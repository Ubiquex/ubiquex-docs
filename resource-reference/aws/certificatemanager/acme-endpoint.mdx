`aws_certificate_manager_acme_endpoint` creates a real, ACMEv2-compatible endpoint in AWS Certificate Manager, letting any standard ACME client automate public TLS certificate issuance and renewal against ACM rather than only through ACM's own API or console. ACM's ACME support was announced in July 2026, roughly a month ago, AWS's response to industry-wide moves toward shorter certificate lifetimes.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	certificatemanager "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/certificatemanager"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own acme endpoint"})

    		ubx.Resource(certificatemanager.AcmeEndpoint, "example", certificatemanager.AcmeEndpointConfig{
    			AuthorizationBehavior: "example",
    			CertificateAuthority:  map[string]any{"public_certificate_authority": map[string]any{"allowed_key_algorithms": []string{"example"}}},
    			CertificateTags:       []map[string]any{{"key": "example"}},
    			Contact:               "example",
    			Tags:                  []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.certificatemanager import AcmeEndpoint, AcmeEndpointConfig
    from ubx.aws.certificatemanager.acme_endpoint import AcmeEndpoint_CertificateAuthority, AcmeEndpoint_CertificateAuthority_PublicCertificateAuthority, AcmeEndpoint_CertificateTags

    def describe():
        ubx.intent("example own acme endpoint")

        ubx.resource(AcmeEndpoint, "example", AcmeEndpointConfig(
            authorization_behavior="example",
            certificate_authority=AcmeEndpoint_CertificateAuthority(public_certificate_authority=AcmeEndpoint_CertificateAuthority_PublicCertificateAuthority(allowed_key_algorithms=["example"])),
            certificate_tags=[AcmeEndpoint_CertificateTags(key="example")],
            contact="example",
            tags=[AcmeEndpoint_CertificateTags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="authorization_behavior" type="string" required>
  The authorization behavior for the ACME endpoint. Required.
</ResponseField>

<ResponseField name="certificate_authority" type="object" required>
  The certificate authority configuration for the ACME endpoint. Required.
  <Expandable title="properties">
    <ResponseField name="public_certificate_authority" type="object" required>
      Configuration for the public certificate authority. Required.
      <Expandable title="properties">
        <ResponseField name="allowed_key_algorithms" type="list(string)">
          The allowed key algorithms for certificates issued via this endpoint. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="certificate_tags" type="list(object)">
  Tags applied to certificates issued via this endpoint. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="contact" type="string">
  Whether contact information is required for the ACME endpoint. Optional.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  Tags associated with the ACME endpoint. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="acme_endpoint_arn" type="string">
  The Amazon Resource Name (ARN) of the ACME endpoint. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="endpoint_url" type="string">
  The ACME directory URL for the endpoint. Computed by AWS after `ship`.
</ResponseField>
