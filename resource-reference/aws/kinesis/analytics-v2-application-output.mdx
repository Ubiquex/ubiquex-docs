`aws_kinesis_analytics_v2_application_output` adds a real external destination, a Kinesis data stream, a Firehose delivery stream, or a Lambda function, that an `aws_kinesis_analytics_v2_application` writes its processed results to.

    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    	kinesis "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/kinesis"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own analytics v2 application output"})

    		ubx.Resource(kinesis.AnalyticsV2ApplicationOutput, "ci-runner", kinesis.AnalyticsV2ApplicationOutputConfig{
    			ApplicationName: "example-application-name",

    		})
    	}))
    }
