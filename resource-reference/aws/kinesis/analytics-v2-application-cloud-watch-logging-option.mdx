`aws_kinesis_analytics_v2_application_cloud_watch_logging_option` attaches a real CloudWatch log stream to an `aws_kinesis_analytics_v2_application`, so the Flink application's own runtime logs land somewhere durable and queryable rather than being discarded.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	kinesisanalyticsv2 "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/kinesisanalyticsv2"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own analytics v2 application cloud watch logging option"})

    		ubx.Resource(kinesisanalyticsv2.ApplicationCloudWatchLoggingOption, "example", kinesisanalyticsv2.ApplicationCloudWatchLoggingOptionConfig{
    			ApplicationName:         "example-application-name",
    			CloudWatchLoggingOption: map[string]any{"log_stream_arn": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { ApplicationCloudWatchLoggingOption } from "@ubx/sdk-aws/aws/kinesisanalyticsv2/application_cloud_watch_logging_option";

    export default stack("example", () => {

        summary: "example own analytics v2 application cloud watch logging option",
      });

        applicationName: "example-application-name",
        cloudWatchLoggingOption: { logStreamArn: "example" },
      });
    });
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.aws.kinesisanalyticsv2.application_cloud_watch_logging_option import ApplicationCloudWatchLoggingOption, ApplicationCloudWatchLoggingOptionConfig

    def describe():
        ubx.intent("example own analytics v2 application cloud watch logging option")

        ubx.resource(ApplicationCloudWatchLoggingOption, "example", ApplicationCloudWatchLoggingOptionConfig(
            application_name="example-application-name",
            cloud_watch_logging_option={"log_stream_arn": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="application_name" type="string" required>
  The name of the Kinesis Data Analytics for Apache Flink application to which this CloudWatch logging option is attached. Required. **(AI-inferred)**
</ResponseField>

<ResponseField name="cloud_watch_logging_option" type="object" required>
  Configures the CloudWatch Logs log stream (via its ARN) to which the Kinesis Analytics v2 application sends operational logs and metrics. Required. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="log_stream_arn" type="string" required>
      The ARN of the CloudWatch log stream that the Kinesis Data Analytics for Apache Flink application writes its diagnostic logs to. Required. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>
