`aws_kinesis_analytics_application_output` defines a real destination, another Kinesis stream, a Lambda function, that an `aws_kinesis_analytics_application`'s own continuous SQL query results actually get delivered to.

## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go

    import (
    	kinesis "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/kinesis"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own analytics application output"})

    		ubx.Resource(kinesis.AnalyticsApplicationOutput, "example", kinesis.AnalyticsApplicationOutputConfig{
    			ApplicationName: "example-application-name",

    		})

    }
