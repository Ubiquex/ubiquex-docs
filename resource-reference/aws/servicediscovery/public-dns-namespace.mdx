`aws_service_discovery_public_dns_namespace` creates a real Cloud Map namespace backed by a public Route 53 hosted zone, resolvable over the internet, the public counterpart to `aws_service_discovery_private_dns_namespace`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	servicediscovery "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/servicediscovery"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own public dns namespace"})

    		ubx.Resource(servicediscovery.PublicDnsNamespace, "example", servicediscovery.PublicDnsNamespaceConfig{
    			Name:        "example",
    			Description: "Managed by ubx.",
    			Properties:  map[string]any{"dns_properties": map[string]any{"soa": map[string]any{"ttl": 1}}},
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    import ubx_sdk as ubx
    from ubx.aws.servicediscovery import PublicDnsNamespace, PublicDnsNamespaceConfig
    from ubx.aws.servicediscovery.public_dns_namespace import PublicDnsNamespace_Properties, PublicDnsNamespace_Properties_DnsProperties, PublicDnsNamespace_Properties_DnsProperties_Soa, PublicDnsNamespace_Tags

    def describe():
        ubx.intent("example own public dns namespace")

        ubx.resource(PublicDnsNamespace, "example", PublicDnsNamespaceConfig(
            name="example",
            description="Managed by ubx.",
            properties=PublicDnsNamespace_Properties(dns_properties=PublicDnsNamespace_Properties_DnsProperties(soa=PublicDnsNamespace_Properties_DnsProperties_Soa(ttl=1))),
            tags=[PublicDnsNamespace_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="description" type="string">
  A description for the namespace. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name that you want to assign to this namespace. Required.
</ResponseField>

<ResponseField name="properties" type="object">
  The `properties` object contains the configuration attributes of the AWS Cloud Map public DNS namespace, including the required `Name`, optional `Description`, and optional `DnsProperties` (which defines SOA record settings). Optional. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="dns_properties" type="object">
      Configures DNS behavior for the public namespace, specifically the SOA record TTL that controls how long resolvers cache the namespace's authoritative data. Optional. **(AI-inferred)**
      <Expandable title="properties">
        <ResponseField name="soa" type="object">
          This object configures the Start of Authority (SOA) record settings for the public DNS namespace, including the TTL value that determines how long DNS resolvers cache the namespace's SOA record. Optional. **(AI-inferred)**
          <Expandable title="properties">
            <ResponseField name="ttl" type="number">
              The time-to-live (TTL) in seconds for the Start of Authority (SOA) record of the public DNS namespace, controlling how long DNS resolvers cache that record. Optional. **(AI-inferred)**
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tags" type="list(object)">
  The tags for the namespace. Each tag consists of a key and an optional value, both of which you define. Tag keys can have a maximum character length of 128 characters, and tag values can have a maximum length of 256 characters. Optional.
