`aws_connect_agent_status` defines a real, custom status, such as "Training" or "Lunch", that an `aws_connect_user` can select in their agent workspace, beyond Connect's own built-in Available/Offline statuses.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	connect "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/connect"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own agent status"})

    		ubx.Resource(connect.AgentStatus, "example", connect.AgentStatusConfig{
    			InstanceArn:      "arn:aws:iam::123456789012:policy/example",
    			Name:             "example",
    			State:            "example",
    			Description:      "Managed by ubx.",
    			DisplayOrder:     1,
    			ResetOrderNumber: true,
    			Tags:             []map[string]any{{"key": "example"}},
    			Type:             "example",
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.connect import AgentStatus, AgentStatusConfig
    from ubx.aws.connect.agent_status import AgentStatus_Tags

    def describe():
        ubx.intent("example own agent status")

        ubx.resource(AgentStatus, "example", AgentStatusConfig(
            instance_arn="arn:aws:iam::123456789012:policy/example",
            name="example",
            state="example",
            description="Managed by ubx.",
            display_order=1,
            reset_order_number=True,
            tags=[AgentStatus_Tags(key="example")],
            type="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="description" type="string">
  The description of the status. Optional.
</ResponseField>

<ResponseField name="display_order" type="number">
  The display order of the status. Optional.
</ResponseField>

<ResponseField name="instance_arn" type="string" required>
  The identifier of the Amazon Connect instance. Required.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the status. Required.
</ResponseField>

<ResponseField name="reset_order_number" type="bool">
  A number indicating the reset order of the agent status. Optional.
</ResponseField>

<ResponseField name="state" type="string" required>
  The state of the status. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  An array of key-value pairs to apply to this resource. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="type" type="string">
  The type of agent status. Optional.
</ResponseField>

## Output properties
