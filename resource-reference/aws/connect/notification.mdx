`aws_connect_notification` configures a real, automated notification, sent by email or in-app, tied to an event on an `aws_connect_instance`, such as an evaluation being completed.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	connect "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/connect"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own notification"})

    		ubx.Resource(connect.Notification, "example", connect.NotificationConfig{
    			Content:     map[string]any{"de_de": "example"},
    			InstanceArn: "arn:aws:iam::123456789012:policy/example",
    			ExpiresAt:   "example",
    			Priority:    "example",
    			Recipients:  []string{"example"},
    			Tags:        []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
