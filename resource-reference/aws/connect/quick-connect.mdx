`aws_connect_quick_connect` creates a real, one-click transfer target for an agent, another queue, a specific user, or an external phone number, so a warm transfer or task assignment doesn't require dialing or searching for the destination each time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	connect "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/connect"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own quick connect"})

    		ubx.Resource(connect.QuickConnect, "example", connect.QuickConnectConfig{
    			InstanceArn:        "arn:aws:iam::123456789012:policy/example",
    			Name:               "example",
    			QuickConnectConfig: map[string]any{"quick_connect_type": "example"},
    			Description:        "Managed by ubx.",
    			Tags:               []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.connect import QuickConnect, QuickConnectConfig
    from ubx.aws.connect.quick_connect import QuickConnect_QuickConnectConfig, QuickConnect_Tags

    def describe():
        ubx.intent("example own quick connect")

        ubx.resource(QuickConnect, "example", QuickConnectConfig(
            instance_arn="arn:aws:iam::123456789012:policy/example",
            name="example",
            quick_connect_config=QuickConnect_QuickConnectConfig(quick_connect_type="example"),
            description="Managed by ubx.",
            tags=[QuickConnect_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="description" type="string">
  The description of the quick connect. Optional.
</ResponseField>

<ResponseField name="instance_arn" type="string" required>
  The identifier of the Amazon Connect instance. Required.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the quick connect. Required.
</ResponseField>

<ResponseField name="quick_connect_config" type="object" required>
  Configuration settings for the quick connect. Required.
  <Expandable title="properties">
    <ResponseField name="flow_config" type="object">
      The flow configuration. This is required only if QuickConnectType is FLOW. Optional.
      <Expandable title="properties">
        <ResponseField name="contact_flow_arn" type="string" required>
          The identifier of the contact flow. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="phone_config" type="object">
      The phone configuration. This is required only if QuickConnectType is PHONE_NUMBER. Optional.
      <Expandable title="properties">
        <ResponseField name="phone_number" type="string" required>
          The phone number in E.164 format. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="queue_config" type="object">
      The queue configuration. This is required only if QuickConnectType is QUEUE. Optional.
      <Expandable title="properties">
        <ResponseField name="contact_flow_arn" type="string" required>
          The identifier of the contact flow. Required.
        </ResponseField>
        <ResponseField name="queue_arn" type="string" required>
          The identifier for the queue. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="quick_connect_type" type="string" required>
      The type of quick connect. In the Amazon Connect console, when you create a quick connect, you are prompted to assign one of the following types: Agent (USER), External (PHONE_NUMBER), or Queue (QUEUE). Required.
    </ResponseField>
    <ResponseField name="user_config" type="object">
      The user configuration. This is required only if QuickConnectType is USER. Optional.
      <Expandable title="properties">
        <ResponseField name="contact_flow_arn" type="string" required>
          The identifier of the contact flow. Required.
        </ResponseField>
        <ResponseField name="user_arn" type="string" required>
          The identifier of the user. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tags" type="list(object)">
  One or more tags. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The key of a tag attached to the Amazon Connect quick connect resource, used to categorize and organize the resource. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="quick_connect_arn" type="string">
  The Amazon Resource Name (ARN) for the quick connect. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="quick_connect_type" type="string">
  The type of quick connect. In the Amazon Connect console, when you create a quick connect, you are prompted to assign one of the following types: Agent (USER), External (PHONE_NUMBER), or Queue (QUEUE). Computed by AWS after `ship`.
</ResponseField>
