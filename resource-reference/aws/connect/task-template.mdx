`aws_connect_task_template` defines the real fields an agent fills in to create or complete a task, a follow-up, a case, a callback, so every task of that type captures the same structured information instead of an agent typing free-form notes.

    ```python
    import ubx_sdk as ubx
    from ubx.aws.connect import TaskTemplate, TaskTemplateConfig
    from ubx.aws.connect.task_template import TaskTemplate_Constraints, TaskTemplate_Constraints_InvisibleFields, TaskTemplate_Constraints_InvisibleFields_Id, TaskTemplate_Defaults, TaskTemplate_Fields

    def describe():
        ubx.intent("example own task template")

        ubx.resource(TaskTemplate, "example", TaskTemplateConfig(
            instance_arn="arn:aws:iam::123456789012:policy/example",
            name="example",
            client_token="example",
            constraints=TaskTemplate_Constraints(invisible_fields=[TaskTemplate_Constraints_InvisibleFields(id=TaskTemplate_Constraints_InvisibleFields_Id(name="example-name"))]),
            contact_flow_arn="arn:aws:iam::123456789012:policy/example",
            defaults=[TaskTemplate_Defaults(default_value="example")],
            description="Managed by ubx.",
            fields=[TaskTemplate_Fields(description="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
