`aws_connect_contact_flow_module_alias` creates a real named pointer to a specific `aws_connect_contact_flow_module_version`, letting a calling contact flow be redirected to a new module version without changing the reference it calls.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	connect "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/connect"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own contact flow module alias"})

    		ubx.Resource(connect.ContactFlowModuleAlias, "ci-runner", connect.ContactFlowModuleAliasConfig{
    			ContactFlowModuleId:      "example",
    			ContactFlowModuleVersion: 1,
    			Name:                     "ci-runner",
    			Description:              "Managed by ubx.",
    		})
    	}))
    }
    ```
  </Tab>

    def describe():

        ubx.resource(ContactFlowModuleAlias, "ci-runner", ContactFlowModuleAliasConfig(

            contact_flow_module_version=1,

## Input properties

<ResponseField name="contact_flow_module_id" type="string" required>
  The identifier of the contact flow module (ARN) this alias is tied to. Required.
</ResponseField>

<ResponseField name="contact_flow_module_version" type="number" required>
  The version number of the contact flow module this alias points to. Required.
</ResponseField>

<ResponseField name="description" type="string">
  The description of the alias. Optional.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the alias. Required.
</ResponseField>

## Output properties

<ResponseField name="alias_id" type="string">
  The unique identifier of the alias. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="contact_flow_module_alias_arn" type="string">
  The identifier of the contact flow module alias (ARN). This is constructed from the ContactFlowModuleArn and AliasId. Computed by AWS after `ship`.
</ResponseField>
