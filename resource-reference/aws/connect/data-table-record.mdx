`aws_connect_data_table_record` is one real row of data in an `aws_connect_data_table`, the actual values a contact flow's Data Table block reads or writes at runtime.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	connect "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/connect"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own data table record"})

    		ubx.Resource(connect.DataTableRecord, "example", connect.DataTableRecordConfig{
    			DataTableArn:    "arn:aws:iam::123456789012:policy/example",
    			DataTableRecord: map[string]any{"values": []map[string]any{{"attribute_id": "example"}}},
    			InstanceArn:     "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
