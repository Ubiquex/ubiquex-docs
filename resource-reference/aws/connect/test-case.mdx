`aws_connect_test_case` defines a real, simulated voice or chat interaction, the starting contact flow, phone number, caller intent, and expected response, that Connect's own native testing and simulation capability runs and scores pass or fail, validating a contact flow change before it reaches real customers.

## Example

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    			Content:            "example",
    			InstanceArn:        "arn:aws:iam::123456789012:policy/example",
    			Name:               "example",

    			EntryPoint:         map[string]any{"chat_entry_point_parameters": map[string]any{"flow_id": "example"}},
    			InitializationData: "example",
    			Status:             "example",
    			Tags:               []map[string]any{{"key": "example"}},
    		})
    	}))
    }

    import ubx_sdk as ubx
    from ubx.aws.connect import TestCase, TestCaseConfig
    from ubx.aws.connect.test_case import TestCase_EntryPoint, TestCase_EntryPoint_ChatEntryPointParameters, TestCase_Tags

    def describe():
        ubx.intent("example own test case")

        ubx.resource(TestCase, "example", TestCaseConfig(
            content="example",
            instance_arn="arn:aws:iam::123456789012:policy/example",
            name="example",

            entry_point=TestCase_EntryPoint(chat_entry_point_parameters=TestCase_EntryPoint_ChatEntryPointParameters(flow_id="example")),
            initialization_data="example",
            status="example",
            tags=[TestCase_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
