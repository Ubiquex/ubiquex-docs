`aws_connect_predefined_attribute` defines a real, named attribute, with a fixed or free-text value type, that can be assigned to `aws_connect_user`s or contacts on an `aws_connect_instance` and referenced in routing logic.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	connect "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/connect"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own predefined attribute"})

    		ubx.Resource(connect.PredefinedAttribute, "example", connect.PredefinedAttributeConfig{
    			InstanceArn:            "arn:aws:iam::123456789012:policy/example",
    			Name:                   "example",
    			AttributeConfiguration: map[string]any{"enable_value_validation_on_association": true},
    			Purposes:               []string{"example"},
    			Values:                 map[string]any{"string_list": []string{"example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    if __name__ == "__main__":
        ubx.run("example", describe)

## Input properties

<ResponseField name="attribute_configuration" type="object">
  Custom metadata associated to a Predefined attribute that controls how the attribute behaves when used by upstream services. Optional.
  <Expandable title="properties">
    <ResponseField name="enable_value_validation_on_association" type="bool">
      Enables customers to enforce strict validation on the specific values that this predefined attribute can hold. Optional.
    </ResponseField>
    <ResponseField name="is_read_only" type="bool">
      Allows the predefined attribute to show up and be managed in the Amazon Connect UI. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="instance_arn" type="string" required>
  The identifier of the Amazon Connect instance. Required.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the predefined attribute. Required.
</ResponseField>

<ResponseField name="purposes" type="list(string)">
  The assigned purposes of the predefined attribute. Optional.
</ResponseField>

<ResponseField name="values" type="object">
  The values of a predefined attribute. Optional.
  <Expandable title="properties">
    <ResponseField name="string_list" type="list(string)">
      Predefined attribute values of type string list. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
