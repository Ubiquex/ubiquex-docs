`aws_connect_user_hierarchy_structure` defines a real, up-to-five-level organizational hierarchy, region, site, team, for an `aws_connect_instance`, the structure that `aws_connect_user_hierarchy_group`s are actually created within.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	connect "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/connect"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own user hierarchy structure"})

    		ubx.Resource(connect.UserHierarchyStructure, "example", connect.UserHierarchyStructureConfig{
    			InstanceArn:            "arn:aws:iam::123456789012:policy/example",
    			UserHierarchyStructure: map[string]any{"level_five": map[string]any{"name": "example-name"}},
    		})
    	}))
    }
    ```
