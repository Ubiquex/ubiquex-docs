`aws_connect_email_address` registers a real, dedicated email address an `aws_connect_instance` sends and receives customer email through, the email counterpart to a phone number for voice or an endpoint for chat.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	connect "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/connect"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own email address"})

    		ubx.Resource(connect.EmailAddress, "example", connect.EmailAddressConfig{
    			EmailAddress:        "example",
    			InstanceArn:         "arn:aws:iam::123456789012:policy/example",
    			AliasConfigurations: []map[string]any{{"email_address_arn": "example"}},
    			Description:         "Managed by ubx.",
    			DisplayName:         "example-display-name",
    			Tags:                []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    import ubx_sdk as ubx
    from ubx.aws.connect import EmailAddress, EmailAddressConfig

    def describe():
        ubx.intent("example own email address")

        ubx.resource(EmailAddress, "example", EmailAddressConfig(
            email_address="example",
            instance_arn="arn:aws:iam::123456789012:policy/example",
            alias_configurations=[{"email_address_arn": "example"}],
            description="Managed by ubx.",
            display_name="example-display-name",
            tags=[{"key": "example"}],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="alias_configurations" type="list(object)">
  List of alias configurations for the email address Optional.
  <Expandable title="properties">
    <ResponseField name="email_address_arn" type="string">
      The Amazon Resource Name (ARN) of an email address that is configured as an alias for the primary Connect email address. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="description" type="string">
  A description for the email address. Optional.
</ResponseField>

<ResponseField name="display_name" type="string">
  The display name for the email address. Optional.
</ResponseField>

<ResponseField name="email_address" type="string" required>
  Email address to be created for this instance Required.
</ResponseField>

<ResponseField name="instance_arn" type="string" required>
  The identifier of the Amazon Connect instance. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  One or more tags. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      The user-defined key for a tag attached to the AWS Connect email address, used to organize and manage the resource. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="value" type="string">
      The value of a tag attached to the Amazon Connect email address resource. Optional; if omitted, computed by AWS. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="email_address_arn" type="string">
  The identifier of the email address. Computed by AWS after `ship`.
</ResponseField>
