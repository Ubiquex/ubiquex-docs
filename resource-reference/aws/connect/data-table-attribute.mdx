`aws_connect_data_table_attribute` defines one real column, its name and data type, in an `aws_connect_data_table`, the schema an `aws_connect_data_table_record` populates values against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	connect "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/connect"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own data table attribute"})

    		ubx.Resource(connect.DataTableAttribute, "example", connect.DataTableAttributeConfig{
    			DataTableArn: "arn:aws:iam::123456789012:policy/example",
    			InstanceArn:  "arn:aws:iam::123456789012:policy/example",
    			Name:         "example",
    			ValueType:    "example",
    			Description:  "Managed by ubx.",
    			Primary:      true,
    			Validation:   map[string]any{"enum": map[string]any{"strict": true}},
    		})
    	}))
    }
    ```
