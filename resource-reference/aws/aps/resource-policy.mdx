`aws_aps_resource_policy` attaches a real resource-based access policy to an `aws_aps_workspace`, controlling which principals, including from other accounts, can read or write metrics to that managed Prometheus workspace.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	aps "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/aps"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resource policy"})

    		ubx.Resource(aps.ResourcePolicy, "example", aps.ResourcePolicyConfig{
    			PolicyDocument: "example",
    			WorkspaceArn:   "arn:aws:iam::123456789012:policy/example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.aws.aps import ResourcePolicy, ResourcePolicyConfig

    def describe():

        ubx.resource(ResourcePolicy, "example", ResourcePolicyConfig(
            policy_document="example",
            workspace_arn="arn:aws:iam::123456789012:policy/example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Properties

<ResponseField name="policy_document" type="string" required>
  The JSON to use as the Resource-based Policy. Required.
</ResponseField>

<ResponseField name="workspace_arn" type="string" required>
  The Arn of an APS Workspace that the PolicyDocument will be attached to. Required.
</ResponseField>
