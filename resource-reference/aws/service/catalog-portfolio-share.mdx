`aws_service_catalog_portfolio_share` shares a real `aws_service_catalog_portfolio` with another AWS account or an AWS Organizations OU, the resource that actually initiates the share an `aws_service_catalog_accepted_portfolio_share` on the other side accepts.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	service "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/service"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own catalog portfolio share"})

    		ubx.Resource(service.CatalogPortfolioShare, "ci-runner", service.CatalogPortfolioShareConfig{
    			AccountId:       "example",
    			PortfolioId:     "example",
    			AcceptLanguage:  "example",
    			ShareTagOptions: true,
    		})
    	}))
    }
    ```
