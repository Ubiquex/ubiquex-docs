`aws_service_catalog_tag_option_association` attaches a real `aws_service_catalog_tag_option` to a specific product or `aws_service_catalog_portfolio`, the resource that actually makes that governed tag available to be applied there.

## Example

<Tabs>

    ```go
    // ubx sdk gen --only aws --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-aws/sdk/go => ./local-sdk/aws/sdk/go
    package main

    import (
    	service "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/service"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own catalog tag option association"})

    		ubx.Resource(service.CatalogTagOptionAssociation, "ci-runner", service.CatalogTagOptionAssociationConfig{
    			ResourceId:  "example",
    			TagOptionId: "example",
    		})
    	}))
    }
    ```
