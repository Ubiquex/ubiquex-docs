`aws_config_connector` connects a real third-party cloud provider's own resources to AWS Config, extending Config's compliance evaluation and configuration history beyond AWS's own resources to a genuinely multi-cloud inventory.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	config "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/config"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own connector"})

    		ubx.Resource(config.Connector, "example", config.ConnectorConfig{
    			ConnectorConfiguration: map[string]any{"azure": map[string]any{"client_identifier": "example", "tenant_identifier": "example"}},
    			Tags:                   []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Connector } from "@ubx/sdk-aws/aws/config/connector";

    export default stack("example", () => {
      intent({ summary: "example own connector" });

      resource(Connector, "example", {
        connectorConfiguration: {
          azure: { clientIdentifier: "example", tenantIdentifier: "example" },
        },
        tags: [{ key: "example" }],
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.aws.config import Connector, ConnectorConfig
    from ubx.aws.config.connector import Connector_ConnectorConfiguration, Connector_ConnectorConfiguration_Azure, Connector_Tags

    def describe():
        ubx.intent("example own connector")

        ubx.resource(Connector, "example", ConnectorConfig(
            connector_configuration=Connector_ConnectorConfiguration(azure=Connector_ConnectorConfiguration_Azure(client_identifier="example", tenant_identifier="example")),
            tags=[Connector_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>

## Input properties

<ResponseField name="connector_configuration" type="object" required>
  The configuration for the connector. Specify the third-party cloud provider configuration. Required.
  <Expandable title="properties">
    <ResponseField name="azure" type="object">
      Configuration for connecting to Microsoft Azure. Optional.
      <Expandable title="properties">
        <ResponseField name="client_identifier" type="string" required>
          The Azure client (application) identifier. Required.
        </ResponseField>
        <ResponseField name="tenant_identifier" type="string" required>
          The Azure tenant identifier. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tags" type="list(object)">
  The tags for the connector. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="arn" type="string">
  The Amazon Resource Name (ARN) of the connector. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="created_time" type="string">
  The time at which the connector was created. Computed by AWS after `ship`.
</ResponseField>

<ResponseField name="name" type="string">
  The name of the connector. AWS Config automatically assigns the name when creating the Connector. Computed by AWS after `ship`.
</ResponseField>
