`aws_config_aggregation_authorization` grants a real, specific source account and Region permission to aggregate this account's AWS Config data into their own `aws_config_configuration_aggregator`, the authorization step required before cross-account aggregation can actually collect this account's data.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	config "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/config"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own aggregation authorization"})

    		ubx.Resource(config.AggregationAuthorization, "example", config.AggregationAuthorizationConfig{
    			AuthorizedAccountId: "example",
    			AuthorizedAwsRegion: "example",
    			Tags:                []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    import ubx_sdk as ubx
    from ubx.aws.config import AggregationAuthorization, AggregationAuthorizationConfig
    from ubx.aws.config.aggregation_authorization import AggregationAuthorization_Tags

    def describe():
        ubx.intent("example own aggregation authorization")

            authorized_account_id="example",
            authorized_aws_region="example",
            tags=[AggregationAuthorization_Tags(key="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="authorized_account_id" type="string" required>
  The 12-digit account ID of the account authorized to aggregate data. Required.
</ResponseField>

<ResponseField name="authorized_aws_region" type="string" required>
  The region authorized to collect aggregated data. Required.
</ResponseField>

<ResponseField name="tags" type="list(object)">
  The tags for the AggregationAuthorization. Optional.
  <Expandable title="properties">
    <ResponseField name="key" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
    <ResponseField name="value" type="string">
      Optional; if omitted, computed by AWS.
    </ResponseField>
  </Expandable>
</ResponseField>

## Output properties

<ResponseField name="aggregation_authorization_arn" type="string">
  The ARN of the AggregationAuthorization. Computed by AWS after `ship`.
</ResponseField>
