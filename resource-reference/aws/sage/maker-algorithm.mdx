`aws_sage_maker_algorithm` registers a real, reusable training algorithm, packaged as a Docker image with defined input channels and hyperparameters, that can be published to AWS Marketplace or reused across SageMaker training jobs without re-specifying the underlying container each time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	sage "github.com/ubiquex/ubx-sdk-aws/sdk/go/aws/sage"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own maker algorithm"})

    		ubx.Resource(sage.MakerAlgorithm, "example", sage.MakerAlgorithmConfig{
    			AlgorithmName:          "example-algorithm-name",
    			TrainingSpecification:  map[string]any{"supported_training_instance_types": []string{"example"}},
    			AlgorithmDescription:   "example",
    			CertifyForMarketplace:  true,
    			InferenceSpecification: map[string]any{"containers": []map[string]any{{"container_hostname": "example"}}},
    			Tags:                   []map[string]any{{"key": "example"}},
    		})
    	}))
    }
    ```
