---

`aws_sage_maker_model` registers a real trained model, its inference container image and model artifacts in S3, the resource an `aws_sage_maker_endpoint_config` references to actually serve predictions.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	sagemaker "github.com/ubiquex/ubx-sdk-aws/sdk/go/v2/aws/sagemaker"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own model"})

    		ubx.Resource(sagemaker.Model, "example", sagemaker.ModelConfig{
    			Containers:               []map[string]any{{"container_hostname": "example"}},
    			EnableNetworkIsolation:   true,
    			ExecutionRoleArn:         "arn:aws:iam::123456789012:policy/example",
    			InferenceExecutionConfig: map[string]any{"mode": "example"},
    			ModelName:                "example-model-name",
    			PrimaryContainer:         map[string]any{"container_hostname": "example"},
    			Tags:                     []map[string]any{{"key": "example"}},
    			VpcConfig:                map[string]any{"security_group_ids": []string{"example"}, "subnets": []string{"example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.aws.sagemaker import Model, ModelConfig
    from ubx.aws.sagemaker.model import Model_Containers, Model_InferenceExecutionConfig, Model_Tags, Model_VpcConfig

    def describe():
