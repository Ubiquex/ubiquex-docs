`kubernetes_node_runtime_class` selects which real container runtime configuration a pod actually runs under -- standard runc, or a sandboxed alternative like gVisor or Kata Containers -- when a cluster's own nodes support more than one.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	node "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/node"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own runtime class"})

    		ubx.Resource(node.RuntimeClass, "example", node.RuntimeClassConfig{
    			Handler: "example",
    			Name:    "example",
    		})
    	}))
    }
    ```
  </Tab>
