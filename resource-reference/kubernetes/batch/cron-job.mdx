`kubernetes_batch_cron_job` creates a real Job on a recurring schedule, expressed in standard cron syntax -- the mechanism behind scheduled backups, report generation, and any other task that needs to run repeatedly rather than once.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	batch "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/batch"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own cron job"})

    		ubx.Resource(batch.CronJob, "example", batch.CronJobConfig{
    			Name:      "example",
    			Namespace: "example",
    			Spec:      map[string]any{"job_template": map[string]any{"metadata": map[string]any{"name": "example-name"}}, "schedule": "example"},
    		})
    	}))
    }
    ```
