`kubernetes_rbac_cluster_role_binding` grants a real ClusterRole's own permissions to a specific user, group, or service account across the whole cluster -- the RBAC resource that actually activates a ClusterRole; the role alone grants nothing on its own.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	rbac "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/rbac"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own cluster role binding"})

    		ubx.Resource(rbac.ClusterRoleBinding, "example", rbac.ClusterRoleBindingConfig{
    			Name:    "example",
    			RoleRef: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>
