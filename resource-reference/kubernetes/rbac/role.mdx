`kubernetes_rbac_role` defines a real set of permissions scoped to one specific namespace -- `edit deployments in this namespace only`, not the whole cluster.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only kubernetes --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-kubernetes/sdk/go => ./local-sdk/kubernetes/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	rbac "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/rbac"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own role"})

    		ubx.Resource(rbac.Role, "ci-runner", rbac.RoleConfig{
    			Name:      "ci-runner",
    			Namespace: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only kubernetes --lang ts --out ./local-sdk

        name: "ci-runner",

    ```
  </Tab>
