`kubernetes_manifest` -- real, typed bindings generated directly from the
`hashicorp/kubernetes` provider schema, in every SDK language.

Every tab below is a complete, runnable program, not a fragment,
real enough to save and run exactly as shown.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	manifest "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/manifest"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own manifest"})

    		ubx.Resource(manifest.Manifest, "ci-runner", manifest.ManifestConfig{
    			Manifest:       "example",
    			ComputedFields: []string{"example"},
    			WaitFor:        map[string]any{"fields": map[string]string{"key": "value"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Manifest } from "jsr:@ubx/sdk-kubernetes/kubernetes/manifest/manifest";

    export default stack("payments", () => {
      intent({ summary: "payments own manifest" });

      resource(Manifest, "ci-runner", {
        manifest: "example",
        computedFields: ["example"],
        waitFor: { fields: { key: "value" } },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.kubernetes.manifest import Manifest, ManifestConfig

    def describe():
        ubx.intent("payments own manifest")

        ubx.resource(Manifest, "ci-runner", ManifestConfig(
            manifest="example",
            computed_fields=["example"],
            wait_for={"fields": {"key": "value"}},
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
  </Tab>
  <Tab title="Markdown">
    ```
    Create kubernetes_manifest, with manifest example, computed
    fields a set value, wait for a set value.
    ```
  </Tab>
</Tabs>
