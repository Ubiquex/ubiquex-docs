`kubernetes_manifest` -- real, typed bindings generated directly from the
`hashicorp/kubernetes` provider schema, in every SDK language.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    import (
        "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/manifest"
    )

    cfg := manifest.ManifestConfig{
        Manifest: "example",
        ComputedFields: []string{"example"},
        WaitFor: map[string]any{"fields": map[string]string{"key": "value"}},
    }
    ubx.Resource(manifest.Manifest, "example", cfg)
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { Manifest } from
      "jsr:@ubx/sdk-kubernetes/kubernetes/manifest/manifest";

    resource(Manifest, "example", {
      manifest: "example",
      computedFields: ["example"],
      waitFor: { fields: { key: "value" } },
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    from ubx.kubernetes.manifest import Manifest, ManifestConfig

    cfg = ManifestConfig(
        manifest="example",
        computed_fields=["example"],
        wait_for={"fields": {"key": "value"}},
    )
    ubx.resource(Manifest, "example", cfg)
    ```
  </Tab>
