The v1beta1 shape of `kubernetes_certificates_pod_certificate_request` -- PodCertificateRequest's own beta API generation (introduced Kubernetes 1.35), kept being served alongside the now-stable v1 API kubelet itself has since moved onto. New certificate requests should target the stable `kubernetes_certificates_pod_certificate_request` instead.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	certificates "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/certificates"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta1 pod certificate request"})

    		ubx.Resource(certificates.V1beta1PodCertificateRequest, "example", certificates.V1beta1PodCertificateRequestConfig{
    			Name:      "example",
    			Namespace: "example",
    			Spec:      map[string]any{"node_name": "example-node-name"},
    		})
    	}))
    }
    ```
