`kubernetes_certificates_pod_certificate_request` is what kubelet creates on a pod's own behalf to request a real certificate from a specific signer, backing the `podCertificate` projected volume type -- Kubernetes itself never issues the certificate, an external controller watching these requests does.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	certificates "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/certificates"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own pod certificate request"})

    		ubx.Resource(certificates.PodCertificateRequest, "example", certificates.PodCertificateRequestConfig{
    			Name:      "example",
    			Namespace: "example",
    			Spec:      map[string]any{"node_name": "example-node-name"},
    		})
    	}))
    }
    ```
