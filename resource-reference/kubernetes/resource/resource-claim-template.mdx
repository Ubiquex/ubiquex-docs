`kubernetes_resource_resource_claim_template` is a real, reusable template for generating a ResourceClaim automatically per pod -- so a Deployment with many replicas doesn't need a hand-written claim for every single one.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only kubernetes --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-kubernetes/sdk/go => ./local-sdk/kubernetes/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	resource "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/resource"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resource claim template"})

    		ubx.Resource(resource.ResourceClaimTemplate, "example", resource.ResourceClaimTemplateConfig{
    			Name:      "example",
    			Namespace: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only kubernetes --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";

    export default stack("example", () => {

      });
    ```
