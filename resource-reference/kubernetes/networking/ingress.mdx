`kubernetes_networking_ingress` routes external HTTP(S) traffic to real Services inside the cluster based on hostname and path -- a single real entry point that can front many different backend Services without giving each one its own external load balancer.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networking "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/networking"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own ingress"})

    		ubx.Resource(networking.Ingress, "example", networking.IngressConfig{
    			Name:      "example",
    			Namespace: "example",
    		})
    	}))
    }
    ```
  </Tab>
