`kubernetes_apps_controller_revision` is a real, immutable snapshot of a StatefulSet or DaemonSet's own pod template at one point in time -- what a rolling update actually rolls back to if it needs to undo a change.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only kubernetes --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-kubernetes/sdk/go => ./local-sdk/kubernetes/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apps "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/apps"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own controller revision"})

    		ubx.Resource(apps.ControllerRevision, "ci-runner", apps.ControllerRevisionConfig{
    			Data:      "example",
    			Name:      "ci-runner",
    			Namespace: "example",
    		})
    	}))
    }
    ```
