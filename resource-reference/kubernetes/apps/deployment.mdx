`kubernetes_apps_deployment` manages a real, replicated set of identical pods and their own rolling update, declaratively -- change the pod template, and the Deployment controller works out how to get from the old real state to the new one without downtime.

    import (

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own deployment"})

    		ubx.Resource(apps.Deployment, "example", apps.DeploymentConfig{

    	}))
