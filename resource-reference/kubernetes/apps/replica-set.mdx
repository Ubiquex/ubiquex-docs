`kubernetes_apps_replica_set` maintains a real, stable count of identical running pods -- the mechanism a Deployment actually manages underneath to implement its own rolling updates, rarely created directly since a Deployment already does that job.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own replica set"})

    		})
    	}))
    }
