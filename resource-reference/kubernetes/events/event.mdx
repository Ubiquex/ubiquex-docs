`kubernetes_events_event` is the same real event record as the original core Event resource, exposed through the newer, dedicated `events.k8s.io` API group -- a real schema cleanup, not a different concept.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	events "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/events"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own event"})

    		ubx.Resource(events.Event, "example", events.EventConfig{
    			EventTime: "example",
    			Name:      "example",
    			Namespace: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript

    import { Event } from "@ubx/sdk-kubernetes/kubernetes/events/event";

    ```
  </Tab>

    import ubx_sdk as ubx

    if __name__ == "__main__":
        ubx.run("example", describe)
