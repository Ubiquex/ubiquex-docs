`kubernetes_certificates_pod_certificate_request` looks up a real request kubelet created on a pod's own behalf for a certificate from a specific signer, backing the `podCertificate` projected volume type.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only kubernetes --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-kubernetes/sdk/go => ./local-sdk/kubernetes/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	certificates "github.com/ubiquex/ubx-sdk-kubernetes/sdk/go/kubernetes/certificates"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own pod certificate request"})

    		ubx.Resource(certificates.PodCertificateRequest, "example", certificates.PodCertificateRequestConfig{
    			Name:      "example",
    			Namespace: "example",
    			Spec:      map[string]any{"node_name": "example-node-name", "node_uid": "example", "pod_name": "example-pod-name", "pod_uid": "example", "service_account_name": "example-service-account-name", "service_account_uid": "example", "signer_name": "example-signer-name", "stub_pkcs10_request": "example"},
    		})
    	}))
    }
    ```
