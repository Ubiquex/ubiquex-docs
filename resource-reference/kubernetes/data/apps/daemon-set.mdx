`kubernetes_apps_daemon_set` looks up a real, already-running DaemonSet -- one that runs exactly one copy of a pod on every node in the cluster (or every node matching a selector), the mechanism behind node-level agents like log collectors and monitoring daemons.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own daemon set"})

    		})
    	}))
    }
