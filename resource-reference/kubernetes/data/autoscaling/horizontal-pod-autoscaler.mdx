`kubernetes_autoscaling_horizontal_pod_autoscaler` looks up a real, already-configured HorizontalPodAutoscaler adjusting a Deployment or StatefulSet's own replica count based on observed CPU, memory, custom, or external metrics. This is the `autoscaling/v2` shape; `v1` (`kubernetes_autoscaling_v1_horizontal_pod_autoscaler`), still served for compatibility, only ever supports a single CPU-utilization target.

    		})
    	}))
    }

    import ubx_sdk as ubx
    from ubx.kubernetes.autoscaling import HorizontalPodAutoscaler, HorizontalPodAutoscalerConfig
    from ubx.kubernetes.autoscaling.horizontal_pod_autoscaler import HorizontalPodAutoscaler_Spec, HorizontalPodAutoscaler_Spec_Metrics_Object_DescribedObject

        ubx.intent("example own horizontal pod autoscaler")

            namespace="example",
            spec=HorizontalPodAutoscaler_Spec(max_replicas=1, scale_target_ref=HorizontalPodAutoscaler_Spec_Metrics_Object_DescribedObject(kind="example", name="example-name")),

    if __name__ == "__main__":
        ubx.run("example", describe)
