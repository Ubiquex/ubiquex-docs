`github_codespace` is a cloud-hosted development environment running in a real Docker container, reachable from a browser, VS Code, or the GitHub CLI with no local setup at all. Organizations customize the container itself via configuration files, so every codespace for a project starts from the same real, consistent environment.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	codespace "github.com/ubiquex/ubx-sdk-github/sdk/go/github/codespace"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own codespace"})

    		ubx.Resource(codespace.Codespace, "example", codespace.CodespaceConfig{
    			CodespaceName: "example-codespace-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only github --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";

    export default stack("example", () => {
      intent({ summary: "example own codespace" });

      resource(Codespace, "example", {

      });
    });
    ```
  </Tab>

    # ubx sdk gen --only github --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.codespace import Codespace, CodespaceConfig

    def describe():
        ubx.intent("example own codespace")

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
