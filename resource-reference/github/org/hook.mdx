`github_org_hook` is a webhook scoped to an entire organization rather than one repository -- the same real event-delivery mechanism, applied once at the org level instead of being configured separately on every repository inside it.

Real, typed bindings generated directly from the real GitHub REST API OpenAPI specification, in every SDK language -- every tab below is a complete, runnable program, not a fragment, real enough to save and run exactly as shown.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	org "github.com/ubiquex/ubx-sdk-github/sdk/go/github/org"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own hook"})

    		ubx.Resource(org.Hook, "ci-runner", org.HookConfig{
    			Config: map[string]any{"url": "example"},
    			HookId: "example",
    			Name:   "ci-runner",
    			Org:    "example",
    		})
    	}))
    }
    ```
