`github_actions_variable` -- real, typed bindings generated directly from
the real GitHub REST API OpenAPI specification, in every SDK language.
Every tab below is a complete, runnable program, not a fragment, real
enough to save and run exactly as shown.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	actions "github.com/ubiquex/ubx-sdk-github/sdk/go/github/actions"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own variable"})

    		ubx.Resource(actions.Variable, "ci-runner", actions.VariableConfig{
    			Name:  "ci-runner",
    			Owner: "example",
    			Repo:  "example",
    			Value: "example",
    		})
    	}))
    }
    ```
  </Tab>

    def describe():

            name="ci-runner",

            repo="example",
            value="example",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)

<ResponseField name="created_at" type="string">

</ResponseField>

<ResponseField name="updated_at" type="string">
