`github_projects_v2_field` is a real piece of metadata attached to items inside a Project -- a built-in field like assignee, or a custom one like priority -- adding context beyond what an issue or pull request already carries on its own, up to 50 fields per project.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	projects "github.com/ubiquex/ubx-sdk-github/sdk/go/github/projects"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v2 field"})

    		ubx.Resource(projects.V2Field, "example", projects.V2FieldConfig{
    			FieldId:       "example",
    			Org:           "example",
    			ProjectNumber: "example",
    		})
    	}))
    }
    ```
