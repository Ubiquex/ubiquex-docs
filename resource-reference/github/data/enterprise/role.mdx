`github_enterprise_role` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	enterprise "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/enterprise"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_enterprise_role"})

    		ubx.Data(enterprise.Role, "example", enterprise.RoleConfig{
    			Enterprise: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript

  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.github.data.enterprise import Role, RoleConfig

    def describe():
        ubx.intent("look up github_enterprise_role")

        ubx.data(Role, "example", RoleConfig(
            enterprise="example",

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
