`github_commit` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	commit "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/commit"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_commit"})

    		ubx.Data(commit.Commit, "example", commit.CommitConfig{
    			CommitSha: "example",
    			Owner:     "example",
    			Repo:      "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Commit } from "@ubx/sdk-github/github/data/commit/commit";

    export default stack("example", () => {

      data(Commit, "example", {
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.commit import Commit, CommitConfig

    def describe():
        ubx.intent("look up github_commit")

        ubx.data(Commit, "example", CommitConfig(
            commit_sha="example",
            owner="example",
            repo="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
