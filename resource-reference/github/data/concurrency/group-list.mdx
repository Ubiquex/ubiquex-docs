`github_concurrency_group_list` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_concurrency_group_list"})

    		ubx.Data(concurrency.GroupList, "example", concurrency.GroupListConfig{
    			Owner:   "example",
    			Repo:    "example",
    			After:   "example",
    			PerPage: 1,
    		})
    	}))
    }
