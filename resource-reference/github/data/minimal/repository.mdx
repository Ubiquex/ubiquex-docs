`github_minimal_repository` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	minimal "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/minimal"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_minimal_repository"})

    		ubx.Data(minimal.Repository, "example", minimal.RepositoryConfig{
    			Org:           "example",
    			RunnerGroupId: 1,
    			Page:          1,
    			PerPage:       1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

    export default stack("example", () => {
      intent({ summary: "look up github_minimal_repository" });

        perPage: 1,
      });

  </Tab>
  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.github.data.minimal import Repository, RepositoryConfig

    def describe():
        ubx.intent("look up github_minimal_repository")

        ubx.data(Repository, "example", RepositoryConfig(
            org="example",
            runner_group_id=1,
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
