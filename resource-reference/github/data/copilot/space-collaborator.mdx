    // ubx sdk gen --only github --lang go --out ./local-sdk

    import (

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_copilot_space_collaborator"})

    			SpaceNumber: 1,
    		})
    	}))
    }

    from ubx.github.data.copilot import SpaceCollaborator, SpaceCollaboratorConfig

    def describe():
        ubx.intent("look up github_copilot_space_collaborator")

        ubx.data(SpaceCollaborator, "example", SpaceCollaboratorConfig(
            org="example",
            space_number=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="org" type="string" required>
  The real organization this result belongs to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="space_number" type="number" required>
  The real, unique number identifying this Copilot Space. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="collaborators" type="list(any)">
  The list of collaborators for this Copilot Space. Returned by the lookup.
</ResponseField>
