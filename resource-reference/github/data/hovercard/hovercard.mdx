`github_hovercard` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	hovercard "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/hovercard"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_hovercard"})

    		ubx.Data(hovercard.Hovercard, "example", hovercard.HovercardConfig{
    			Username:    "example",
    			SubjectId:   "example",
    			SubjectType: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Hovercard } from "@ubx/sdk-github/github/data/hovercard/hovercard";

    export default stack("example", () => {
      intent({ summary: "look up github_hovercard" });

      data(Hovercard, "example", {
        username: "example",
        subjectId: "example",
        subjectType: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.hovercard import Hovercard, HovercardConfig

    def describe():
        ubx.intent("look up github_hovercard")

        ubx.data(Hovercard, "example", HovercardConfig(
            username="example",
            subject_id="example",
            subject_type="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
