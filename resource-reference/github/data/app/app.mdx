`github_app` looks up a real, registered GitHub App -- its own name, owner, and permissions -- as opposed to `github_installation`, which represents one specific install of an app on an account.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	app "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/app"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_app"})

    		ubx.Data(app.App, "example", app.AppConfig{
    			Page:    1,
    			PerPage: 1,
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.github.data.app import App, AppConfig

    def describe():
        ubx.intent("look up github_app")

        ubx.data(App, "example", AppConfig(
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
