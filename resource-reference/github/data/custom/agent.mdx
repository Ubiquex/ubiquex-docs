## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	custom "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/custom"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_custom_agent"})

    		ubx.Data(custom.Agent, "example", custom.AgentConfig{
    			Enterprise: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only github --lang ts --out ./local-sdk
    import { Agent } from "./local-sdk/github/sdk/typescript/github/data/custom/agent.ts";

    export default stack("example", () => {
      intent({ summary: "look up github_custom_agent" });

      data(Agent, "example", {
        enterprise: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only github --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.data.custom import Agent, AgentConfig

    def describe():
        ubx.intent("look up github_custom_agent")

        ubx.data(Agent, "example", AgentConfig(
            enterprise="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
