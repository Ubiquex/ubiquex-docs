`github_release` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	release "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/release"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_release"})

    		ubx.Data(release.Release, "example", release.ReleaseConfig{
    			Owner:     "example",
    			ReleaseId: 1,
    			Repo:      "example",
    			Page:      1,
    			PerPage:   1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Release } from "@ubx/sdk-github/github/data/release/release";

    export default stack("example", () => {
      intent({ summary: "look up github_release" });

      data(Release, "example", {

        releaseId: 1,
        repo: "example",
        page: 1,
        perPage: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.release import Release, ReleaseConfig

    def describe():
        ubx.intent("look up github_release")

        ubx.data(Release, "example", ReleaseConfig(
            owner="example",
            release_id=1,
            repo="example",
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
