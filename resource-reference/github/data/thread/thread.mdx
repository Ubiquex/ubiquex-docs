---
title: "github_thread"
description: "`github_thread` looks up a real notification thread -- the issue, pull request, or discussion it's about, and whether it's been read."
---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	thread "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/thread"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_thread"})

    		ubx.Data(thread.Thread, "example", thread.ThreadConfig{
    			ThreadId: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only github --lang ts --out ./local-sdk
    import { Thread } from "./local-sdk/github/sdk/typescript/github/data/thread/thread.ts";

    export default stack("example", () => {
      intent({ summary: "look up github_thread" });

      data(Thread, "example", {
        threadId: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only github --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.data.thread import Thread, ThreadConfig

    def describe():
        ubx.intent("look up github_thread")

        ubx.data(Thread, "example", ThreadConfig(
            thread_id=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

  The real, numeric ID of this notification thread. **(AI-inferred)** Required lookup argument.
</ResponseField>
