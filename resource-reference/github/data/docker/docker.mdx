## Example

  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	docker "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/docker"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_docker"})

    		ubx.Data(docker.Docker, "example", docker.DockerConfig{
    			Org: "example",
    		})
    	}))
    }
    ```
