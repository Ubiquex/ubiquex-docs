`github_check_run` looks up one real, already-created individual test result inside a check suite, created and updated by a GitHub App as it moves through `queued`, `in_progress`, and `completed`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	check "github.com/ubiquex/ubx-sdk-github/sdk/go/github/check"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own run"})

    		ubx.Resource(check.Run, "example", check.RunConfig{
    			CheckRunId: "example",
    			HeadSha:    "example",
    			Name:       "example",
    			Owner:      "example",
    			Repo:       "example",
    			Actions:    []map[string]any{{"description": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    import ubx_sdk as ubx
    from ubx.github.check import Run, RunConfig
    from ubx.github.check.run import Run_Actions

    def describe():
        ubx.intent("example own run")

        ubx.resource(Run, "example", RunConfig(
            check_run_id="example",
            head_sha="example",
            name="example",
            owner="example",
            repo="example",
            actions=[Run_Actions(description="example")],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="check_run_id" type="number" required>
  The real, numeric ID of this check run. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="owner" type="string" required>
  The real account (user or organization) that owns the repository. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="page" type="number">
  The page number of results to fetch, for a paginated response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="per_page" type="number">
  The number of results per page, for a paginated response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="repo" type="string" required>
  The real repository's own name, without its owner. **(AI-inferred)** Required lookup argument.
</ResponseField>
