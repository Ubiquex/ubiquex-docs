`github_check_run` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	check "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/check"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_check_run"})

    		ubx.Data(check.Run, "example", check.RunConfig{
    			CheckRunId: 1,
    			Owner:      "example",
    			Repo:       "example",
    			Page:       1,
    			PerPage:    1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.github.data.check import Run, RunConfig

    def describe():

        ubx.data(Run, "example", RunConfig(
            check_run_id=1,
            owner="example",
            repo="example",
            page=1,
            per_page=1,

    if __name__ == "__main__":
        ubx.run("example", describe)
