`github_installation` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	installation "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/installation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_installation"})

    		ubx.Data(installation.Installation, "example", installation.InstallationConfig{
    			Enterprise:     "example",
    			InstallationId: 1,
    			Org:            "example",
    			Page:           1,
    			PerPage:        1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    import ubx_sdk as ubx
    from ubx.github.data.installation import Installation, InstallationConfig

    def describe():
        ubx.intent("look up github_installation")

        ubx.data(Installation, "example", InstallationConfig(
            enterprise="example",
            installation_id=1,
            org="example",
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
