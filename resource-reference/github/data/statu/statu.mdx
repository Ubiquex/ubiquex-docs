## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	statu "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/statu"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_statu"})

    		ubx.Data(statu.Statu, "example", statu.StatuConfig{
    			Enterprise: "example",
    			JobId:      "example",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.github.data.statu import Statu, StatuConfig

    def describe():
        ubx.intent("look up github_statu")

        ubx.data(Statu, "example", StatuConfig(
            enterprise="example",
            job_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="enterprise" type="string" required>
  The real enterprise account this result belongs to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="job_id" type="string" required>
  The real, numeric ID of this Actions workflow job. **(AI-inferred)** Required lookup argument.
</ResponseField>
