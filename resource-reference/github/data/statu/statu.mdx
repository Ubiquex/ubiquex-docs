`github_statu` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	statu "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/statu"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_statu"})

    		ubx.Data(statu.Statu, "example", statu.StatuConfig{
    			Enterprise: "example",
    			JobId:      "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Statu } from "@ubx/sdk-github/github/data/statu/statu";

    export default stack("example", () => {

      data(Statu, "example", {

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.statu import Statu, StatuConfig

    def describe():
        ubx.intent("look up github_statu")

        ubx.data(Statu, "example", StatuConfig(
            enterprise="example",
            job_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
