`github_comment` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	comment "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/comment"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_comment"})

    		ubx.Data(comment.Comment, "example", comment.CommentConfig{
    			CommentId: 1,
    			Owner:     "example",
    			Repo:      "example",
    			Content:   "example",
    			Page:      1,
    			PerPage:   1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript

  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.github.data.comment import Comment, CommentConfig

    def describe():
        ubx.intent("look up github_comment")

        ubx.data(Comment, "example", CommentConfig(
            comment_id=1,
            owner="example",
            repo="example",
            content="example",
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
