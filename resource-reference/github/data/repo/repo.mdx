---
title: "github_repo"
description: "`github_repo` is a real, live lookup against GitHub, generated from github's own real schema -- read-only, never created or destroyed."
---

`github_repo` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	repo "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/repo"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_repo"})

    		ubx.Data(repo.Repo, "example", repo.RepoConfig{
    			Owner:      "example",
    			Repo:       "example",
    			CreatorId:  []string{"example"},
    			Direction:  "example",
    			IsArchived: true,
    			Page:       1,
    			PerPage:    1,
    			Since:      "example",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.github.data.repo import Repo, RepoConfig

    def describe():
        ubx.intent("look up github_repo")

        ubx.data(Repo, "example", RepoConfig(
            owner="example",
            repo="example",
            creator_id=["example"],
            direction="example",
            is_archived=True,
            page=1,

    if __name__ == "__main__":
        ubx.run("example", describe)
