`github_image` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	image "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/image"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_image"})

    		ubx.Data(image.Image, "example", image.ImageConfig{
    			Enterprise: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Image } from "@ubx/sdk-github/github/data/image/image";

    export default stack("example", () => {
      intent({ summary: "look up github_image" });

      data(Image, "example", {
        enterprise: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.image import Image, ImageConfig

    def describe():
        ubx.intent("look up github_image")

        ubx.data(Image, "example", ImageConfig(
            enterprise="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
