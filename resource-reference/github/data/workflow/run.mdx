`github_workflow_run` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    ```go

    import (
    	workflow "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/workflow"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_workflow_run"})

    		ubx.Data(workflow.Run, "example", workflow.RunConfig{
    			Owner:               "example",
    			Repo:                "example",
    			Actor:               "example",
    			Branch:              "example",
    			CheckSuiteId:        1,
    			Created:             "example",
    			Event:               "example",
    			ExcludePullRequests: true,
    		})
    	}))
    }
    ```
