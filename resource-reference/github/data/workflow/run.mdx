## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	workflow "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/workflow"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_workflow_run"})

    		ubx.Data(workflow.Run, "example", workflow.RunConfig{
    			Owner:               "example",
    			Repo:                "example",
    			Actor:               "example",
    			Branch:              "example",
    			CheckSuiteId:        1,
    			Created:             "example",
    			Event:               "example",
    			ExcludePullRequests: true,
    		})
    	}))
    }
    ```

    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.data.workflow import Run, RunConfig

    def describe():
        ubx.intent("look up github_workflow_run")

        ubx.data(Run, "example", RunConfig(
            owner="example",
            repo="example",
            actor="example",
            branch="example",
            check_suite_id=1,
            created="example",
            event="example",
            exclude_pull_requests=True,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="actor" type="string">
  The real user or app that performed this action. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="branch" type="string">
  The real branch name this result is scoped to. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="check_suite_id" type="number">
  The real, numeric ID of this check suite. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="created" type="string">
  A real qualifier restricting results by their own creation date. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="event" type="string">
  The real event that triggered this result. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="exclude_pull_requests" type="bool">
  Whether commits reachable only through open pull requests are excluded from these results. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="head_sha" type="string">
  The real SHA of the commit at the head of this check run or pull request. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="owner" type="string" required>
  The real account (user or organization) that owns the repository. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="page" type="number">
  The page number of results to fetch, for a paginated response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="per_page" type="number">
  The number of results per page, for a paginated response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="repo" type="string" required>
  The real repository's own name, without its owner. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="status" type="string">
  The real, current status of this object. **(AI-inferred)** Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="total_count" type="number">
  The total number of results matching the request, across every page. **(AI-inferred)** Returned by the lookup.
</ResponseField>

<ResponseField name="workflow_runs" type="list(object)">
  The real Actions workflow runs included in this result. **(AI-inferred)** Returned by the lookup.
</ResponseField>
