`github_review` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	review "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/review"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_review"})

    		ubx.Data(review.Review, "example", review.ReviewConfig{
    			Owner:      "example",
    			PullNumber: 1,
    			Repo:       "example",
    			ReviewId:   1,
    			Page:       1,
    			PerPage:    1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Review } from "@ubx/sdk-github/github/data/review/review";

    export default stack("example", () => {
      intent({ summary: "look up github_review" });

      data(Review, "example", {
        owner: "example",
        pullNumber: 1,
        repo: "example",
        reviewId: 1,
        page: 1,
        perPage: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.review import Review, ReviewConfig

    def describe():
        ubx.intent("look up github_review")

        ubx.data(Review, "example", ReviewConfig(
            owner="example",
            pull_number=1,
            repo="example",
            review_id=1,
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
