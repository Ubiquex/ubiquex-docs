`github_actions_hosted_runner` looks up a real, already-provisioned GitHub-hosted runner -- the virtual machine GitHub provisions and maintains to run one Actions job, Ubuntu, Windows, or macOS.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	actions "github.com/ubiquex/ubx-sdk-github/sdk/go/github/actions"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own hosted runner"})

    		ubx.Resource(actions.HostedRunner, "example", actions.HostedRunnerConfig{
    			Enterprise:     "example",
    			HostedRunnerId: "example",
    			Image:          map[string]any{"id": "example"},
    			Name:           "example",
    			RunnerGroupId:  1,
    			Size:           "example",
    			EnableStaticIp: true,
    		})
    	}))
    }
    ```

    # ubx sdk gen --only github --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.actions import HostedRunner, HostedRunnerConfig
    from ubx.github.actions.hosted_runner import HostedRunner_Image

    def describe():
        ubx.intent("example own hosted runner")

        ubx.resource(HostedRunner, "example", HostedRunnerConfig(
            enterprise="example",
            hosted_runner_id="example",
            image=HostedRunner_Image(id="example"),
            name="example",
            runner_group_id=1,
            size="example",
            enable_static_ip=True,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
