`github_user` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	user "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/user"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_user"})

    		ubx.Data(user.User, "example", user.UserConfig{
    			Page:    1,
    			PerPage: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { User } from "@ubx/sdk-github/github/data/user/user";

    export default stack("example", () => {
      intent({ summary: "look up github_user" });

      data(User, "example", {
        page: 1,
        perPage: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.user import User, UserConfig

    def describe():
        ubx.intent("look up github_user")

        ubx.data(User, "example", UserConfig(
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
