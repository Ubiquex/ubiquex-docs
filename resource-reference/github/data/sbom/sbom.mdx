---

`github_sbom` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	sbom "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/sbom"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_sbom"})

    		ubx.Data(sbom.Sbom, "example", sbom.SbomConfig{
    			Owner: "example",
    			Repo:  "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Sbom } from "@ubx/sdk-github/github/data/sbom/sbom";

    export default stack("example", () => {
      intent({ summary: "look up github_sbom" });

      data(Sbom, "example", {
        owner: "example",
        repo: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.sbom import Sbom, SbomConfig

    def describe():
        ubx.intent("look up github_sbom")

        ubx.data(Sbom, "example", SbomConfig(
            owner="example",
            repo="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
