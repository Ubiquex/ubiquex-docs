    package main

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_assignment"})

    		ubx.Data(assignment.Assignment, "example", assignment.AssignmentConfig{
    			AssignmentId: 1,
    			Page:         1,
    			PerPage:      1,
    		})
    	}))
    }
