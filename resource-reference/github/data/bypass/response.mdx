## Lookup arguments

<ResponseField name="bypass_request_number" type="number" required>
  The real, repository-scoped number identifying this bypass request. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="owner" type="string" required>
  The real account (user or organization) that owns the repository. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="repo" type="string" required>
  The real repository's own name, without its owner. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="created_at" type="string">
  The date and time the bypass request was created. Returned by the lookup.
</ResponseField>

<ResponseField name="data" type="list(object)">
  Data describing the push rules that are being requested to be bypassed. Returned by the lookup.
</ResponseField>

<ResponseField name="expires_at" type="string">
  The date and time the bypass request will expire. Returned by the lookup.
</ResponseField>

<ResponseField name="html_url" type="string">
  The URL to view the bypass request in a browser. Returned by the lookup.
</ResponseField>

<ResponseField name="id" type="number">
  The unique identifier of the bypass request. Returned by the lookup.
</ResponseField>

<ResponseField name="number" type="number">
  The number uniquely identifying the bypass request within its repository. Returned by the lookup.
</ResponseField>

<ResponseField name="organization" type="object">
  The organization associated with the repository the bypass request is for. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="id" type="number">
      The ID of the organization. Optional lookup argument.
    </ResponseField>
    <ResponseField name="name" type="string">
      The name of the organization. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="repository" type="object">
  The repository the bypass request is for. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="full_name" type="string">
      The full name of the repository the bypass request is for. Optional lookup argument.
    </ResponseField>
    <ResponseField name="id" type="number">
      The ID of the repository the bypass request is for. Optional lookup argument.
    </ResponseField>
    <ResponseField name="name" type="string">
      The name of the repository the bypass request is for. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="request_type" type="string">
  The type of request. Returned by the lookup.
</ResponseField>

<ResponseField name="requester" type="object">
  The user who requested the bypass. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="actor_id" type="number">

    <ResponseField name="actor_name" type="string">
      The name of the GitHub user who requested the bypass. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="requester_comment" type="string">

<ResponseField name="resource_identifier" type="string">

<ResponseField name="responses" type="list(object)">
  The responses to the bypass request. Returned by the lookup.

<ResponseField name="status" type="string">
