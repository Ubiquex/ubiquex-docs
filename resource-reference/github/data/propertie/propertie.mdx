## Example

    package main

    import (
    	propertie "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/propertie"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_propertie"})

    		ubx.Data(propertie.Propertie, "example", propertie.PropertieConfig{
    			Enterprise: "example",
    		})
    	}))
    }
