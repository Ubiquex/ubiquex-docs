---
title: "github_artifact"

---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	artifact "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/artifact"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_artifact"})

    		ubx.Data(artifact.Artifact, "example", artifact.ArtifactConfig{
    			Owner:   "example",
    			Repo:    "example",
    			Name:    "example",
    			Page:    1,
    			PerPage: 1,
    		})
    	}))
    }
    ```

    # ubx sdk gen --only github --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.data.artifact import Artifact, ArtifactConfig

    def describe():
        ubx.intent("look up github_artifact")

        ubx.data(Artifact, "example", ArtifactConfig(
            owner="example",
            repo="example",
            name="example",
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="name" type="string">
  The real, given name of this object. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="owner" type="string" required>
  The real account (user or organization) that owns the repository. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="page" type="number">
  The page number of results to fetch, for a paginated response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="per_page" type="number">
  The number of results per page, for a paginated response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="repo" type="string" required>
  The real repository's own name, without its owner. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="artifacts" type="list(object)">
  The real Actions build artifacts included in this result. **(AI-inferred)** Returned by the lookup.
</ResponseField>

<ResponseField name="total_count" type="number">
  The total number of results matching the request, across every page. **(AI-inferred)** Returned by the lookup.
</ResponseField>
