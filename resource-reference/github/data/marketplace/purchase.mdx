    import (
    	marketplace "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/marketplace"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_marketplace_purchase"})

    		ubx.Data(marketplace.Purchase, "example", marketplace.PurchaseConfig{
    			AccountId: 1,
    		})
    	}))
    }

    from ubx.github.data.marketplace import Purchase, PurchaseConfig

    def describe():
        ubx.intent("look up github_marketplace_purchase")

    if __name__ == "__main__":
        ubx.run("example", describe)
