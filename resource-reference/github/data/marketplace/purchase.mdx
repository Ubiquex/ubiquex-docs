`github_marketplace_purchase` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	marketplace "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/marketplace"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_marketplace_purchase"})

    		ubx.Data(marketplace.Purchase, "example", marketplace.PurchaseConfig{
    			AccountId: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Purchase } from "@ubx/sdk-github/github/data/marketplace/purchase";

    export default stack("example", () => {
      intent({ summary: "look up github_marketplace_purchase" });

      data(Purchase, "example", {
        accountId: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.marketplace import Purchase, PurchaseConfig

    def describe():
        ubx.intent("look up github_marketplace_purchase")

        ubx.data(Purchase, "example", PurchaseConfig(
            account_id=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
