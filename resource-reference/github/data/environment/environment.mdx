  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	environment "github.com/ubiquex/ubx-sdk-github/sdk/go/github/environment"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own environment"})

    		ubx.Resource(environment.Environment, "example", environment.EnvironmentConfig{
    			EnvironmentName:   "example-environment-name",
    			Owner:             "example",
    			Repo:              "example",
    			PreventSelfReview: true,
    			Reviewers:         []map[string]any{{"id": 1}},
    			WaitTimer:         1,
    		})
    	}))
    }
    ```

    ```python
    # ubx sdk gen --only github --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.environment import Environment, EnvironmentConfig
    from ubx.github.environment.environment import Environment_Reviewers

    def describe():
        ubx.intent("example own environment")

        ubx.resource(Environment, "example", EnvironmentConfig(
            environment_name="example-environment-name",
            owner="example",
            repo="example",
            prevent_self_review=True,
            reviewers=[Environment_Reviewers(id=1)],
            wait_timer=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="owner" type="string" required>
  The real account (user or organization) that owns the repository. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="page" type="number">
  The page number of results to fetch, for a paginated response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="per_page" type="number">
  The number of results per page, for a paginated response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="repo" type="string" required>
  The real repository's own name, without its owner. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="environments" type="list(object)">
  The real deployment environments included in this result. **(AI-inferred)** Returned by the lookup.
</ResponseField>

<ResponseField name="total_count" type="number">
  The number of environments in this repository Returned by the lookup.
</ResponseField>
