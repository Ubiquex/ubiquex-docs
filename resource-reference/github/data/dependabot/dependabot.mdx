`github_dependabot` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    ```go
    package main

    import (
    	dependabot "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/dependabot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_dependabot"})

    		ubx.Data(dependabot.Dependabot, "example", dependabot.DependabotConfig{
    			Enterprise:     "example",

    			Assignee:       "example",

    			Classification: "example",
    			Direction:      "example",

    		})
    	}))
    }
    ```
