`github_organization_full` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	organization "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/organization"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_organization_full"})

    		ubx.Data(organization.Full, "example", organization.FullConfig{
    			Org: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Full } from "@ubx/sdk-github/github/data/organization/full";

    export default stack("example", () => {
      intent({ summary: "look up github_organization_full" });

      data(Full, "example", {
        org: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.organization import Full, FullConfig

    def describe():
        ubx.intent("look up github_organization_full")

        ubx.data(Full, "example", FullConfig(
            org="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

</Tabs>
