## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	page "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/page"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_page"})

    		ubx.Data(page.Page, "example", page.PageConfig{
    			Owner: "example",
    			Repo:  "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```typescript
    export default stack("example", () => {
    ```

  <Tab title="Python">
    ```python
    # ubx sdk gen --only github --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.data.page import Page, PageConfig

    def describe():
        ubx.intent("look up github_page")

        ubx.data(Page, "example", PageConfig(
            owner="example",
            repo="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="owner" type="string" required>
  The real account (user or organization) that owns the repository. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="repo" type="string" required>
  The real repository's own name, without its owner. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="build_type" type="string">
  The process in which the Page will be built. Returned by the lookup.
</ResponseField>

<ResponseField name="cname" type="string">
  The Pages site's custom domain Returned by the lookup.
</ResponseField>

<ResponseField name="custom_404" type="bool">
  Whether the Page has a custom 404 page. Returned by the lookup.
</ResponseField>

<ResponseField name="html_url" type="string">
  The web address the Page can be accessed from. Returned by the lookup.
</ResponseField>

<ResponseField name="https_certificate" type="object">
  The real status of the HTTPS certificate provisioned for this custom domain. **(AI-inferred)** Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="description" type="string" required>
      Required lookup argument.
    </ResponseField>
    <ResponseField name="domains" type="list(string)" required>
      Array of the domain set and its alternate name (if it is configured) Required lookup argument.
    </ResponseField>
    <ResponseField name="expires_at" type="string">
      Optional lookup argument.
    </ResponseField>
    <ResponseField name="state" type="string" required>
      Required lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="https_enforced" type="bool">
  Whether https is enabled on the domain Returned by the lookup.
</ResponseField>

<ResponseField name="pending_domain_unverified_at" type="string">
  The timestamp when a pending domain becomes unverified. Returned by the lookup.
</ResponseField>

<ResponseField name="protected_domain_state" type="string">
  The state if the domain is verified Returned by the lookup.
</ResponseField>

<ResponseField name="public" type="bool">
  Whether the GitHub Pages site is publicly visible. If set to `true`, the site is accessible to anyone on the internet. If set to `false`, the site will only be accessible to users who have at least `read` access to the repository that published the site. Returned by the lookup.
</ResponseField>

<ResponseField name="source" type="object">
  The real source this object was derived or imported from. **(AI-inferred)** Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="branch" type="string" required>
      Required lookup argument.
    </ResponseField>
    <ResponseField name="path" type="string" required>
      Required lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="status" type="string">
  The status of the most recent build of the Page. Returned by the lookup.
</ResponseField>

<ResponseField name="url" type="string">
  The API address for accessing this Page resource. Returned by the lookup.
</ResponseField>
