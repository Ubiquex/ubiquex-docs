`github_page_build` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	page "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/page"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_page_build"})

    		ubx.Data(page.Build, "example", page.BuildConfig{
    			Owner: "example",
    			Repo:  "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Build } from "@ubx/sdk-github/github/data/page/build";

    export default stack("example", () => {
      intent({ summary: "look up github_page_build" });

      data(Build, "example", {
        owner: "example",
        repo: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.page import Build, BuildConfig

    def describe():
        ubx.intent("look up github_page_build")

        ubx.data(Build, "example", BuildConfig(
            owner="example",
            repo="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
