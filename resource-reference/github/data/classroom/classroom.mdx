`github_classroom` looks up a real GitHub Classroom -- the organization-linked container an instructor manages assignments and student repositories through.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	classroom "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/classroom"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_classroom"})

    		ubx.Data(classroom.Classroom, "example", classroom.ClassroomConfig{
    			ClassroomId: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Classroom } from "@ubx/sdk-github/github/data/classroom/classroom";

    export default stack("example", () => {
      intent({ summary: "look up github_classroom" });

      data(Classroom, "example", {
        classroomId: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.classroom import Classroom, ClassroomConfig

    def describe():
        ubx.intent("look up github_classroom")

        ubx.data(Classroom, "example", ClassroomConfig(
            classroom_id=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
