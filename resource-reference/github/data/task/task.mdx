`github_task` looks up one real repository's own already-recorded result within a broader Multi-Repository Variant Analysis run -- its analysis status, its result count, and the reason it failed if it did.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	task "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/task"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_task"})

    		ubx.Data(task.Task, "example", task.TaskConfig{
    			TaskId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.github.data.task import Task, TaskConfig

    def describe():

        ubx.data(Task, "example", TaskConfig(

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
