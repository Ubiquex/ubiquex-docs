`github_alert` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	alert "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/alert"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_alert"})

    		ubx.Data(alert.Alert, "example", alert.AlertConfig{
    			Owner:   "example",
    			Repo:    "example",
    			Page:    1,
    			PerPage: 1,
    			Pr:      1,
    			Ref:     "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.alert import Alert, AlertConfig

    def describe():
        ubx.intent("look up github_alert")

        ubx.data(Alert, "example", AlertConfig(
            owner="example",
            repo="example",
            page=1,
            per_page=1,
            pr=1,
            ref="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

<ResponseField name="owner" type="string" required>
  Required lookup argument.

<ResponseField name="page" type="number">
  Optional lookup argument.
</ResponseField>

<ResponseField name="per_page" type="number">
  Optional lookup argument.
</ResponseField>

<ResponseField name="pr" type="number">
  Optional lookup argument.
</ResponseField>

<ResponseField name="ref" type="string">
  The Git reference, formatted as `refs/pull/&lt;number&gt;/merge`, `refs/pull/&lt;number&gt;/head`, `refs/heads/&lt;branch name&gt;` or simply `&lt;branch name&gt;`. Optional lookup argument.
</ResponseField>

<ResponseField name="repo" type="string" required>
  Required lookup argument.
</ResponseField>
