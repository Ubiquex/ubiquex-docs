    from ubx.github.data.team import Team, TeamConfig

    def describe():
        ubx.intent("look up github_team")

        ubx.data(Team, "example", TeamConfig(
            enterprise="example",
            enterprise_team="example",
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

</ResponseField>

<ResponseField name="page" type="number">
  The page number of results to fetch, for a paginated response. **(AI-inferred)** Optional lookup argument.
</ResponseField>
