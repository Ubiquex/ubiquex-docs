title: "github_team_full"

---

`github_team_full` looks up a real, already-created group of organization members, along with the access permissions that cascade down through any nested teams.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	team "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/team"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_team_full"})

    		ubx.Data(team.Full, "example", team.FullConfig{
    			TeamId: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Full } from "@ubx/sdk-github/github/data/team/full";

    export default stack("example", () => {
      intent({ summary: "look up github_team_full" });

      data(Full, "example", {
        teamId: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.team import Full, FullConfig

    def describe():
        ubx.intent("look up github_team_full")

        ubx.data(Full, "example", FullConfig(
            team_id=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
