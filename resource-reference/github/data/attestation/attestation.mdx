## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	attestation "github.com/ubiquex/ubx-sdk-github/sdk/go/github/attestation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own attestation"})

    		ubx.Resource(attestation.Attestation, "example", attestation.AttestationConfig{
    			Bundle:        map[string]any{"dsse_envelope": "example"},
    			Owner:         "example",
    			Repo:          "example",
    			SubjectDigest: "example",
    		})
    	}))
    }
    ```
