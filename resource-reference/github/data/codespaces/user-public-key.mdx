---
title: "github_codespaces_user_public_key"
description: "`github_codespaces_user_public_key` reads the real public key used to encrypt a user-level Codespaces secret's value before it's sent to GitHub."
---

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_codespaces_user_public_key"})

    	}))
    }
