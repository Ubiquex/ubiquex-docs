---
title: "github_codespaces_secret"

---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	codespaces "github.com/ubiquex/ubx-sdk-github/sdk/go/github/codespaces"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own secret"})

    		ubx.Resource(codespaces.Secret, "example", codespaces.SecretConfig{
    			KeyId:                 "example",
    			SecretName:            "example-secret-name",
    			EncryptedValue:        "example",
    			SelectedRepositoryIds: []any{"example"},
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    # ubx sdk gen --only github --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.codespaces import Secret, SecretConfig

    def describe():
        ubx.intent("example own secret")

        ubx.resource(Secret, "example", SecretConfig(
            key_id="example",
            secret_name="example-secret-name",
            encrypted_value="example",
            selected_repository_ids=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
