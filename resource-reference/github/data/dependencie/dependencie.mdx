## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	dependencie "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/dependencie"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_dependencie"})

    		ubx.Data(dependencie.Dependencie, "example", dependencie.DependencieConfig{
    			IssueNumber: 1,
    			Owner:       "example",
    			Repo:        "example",
    			Page:        1,
    			PerPage:     1,
    		})
    	}))
    }
    ```

    # ubx sdk gen --only github --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.data.dependencie import Dependencie, DependencieConfig

    def describe():
        ubx.intent("look up github_dependencie")

        ubx.data(Dependencie, "example", DependencieConfig(
            issue_number=1,
            owner="example",
            repo="example",
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="issue_number" type="number" required>
  The real, repository-scoped number identifying this issue. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="owner" type="string" required>
  The real account (user or organization) that owns the repository. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="page" type="number">
  The page number of results to fetch, for a paginated response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="per_page" type="number">
  The number of results per page, for a paginated response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="repo" type="string" required>
  The real repository's own name, without its owner. **(AI-inferred)** Required lookup argument.
</ResponseField>
