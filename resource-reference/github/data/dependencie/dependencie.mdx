`github_dependencie` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dependencie "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/dependencie"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_dependencie"})

    		ubx.Data(dependencie.Dependencie, "example", dependencie.DependencieConfig{
    			IssueNumber: 1,
    			Owner:       "example",
    			Repo:        "example",
    			Page:        1,
    			PerPage:     1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.dependencie import Dependencie, DependencieConfig

    def describe():
        ubx.intent("look up github_dependencie")

        ubx.data(Dependencie, "example", DependencieConfig(
            issue_number=1,
            owner="example",
            repo="example",
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
