`github_content_file` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	content "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/content"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_content_file"})

    		ubx.Data(content.File, "example", content.FileConfig{
    			Owner: "example",
    			Repo:  "example",
    			Ref:   "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { File } from "@ubx/sdk-github/github/data/content/file";

    export default stack("example", () => {
      intent({ summary: "look up github_content_file" });

      data(File, "example", {
        owner: "example",
        repo: "example",
        ref: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.content import File, FileConfig

    def describe():
        ubx.intent("look up github_content_file")

        ubx.data(File, "example", FileConfig(
            owner="example",
            repo="example",
            ref="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
