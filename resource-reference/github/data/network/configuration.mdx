`github_network_configuration` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	network "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/network"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_network_configuration"})

    		ubx.Data(network.Configuration, "example", network.ConfigurationConfig{
    			Enterprise: "example",
    			Page:       1,
    			PerPage:    1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Configuration } from "@ubx/sdk-github/github/data/network/configuration";

    export default stack("example", () => {
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.network import Configuration, ConfigurationConfig

    def describe():

        ubx.data(Configuration, "example", ConfigurationConfig(
            enterprise="example",
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
