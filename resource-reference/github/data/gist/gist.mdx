`github_gist` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	gist "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/gist"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_gist"})

    		ubx.Data(gist.Gist, "example", gist.GistConfig{
    			Page:    1,
    			PerPage: 1,
    			Since:   "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Gist } from "@ubx/sdk-github/github/data/gist/gist";

    export default stack("example", () => {
      intent({ summary: "look up github_gist" });

      data(Gist, "example", {
        page: 1,
        perPage: 1,
        since: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.gist import Gist, GistConfig

    def describe():
        ubx.intent("look up github_gist")

        ubx.data(Gist, "example", GistConfig(
            page=1,
            per_page=1,
            since="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
