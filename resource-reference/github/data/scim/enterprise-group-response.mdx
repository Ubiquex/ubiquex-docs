`github_scim_enterprise_group_response` looks up a real, SCIM-provisioned group at the enterprise level, kept in sync by the enterprise's own identity provider.

## Example

    ```go
    	scim "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/scim"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_scim_enterprise_group_response"})

    		ubx.Data(scim.EnterpriseGroupResponse, "example", scim.EnterpriseGroupResponseConfig{
    			Enterprise:         "example",
    			Count:              1,
    			ExcludedAttributes: "example",
    			Filter:             "example",
    		})
    	}))
    }
    ```
