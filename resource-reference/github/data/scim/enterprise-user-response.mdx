`github_scim_enterprise_user_response` looks up a real, SCIM-provisioned enterprise member's own identity attributes as synced from the enterprise's identity provider.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	scim "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/scim"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_scim_enterprise_user_response"})

    		ubx.Data(scim.EnterpriseUserResponse, "example", scim.EnterpriseUserResponseConfig{
    			Enterprise: "example",
    			Count:      1,
    			Filter:     "example",
    		})
    	}))
    }
    ```
