`github_hosted_runner` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	hosted "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/hosted"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_hosted_runner"})

    		ubx.Data(hosted.Runner, "example", hosted.RunnerConfig{
    			Enterprise: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Runner } from "@ubx/sdk-github/github/data/hosted/runner";

    export default stack("example", () => {
      intent({ summary: "look up github_hosted_runner" });

      data(Runner, "example", {
        enterprise: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.hosted import Runner, RunnerConfig

    def describe():
        ubx.intent("look up github_hosted_runner")

        ubx.data(Runner, "example", RunnerConfig(
            enterprise="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

</Tabs>
