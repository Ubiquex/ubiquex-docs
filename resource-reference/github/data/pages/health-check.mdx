`github_pages_health_check` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	pages "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/pages"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_pages_health_check"})

    		ubx.Data(pages.HealthCheck, "example", pages.HealthCheckConfig{
    			Owner: "example",
    			Repo:  "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { HealthCheck } from "@ubx/sdk-github/github/data/pages/health_check";

    export default stack("example", () => {
      intent({ summary: "look up github_pages_health_check" });

      data(HealthCheck, "example", {
        owner: "example",
        repo: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.pages import HealthCheck, HealthCheckConfig

    def describe():

            owner="example",
            repo="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
