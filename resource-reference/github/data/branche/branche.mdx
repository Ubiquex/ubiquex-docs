`github_branche` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	branche "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/branche"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_branche"})

    		ubx.Data(branche.Branche, "example", branche.BrancheConfig{
    			Branch:  "example",
    			Owner:   "example",
    			Repo:    "example",
    			Page:    1,
    			PerPage: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.github.data.branche import Branche, BrancheConfig

    def describe():
        ubx.intent("look up github_branche")

        ubx.data(Branche, "example", BrancheConfig(
            branch="example",
            owner="example",
            repo="example",
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
