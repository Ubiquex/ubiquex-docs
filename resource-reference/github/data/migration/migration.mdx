`github_migration` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	migration "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/migration"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_migration"})

    		ubx.Data(migration.Migration, "example", migration.MigrationConfig{
    			MigrationId: 1,
    			Org:         "example",
    			Page:        1,
    			PerPage:     1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript

    import ubx_sdk as ubx
    from ubx.github.data.migration import Migration, MigrationConfig

    def describe():
        ubx.intent("look up github_migration")

        ubx.data(Migration, "example", MigrationConfig(
            migration_id=1,
            org="example",
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
