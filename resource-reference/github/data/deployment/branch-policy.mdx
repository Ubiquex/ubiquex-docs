`github_deployment_branch_policy` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

    ```go
    package main

    import (
    	deployment "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/deployment"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_deployment_branch_policy"})

    		ubx.Data(deployment.BranchPolicy, "example", deployment.BranchPolicyConfig{
    			EnvironmentName: "example-environment-name",
    			Owner:           "example",
    			Repo:            "example",
    			Page:            1,
    			PerPage:         1,
    		})
    	}))
    }
    ```
