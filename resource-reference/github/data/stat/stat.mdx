---

`github_stat` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against GitHub's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	stat "github.com/ubiquex/ubx-sdk-github/sdk/go/github/data/stat"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up github_stat"})

    		ubx.Data(stat.Stat, "example", stat.StatConfig{
    			Owner: "example",
    			Repo:  "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Stat } from "@ubx/sdk-github/github/data/stat/stat";

    export default stack("example", () => {
      intent({ summary: "look up github_stat" });

      data(Stat, "example", {
        owner: "example",
        repo: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.github.data.stat import Stat, StatConfig

    def describe():
        ubx.intent("look up github_stat")

        ubx.data(Stat, "example", StatConfig(
            owner="example",
            repo="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
