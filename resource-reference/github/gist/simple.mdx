`github_gist_simple` is a real, shareable code snippet that's actually its own git repository underneath -- forkable and clonable like any other -- published either public (discoverable and searchable) or secret (reachable only by anyone who has the exact URL).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	gist "github.com/ubiquex/ubx-sdk-github/sdk/go/github/gist"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own simple"})

    		ubx.Resource(gist.Simple, "example", gist.SimpleConfig{
    			Files:  map[string]any{"key": map[string]any{"content": "example"}},
    			GistId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
