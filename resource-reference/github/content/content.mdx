## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	content "github.com/ubiquex/ubx-sdk-github/sdk/go/github/content"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own content"})

    		ubx.Resource(content.Content, "example", content.ContentConfig{
    			Content:   "example",
    			Message:   "example",
    			Owner:     "example",
    			Path:      "/example/",
    			Repo:      "example",
    			Author:    map[string]any{"email": "example", "name": "example-name"},
    			Branch:    "example",
    			Committer: map[string]any{"email": "example", "name": "example-name"},
    		})
    	}))
    }
    ```

    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.content import Content, ContentConfig
    from ubx.github.content.content import Content_Author

    def describe():
        ubx.intent("example own content")

        ubx.resource(Content, "example", ContentConfig(
            content="example",
            message="example",
            owner="example",
            path="/example/",
            repo="example",
            author=Content_Author(email="example", name="example-name"),
            branch="example",
            committer=Content_Author(email="example", name="example-name"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Properties

<ResponseField name="author" type="object">
  The author of the file. Default: The `committer` or the authenticated user if you omit `committer`. Optional.
  <Expandable title="properties">
    <ResponseField name="date" type="string">
      Optional.
    </ResponseField>
    <ResponseField name="email" type="string" required>
      The email of the author or committer of the commit. You'll receive a `422` status code if `email` is omitted. Required.
    </ResponseField>
    <ResponseField name="name" type="string" required>
      The name of the author or committer of the commit. You'll receive a `422` status code if `name` is omitted. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="branch" type="string">
  The branch name. Default: the repository’s default branch. Optional.
</ResponseField>

<ResponseField name="committer" type="object">
  The person that committed the file. Default: the authenticated user. Optional.
  <Expandable title="properties">
    <ResponseField name="date" type="string">
      Optional.
    </ResponseField>
    <ResponseField name="email" type="string" required>
      The email of the author or committer of the commit. You'll receive a `422` status code if `email` is omitted. Required.
    </ResponseField>
    <ResponseField name="name" type="string" required>
      The name of the author or committer of the commit. You'll receive a `422` status code if `name` is omitted. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="content" type="string" required>
  The new file content, using Base64 encoding. Required.
</ResponseField>

<ResponseField name="message" type="string" required>
  The commit message. Required.
</ResponseField>

<ResponseField name="owner" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="path" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="repo" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="sha" type="string">
  **Required if you are updating a file**. The blob SHA of the file being replaced. Optional.
</ResponseField>
