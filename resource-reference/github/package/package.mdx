## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	package_ "github.com/ubiquex/ubx-sdk-github/sdk/go/github/package_"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own package"})

    		ubx.Resource(package_.Package, "example", package_.PackageConfig{
    			Org:         "example",
    			PackageName: "example-package-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only github --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Package } from "./local-sdk/github/sdk/typescript/github/package/package.ts";

    export default stack("example", () => {
      intent({ summary: "example own package" });

      resource(Package, "example", {
        org: "example",
        packageName: "example-package-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only github --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.package import Package, PackageConfig

    def describe():
        ubx.intent("example own package")

        ubx.resource(Package, "example", PackageConfig(
            org="example",
            package_name="example-package-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
