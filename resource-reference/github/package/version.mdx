## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only github --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-github/sdk/go => ./local-sdk/github/sdk/go
    package main

    import (
    	package_ "github.com/ubiquex/ubx-sdk-github/sdk/go/github/package_"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own version"})

    		ubx.Resource(package_.Version, "example", package_.VersionConfig{
    			Org:              "example",
    			PackageName:      "example-package-name",
    			PackageType:      "example",
    			PackageVersionId: "example",
    		})
    	}))
    }
    ```
  </Tab>

    # ubx sdk gen --only github --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/github/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.github.package import Version, VersionConfig

    def describe():
        ubx.intent("example own version")

        ubx.resource(Version, "example", VersionConfig(
            org="example",
            package_name="example-package-name",
            package_type="example",
            package_version_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
