`digitalocean_api_get_model_router_output` looks up a real, already-configured model router -- the policy that dispatches a Gradient AI Platform request across a set of fallback models in a defined failover order.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	gradientaiplatform "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/gradientaiplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own api get model router output"})

    		ubx.Resource(gradientaiplatform.ApiGetModelRouterOutput, "example", gradientaiplatform.ApiGetModelRouterOutputConfig{
    			Uuid:           "example",
    			Name:           "example",
    			Description:    "Managed by ubx.",
    			FallbackModels: []string{"example"},
    			Policies:       []map[string]any{{"custom_task": map[string]any{"name": "example-name"}}},
    			Regions:        []string{"example"},
    		})
    	}))
    }
    ```

<ResponseField name="description" type="string">
  Model router description Optional.

<ResponseField name="fallback_models" type="list(string)">
  At least one fallback model is required; order defines failover priority Optional.
</ResponseField>

<ResponseField name="name" type="string">
  Model router name: lowercase, at most 255 characters, only a-z, 0-9, and hyphens Optional.
</ResponseField>
