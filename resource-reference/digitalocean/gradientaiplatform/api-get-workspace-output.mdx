`digitalocean_api_get_workspace_output` looks up a real, already-created Gradient AI Platform workspace -- the grouping that organizes a set of related agents together.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own api get workspace output"})

    		ubx.Resource(gradientaiplatform.ApiGetWorkspaceOutput, "example", gradientaiplatform.ApiGetWorkspaceOutputConfig{

    			Description:   "Managed by ubx.",
    		})
    	}))
    }
