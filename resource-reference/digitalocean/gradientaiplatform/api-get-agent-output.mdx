`digitalocean_api_get_agent_output` looks up a real, already-created Gradient AI Platform agent -- the model, instructions, and attached knowledge bases that define one deployable AI assistant.

## Example

<Tabs>
  <Tab title="Go">
    ```go

    import (
    	gradientaiplatform "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/gradientaiplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own api get agent output"})

    		ubx.Resource(gradientaiplatform.ApiGetAgentOutput, "example", gradientaiplatform.ApiGetAgentOutputConfig{
    			Uuid:                 "example",
    			Name:                 "example",
    			AnthropicKeyUuid:     "example",
    			Description:          "Managed by ubx.",
    			Instruction:          "example",
    			KnowledgeBaseUuid:    []string{"example"},
    			McpServers:           []map[string]any{{"allowed_tools": []string{"example"}}},
    			ModelProviderKeyUuid: "example",
    		})
    	}))
    }
    ```
