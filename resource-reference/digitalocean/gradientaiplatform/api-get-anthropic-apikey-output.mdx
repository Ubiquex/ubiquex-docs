`digitalocean_api_get_anthropic_apikey_output` looks up a real, already-stored Anthropic API key registered with the Gradient AI Platform, so an agent can call Anthropic's own models without embedding the raw key in its own configuration.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	gradientaiplatform "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/gradientaiplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own api get anthropic apikey output"})

    		ubx.Resource(gradientaiplatform.ApiGetAnthropicApikeyOutput, "example", gradientaiplatform.ApiGetAnthropicApikeyOutputConfig{
    			ApiKeyUuid: "example",
    			Name:       "example",
    			ApiKey:     "example",
    		})
    	}))
    }
    ```
  </Tab>
