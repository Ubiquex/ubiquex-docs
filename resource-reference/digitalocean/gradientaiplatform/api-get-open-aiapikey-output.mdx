`digitalocean_api_get_open_aiapikey_output` looks up a real, already-stored OpenAI API key registered with the Gradient AI Platform, so an agent can call OpenAI's own models without embedding the raw key in its own configuration.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	gradientaiplatform "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/gradientaiplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own api get open aiapikey output"})

    		ubx.Resource(gradientaiplatform.ApiGetOpenAiapikeyOutput, "example", gradientaiplatform.ApiGetOpenAiapikeyOutputConfig{
    			ApiKeyUuid: "example",
    			Name:       "example",
    			ApiKey:     "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { ApiGetOpenAiapikeyOutput } from "@ubx/sdk-digitalocean/digitalocean/gradientaiplatform/api_get_open_aiapikey_output";

    export default stack("example", () => {
      intent({ summary: "example own api get open aiapikey output" });

      resource(ApiGetOpenAiapikeyOutput, "example", {
        apiKeyUuid: "example",
        name: "example",
        apiKey: "example",
      });
    });
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.digitalocean.gradientaiplatform import ApiGetOpenAiapikeyOutput, ApiGetOpenAiapikeyOutputConfig

    def describe():
        ubx.intent("example own api get open aiapikey output")

        ubx.resource(ApiGetOpenAiapikeyOutput, "example", ApiGetOpenAiapikeyOutputConfig(
            api_key_uuid="example",
            name="example",
            api_key="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
