`digitalocean_vpc_nat_gateway` looks up a real, already-created VPC NAT gateway -- the managed egress point that lets resources inside a VPC reach the public internet through one shared, defined address without each needing a public IP of its own.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	vpcnatgateways "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/vpcnatgateways"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own vpc nat gateway"})

    		ubx.Resource(vpcnatgateways.VpcNatGateway, "example", vpcnatgateways.VpcNatGatewayConfig{
    			Id:                 "example",
    			Name:               "example",
    			Region:             "example",
    			Size:               1,
    			Type:               "example",
    			Vpcs:               []map[string]any{{"default_gateway": true}},
    			IcmpTimeoutSeconds: 7200,
    			TcpTimeoutSeconds:  7200,
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.digitalocean.vpcnatgateways import VpcNatGateway, VpcNatGatewayConfig
    from ubx.digitalocean.vpcnatgateways.vpc_nat_gateway import VpcNatGateway_Vpcs

    def describe():
        ubx.intent("example own vpc nat gateway")

        ubx.resource(VpcNatGateway, "example", VpcNatGatewayConfig(
            id="example",
            name="example",
            region="example",
            size=1,
            type="example",

## Input properties

<ResponseField name="icmp_timeout_seconds" type="number">
  The ICMP timeout in seconds for the VPC NAT gateway. Optional.
</ResponseField>

</ResponseField>

</ResponseField>

<ResponseField name="region" type="string" required>

</ResponseField>

<ResponseField name="size" type="number" required>
  The size of the VPC NAT gateway. Required.
</ResponseField>

<ResponseField name="tcp_timeout_seconds" type="number">
  The TCP timeout in seconds for the VPC NAT gateway. Optional.
</ResponseField>

<ResponseField name="type" type="string" required>

<ResponseField name="udp_timeout_seconds" type="number">
