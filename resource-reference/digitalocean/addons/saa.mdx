`digitalocean_saa` looks up a real, already-enabled Add-on for a resource -- the association between an account's own resource and one of DigitalOcean's optional add-on services (such as Scheduled Snapshot Protection).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	addons "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/addons"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own saa"})

    		ubx.Resource(addons.Saa, "example", addons.SaaConfig{
    			ResourceUuid: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only digitalocean --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";

    export default stack("example", () => {

  </Tab>

    # ubx sdk gen --only digitalocean --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/digitalocean/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.digitalocean.addons import Saa, SaaConfig

    def describe():
        ubx.intent("example own saa")

        ubx.resource(Saa, "example", SaaConfig(
            resource_uuid="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
