## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	databases "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/databases"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own topic"})

    		ubx.Resource(databases.Topic, "example", databases.TopicConfig{
    			DatabaseClusterUuid: "example",
    			TopicName:           "example-topic-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only digitalocean --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Topic } from "./local-sdk/digitalocean/sdk/typescript/digitalocean/databases/topic.ts";

    export default stack("example", () => {
      intent({ summary: "example own topic" });

      resource(Topic, "example", {
        databaseClusterUuid: "example",
        topicName: "example-topic-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only digitalocean --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/digitalocean/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.digitalocean.databases import Topic, TopicConfig

    def describe():
        ubx.intent("example own topic")

        ubx.resource(Topic, "example", TopicConfig(
            database_cluster_uuid="example",
            topic_name="example-topic-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
