`digitalocean_database` looks up a real, already-created logical database inside a managed database cluster -- the individual database a client actually connects to, distinct from the cluster hosting it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	databases "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/databases"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own database"})

    		ubx.Resource(databases.Database, "example", databases.DatabaseConfig{
    			DatabaseClusterUuid: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only digitalocean --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Database } from "./local-sdk/digitalocean/sdk/typescript/digitalocean/databases/database.ts";

    export default stack("example", () => {
      intent({ summary: "example own database" });

      resource(Database, "example", {
        databaseClusterUuid: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only digitalocean --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/digitalocean/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.digitalocean.databases import Database, DatabaseConfig

    def describe():
        ubx.intent("example own database")

        ubx.resource(Database, "example", DatabaseConfig(
            database_cluster_uuid="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
