title: "digitalocean_db"

---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	databases "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/databases"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own db"})

    		ubx.Resource(databases.Db, "example", databases.DbConfig{
    			DatabaseClusterUuid: "example",
    			DatabaseName:        "example-database-name",
    			Name:                "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only digitalocean --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Db } from "./local-sdk/digitalocean/sdk/typescript/digitalocean/databases/db.ts";

    export default stack("example", () => {
      intent({ summary: "example own db" });

      resource(Db, "example", {
        databaseClusterUuid: "example",
        databaseName: "example-database-name",
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only digitalocean --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/digitalocean/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.digitalocean.databases import Db, DbConfig

    def describe():
        ubx.intent("example own db")

        ubx.resource(Db, "example", DbConfig(
            database_cluster_uuid="example",
            database_name="example-database-name",
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
