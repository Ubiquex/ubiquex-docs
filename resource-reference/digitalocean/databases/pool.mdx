`digitalocean_pool` looks up a real, already-created PgBouncer connection pool for a managed PostgreSQL cluster -- the pooling mode and size that let many client connections share a smaller number of real backend connections.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	databases "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/databases"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own pool"})

    		ubx.Resource(databases.Pool, "example", databases.PoolConfig{
    			DatabaseClusterUuid: "example",
    			Db:                  "example",
    			Mode:                "example",
    			Name:                "example",
    			PoolName:            "example-pool-name",
    			Size:                1,
    			Connection:          map[string]any{"database": "example"},
    			PrivateConnection:   map[string]any{"database": "example"},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.digitalocean.databases import Pool, PoolConfig
    from ubx.digitalocean.databases.pool import Pool_Connection

    def describe():
        ubx.intent("example own pool")

        ubx.resource(Pool, "example", PoolConfig(
            database_cluster_uuid="example",
            db="example",
            mode="example",
            name="example",
            pool_name="example-pool-name",
            size=1,
            connection=Pool_Connection(database="example"),
            private_connection=Pool_Connection(database="example"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

<ResponseField name="size" type="number" required>
  The desired size of the PGBouncer connection pool. The maximum allowed size is determined by the size of the cluster's primary node. 25 backend server connections are allowed for every 1GB of RAM. Three are reserved for maintenance. For example, a primary node with 1 GB of RAM allows for a maximum of 22 backend server connections while one with 4 GB would allow for 97. Note that these are shared across all connection pools in a cluster. Required.
</ResponseField>
