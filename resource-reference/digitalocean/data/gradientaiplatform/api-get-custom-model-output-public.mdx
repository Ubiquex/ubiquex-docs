`data_digitalocean_api_get_custom_model_output_public` looks up a real, already-imported custom model by ID -- one uploaded from an external source such as Hugging Face.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	gradientaiplatform "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/gradientaiplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_api_get_custom_model_output_public"})

    		ubx.Data(gradientaiplatform.ApiGetCustomModelOutputPublic, "example", gradientaiplatform.ApiGetCustomModelOutputPublicConfig{
    			Uuid: "example",
    		})
    	}))
    }
    ```
  </Tab>
