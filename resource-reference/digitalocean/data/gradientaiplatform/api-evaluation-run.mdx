## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	gradientaiplatform "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/gradientaiplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_api_evaluation_run"})

    		ubx.Data(gradientaiplatform.ApiEvaluationRun, "example", gradientaiplatform.ApiEvaluationRunConfig{
    			EvaluationTestCaseUuid:    "example",
    			EvaluationTestCaseVersion: 1,
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.digitalocean.data.gradientaiplatform import ApiEvaluationRun, ApiEvaluationRunConfig

    def describe():
        ubx.intent("look up digitalocean_api_evaluation_run")

        ubx.data(ApiEvaluationRun, "example", ApiEvaluationRunConfig(
            evaluation_test_case_uuid="example",
            evaluation_test_case_version=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="evaluation_test_case_uuid" type="string" required>
  The UUID of the evaluation test case these runs belong to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="evaluation_test_case_version" type="number">
  The version of the evaluation test case these runs were executed against. **(AI-inferred)** Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="evaluation_runs" type="list(object)">
  List of evaluation runs. Returned by the lookup.
</ResponseField>
