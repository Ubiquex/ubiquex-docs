title: "digitalocean_api_prompt"

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	gradientaiplatform "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/gradientaiplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_api_prompt"})

    		ubx.Data(gradientaiplatform.ApiPrompt, "example", gradientaiplatform.ApiPromptConfig{
    			EvaluationRunUuid: "example",
    			Page:              1,
    			PerPage:           1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.digitalocean.data.gradientaiplatform import ApiPrompt, ApiPromptConfig

    def describe():
        ubx.intent("look up digitalocean_api_prompt")

        ubx.data(ApiPrompt, "example", ApiPromptConfig(
            evaluation_run_uuid="example",
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="evaluation_run_uuid" type="string" required>
  The UUID of the evaluation run these prompt results belong to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="page" type="number">
  The page number to return, for a paginated list of results. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="per_page" type="number">
  The number of items to return per page. **(AI-inferred)** Optional lookup argument.
</ResponseField>
