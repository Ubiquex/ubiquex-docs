`data_digitalocean_api_get_indexing_job_details_signed_urloutput` generates a real, signed URL for downloading a knowledge base indexing job's own detailed results.

## Example

  <Tab title="Go">

    ```go
    import (
    	gradientaiplatform "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/gradientaiplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_api_get_indexing_job_details_signed_urloutput"})

    		ubx.Data(gradientaiplatform.ApiGetIndexingJobDetailsSignedUrloutput, "example", gradientaiplatform.ApiGetIndexingJobDetailsSignedUrloutputConfig{
    			IndexingJobUuid: "example",
    		})
    	}))
    }
    ```
