title: "digitalocean_app"

---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	addons "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/addons"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_app"})

    		ubx.Data(addons.App, "example", addons.AppConfig{
    			AppSlug: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { App } from "@ubx/sdk-digitalocean/digitalocean/data/addons/app";

    export default stack("example", () => {
      intent({ summary: "look up digitalocean_app" });

      data(App, "example", {
        appSlug: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.digitalocean.data.addons import App, AppConfig

    def describe():
        ubx.intent("look up digitalocean_app")

        ubx.data(App, "example", AppConfig(
            app_slug="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
