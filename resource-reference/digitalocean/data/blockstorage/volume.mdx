## Example

    ```go
    package main

    import (
    	blockstorage "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/blockstorage"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_volume"})

    		ubx.Data(blockstorage.Volume, "example", blockstorage.VolumeConfig{
    			VolumeId: "example",
    			Page:     1,
    			PerPage:  1,
    		})
    	}))
    }
    ```
