## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	databases "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/databases"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_kafka_schema_verbose"})

    		ubx.Data(databases.KafkaSchemaVerbose, "example", databases.KafkaSchemaVerboseConfig{
    			DatabaseClusterUuid: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.digitalocean.data.databases import KafkaSchemaVerbose, KafkaSchemaVerboseConfig

    def describe():
        ubx.intent("look up digitalocean_kafka_schema_verbose")

        ubx.data(KafkaSchemaVerbose, "example", KafkaSchemaVerboseConfig(
            database_cluster_uuid="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

<ResponseField name="subjects" type="list(object)">
