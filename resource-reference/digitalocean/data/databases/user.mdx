    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_user"})

    		})
    	}))

## Lookup arguments

<ResponseField name="cluster_id" type="string" required>
  The ID of the Kubernetes cluster this user identity is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
