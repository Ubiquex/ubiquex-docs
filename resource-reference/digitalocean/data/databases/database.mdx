## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	databases "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/databases"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_database"})

    		ubx.Data(databases.Database, "example", databases.DatabaseConfig{
    			DatabaseClusterUuid: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Database } from "@ubx/sdk-digitalocean/digitalocean/data/databases/database";

    export default stack("example", () => {
      intent({ summary: "look up digitalocean_database" });

      data(Database, "example", {
        databaseClusterUuid: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.digitalocean.data.databases import Database, DatabaseConfig

    def describe():
        ubx.intent("look up digitalocean_database")

        ubx.data(Database, "example", DatabaseConfig(
            database_cluster_uuid="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
