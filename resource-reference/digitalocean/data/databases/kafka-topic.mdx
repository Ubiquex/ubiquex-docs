    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_kafka_topic"})

    		ubx.Data(databases.KafkaTopic, "example", databases.KafkaTopicConfig{

    		})
    	}))
    }
