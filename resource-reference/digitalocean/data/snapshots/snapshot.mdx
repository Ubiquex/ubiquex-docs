## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	snapshots "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/snapshots"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_snapshot"})

    		ubx.Data(snapshots.Snapshot, "example", snapshots.SnapshotConfig{
    			SnapshotId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
