`data_digitalocean_settings` lists the real, current managed database settings -- available plan downgrades and real tier coverage -- for an account, paginated.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	security "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/security"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_settings"})

    		ubx.Data(security.Settings, "example", security.SettingsConfig{
    			Page:    1,
    			PerPage: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    ```

  </Tab>
  <Tab title="Python">

    ```python
    import ubx_sdk as ubx
    from ubx.digitalocean.data.security import Settings, SettingsConfig

    def describe():
        ubx.intent("look up digitalocean_settings")

        ubx.data(Settings, "example", SettingsConfig(
            page=1,
            per_page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
