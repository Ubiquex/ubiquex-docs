    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_key"})

    		ubx.Data(sshkeys.Key, "example", sshkeys.KeyConfig{
    			AccessKey: "example",
    		})
    	}))
    }
