    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_droplet"})

    		ubx.Data(droplets.Droplet, "example", droplets.DropletConfig{
    			Name:    "example",
    			Page:    1,
    			PerPage: 1,
    		})
    	}))
    }
