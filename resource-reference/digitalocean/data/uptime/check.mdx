    package main

    import (
    	uptime "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/uptime"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_check"})

    		ubx.Data(uptime.Check, "example", uptime.CheckConfig{
    			CheckId: "example",
    			Page:    1,
    			PerPage: 1,
    		})
    	}))
    }
