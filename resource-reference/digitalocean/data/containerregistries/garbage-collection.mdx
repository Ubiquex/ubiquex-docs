`data_digitalocean_garbage_collection` lists the real, already-run or scheduled garbage collection passes for a container registry -- the process that reclaims storage from unreferenced image layers.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	containerregistries "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/containerregistries"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_garbage_collection"})

    		ubx.Data(containerregistries.GarbageCollection, "example", containerregistries.GarbageCollectionConfig{
    			RegistryName: "example-registry-name",
    			Page:         1,
    			PerPage:      1,
    		})
    	}))
    }
    ```
