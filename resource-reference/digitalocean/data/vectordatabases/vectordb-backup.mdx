## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	vectordatabases "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/vectordatabases"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_vectordb_backup"})

    		ubx.Data(vectordatabases.VectordbBackup, "example", vectordatabases.VectordbBackupConfig{
    			Id: "example",
    		})
    	}))
    }
    ```
  </Tab>
