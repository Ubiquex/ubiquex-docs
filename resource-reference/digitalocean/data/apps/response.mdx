`data_digitalocean_apps_response` lists the real, already-created App Platform applications on the account, paginated, optionally including project associations.

## Example

  <Tab title="Go">

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_apps_response"})

    		ubx.Data(apps.Response, "example", apps.ResponseConfig{
    			Page:         1,
    			PerPage:      1,
    			WithProjects: true,
    		})
    	}))
    }
