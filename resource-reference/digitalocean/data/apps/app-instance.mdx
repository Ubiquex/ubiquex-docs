## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	apps "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/apps"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_app_instance"})

    		ubx.Data(apps.AppInstance, "example", apps.AppInstanceConfig{
    			AppId: "example",
    		})
    	}))
    }
    ```
