## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	apps "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/data/apps"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up digitalocean_app_alert"})

    		ubx.Data(apps.AppAlert, "example", apps.AppAlertConfig{
    			AppId: "example",
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">

    ```python
    import ubx_sdk as ubx
    from ubx.digitalocean.data.apps import AppAlert, AppAlertConfig

    def describe():
        ubx.intent("look up digitalocean_app_alert")

        ubx.data(AppAlert, "example", AppAlertConfig(
            app_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
