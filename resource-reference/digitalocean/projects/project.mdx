`digitalocean_project` looks up a real, already-created project -- the grouping DigitalOcean uses to organize a set of resources (Droplets, volumes, domains, and more) together for billing and organizational visibility, independent of what type each resource is.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	projects "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/projects"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own project"})

    		ubx.Resource(projects.Project, "example", projects.ProjectConfig{
    			ProjectId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    # ubx sdk gen --only digitalocean --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/digitalocean/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.digitalocean.projects import Project, ProjectConfig

    def describe():
        ubx.intent("example own project")

        ubx.resource(Project, "example", ProjectConfig(
            project_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
