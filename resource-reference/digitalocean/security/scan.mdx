`digitalocean_scan` looks up a real, already-run security scan -- the result of DigitalOcean's own scanning service checking a resource for known vulnerabilities.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	security "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/security"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own scan"})

    		ubx.Resource(security.Scan, "example", security.ScanConfig{
    			ScanId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only digitalocean --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Scan } from "./local-sdk/digitalocean/sdk/typescript/digitalocean/security/scan.ts";

    export default stack("example", () => {
      intent({ summary: "example own scan" });

      resource(Scan, "example", {
        scanId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only digitalocean --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/digitalocean/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.digitalocean.security import Scan, ScanConfig

    def describe():
        ubx.intent("example own scan")

        ubx.resource(Scan, "example", ScanConfig(
            scan_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
