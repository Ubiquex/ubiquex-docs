`digitalocean_dedicated_inference` looks up a real, already-provisioned Gradient AI Platform dedicated inference deployment -- a model served on capacity reserved for one account, rather than shared serverless inference.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	dedicatedinference "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/dedicatedinference"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dedicatedinference"})

    		ubx.Resource(dedicatedinference.Dedicatedinference, "example", dedicatedinference.DedicatedinferenceConfig{
    			DedicatedInferenceId: "example",
    			Spec:                 map[string]any{"enable_public_endpoint": true, "model_deployments": []map[string]any{{"accelerators": []map[string]any{{"accelerator_slug": "example"}}}}, "name": "example-name", "region": "example", "version": 1, "vpc": map[string]any{"uuid": "example"}},
    			AccessTokens:         map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```

    ```python
    # ubx sdk gen --only digitalocean --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/digitalocean/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.digitalocean.dedicatedinference import Dedicatedinference, DedicatedinferenceConfig
    from ubx.digitalocean.dedicatedinference.dedicatedinference import Dedicatedinference_DedicatedInference_Spec, Dedicatedinference_DedicatedInference_PendingDeploymentSpec_ModelDeployments, Dedicatedinference_DedicatedInference_PendingDeploymentSpec_ModelDeployments_Accelerators, Dedicatedinference_DedicatedInference_PendingDeploymentSpec_Vpc

    def describe():
        ubx.intent("example own dedicatedinference")

        ubx.resource(Dedicatedinference, "example", DedicatedinferenceConfig(
            dedicated_inference_id="example",
            spec=Dedicatedinference_DedicatedInference_Spec(enable_public_endpoint=True, model_deployments=[Dedicatedinference_DedicatedInference_PendingDeploymentSpec_ModelDeployments(accelerators=[Dedicatedinference_DedicatedInference_PendingDeploymentSpec_ModelDeployments_Accelerators(accelerator_slug="example")])], name="example-name", region="example", version=1, vpc=Dedicatedinference_DedicatedInference_PendingDeploymentSpec_Vpc(uuid="example")),
            access_tokens={"managed-by": "ubx"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="access_tokens" type="map(string)">
  Key-value pairs for provider tokens (e.g. Hugging Face). Optional.
</ResponseField>

<ResponseField name="dedicated_inference_id" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="spec" type="object" required>
  Structured configuration for a Dedicated Inference deployment. Required.
  <Expandable title="properties">
    <ResponseField name="enable_public_endpoint" type="bool" required>
      Whether to expose a public LLM endpoint. Required.
    </ResponseField>
    <ResponseField name="model_deployments" type="list(object)" required>
      At least one model deployment is required. Required.
      <Expandable title="properties">

            </ResponseField>

              Optional; if omitted, computed by DigitalOcean.
            </ResponseField>
            <ResponseField name="status" type="string">
              Optional; if omitted, computed by DigitalOcean.
            </ResponseField>
            <ResponseField name="type" type="string">
              Optional; if omitted, computed by DigitalOcean.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="model_id" type="string">
          Optional; if omitted, computed by DigitalOcean.
        </ResponseField>
        <ResponseField name="model_provider" type="string">
          Optional; if omitted, computed by DigitalOcean.
        </ResponseField>
        <ResponseField name="model_slug" type="string">
          Optional; if omitted, computed by DigitalOcean.
        </ResponseField>
        <ResponseField name="workload_config" type="any">
          Optional; if omitted, computed by DigitalOcean.
        </ResponseField>
      </Expandable>
    </ResponseField>

      Name of the Dedicated Inference. Must be unique within the team. Required.
    </ResponseField>

      DigitalOcean region where the Dedicated Inference is hosted. Required.
    </ResponseField>
    <ResponseField name="version" type="number" required>
      Spec version. Required.
    </ResponseField>
    <ResponseField name="vpc" type="object" required>
      Required.
      <Expandable title="properties">

          VPC UUID for the Dedicated Inference. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
