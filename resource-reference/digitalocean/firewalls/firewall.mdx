`digitalocean_firewall` looks up a real, already-created cloud firewall -- the inbound/outbound rule set DigitalOcean enforces at the network edge for whichever Droplets or tags it's applied to.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	firewalls "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/firewalls"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own firewall"})

    		ubx.Resource(firewalls.Firewall, "example", firewalls.FirewallConfig{
    			FirewallId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only digitalocean --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Firewall } from "./local-sdk/digitalocean/sdk/typescript/digitalocean/firewalls/firewall.ts";

    export default stack("example", () => {
      intent({ summary: "example own firewall" });

      resource(Firewall, "example", {
        firewallId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only digitalocean --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/digitalocean/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.digitalocean.firewalls import Firewall, FirewallConfig

    def describe():
        ubx.intent("example own firewall")

        ubx.resource(Firewall, "example", FirewallConfig(
            firewall_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
