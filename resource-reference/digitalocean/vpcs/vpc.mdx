`digitalocean_vpc` looks up a real, already-created VPC -- the private, isolated network a region's own Droplets and other resources can be placed inside, with no public routing by default.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	vpcs "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/vpcs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own vpc"})

    		ubx.Resource(vpcs.Vpc, "example", vpcs.VpcConfig{
    			VpcId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only digitalocean --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Vpc } from "./local-sdk/digitalocean/sdk/typescript/digitalocean/vpcs/vpc.ts";

    export default stack("example", () => {
      intent({ summary: "example own vpc" });

      resource(Vpc, "example", {
        vpcId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only digitalocean --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/digitalocean/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.digitalocean.vpcs import Vpc, VpcConfig

    def describe():
        ubx.intent("example own vpc")

        ubx.resource(Vpc, "example", VpcConfig(
            vpc_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
