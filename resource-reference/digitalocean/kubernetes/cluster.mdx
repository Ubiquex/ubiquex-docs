`digitalocean_cluster` looks up a real, already-provisioned DigitalOcean Kubernetes cluster (DOKS) -- its control plane configuration, networking, and add-ons, independent of the node pools that actually run workloads on it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	kubernetes "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/kubernetes"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own cluster"})

    		ubx.Resource(kubernetes.Cluster, "example", kubernetes.ClusterConfig{
    			ClusterId:                         "example",
    			Name:                              "example",
    			NodePools:                         []map[string]any{{"name": "example-name"}},
    			Region:                            "example",
    			Version:                           "example",
    			AmdGpuDeviceMetricsExporterPlugin: map[string]any{"enabled": true},
    			AmdGpuDevicePlugin:                map[string]any{"enabled": true},
    			AmdGpuDraDriver:                   map[string]any{"enabled": true},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.digitalocean.kubernetes import Cluster, ClusterConfig
    from ubx.digitalocean.kubernetes.cluster import Cluster_KubernetesCluster_NodePools, Cluster_AmdGpuDeviceMetricsExporterPlugin

    def describe():
        ubx.intent("example own cluster")

        ubx.resource(Cluster, "example", ClusterConfig(
            cluster_id="example",
            name="example",
            node_pools=[Cluster_KubernetesCluster_NodePools(name="example-name")],
            region="example",
            version="example",
            amd_gpu_device_metrics_exporter_plugin=Cluster_AmdGpuDeviceMetricsExporterPlugin(enabled=True),
            amd_gpu_device_plugin=Cluster_AmdGpuDeviceMetricsExporterPlugin(enabled=True),
            amd_gpu_dra_driver=Cluster_AmdGpuDeviceMetricsExporterPlugin(enabled=True),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

<ResponseField name="cluster_autoscaler_configuration" type="object">
  An object specifying custom cluster autoscaler configuration. Optional.
  <Expandable title="properties">
    <ResponseField name="expanders" type="list(string)">
      Customizes expanders used by cluster-autoscaler. The autoscaler will apply each expander from the provided list to narrow down the selection of node types created to scale up, until either a single node type is left, or the list of expanders is exhausted. If this flag is unset, autoscaler will use its default expander `random`. Passing an empty list (_not_ `null`) will unset any previous expander customizations. Available expanders: - `random`: Randomly selects a node group to scale. - `priority`: Selects the node group with the highest priority as per [user-provided configuration](https://docs.digitalocean.com/products/kubernetes/how-to/autoscale/#configuring-priority-expander) - `least_waste`: Selects the node group that will result in the least amount of idle resources. Optional.
    </ResponseField>
    <ResponseField name="scale_down_unneeded_time" type="string">
      Used to customize how long a node is unneeded before being scaled down. Optional.
    </ResponseField>
    <ResponseField name="scale_down_utilization_threshold" type="number">
      Used to customize when cluster autoscaler scales down non-empty nodes by setting the node utilization threshold. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
