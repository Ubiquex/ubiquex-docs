`digitalocean_node_pool` looks up a real, already-created node pool inside a DigitalOcean Kubernetes cluster -- the group of worker nodes sharing one machine size and autoscaling configuration that actually runs cluster workloads.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	kubernetes "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/kubernetes"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own node pool"})

    		ubx.Resource(kubernetes.NodePool, "example", kubernetes.NodePoolConfig{
    			ClusterId:  "example",
    			NodePoolId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
