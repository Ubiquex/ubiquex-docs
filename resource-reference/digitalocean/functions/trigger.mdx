`digitalocean_trigger` looks up a real, already-configured Functions trigger -- the schedule or event source that invokes a specific function inside a namespace without a direct HTTP call.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	functions "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/functions"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own trigger"})

    		ubx.Resource(functions.Trigger, "example", functions.TriggerConfig{
    			Function:         "example",
    			IsEnabled:        true,
    			Name:             "example",
    			NamespaceId:      "example",
    			ScheduledDetails: map[string]any{"cron": "example"},
    			TriggerName:      "example-trigger-name",
    			Type:             "example",
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    # ubx sdk gen --only digitalocean --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/digitalocean/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.digitalocean.functions import Trigger, TriggerConfig
    from ubx.digitalocean.functions.trigger import Trigger_ScheduledDetails

    def describe():
        ubx.intent("example own trigger")

        ubx.resource(Trigger, "example", TriggerConfig(
            function="example",
            is_enabled=True,
            name="example",
            namespace_id="example",
            scheduled_details=Trigger_ScheduledDetails(cron="example"),
            trigger_name="example-trigger-name",
            type="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="function" type="string" required>
  Name of function(action) that exists in the given namespace. Required.
</ResponseField>

<ResponseField name="is_enabled" type="bool" required>
  Indicates weather the trigger is paused or unpaused. Required.
</ResponseField>

<ResponseField name="name" type="string" required>
  The trigger's unique name within the namespace. Required.
</ResponseField>

<ResponseField name="namespace_id" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="scheduled_details" type="object" required>
  Trigger details for SCHEDULED type, where body is optional. Required.
  <Expandable title="properties">
    <ResponseField name="body" type="object">
      Optional data to be sent to function while triggering the function. Optional.
      <Expandable title="properties">
        <ResponseField name="name" type="string">
          Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="cron" type="string" required>
      valid cron expression string which is required for SCHEDULED type triggers. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="trigger_name" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="type" type="string" required>
  One of different type of triggers. Currently only SCHEDULED is supported. Required.
</ResponseField>

## Output properties

<ResponseField name="trigger" type="object">
  The Functions trigger's own real, current representation. Computed by DigitalOcean after `ship`. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="created_at" type="string">
      UTC time string. Optional.
    </ResponseField>
    <ResponseField name="function" type="string">
      Name of function(action) that exists in the given namespace. Optional.
    </ResponseField>
    <ResponseField name="is_enabled" type="bool">
      Indicates weather the trigger is paused or unpaused. Optional.
    </ResponseField>
    <ResponseField name="name" type="string">
      The trigger's unique name within the namespace. Optional.
    </ResponseField>
    <ResponseField name="namespace" type="string">
      A unique string format of UUID with a prefix fn-. Optional.
    </ResponseField>
    <ResponseField name="scheduled_details" type="object">
      Trigger details for SCHEDULED type, where body is optional. Optional.
      <Expandable title="properties">
        <ResponseField name="body" type="object">
          Optional data to be sent to function while triggering the function. Optional.
          <Expandable title="properties">
            <ResponseField name="name" type="string">
              Optional.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="cron" type="string" required>
          valid cron expression string which is required for SCHEDULED type triggers. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="scheduled_runs" type="object">
      Optional.
      <Expandable title="properties">
        <ResponseField name="last_run_at" type="string">
          Indicates last run time. null value indicates trigger not run yet. Optional.
        </ResponseField>
        <ResponseField name="next_run_at" type="string">
          Indicates next run time. null value indicates trigger will not run. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="type" type="string">
      String which indicates the type of trigger source like SCHEDULED. Optional.
    </ResponseField>
    <ResponseField name="updated_at" type="string">
      UTC time string. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
