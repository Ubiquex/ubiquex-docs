`digitalocean_floating_ip` looks up a real, already-reserved floating IP -- a publicly routable IPv4 address that can be reassigned between Droplets in a region without any DNS propagation delay.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	floatingips "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/floatingips"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own floating ip"})

    		ubx.Resource(floatingips.FloatingIp, "example", floatingips.FloatingIpConfig{
    			FloatingIpPath: "example",
    		})
    	}))
    }
    ```
