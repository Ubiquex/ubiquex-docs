`digitalocean_volume` looks up a real, already-created Block Storage volume -- an independently resizable SSD volume attachable to a Droplet, outliving the Droplet's own lifecycle.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	blockstorage "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/blockstorage"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own volume"})

    		ubx.Resource(blockstorage.Volume, "example", blockstorage.VolumeConfig{
    			VolumeId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only digitalocean --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Volume } from "./local-sdk/digitalocean/sdk/typescript/digitalocean/blockstorage/volume.ts";

    export default stack("example", () => {
      intent({ summary: "example own volume" });

      resource(Volume, "example", {
        volumeId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">

    ```python
    # ubx sdk gen --only digitalocean --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/digitalocean/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.digitalocean.blockstorage import Volume, VolumeConfig

    def describe():
        ubx.intent("example own volume")

        ubx.resource(Volume, "example", VolumeConfig(
            volume_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
