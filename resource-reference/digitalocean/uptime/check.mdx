`digitalocean_check` looks up a real, already-configured Uptime check -- the endpoint, protocol, and regions DigitalOcean's own synthetic monitoring probes to track an external service's own real availability and latency.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	uptime "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/uptime"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own check"})

    		ubx.Resource(uptime.Check, "example", uptime.CheckConfig{
    			CheckId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only digitalocean --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Check } from "./local-sdk/digitalocean/sdk/typescript/digitalocean/uptime/check.ts";

    export default stack("example", () => {
      intent({ summary: "example own check" });

      resource(Check, "example", {
        checkId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only digitalocean --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/digitalocean/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.digitalocean.uptime import Check, CheckConfig

    def describe():
        ubx.intent("example own check")

        ubx.resource(Check, "example", CheckConfig(
            check_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
