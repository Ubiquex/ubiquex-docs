`digitalocean_droplet` looks up a real, already-created Droplet -- DigitalOcean's own virtual machine, with its real current size, image, region, and networking.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	droplets "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/droplets"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own droplet"})

    		ubx.Resource(droplets.Droplet, "example", droplets.DropletConfig{
    			DropletId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Droplet } from "@ubx/sdk-digitalocean/digitalocean/droplets/droplet";

    export default stack("example", () => {
      intent({ summary: "example own droplet" });

      resource(Droplet, "example", {
        dropletId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.digitalocean.droplets import Droplet, DropletConfig

    def describe():
        ubx.intent("example own droplet")

        ubx.resource(Droplet, "example", DropletConfig(
            droplet_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
  <Tab title="Markdown">
    ```
    Create digitalocean_droplet, with droplet id example.
    ```
  </Tab>
</Tabs>
