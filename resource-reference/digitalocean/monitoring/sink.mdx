`digitalocean_sink` looks up a real, already-configured metrics or log sink for a managed database cluster -- one of possibly several forwarding destinations for that cluster's own operational data.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	monitoring "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/monitoring"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own sink"})

    		ubx.Resource(monitoring.Sink, "example", monitoring.SinkConfig{
    			DatabaseClusterUuid: "example",
    			SinkUuid:            "example",
    		})
    	}))
    }
    ```
  </Tab>
