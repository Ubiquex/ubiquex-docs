`digitalocean_destination` looks up a real, already-configured log or metrics sink destination for a managed database cluster -- where DigitalOcean forwards that cluster's own operational data (a managed OpenSearch instance, or an externally managed one).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only digitalocean --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-digitalocean/sdk/go => ./local-sdk/digitalocean/sdk/go
    package main

    import (
    	monitoring "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/monitoring"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own destination"})

    		ubx.Resource(monitoring.Destination, "example", monitoring.DestinationConfig{
    			Config:          map[string]any{"endpoint": "example"},
    			DestinationUuid: "example",
    			Type:            "example",
    			Name:            "example",
    		})
    	}))
    }
    ```

    from ubx.digitalocean.monitoring import Destination, DestinationConfig
    from ubx.digitalocean.monitoring.destination import Destination_Config

        ubx.intent("example own destination")

            type="example",
            name="example",

## Input properties

<ResponseField name="config" type="object" required>
  The real, type-specific configuration for this sink destination. Required. **(AI-inferred)**
  <Expandable title="properties">
    <ResponseField name="cluster_name" type="string">
      Name of a managed OpenSearch cluster. Optional.
    </ResponseField>
    <ResponseField name="cluster_uuid" type="string">
      A unique identifier for a managed OpenSearch cluster. Optional.
    </ResponseField>
    <ResponseField name="credentials" type="object">
      Credentials for an OpenSearch cluster user. Optional if `cluster_uuid` is passed. Optional.
      <Expandable title="properties">
        <ResponseField name="password" type="string">
          Optional.
        </ResponseField>
        <ResponseField name="username" type="string">
          Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="endpoint" type="string" required>
      host of the OpenSearch cluster Required.
    </ResponseField>
    <ResponseField name="index_name" type="string">
      OpenSearch index to send logs to. Optional.
    </ResponseField>
    <ResponseField name="retention_days" type="number">
      Number of days to retain logs in an OpenSearch cluster. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="destination_uuid" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="name" type="string">
  destination name Optional.
</ResponseField>

<ResponseField name="type" type="string" required>
  The destination type. `opensearch_dbaas` for a DigitalOcean managed OpenSearch cluster or `opensearch_ext` for an externally managed one. Required.
</ResponseField>
