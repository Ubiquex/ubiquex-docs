`digitalocean_load_balancer` looks up a real, already-created Load Balancer -- the forwarding rules, health checks, and backend Droplet set it distributes traffic across.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	loadbalancers "github.com/ubiquex/ubx-sdk-digitalocean/sdk/go/digitalocean/loadbalancers"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own load balancer"})

    		ubx.Resource(loadbalancers.LoadBalancer, "example", loadbalancers.LoadBalancerConfig{
    			LbId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { LoadBalancer } from "@ubx/sdk-digitalocean/digitalocean/loadbalancers/load_balancer";

    export default stack("example", () => {
      intent({ summary: "example own load balancer" });

      resource(LoadBalancer, "example", {
        lbId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.digitalocean.loadbalancers import LoadBalancer, LoadBalancerConfig

    def describe():
        ubx.intent("example own load balancer")

        ubx.resource(LoadBalancer, "example", LoadBalancerConfig(
            lb_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
  <Tab title="Markdown">

  </Tab>
</Tabs>
