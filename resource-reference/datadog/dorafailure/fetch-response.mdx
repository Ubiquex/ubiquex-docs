`datadog_dorafailure_fetch_response` is the real, queried change-failure-rate half of Datadog's DORA metrics -- how often a deployment led to a real incident or rollback, computed alongside `datadog_doradeployment_fetch_response`'s own deployment-frequency data.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	dorafailure "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/dorafailure"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own fetch response"})

    		ubx.Resource(dorafailure.FetchResponse, "example", dorafailure.FetchResponseConfig{
    			Data:      map[string]any{"attributes": map[string]any{"from": "example"}},
    			FailureId: "example",
    		})
    	}))
    }
    ```
