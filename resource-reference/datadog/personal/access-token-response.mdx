`datadog_personal_access_token_response` is a real, user-scoped Datadog API credential -- unlike an application key (tied to a user but usable org-wide within that user's permissions), a personal access token is explicitly scoped to specific real permissions the user grants it at creation.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	personal "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/personal"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own access token response"})

    		ubx.Resource(personal.AccessTokenResponse, "example", personal.AccessTokenResponseConfig{
    			Data:    map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			TokenId: "example",
    		})
    	}))
    }
    ```
  </Tab>
