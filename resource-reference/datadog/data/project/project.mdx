## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	project "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/project"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_project"})

    		ubx.Data(project.Project, "example", project.ProjectConfig{})
    	}))
    }
    ```
