## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	log "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/log"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_log"})

    		ubx.Data(log.Log, "example", log.LogConfig{
    			Sort: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Log } from "@ubx/sdk-datadog/datadog/data/log/log";

    export default stack("example", () => {
      intent({ summary: "look up datadog_log" });

      data(Log, "example", {
        sort: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.log import Log, LogConfig

    def describe():
        ubx.intent("look up datadog_log")

        ubx.data(Log, "example", LogConfig(
            sort="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
