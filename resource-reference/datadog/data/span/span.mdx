## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	span "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/span"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_span"})

    		ubx.Data(span.Span, "example", span.SpanConfig{
    			Sort: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { Span } from "./local-sdk/datadog/sdk/typescript/datadog/data/span/span.ts";

    export default stack("example", () => {
      intent({ summary: "look up datadog_span" });

      data(Span, "example", {
        sort: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.span import Span, SpanConfig

    def describe():
        ubx.intent("look up datadog_span")

        ubx.data(Span, "example", SpanConfig(
            sort="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Lookup arguments

<ResponseField name="sort" type="string">
  Sort parameters when querying spans. Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="data" type="list(object)">
  Array of spans matching the request. Returned by the lookup.
</ResponseField>

<ResponseField name="links" type="object">
  Links attributes. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="next" type="string">
      Link for the next set of results. Note that the request can also be made using the POST endpoint. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="meta" type="object">
  The metadata associated with a request. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="elapsed" type="number">
      The time elapsed in milliseconds. Optional lookup argument.
    </ResponseField>
    <ResponseField name="page" type="object">
      Paging attributes. Optional lookup argument.
      <Expandable title="properties">
        <ResponseField name="after" type="string">
          The cursor to use to get the next results, if any. To make the next request, use the same parameters with the addition of the `page[cursor]`. Optional lookup argument.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="request_id" type="string">
      The identifier of the request. Optional lookup argument.
    </ResponseField>
    <ResponseField name="status" type="string">
      The status of the response. Optional lookup argument.
    </ResponseField>
    <ResponseField name="warnings" type="list(object)">
      A list of warnings (non fatal errors) encountered, partial results might be returned if warnings are present in the response. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>
