## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	report "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/report"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_report"})

    		ubx.Data(report.Report, "example", report.ReportConfig{
    			ReportId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

    export default stack("example", () => {
      intent({ summary: "look up datadog_report" });

      data(Report, "example", {
        reportId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.report import Report, ReportConfig

    def describe():
        ubx.intent("look up datadog_report")

        ubx.data(Report, "example", ReportConfig(
            report_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
