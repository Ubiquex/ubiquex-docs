    ```go
    			ProductKeys: "example",
    		})
    	}))
    }
    ```

    ```python
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.tenancy import ProductsData, ProductsDataConfig

    def describe():
        ubx.intent("look up datadog_tenancy_products_data")

        ubx.data(ProductsData, "example", ProductsDataConfig(
            product_keys="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
