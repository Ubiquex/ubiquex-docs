## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	integration "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/integration"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_integration"})

    		ubx.Data(integration.Integration, "example", integration.IntegrationConfig{})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```typescript
    export default stack("example", () => {
      intent({ summary: "look up datadog_integration" });

      data(Integration, "example", {});
    });
    ```
  </Tab>
