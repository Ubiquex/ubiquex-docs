## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	fastly "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/fastly"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_fastly_service_data"})

    		ubx.Data(fastly.ServiceData, "example", fastly.ServiceDataConfig{
    			AccountId: "example",
    		})
    	}))
    }
    ```

    export default stack("example", () => {

  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.fastly import ServiceData, ServiceDataConfig

    def describe():
        ubx.intent("look up datadog_fastly_service_data")

        ubx.data(ServiceData, "example", ServiceDataConfig(
            account_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
