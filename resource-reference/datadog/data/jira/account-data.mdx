## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	jira "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/jira"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_jira_account_data"})

    		ubx.Data(jira.AccountData, "example", jira.AccountDataConfig{})
    	}))
    }
