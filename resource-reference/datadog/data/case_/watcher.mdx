## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	case_ "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/case_"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_case_watcher"})

    		ubx.Data(case_.Watcher, "example", case_.WatcherConfig{
    			CaseId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

    export default stack("example", () => {
      intent({ summary: "look up datadog_case_watcher" });

      data(Watcher, "example", {
        caseId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.case_ import Watcher, WatcherConfig

    def describe():
        ubx.intent("look up datadog_case_watcher")

        ubx.data(Watcher, "example", WatcherConfig(
            case_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
