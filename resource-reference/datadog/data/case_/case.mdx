`datadog_case` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	case_ "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/case_"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_case"})

    		ubx.Data(case_.Case, "example", case_.CaseConfig{
    			Filter: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Case } from "@ubx/sdk-datadog/datadog/data/case_/case";

    export default stack("example", () => {
      intent({ summary: "look up datadog_case" });

      data(Case, "example", {
        filter: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.case_ import Case, CaseConfig

    def describe():
        ubx.intent("look up datadog_case")

        ubx.data(Case, "example", CaseConfig(
            filter="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
