## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	timeline "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/timeline"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_timeline_cell_resource"})

    		ubx.Data(timeline.CellResource, "example", timeline.CellResourceConfig{
    			CaseId: "example",
    		})
    	}))
    }
    ```

    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.timeline import CellResource, CellResourceConfig

    def describe():
        ubx.intent("look up datadog_timeline_cell_resource")

        ubx.data(CellResource, "example", CellResourceConfig(
            case_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
