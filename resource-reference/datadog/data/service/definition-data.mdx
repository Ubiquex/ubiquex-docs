`datadog_service_definition_data` looks up a real Service Catalog service definition -- the metadata describing one service's ownership, dependencies, and links.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	service "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/service"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_service_definition_data"})

    		ubx.Data(service.DefinitionData, "example", service.DefinitionDataConfig{
    			SchemaVersion: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { DefinitionData } from "@ubx/sdk-datadog/datadog/data/service/definition_data";

    export default stack("example", () => {
      intent({ summary: "look up datadog_service_definition_data" });

      data(DefinitionData, "example", {
        schemaVersion: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.service import DefinitionData, DefinitionDataConfig

    def describe():
        ubx.intent("look up datadog_service_definition_data")

        ubx.data(DefinitionData, "example", DefinitionDataConfig(
            schema_version="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
