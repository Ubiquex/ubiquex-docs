`datadog_commitments_coverage_timeseries_response` reads the real, time-series coverage of cloud spend by commitments (Reserved Instances, Savings Plans) over time.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_commitments_coverage_timeseries_response"})

    		ubx.Data(commitments.CoverageTimeseriesResponse, "example", commitments.CoverageTimeseriesResponseConfig{
    			End:      1,
    			Product:  "example",

    		})
    	}))
