## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	slack "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/slack"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_slack_user_binding_data"})

    		ubx.Data(slack.UserBindingData, "example", slack.UserBindingDataConfig{
    			UserUuid: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { UserBindingData } from "./local-sdk/datadog/sdk/typescript/datadog/data/slack/user_binding_data.ts";

    export default stack("example", () => {
      intent({ summary: "look up datadog_slack_user_binding_data" });

      data(UserBindingData, "example", {
        userUuid: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.slack import UserBindingData, UserBindingDataConfig

    def describe():
        ubx.intent("look up datadog_slack_user_binding_data")

        ubx.data(UserBindingData, "example", UserBindingDataConfig(
            user_uuid="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="user_uuid" type="string" required>
  The real, unique identifier of this user. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="data" type="list(object)">
  An array of Slack user bindings. Returned by the lookup.
</ResponseField>
