`datadog_ticket_creation_rule_data_response` looks up a real rule that automatically creates an external ticket (Jira, ServiceNow) from a Datadog case or incident.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_ticket_creation_rule_data_response"})

    	}))
