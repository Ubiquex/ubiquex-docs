## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	rumevent "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/rumevent"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_rumevent"})

    		ubx.Data(rumevent.Rumevent, "example", rumevent.RumeventConfig{
    			Sort: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Rumevent } from "@ubx/sdk-datadog/datadog/data/rumevent/rumevent";

    export default stack("example", () => {
      intent({ summary: "look up datadog_rumevent" });

      data(Rumevent, "example", {
        sort: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.rumevent import Rumevent, RumeventConfig

    def describe():
        ubx.intent("look up datadog_rumevent")

        ubx.data(Rumevent, "example", RumeventConfig(
            sort="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
