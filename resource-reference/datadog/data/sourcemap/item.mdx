`datadog_sourcemap_item` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	sourcemap "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/sourcemap"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_sourcemap_item"})

    		ubx.Data(sourcemap.Item, "example", sourcemap.ItemConfig{
    			Mapkind: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Item } from "@ubx/sdk-datadog/datadog/data/sourcemap/item";

    export default stack("example", () => {
      intent({ summary: "look up datadog_sourcemap_item" });

      data(Item, "example", {
        mapkind: "example",

    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.sourcemap import Item, ItemConfig

    def describe():
        ubx.intent("look up datadog_sourcemap_item")

        ubx.data(Item, "example", ItemConfig(
            mapkind="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
