## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	sourcemap "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/sourcemap"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_sourcemap_item"})

    		ubx.Data(sourcemap.Item, "example", sourcemap.ItemConfig{
    			Mapkind: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { Item } from "./local-sdk/datadog/sdk/typescript/datadog/data/sourcemap/item.ts";

    export default stack("example", () => {
      intent({ summary: "look up datadog_sourcemap_item" });

      data(Item, "example", {
        mapkind: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.sourcemap import Item, ItemConfig

    def describe():
        ubx.intent("look up datadog_sourcemap_item")

        ubx.data(Item, "example", ItemConfig(
            mapkind="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Lookup arguments

<ResponseField name="mapkind" type="string">
  The type of source map. Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="data" type="list(object)">
  List of source map data objects. Returned by the lookup.
</ResponseField>

<ResponseField name="meta" type="object">
  Pagination metadata for the source maps list response. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="page" type="object" required>
      Page information for the source maps list response. Required lookup argument.
      <Expandable title="properties">
        <ResponseField name="has_more_results" type="bool" required>
          Whether there are more results available beyond the current page. Required lookup argument.
        </ResponseField>
        <ResponseField name="total_filtered_count" type="number" required>
          Total number of source maps matching the filter criteria. Required lookup argument.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
