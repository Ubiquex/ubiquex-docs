`datadog_v1` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	v1 "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/v1"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_v1"})

    		ubx.Data(v1.V1, "example", v1.V1Config{
    			CurrentOnly: true,
    			WithCreator: true,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { V1 } from "@ubx/sdk-datadog/datadog/data/v1/v1";

    export default stack("example", () => {
      intent({ summary: "look up datadog_v1" });

      data(V1, "example", {
        currentOnly: true,
        withCreator: true,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.v1 import V1, V1Config

    def describe():
        ubx.intent("look up datadog_v1")

        ubx.data(V1, "example", V1Config(
            current_only=True,
            with_creator=True,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
