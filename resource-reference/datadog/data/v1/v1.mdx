    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	v1 "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/v1"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_v1"})

    		ubx.Data(v1.V1, "example", v1.V1Config{
    			CurrentOnly: true,
    			WithCreator: true,
    		})
    	}))
    }

    from ubx.datadog.data.v1 import V1, V1Config

    def describe():
        ubx.intent("look up datadog_v1")

        ubx.data(V1, "example", V1Config(
            current_only=True,

        ))

    if __name__ == "__main__":
