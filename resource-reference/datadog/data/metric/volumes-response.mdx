## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	metric "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/metric"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_metric_volumes_response"})

    		ubx.Data(metric.VolumesResponse, "example", metric.VolumesResponseConfig{
    			MetricName: "example-metric-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.metric import VolumesResponse, VolumesResponseConfig

    def describe():
        ubx.intent("look up datadog_metric_volumes_response")

        ubx.data(VolumesResponse, "example", VolumesResponseConfig(
            metric_name="example-metric-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
