    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go

    	tag "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/tag"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_tag_indexing_rule_data"})

    		ubx.Data(tag.IndexingRuleData, "example", tag.IndexingRuleDataConfig{
    			Search: "example",
    		})
    	}))
    }

    import ubx_sdk as ubx
    from ubx.datadog.data.tag import IndexingRuleData, IndexingRuleDataConfig

    def describe():
        ubx.intent("look up datadog_tag_indexing_rule_data")

        ubx.data(IndexingRuleData, "example", IndexingRuleDataConfig(
            search="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
