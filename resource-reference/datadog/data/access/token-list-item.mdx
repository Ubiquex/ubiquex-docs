## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	access "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/access"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_access_token_list_item"})

    		ubx.Data(access.TokenListItem, "example", access.TokenListItemConfig{
    			Filter: "example",
    			Sort:   "example",
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.datadog.data.access import TokenListItem, TokenListItemConfig

    def describe():
        ubx.intent("look up datadog_access_token_list_item")

        ubx.data(TokenListItem, "example", TokenListItemConfig(
            filter="example",
            sort="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="filter" type="string">

</ResponseField>

<ResponseField name="sort" type="string">
  Sorting options Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="data" type="list(object)">
  Array of access tokens. Includes both personal and service access tokens. Returned by the lookup.
</ResponseField>
