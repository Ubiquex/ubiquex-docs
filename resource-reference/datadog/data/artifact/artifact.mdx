`datadog_artifact` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	artifact "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/artifact"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_artifact"})

    		ubx.Data(artifact.Artifact, "example", artifact.ArtifactConfig{
    			ArtifactPath: "example",
    			ProjectId:    "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Artifact } from "@ubx/sdk-datadog/datadog/data/artifact/artifact";

    export default stack("example", () => {
      intent({ summary: "look up datadog_artifact" });

      data(Artifact, "example", {
        artifactPath: "example",
        projectId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.artifact import Artifact, ArtifactConfig

    def describe():
        ubx.intent("look up datadog_artifact")

        ubx.data(Artifact, "example", ArtifactConfig(
            artifact_path="example",
            project_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
