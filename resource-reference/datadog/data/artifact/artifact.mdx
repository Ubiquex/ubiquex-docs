## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	artifact "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/artifact"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_artifact"})

    		ubx.Data(artifact.Artifact, "example", artifact.ArtifactConfig{
    			ArtifactPath: "example",
    			ProjectId:    "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { Artifact } from "./local-sdk/datadog/sdk/typescript/datadog/data/artifact/artifact.ts";

    export default stack("example", () => {
      intent({ summary: "look up datadog_artifact" });

      data(Artifact, "example", {
        artifactPath: "example",
        projectId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.artifact import Artifact, ArtifactConfig

    def describe():
        ubx.intent("look up datadog_artifact")

        ubx.data(Artifact, "example", ArtifactConfig(
            artifact_path="example",
            project_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Lookup arguments

<ResponseField name="artifact_path" type="string" required>
  The real, storage path of this artifact. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="project_id" type="string" required>
  The real, unique identifier of this project. **(AI-inferred)** Required lookup argument.
