`datadog_watcher_data` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	watcher "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/watcher"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_watcher_data"})

    		ubx.Data(watcher.Data, "example", watcher.DataConfig{
    			SessionId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Data } from "@ubx/sdk-datadog/datadog/data/watcher/data";

    export default stack("example", () => {
      intent({ summary: "look up datadog_watcher_data" });

      data(Data, "example", {
        sessionId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.watcher import Data, DataConfig

    def describe():
        ubx.intent("look up datadog_watcher_data")

        ubx.data(Data, "example", DataConfig(
            session_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
