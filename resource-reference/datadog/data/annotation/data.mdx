`datadog_annotation_data` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	annotation "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/annotation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_annotation_data"})

    		ubx.Data(annotation.Data, "example", annotation.DataConfig{
    			EndTime:   1,
    			PageId:    "example",
    			StartTime: 1,
    			WidgetId:  "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    def describe():

        ubx.data(Data, "example", DataConfig(
            end_time=1,
            page_id="example",
            start_time=1,
            widget_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
