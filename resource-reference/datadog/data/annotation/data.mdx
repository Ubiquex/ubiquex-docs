## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	annotation "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/annotation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_annotation_data"})

    		ubx.Data(annotation.Data, "example", annotation.DataConfig{
    			EndTime:   1,
    			PageId:    "example",
    			StartTime: 1,
    			WidgetId:  "example",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.datadog.data.annotation import Data, DataConfig

    def describe():
        ubx.intent("look up datadog_annotation_data")

        ubx.data(Data, "example", DataConfig(
            end_time=1,
            page_id="example",
            start_time=1,
            widget_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="end_time" type="number" required>
  The real, inclusive end of the time range this request covers. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="page_id" type="string" required>
  The real, unique identifier of this status page. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="start_time" type="number" required>
  The real, inclusive start of the time range this request covers. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="widget_id" type="string">
  The real, unique identifier of this widget. **(AI-inferred)** Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="data" type="list(object)">
  List of annotation resources. Returned by the lookup.
</ResponseField>
