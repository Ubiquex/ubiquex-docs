## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	api "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/api"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_api"})

    		ubx.Data(api.Api, "example", api.ApiConfig{
    			Id: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Api } from "@ubx/sdk-datadog/datadog/data/api/api";

    export default stack("example", () => {
      intent({ summary: "look up datadog_api" });

      data(Api, "example", {
        id: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.api import Api, ApiConfig

    def describe():
        ubx.intent("look up datadog_api")

        ubx.data(Api, "example", ApiConfig(
            id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
