    package main

    import (

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_shared_dashboard_invites"})

    		ubx.Data(shared.DashboardInvites, "example", shared.DashboardInvitesConfig{
    			Token:      "example",
    			PageNumber: 1,
    			PageSize:   1,
    		})
    	}))
    }
