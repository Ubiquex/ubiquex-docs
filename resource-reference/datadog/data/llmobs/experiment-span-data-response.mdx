## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	llmobs "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/llmobs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_llmobs_experiment_span_data_response"})

    		ubx.Data(llmobs.ExperimentSpanDataResponse, "example", llmobs.ExperimentSpanDataResponseConfig{
    			ExperimentId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

    import { ExperimentSpanDataResponse } from "./local-sdk/datadog/sdk/typescript/datadog/data/llmobs/experiment_span_data_response.ts";

    export default stack("example", () => {
      intent({ summary: "look up datadog_llmobs_experiment_span_data_response" });

      data(ExperimentSpanDataResponse, "example", {
        experimentId: "example",
      });
    });
    ```

  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.llmobs import ExperimentSpanDataResponse, ExperimentSpanDataResponseConfig

    def describe():
        ubx.intent("look up datadog_llmobs_experiment_span_data_response")

        ubx.data(ExperimentSpanDataResponse, "example", ExperimentSpanDataResponseConfig(
            experiment_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="experiment_id" type="string" required>
  The real, unique identifier of this experiment. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="data" type="list(object)">
  List of experiment spans with their evaluation metrics. Returned by the lookup.
</ResponseField>
