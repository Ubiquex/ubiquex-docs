`datadog_llmobs_annotation_queue_data_response` looks up a real LLM Observability annotation queue -- a curated list of interactions routed for human review.

## Example

  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	llmobs "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/llmobs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_llmobs_annotation_queue_data_response"})

    		ubx.Data(llmobs.AnnotationQueueDataResponse, "example", llmobs.AnnotationQueueDataResponseConfig{
    			ProjectId: "example",
    			QueueIds:  []string{"example"},
    		})
    	}))
    }
    ```
