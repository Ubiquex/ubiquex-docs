## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	llmobs "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/llmobs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_llmobs_dataset_data_response"})

    		ubx.Data(llmobs.DatasetDataResponse, "example", llmobs.DatasetDataResponseConfig{
    			ProjectId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    export default stack("example", () => {
      intent({ summary: "look up datadog_llmobs_dataset_data_response" });

  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.llmobs import DatasetDataResponse, DatasetDataResponseConfig

    def describe():
        ubx.intent("look up datadog_llmobs_dataset_data_response")

        ubx.data(DatasetDataResponse, "example", DatasetDataResponseConfig(
            project_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
