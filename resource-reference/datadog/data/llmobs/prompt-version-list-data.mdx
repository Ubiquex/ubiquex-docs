## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	llmobs "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/llmobs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_llmobs_prompt_version_list_data"})

    		ubx.Data(llmobs.PromptVersionListData, "example", llmobs.PromptVersionListDataConfig{
    			PromptId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { PromptVersionListData } from "./local-sdk/datadog/sdk/typescript/datadog/data/llmobs/prompt_version_list_data.ts";

    export default stack("example", () => {
      intent({ summary: "look up datadog_llmobs_prompt_version_list_data" });

      data(PromptVersionListData, "example", {
        promptId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.llmobs import PromptVersionListData, PromptVersionListDataConfig

    def describe():
        ubx.intent("look up datadog_llmobs_prompt_version_list_data")

        ubx.data(PromptVersionListData, "example", PromptVersionListDataConfig(
            prompt_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="prompt_id" type="string" required>
  The real, unique identifier of this prompt. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="data" type="list(object)">
  Prompt versions ordered from newest to oldest. Returned by the lookup.
</ResponseField>
