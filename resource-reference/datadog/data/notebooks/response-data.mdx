`datadog_notebooks_response_data` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	notebooks "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/notebooks"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_notebooks_response_data"})

    		ubx.Data(notebooks.ResponseData, "example", notebooks.ResponseDataConfig{
    			AuthorHandle:        "example",
    			Count:               1,
    			ExcludeAuthorHandle: "example",
    			IncludeCells:        true,
    			IsTemplate:          true,
    			Query:               "example",
    			SortDir:             "example",
    			SortField:           "example",
    		})
    	}))
    }
    ```
