    from ubx.datadog.data.seat import UserData, UserDataConfig

    def describe():
        ubx.intent("look up datadog_seat_user_data")

        ubx.data(UserData, "example", UserDataConfig(
            product_code="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
