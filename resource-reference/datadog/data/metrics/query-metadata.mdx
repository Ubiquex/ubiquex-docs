    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    			Query: "example",

    		})
    	}))
    }

    from ubx.datadog.data.metrics import QueryMetadata, QueryMetadataConfig

    def describe():
        ubx.intent("look up datadog_metrics_query_metadata")

        ubx.data(QueryMetadata, "example", QueryMetadataConfig(
            from_=1,
            query="example",
            to=1,
        ))

    if __name__ == "__main__":
