`data_datadog_metrics_query_metadata` is a real, live lookup against Datadog, generated from datadog's own real schema -- read-only, never created or destroyed.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	metrics "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/metrics"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_metrics_query_metadata"})

    		ubx.Data(metrics.QueryMetadata, "example", metrics.QueryMetadataConfig{
    			From:  1,
    			Query: "example",
    			To:    1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.metrics import QueryMetadata, QueryMetadataConfig

    def describe():
        ubx.intent("look up datadog_metrics_query_metadata")

        ubx.data(QueryMetadata, "example", QueryMetadataConfig(
            from_=1,
            query="example",
            to=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
