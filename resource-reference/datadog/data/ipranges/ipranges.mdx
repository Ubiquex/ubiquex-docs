## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	ipranges "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/ipranges"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_ipranges"})

    		ubx.Data(ipranges.Ipranges, "example", ipranges.IprangesConfig{})
    	}))
    }
    ```
