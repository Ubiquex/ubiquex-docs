`datadog_finding` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	finding "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/finding"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_finding"})

    		ubx.Data(finding.Finding, "example", finding.FindingConfig{
    			DetailedFindings:  true,
    			SnapshotTimestamp: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Finding } from "@ubx/sdk-datadog/datadog/data/finding/finding";

    export default stack("example", () => {
      intent({ summary: "look up datadog_finding" });

      data(Finding, "example", {
        detailedFindings: true,
        snapshotTimestamp: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.finding import Finding, FindingConfig

    def describe():
        ubx.intent("look up datadog_finding")

        ubx.data(Finding, "example", FindingConfig(
            detailed_findings=True,
            snapshot_timestamp=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Result properties

<ResponseField name="data" type="list(object)">
  Array of findings. Returned by the lookup.
</ResponseField>

<ResponseField name="meta" type="object">
  Metadata for pagination. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="page" type="object">
      Pagination and findings count information. Optional lookup argument.
      <Expandable title="properties">
        <ResponseField name="cursor" type="string">
          The cursor used to paginate requests. Optional lookup argument.
        </ResponseField>
        <ResponseField name="total_filtered_count" type="number">

        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="snapshot_timestamp" type="number">
      The point in time corresponding to the listed findings. Optional lookup argument.

  </Expandable>
</ResponseField>
