## Example

    package main

    import (
    	confluent "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/confluent"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_confluent_resource_response_data"})

    		ubx.Data(confluent.ResourceResponseData, "example", confluent.ResourceResponseDataConfig{
    			AccountId: "example",
    		})
    	}))
    }
