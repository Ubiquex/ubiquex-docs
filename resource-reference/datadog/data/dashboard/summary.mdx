`datadog_dashboard_summary` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dashboard "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/dashboard"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_dashboard_summary"})

    		ubx.Data(dashboard.Summary, "example", dashboard.SummaryConfig{
    			Count: 1,
    			Start: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Summary } from "@ubx/sdk-datadog/datadog/data/dashboard/summary";

    export default stack("example", () => {
      intent({ summary: "look up datadog_dashboard_summary" });

      data(Summary, "example", {
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.dashboard import Summary, SummaryConfig

    def describe():
        ubx.intent("look up datadog_dashboard_summary")

        ubx.data(Summary, "example", SummaryConfig(
            count=1,
            start=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
