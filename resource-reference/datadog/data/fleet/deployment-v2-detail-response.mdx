## Result properties

<ResponseField name="data" type="object">
  Detailed information about a deployment. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="attributes" type="object" required>
      Attributes of a deployment detail response. Required lookup argument.
      <Expandable title="properties">
        <ResponseField name="author" type="string">
          Handle of the user who triggered the deployment. Optional lookup argument.
        </ResponseField>
        <ResponseField name="canceled_hosts" type="number">
          Number of hosts on which the deployment was canceled. Optional lookup argument.
        </ResponseField>
        <ResponseField name="config_operations" type="list(object)">
          Ordered list of configuration file operations applied by this deployment. Absent for package deployments, which have no configuration file operations. Optional lookup argument.
        </ResponseField>
        <ResponseField name="duration_seconds" type="number">
          Duration of the deployment in seconds, computed as `finished_at - started_at`. Zero if the deployment has not finished. Optional lookup argument.
        </ResponseField>
        <ResponseField name="error_summary" type="string">
          Top-level error message for the deployment. Populated only when the deployment has failed. Optional lookup argument.
        </ResponseField>
        <ResponseField name="estimated_finished_at" type="number">
          Estimated completion time of the deployment as a Unix timestamp. Zero if not available. Optional lookup argument.
        </ResponseField>
        <ResponseField name="failed_hosts" type="number">
          Number of hosts on which the deployment failed. Optional lookup argument.
        </ResponseField>
        <ResponseField name="high_level_status" type="string">
          Current high-level status of the deployment (for example, "pending", "running", "completed", "failed"). Optional lookup argument.
        </ResponseField>
        <ResponseField name="hosts" type="list(object)">
          Per-host status list for this deployment. Optional lookup argument.
        </ResponseField>
        <ResponseField name="is_scheduled" type="bool">
          Whether this deployment was triggered by a schedule (`schedule_id` is non-empty). Optional lookup argument.
        </ResponseField>
        <ResponseField name="query" type="string">
          Query used to filter and select target hosts for the deployment. Optional lookup argument.
        </ResponseField>
        <ResponseField name="running_hosts" type="number">
          Number of hosts on which the deployment is currently running. Optional lookup argument.
        </ResponseField>
        <ResponseField name="schedule_id" type="string">
          Identifier of the schedule that triggered this deployment. Empty if triggered manually. Optional lookup argument.
        </ResponseField>
        <ResponseField name="skipped_hosts" type="number">
          Number of hosts that were skipped during the deployment. Optional lookup argument.
        </ResponseField>
        <ResponseField name="succeeded_hosts" type="number">
          Number of hosts on which the deployment succeeded. Optional lookup argument.
        </ResponseField>
        <ResponseField name="target_versions" type="list(string)">
          Distinct package versions targeted by this deployment, in first-seen order. Optional lookup argument.
        </ResponseField>
        <ResponseField name="total_hosts" type="number">
          Total number of hosts targeted by this deployment. Optional lookup argument.
        </ResponseField>
        <ResponseField name="update_type" type="string">
          Type of update operation performed by this deployment (for example, "update_config_operations", "update_package"). Optional lookup argument.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="id" type="string" required>
      Unique identifier for the deployment. Required lookup argument.
    </ResponseField>
    <ResponseField name="type" type="string" required>
      The type of deployment resource. Required lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>
