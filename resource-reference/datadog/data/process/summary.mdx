## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	process "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/process"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_process_summary"})

    		ubx.Data(process.Summary, "example", process.SummaryConfig{
    			From:   1,
    			Search: "example",
    			Tags:   "example",
    			To:     1,
    		})
    	}))
    }
    ```
  </Tab>
