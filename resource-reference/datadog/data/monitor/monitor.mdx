`datadog_monitor` looks up a real Monitor -- Datadog's own core detection mechanism, watching a metric, an integration's own availability, or a network endpoint against a threshold and alerting when it's crossed.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	monitor "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/monitor"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_monitor"})

    		ubx.Data(monitor.Monitor, "example", monitor.MonitorConfig{
    			MonitorId: 1,
    		})
    	}))
    }
    ```
