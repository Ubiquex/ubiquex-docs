`datadog_monitor` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	monitor "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/monitor"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_monitor"})

    		ubx.Data(monitor.Monitor, "example", monitor.MonitorConfig{
    			MonitorId: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Monitor } from "@ubx/sdk-datadog/datadog/data/monitor/monitor";

    export default stack("example", () => {
      intent({ summary: "look up datadog_monitor" });

      data(Monitor, "example", {
        monitorId: 1,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.monitor import Monitor, MonitorConfig

    def describe():
        ubx.intent("look up datadog_monitor")

        ubx.data(Monitor, "example", MonitorConfig(
            monitor_id=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
