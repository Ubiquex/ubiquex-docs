## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	incident "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/incident"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_incident_rule_data_response"})

    		ubx.Data(incident.RuleDataResponse, "example", incident.RuleDataResponseConfig{
    			IncidentTypeUuid: "example",
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.datadog.data.incident import RuleDataResponse, RuleDataResponseConfig

    def describe():
        ubx.intent("look up datadog_incident_rule_data_response")

        ubx.data(RuleDataResponse, "example", RuleDataResponseConfig(
            incident_type_uuid="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
