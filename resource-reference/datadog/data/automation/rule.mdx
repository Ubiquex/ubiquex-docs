## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_automation_rule"})

    		ubx.Data(automation.Rule, "example", automation.RuleConfig{
    			ProjectId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```typescript
    export default stack("example", () => {
      intent({ summary: "look up datadog_automation_rule" });

      });

    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.automation import Rule, RuleConfig

    def describe():
        ubx.intent("look up datadog_automation_rule")

        ubx.data(Rule, "example", RuleConfig(
            project_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
