`datadog_security_entity_risk_score` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_security_entity_risk_score"})

    		ubx.Data(security.EntityRiskScore, "example", security.EntityRiskScoreConfig{
    			EntityType: []string{"example"},
    			From:       1,
    			To:         1,
    		})
    	}))
    }
