## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	security "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/security"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_security_findings_data"})

    		ubx.Data(security.FindingsData, "example", security.FindingsDataConfig{
    			Sort: "example",
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx

    def describe():
        ubx.intent("look up datadog_security_findings_data")
