## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_web_integration_account_response_data"})

    		ubx.Data(web.IntegrationAccountResponseData, "example", web.IntegrationAccountResponseDataConfig{
    			IntegrationName: "example-integration-name",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
