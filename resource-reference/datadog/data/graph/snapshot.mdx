    			EventQuery: "example",

    		})

    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.graph import Snapshot, SnapshotConfig

    def describe():
        ubx.intent("look up datadog_graph_snapshot")

        ubx.data(Snapshot, "example", SnapshotConfig(
            end=1,
            start=1,
            event_query="example",
            height=1,
            title="example",
            width=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="end" type="number" required>
  The real, inclusive end of the time range this request covers. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="event_query" type="string">
  The real query string evaluated against matching events. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="height" type="number">
  The real height of this object, in pixels. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="start" type="number" required>
  The real, inclusive start of the time range this request covers. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="title" type="string">
  The real, human-readable title of this object. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="width" type="number">
  The real width of this object, in pixels. **(AI-inferred)** Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="graph_def" type="string">
  The real, underlying graph definition (widget query and display options) this request is scoped to. **(AI-inferred)** Returned by the lookup.
</ResponseField>

<ResponseField name="metric_query" type="string">
  The real metrics query string being evaluated. **(AI-inferred)** Returned by the lookup.
</ResponseField>

<ResponseField name="snapshot_url" type="string">
  URL of your [graph snapshot](https://docs.datadoghq.com/metrics/explorer/#snapshot). Returned by the lookup.
</ResponseField>
