`datadog_graph_snapshot` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	graph "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/graph"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_graph_snapshot"})

    		ubx.Data(graph.Snapshot, "example", graph.SnapshotConfig{
    			End:        1,
    			Start:      1,
    			EventQuery: "example",
    			Height:     1,
    			Title:      "example",
    			Width:      1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.datadog.data.graph import Snapshot, SnapshotConfig

    def describe():
