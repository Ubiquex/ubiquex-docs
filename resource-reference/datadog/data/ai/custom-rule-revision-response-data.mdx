    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	ai "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/ai"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_ai_custom_rule_revision_response_data"})

    		ubx.Data(ai.CustomRuleRevisionResponseData, "example", ai.CustomRuleRevisionResponseDataConfig{
    			RuleName:    "example-rule-name",
    			RulesetName: "example-ruleset-name",
    		})
    	}))
    }
    ```

## Lookup arguments

<ResponseField name="rule_name" type="string" required>
  The real name of this rule. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="ruleset_name" type="string" required>
  The real name of this ruleset. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="data" type="list(object)">
  The list of AI custom rule revisions. Returned by the lookup.
</ResponseField>
