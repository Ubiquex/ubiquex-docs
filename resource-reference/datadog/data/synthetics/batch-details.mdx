## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	synthetics "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/synthetics"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_synthetics_batch_details"})

    		ubx.Data(synthetics.BatchDetails, "example", synthetics.BatchDetailsConfig{
    			BatchId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { BatchDetails } from "./local-sdk/datadog/sdk/typescript/datadog/data/synthetics/batch_details.ts";

    export default stack("example", () => {
      intent({ summary: "look up datadog_synthetics_batch_details" });

      data(BatchDetails, "example", {
        batchId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.synthetics import BatchDetails, BatchDetailsConfig

    def describe():
        ubx.intent("look up datadog_synthetics_batch_details")

        ubx.data(BatchDetails, "example", BatchDetailsConfig(
            batch_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Lookup arguments

<ResponseField name="batch_id" type="string" required>
  The real, unique identifier of this batch of test runs. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="data" type="object">
  Wrapper object that contains the details of a batch. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="metadata" type="object">
      Metadata for the Synthetic tests run. Optional lookup argument.
      <Expandable title="properties">
        <ResponseField name="ci" type="object">
          Description of the CI provider. Optional lookup argument.
          <Expandable title="properties">
            <ResponseField name="pipeline" type="object">
              Description of the CI pipeline. Optional lookup argument.
              <Expandable title="properties">
                <ResponseField name="url" type="string">
                  URL of the pipeline. Optional lookup argument.
                </ResponseField>
              </Expandable>
            </ResponseField>
            <ResponseField name="provider" type="object">
              Description of the CI provider. Optional lookup argument.
              <Expandable title="properties">
                <ResponseField name="name" type="string">
                  Name of the CI provider. Optional lookup argument.
                </ResponseField>
              </Expandable>
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="git" type="object">
          Git information. Optional lookup argument.
          <Expandable title="properties">
            <ResponseField name="branch" type="string">
              Branch name. Optional lookup argument.
            </ResponseField>
            <ResponseField name="commit_sha" type="string">
              The commit SHA. Optional lookup argument.
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="results" type="list(object)">
      List of results for the batch. Optional lookup argument.
    </ResponseField>
    <ResponseField name="status" type="string">
      Determines whether the batch has passed, failed, or is in progress. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>
