`datadog_google_chat_organization_handle_response_data` reads the real handle identifying a Google Workspace organization connected via Datadog's Google Chat integration.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	google "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/google"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_google_chat_organization_handle_response_data"})

    		ubx.Data(google.ChatOrganizationHandleResponseData, "example", google.ChatOrganizationHandleResponseDataConfig{
    			OrganizationBindingId: "example",
    		})
    	}))
    }
    ```
