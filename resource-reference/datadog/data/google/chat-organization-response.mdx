## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	google "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/google"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_google_chat_organization_response"})

    		ubx.Data(google.ChatOrganizationResponse, "example", google.ChatOrganizationResponseConfig{
    			OrganizationBindingId: "example",
    		})
    	}))
    }
    ```
