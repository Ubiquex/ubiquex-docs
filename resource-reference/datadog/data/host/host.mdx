`datadog_host` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	host "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/host"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_host"})

    		ubx.Data(host.Host, "example", host.HostConfig{
    			Count:                 1,
    			Filter:                "example",
    			From:                  1,
    			IncludeHostsMetadata:  true,
    			IncludeMutedHostsData: true,
    			SortDir:               "example",
    			SortField:             "example",
    			Start:                 1,
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.datadog.data.host import Host, HostConfig

    def describe():
        ubx.intent("look up datadog_host")

        ubx.data(Host, "example", HostConfig(
            count=1,
            filter="example",
            from_=1,
            include_hosts_metadata=True,
            include_muted_hosts_data=True,

    if __name__ == "__main__":
