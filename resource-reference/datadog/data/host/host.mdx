    			From:                  1,

    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.host import Host, HostConfig

    def describe():
        ubx.intent("look up datadog_host")

        ubx.data(Host, "example", HostConfig(
            count=1,
            filter="example",
            from_=1,
            include_hosts_metadata=True,
            include_muted_hosts_data=True,
            sort_dir="example",
            sort_field="example",
            start=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

<ResponseField name="sort_field" type="string">

</ResponseField>
