`datadog_host_totals` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Datadog's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	host "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/host"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_host_totals"})

    		ubx.Data(host.Totals, "example", host.TotalsConfig{
    			From: 1,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Totals } from "@ubx/sdk-datadog/datadog/data/host/totals";

    export default stack("example", () => {
      intent({ summary: "look up datadog_host_totals" });

      data(Totals, "example", {

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.host import Totals, TotalsConfig

    def describe():
        ubx.intent("look up datadog_host_totals")

        ubx.data(Totals, "example", TotalsConfig(
            from_=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
