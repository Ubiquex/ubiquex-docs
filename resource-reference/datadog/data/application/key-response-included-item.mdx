`datadog_application_key_response_included_item` reads a real, related application key item included alongside a broader lookup -- the same redacted view `datadog_partial_application_key_response` represents as a resource, never exposing the real secret value.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	application "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/application"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_application_key_response_included_item"})

    		ubx.Data(application.KeyResponseIncludedItem, "example", application.KeyResponseIncludedItemConfig{
    			AppKeyId: "example",
    		})
    	}))
    }
    ```
