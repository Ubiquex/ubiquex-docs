## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	team "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/team"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_team_permission_setting"})

    		ubx.Data(team.PermissionSetting, "example", team.PermissionSettingConfig{
    			TeamId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk

    export default stack("example", () => {
      intent({ summary: "look up datadog_team_permission_setting" });

      data(PermissionSetting, "example", {
        teamId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.team import PermissionSetting, PermissionSettingConfig

    def describe():
        ubx.intent("look up datadog_team_permission_setting")

        ubx.data(PermissionSetting, "example", PermissionSettingConfig(
            team_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
