title: "datadog_team_link"

---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	team "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/team"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_team_link"})

    		ubx.Data(team.Link, "example", team.LinkConfig{
    			TeamId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Link } from "@ubx/sdk-datadog/datadog/data/team/link";

    export default stack("example", () => {
      intent({ summary: "look up datadog_team_link" });

      data(Link, "example", {
        teamId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.team import Link, LinkConfig

    def describe():
        ubx.intent("look up datadog_team_link")

        ubx.data(Link, "example", LinkConfig(
            team_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
