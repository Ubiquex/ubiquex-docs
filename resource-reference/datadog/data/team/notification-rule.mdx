## Example

<Tabs>

    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	team "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/team"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_team_notification_rule"})

    		ubx.Data(team.NotificationRule, "example", team.NotificationRuleConfig{
    			TeamId: "example",
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.datadog.data.team import NotificationRule, NotificationRuleConfig

    def describe():
        ubx.intent("look up datadog_team_notification_rule")

        ubx.data(NotificationRule, "example", NotificationRuleConfig(
            team_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
