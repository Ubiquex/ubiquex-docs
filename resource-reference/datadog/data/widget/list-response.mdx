    import ubx_sdk as ubx
    from ubx.datadog.widget import ListResponse, ListResponseConfig
    from ubx.datadog.widget.list_response import ListResponse_Data, ListResponse_Data_Attributes, ListResponse_Data_Attributes_Definition

    def describe():
        ubx.intent("example own list response")

        ubx.resource(ListResponse, "example", ListResponseConfig(
            data=ListResponse_Data(attributes=ListResponse_Data_Attributes(definition=ListResponse_Data_Attributes_Definition(title="example", type="example")), type="example"),
            experience_type="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="experience_type" type="string" required>
  Widget experience types that differentiate between the products using the specific widget. Required lookup argument.
</ResponseField>

<ResponseField name="sort" type="string">
  The field results are sorted by. **(AI-inferred)** Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="data" type="list(object)">
  List of widget resources. Returned by the lookup.
</ResponseField>

<ResponseField name="included" type="list(object)">
  Array of user resources related to the widgets. Returned by the lookup.
</ResponseField>

<ResponseField name="meta" type="object">
  Metadata about the search results. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="created_by_anyone_total" type="number">
      Total number of widgets created by anyone. Optional lookup argument.
    </ResponseField>
    <ResponseField name="created_by_you_total" type="number">
      Total number of widgets created by the current user. Optional lookup argument.
    </ResponseField>
    <ResponseField name="favorited_by_you_total" type="number">
      Total number of widgets favorited by the current user. Optional lookup argument.
    </ResponseField>
    <ResponseField name="filtered_total" type="number">
      Total number of widgets matching the current filter criteria. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>
