    		})
    	}))
    }

    from ubx.datadog.data.trace import Response, ResponseConfig

        ubx.intent("look up datadog_trace_response")

            trace_id="example",

        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="include_fields" type="list(string)">
  Which real, additional fields to include alongside the response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="trace_id" type="string" required>
  The real, unique identifier of this APM trace. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="data" type="object">
  A trace resource document. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="attributes" type="object" required>
      The attributes of a trace returned by the Get trace by ID endpoint. Required lookup argument.
      <Expandable title="properties">
        <ResponseField name="is_truncated" type="bool" required>
          Indicates whether the trace was truncated because its size exceeded the maximum response payload. Required lookup argument.
        </ResponseField>
        <ResponseField name="spans" type="list(object)" required>
          The list of spans that compose the trace. Required lookup argument.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="id" type="string" required>
      The full 128-bit trace ID, encoded as a 32-character hexadecimal string. Required lookup argument.
    </ResponseField>
    <ResponseField name="type" type="string" required>
      The type of the trace resource. The value is always `trace`. Required lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>
