    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	blueprint "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/blueprint"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_blueprint_metadata_data"})

    		ubx.Data(blueprint.MetadataData, "example", blueprint.MetadataDataConfig{
    			Limit: 1,
    			Page:  1,
    		})
    	}))
    }

    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.blueprint import MetadataData, MetadataDataConfig

    def describe():
        ubx.intent("look up datadog_blueprint_metadata_data")

        ubx.data(MetadataData, "example", MetadataDataConfig(
            limit=1,
            page=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
