    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	dataset "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/dataset"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_dataset"})

    		ubx.Data(dataset.Dataset, "example", dataset.DatasetConfig{
    			DatasetId: "example",
    			ProjectId: "example",
    			Format:    "example",
    			Version:   1,
    		})
    	}))
    }

    from ubx.datadog.data.dataset import Dataset, DatasetConfig

    def describe():
        ubx.intent("look up datadog_dataset")

        ubx.data(Dataset, "example", DatasetConfig(
            dataset_id="example",
            project_id="example",
            format="example",
            version=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
