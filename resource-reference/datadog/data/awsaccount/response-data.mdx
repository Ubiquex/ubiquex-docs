## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	awsaccount "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/awsaccount"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_awsaccount_response_data"})

    		ubx.Data(awsaccount.ResponseData, "example", awsaccount.ResponseDataConfig{
    			AwsAccountId: "example",
    		})
    	}))
    }
    ```

    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.awsaccount import ResponseData, ResponseDataConfig

    def describe():
        ubx.intent("look up datadog_awsaccount_response_data")

        ubx.data(ResponseData, "example", ResponseDataConfig(
            aws_account_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
