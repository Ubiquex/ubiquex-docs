## Lookup arguments

_This lookup takes no arguments._

## Result properties

<ResponseField name="data" type="object">
  The RUM configuration data. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="attributes" type="object" required>
      Attributes of the RUM configuration. Required lookup argument.
      <Expandable title="properties">
        <ResponseField name="disabled" type="bool">
          Whether the RUM configuration is disabled for the organization. Optional lookup argument.
        </ResponseField>
        <ResponseField name="enforced_application_tags" type="bool" required>
          Whether application tags are enforced for the RUM applications in the organization. Required lookup argument.
        </ResponseField>
        <ResponseField name="enforced_application_tags_updated_at" type="string">
          Timestamp of when the enforced application tags setting was last updated. Optional lookup argument.
        </ResponseField>
        <ResponseField name="enforced_application_tags_updated_by" type="string">
          Handle of the user who last updated the enforced application tags setting. Optional lookup argument.
        </ResponseField>
        <ResponseField name="ootb_metrics_version" type="number">
          Version of the out-of-the-box metrics installed for the organization. Optional lookup argument.
        </ResponseField>
        <ResponseField name="ootb_metrics_version_installed_at" type="string">
          Timestamp of when the out-of-the-box metrics version was installed. Optional lookup argument.
        </ResponseField>
        <ResponseField name="retention_filters_enabled" type="bool" required>
          Whether retention filters are enabled for the organization. Required lookup argument.
        </ResponseField>
        <ResponseField name="retention_filters_enabled_updated_at" type="string">
          Timestamp of when the retention filters setting was last updated. Optional lookup argument.
        </ResponseField>
        <ResponseField name="retention_filters_enabled_updated_by" type="string">
          Handle of the user or job who last updated the retention filters setting. Optional lookup argument.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="id" type="string" required>
      The organization ID associated with the RUM configuration. Required lookup argument.
    </ResponseField>
    <ResponseField name="type" type="string" required>
      The type of the resource. The value should always be `rum_config`. Required lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>
