`datadog_rum_permanent_retention_filter_data` looks up a real RUM permanent retention filter -- a rule exempting matching RUM events from standard retention limits.

## Example

    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	rum "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/rum"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_rum_permanent_retention_filter_data"})

    		})
