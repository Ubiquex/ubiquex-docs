    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go

    }

## Lookup arguments

<ResponseField name="page" type="number">
  The page number of results to fetch, for a paginated response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="query" type="string">
  The real query string evaluated against the underlying data. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="size" type="number">
  The maximum number of results to return. **(AI-inferred)** Optional lookup argument.
</ResponseField>

## Result properties

<ResponseField name="data" type="list(object)">
  The list of unified hosts for the current page. Returned by the lookup.
</ResponseField>

<ResponseField name="meta" type="object">
  Pagination metadata for a unified hosts list response. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="page_index" type="number" required>
      The current page index (zero-based). Required lookup argument.
    </ResponseField>
    <ResponseField name="page_size" type="number" required>
      The number of hosts returned per page. Required lookup argument.
    </ResponseField>
    <ResponseField name="total_filtered" type="number" required>
      The total number of hosts matching the filter criteria. Required lookup argument.
    </ResponseField>
    <ResponseField name="total_pages" type="number" required>
      The total number of pages available. Required lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>
