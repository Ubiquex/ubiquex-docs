## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	get "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/data/get"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_get_device_response"})

    		ubx.Data(get.DeviceResponse, "example", get.DeviceResponseConfig{
    			DeviceId: "example",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
