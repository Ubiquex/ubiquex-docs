    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up datadog_get_interfaces_data"})

    		ubx.Data(get.InterfacesData, "example", get.InterfacesDataConfig{
    			DeviceId:       "example",
    			GetIpAddresses: true,
    		})
    	}))
    }

    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.get import InterfacesData, InterfacesDataConfig

    def describe():
        ubx.intent("look up datadog_get_interfaces_data")

        ubx.data(InterfacesData, "example", InterfacesDataConfig(
            device_id="example",
            get_ip_addresses=True,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
