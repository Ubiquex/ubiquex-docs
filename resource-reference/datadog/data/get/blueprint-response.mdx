    package main

    		})

    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.data.get import BlueprintResponse, BlueprintResponseConfig

    def describe():
        ubx.intent("look up datadog_get_blueprint_response")

        ubx.data(BlueprintResponse, "example", BlueprintResponseConfig(
            blueprint_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
