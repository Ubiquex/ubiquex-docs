`datadog_pager_duty_service_name` maps one PagerDuty service into Datadog by its own service name and integration key, which creates a real `@pagerduty-<serviceName>` notification handle. Adding that handle to a monitor's own alert text is what actually pages the right PagerDuty service when the monitor breaches, with incident severity mapped automatically from the monitor's own status.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	pager "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/pager"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own duty service name"})

    		ubx.Resource(pager.DutyServiceName, "ci-runner", pager.DutyServiceNameConfig{
    			ServiceKey:  "example",
    			ServiceName: "example-service-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { DutyServiceName } from "./local-sdk/datadog/sdk/typescript/datadog/pager/duty_service_name.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own duty service name" });

      resource(DutyServiceName, "ci-runner", {
        serviceKey: "example",
        serviceName: "example-service-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.pager import DutyServiceName, DutyServiceNameConfig

    def describe():
        ubx.intent("payments own duty service name")

        ubx.resource(DutyServiceName, "ci-runner", DutyServiceNameConfig(
            service_key="example",
            service_name="example-service-name",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
