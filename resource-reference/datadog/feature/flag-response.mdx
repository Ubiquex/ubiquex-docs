`datadog_feature_flag_response` is a real feature flag as tracked by Datadog's own feature-flag correlation surface -- letting a team see a flag's real rollout state alongside the metrics, traces, and errors it may be causing.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	feature "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/feature"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own flag response"})

    		ubx.Resource(feature.FlagResponse, "example", feature.FlagResponseConfig{
    			Data:          map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			FeatureFlagId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.feature import FlagResponse, FlagResponseConfig

    def describe():
        ubx.intent("example own flag response")

        ubx.resource(FlagResponse, "example", FlagResponseConfig(
            data={"attributes": {"name": "example-name"}},
            feature_flag_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
