`datadog_incident_responder_response` is a real person or team assigned a responder role on a Datadog incident -- who's actively working the incident, distinct from who merely has visibility into it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	incident "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/incident"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own responder response"})

    		ubx.Resource(incident.ResponderResponse, "example", incident.ResponderResponseConfig{
    			Data:        map[string]any{"relationships": map[string]any{"user": map[string]any{"data": map[string]any{"id": "example"}}}},
    			IncidentId:  "example",
    			ResponderId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import { intent, resource, stack } from "@ubx/sdk";
