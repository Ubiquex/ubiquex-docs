`datadog_incident_type_response` is a real, organization-defined category of incident (a security incident vs. an availability incident, for example) -- used to route and report on incidents differently depending on what kind of event they actually are.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	incident "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/incident"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own type response"})

    		ubx.Resource(incident.TypeResponse, "example", incident.TypeResponseConfig{
    			Data:           map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			IncidentTypeId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { TypeResponse } from "@ubx/sdk-datadog/datadog/incident/type_response";

    export default stack("example", () => {
    ```

    ```python
    import ubx_sdk as ubx

        ubx.run("example", describe)
    ```
