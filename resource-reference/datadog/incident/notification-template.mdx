`datadog_incident_notification_template` is a real, reusable message template Datadog Incident Management notifications are rendered from -- keeping incident update messages consistent in format across a team rather than hand-written each time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	incident "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/incident"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own notification template"})

    		ubx.Resource(incident.NotificationTemplate, "example", incident.NotificationTemplateConfig{
    			Data: map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			Id:   "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";

    export default stack("example", () => {
      intent({ summary: "example own notification template" });

      resource(NotificationTemplate, "example", {
    ```

    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.incident import NotificationTemplate, NotificationTemplateConfig

    def describe():
        ubx.intent("example own notification template")

        ubx.resource(NotificationTemplate, "example", NotificationTemplateConfig(
            data={"attributes": {"name": "example-name"}},
            id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="data" type="object" required>
  Notification template data for a create request. Required.
  <Expandable title="properties">
    <ResponseField name="attributes" type="object" required>
      The attributes for creating a notification template. Required.
      <Expandable title="properties">
        <ResponseField name="category" type="string" required>
          The category of the notification template. Required.
        </ResponseField>
        <ResponseField name="content" type="string" required>
          The content body of the notification template. Required.
        </ResponseField>
        <ResponseField name="name" type="string" required>
          The name of the notification template. Required.
        </ResponseField>
        <ResponseField name="subject" type="string" required>
          The subject line of the notification template. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="relationships" type="object">
      The definition of `NotificationTemplateCreateDataRelationships` object. Optional.
      <Expandable title="properties">
        <ResponseField name="incident_type" type="object">
          Relationship to an incident type. Optional.
          <Expandable title="properties">
            <ResponseField name="data" type="object" required>
              Relationship to incident type object. Required.
              <Expandable title="properties">
                <Note>`data` recurs here (its own type contains itself); see the `data` field's own properties above for the repeating pattern.</Note>
              </Expandable>
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="type" type="string" required>
      Notification templates resource type. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="id" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>
