`datadog_user_invitation_response` is a real, pending invitation for someone to join a Datadog organization -- the invited email, the role they'll be granted, and the invitation's own real status until it's accepted.

## Example

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own invitation response"})

    		ubx.Resource(user.InvitationResponse, "example", user.InvitationResponseConfig{
    			Data:               []map[string]any{{"relationships": map[string]any{"user": map[string]any{"data": map[string]any{"id": "example"}}}}},
    			UserInvitationUuid: "example",
    		})
    	}))
    }
