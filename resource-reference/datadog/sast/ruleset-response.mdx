`datadog_sast_ruleset_response` is a real, named collection of Static Application Security Testing (SAST) rules -- the ruleset used to scan source code for real vulnerability patterns as part of Datadog Code Security.

## Example

    package main

    	sast "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/sast"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own ruleset response"})

    		ubx.Resource(sast.RulesetResponse, "example", sast.RulesetResponseConfig{
    			RulesetName: "example-ruleset-name",
    		})
    	}))
    }
