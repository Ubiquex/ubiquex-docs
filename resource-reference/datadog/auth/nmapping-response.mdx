`datadog_auth_nmapping_response` is a real authentication-to-identity mapping used by Datadog's own auth-n integration layer -- associating an external identity (from an upstream auth provider) with the internal principal Datadog authorization decisions are made against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	auth "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/auth"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own nmapping response"})

    		ubx.Resource(auth.NmappingResponse, "example", auth.NmappingResponseConfig{
    			AuthnMappingId: "example",
    			Data:           map[string]any{"type": "example"},
    		})
    	}))
    }
    ```
  </Tab>
