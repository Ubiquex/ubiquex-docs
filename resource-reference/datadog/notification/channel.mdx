`datadog_notification_channel` is a real, general-purpose configured destination for Datadog notifications -- an umbrella concept spanning the many concrete channel types (Slack, email, PagerDuty, webhook, and similar) a monitor, monitor, or workflow can notify.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	notification "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/notification"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own channel"})

    		ubx.Resource(notification.Channel, "example", notification.ChannelConfig{
    			ChannelId: "example",
    			Data:      map[string]any{"type": "example"},
    			UserId:    "example",
    		})
    	}))
    }
    ```
  </Tab>
