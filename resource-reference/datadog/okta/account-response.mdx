`datadog_okta_account_response` is a real Okta account connected to Datadog's own Okta integration -- the credentials Datadog uses to pull Okta identity and authentication events into Datadog, commonly for Cloud SIEM detection.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	okta "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/okta"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own account response"})

    		ubx.Resource(okta.AccountResponse, "example", okta.AccountResponseConfig{
    			AccountId: "example",
    			Data:      map[string]any{"attributes": map[string]any{"name": "example-name"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    import ubx_sdk as ubx
    from ubx.datadog.okta.account_response import AccountResponse, AccountResponseConfig

    def describe():
        ubx.intent("example own account response")

            account_id="example",

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
