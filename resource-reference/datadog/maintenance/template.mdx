`datadog_maintenance_template` is a reusable template a `datadog_maintenance` window is created from -- the real, saved scope and suppression rules a team applies repeatedly rather than reconfiguring a maintenance window from scratch each time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	maintenance "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/maintenance"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own template"})

    		ubx.Resource(maintenance.Template, "example", maintenance.TemplateConfig{
    			PageId:     "example",
    			TemplateId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Template } from "./local-sdk/datadog/sdk/typescript/datadog/maintenance/template.ts";

    export default stack("example", () => {
      intent({ summary: "example own template" });

      resource(Template, "example", {
        pageId: "example",
        templateId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.maintenance import Template, TemplateConfig

    def describe():
        ubx.intent("example own template")

        ubx.resource(Template, "example", TemplateConfig(
            page_id="example",
            template_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
