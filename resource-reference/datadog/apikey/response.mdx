`datadog_apikey_response` is a real Datadog organization API key -- the credential the Datadog Agent and integrations use to submit metrics, traces, and logs. Distinct from `datadog_api_key_response` (v1): this is v2's own richer representation of the identical real concept, with `datadog_partial_application_key_response` and `datadog_application_key_response` as its companion credential.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	apikey "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/apikey"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own response"})

    		ubx.Resource(apikey.Response, "example", apikey.ResponseConfig{
    			ApiKeyId: "example",
    			Data:     map[string]any{"attributes": map[string]any{"name": "example-name"}},
    		})
    	}))
    }
    ```
  </Tab>

    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.apikey import Response, ResponseConfig

    def describe():
        ubx.intent("example own response")

        ubx.resource(Response, "example", ResponseConfig(
            api_key_id="example",
            data={"attributes": {"name": "example-name"}},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

</ResponseField>
