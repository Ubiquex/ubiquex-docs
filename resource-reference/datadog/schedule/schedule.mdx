## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	schedule "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/schedule"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own schedule"})

    		ubx.Resource(schedule.Schedule, "example", schedule.ScheduleConfig{
    			Data:       map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			ScheduleId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Schedule } from "./local-sdk/datadog/sdk/typescript/datadog/schedule/schedule.ts";

    export default stack("example", () => {
      intent({ summary: "example own schedule" });

      resource(Schedule, "example", {
        data: { attributes: { name: "example-name" } },
        scheduleId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.schedule import Schedule, ScheduleConfig

    def describe():
        ubx.intent("example own schedule")

        ubx.resource(Schedule, "example", ScheduleConfig(
            data={"attributes": {"name": "example-name"}},
            schedule_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
