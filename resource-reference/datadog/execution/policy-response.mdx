`datadog_execution_policy_response` is a real policy governing how a Datadog Workflow Automation workflow is allowed to run -- real constraints like which connections/credentials it may use and under what conditions, enforced by Datadog before a workflow executes.

## Example

<Tabs>

    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	execution "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/execution"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own policy response"})

    		ubx.Resource(execution.PolicyResponse, "example", execution.PolicyResponseConfig{
    			Data:     map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			PolicyId: "example",
    		})
    	}))
    }
    ```
