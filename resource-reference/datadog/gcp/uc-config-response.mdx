`datadog_gcp_uc_config_response` is Datadog's own configured connection to Google Cloud's Usage/Cost data export -- the GCP counterpart to `datadog_aws_cur_config_response`, powering granular GCP cost visibility inside Cloud Cost Management.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	gcp "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/gcp"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own uc config response"})

    		ubx.Resource(gcp.UcConfigResponse, "example", gcp.UcConfigResponseConfig{
    			CloudAccountId: "example",
    			Data:           map[string]any{"type": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
