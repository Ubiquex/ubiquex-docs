`datadog_gcp_scan_options` is the GCP counterpart to `datadog_aws_scan_options_response`/`datadog_azure_scan_options` -- the real, configured scope and cadence Datadog Cloud Security Management uses when scanning a connected Google Cloud project.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	gcp "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/gcp"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own scan options"})

    		ubx.Resource(gcp.ScanOptions, "example", gcp.ScanOptionsConfig{
    			ProjectId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { ScanOptions } from "@ubx/sdk-datadog/datadog/gcp/scan_options";

    export default stack("example", () => {
      intent({ summary: "example own scan options" });

      resource(ScanOptions, "example", {
        projectId: "example",

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.gcp.scan_options import ScanOptions, ScanOptionsConfig

    def describe():
        ubx.intent("example own scan options")

            project_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="data" type="object">
  Single GCP scan options entry. Optional; if omitted, computed by Datadog.
  <Expandable title="properties">
    <ResponseField name="attributes" type="object">
      Attributes for GCP scan options configuration. Optional.
      <Expandable title="properties">
        <ResponseField name="cloud_function" type="bool">
          Indicates if scanning of Cloud Functions is enabled. Optional.
        </ResponseField>
        <ResponseField name="compliance_host" type="bool">
          Indicates whether host compliance scanning is enabled. Optional.
        </ResponseField>
        <ResponseField name="vuln_containers_os" type="bool">
          Indicates if scanning for vulnerabilities in containers is enabled. Optional.
        </ResponseField>
        <ResponseField name="vuln_host_os" type="bool">
          Indicates if scanning for vulnerabilities in hosts is enabled. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="id" type="string" required>
      The GCP project ID. Required.
    </ResponseField>
    <ResponseField name="type" type="string" required>
      GCP scan options resource type. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="project_id" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>
