`datadog_awsaccount_response` is a real AWS account connected to Datadog's own AWS integration -- the account ID, IAM role Datadog assumes, and which AWS namespaces/services are enabled for metric collection from that specific account.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	awsaccount "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/awsaccount"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own response"})

    		ubx.Resource(awsaccount.Response, "example", awsaccount.ResponseConfig{
    			AwsAccountConfigId: "example",
    			Data:               map[string]any{"attributes": map[string]any{"auth_config": map[string]any{"access_key_id": "example"}}},
    		})
    	}))
    }
    ```
  </Tab>

    import { intent, resource, stack } from "@ubx/sdk";
