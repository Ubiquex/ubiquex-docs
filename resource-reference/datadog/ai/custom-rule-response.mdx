`datadog_ai_custom_rule_response` is a custom detection rule for Datadog's AI/LLM observability surface -- a condition written against LLM Observability trace data (prompts, completions, tool calls) that flags a real problem, like a policy violation or a quality regression, as traces flow through a monitored application.

## Example

    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	ai "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/ai"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own custom rule response"})

    		ubx.Resource(ai.CustomRuleResponse, "example", ai.CustomRuleResponseConfig{
    			RuleName:    "example-rule-name",
    			RulesetName: "example-ruleset-name",
    		})
    	}))
    }
    ```
