`datadog_ai_custom_rule_revision_response` is one saved, versioned revision of an `datadog_ai_custom_rule_response` -- the rule's own real edit history, letting a team roll a custom AI detection rule back to a prior version or see exactly what changed and when.

## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	ai "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/ai"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own custom rule revision response"})

    		ubx.Resource(ai.CustomRuleRevisionResponse, "example", ai.CustomRuleRevisionResponseConfig{
    			Id:          "example",
    			RuleName:    "example-rule-name",
    			RulesetName: "example-ruleset-name",
    		})
    	}))
    }
