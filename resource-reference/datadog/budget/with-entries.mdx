`datadog_budget_with_entries` is a real Datadog Cloud Cost Management budget -- a spend target broken down by real budget entries (per team, per time period), used to track actual cloud spend against a plan and alert when a team is trending over budget.

## Example

    package main

    import (
    	budget "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/budget"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own with entries"})

    		ubx.Resource(budget.WithEntries, "example", budget.WithEntriesConfig{
    			BudgetId: "example",
    		})
    	}))
    }
