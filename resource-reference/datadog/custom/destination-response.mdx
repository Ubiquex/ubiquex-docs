`datadog_custom_destination_response` is a real, configured forwarding destination for Datadog Log Forwarding -- an external endpoint (Splunk HEC, a generic HTTP endpoint, cloud storage) logs matching a filter get re-forwarded to, alongside Datadog's own log pipeline.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	custom "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/custom"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own destination response"})

    		ubx.Resource(custom.DestinationResponse, "example", custom.DestinationResponseConfig{
    			CustomDestinationId: "example",
    		})
    	}))
    }
    ```
  </Tab>
