`datadog_custom_rule_response` is a real, general-purpose custom detection rule -- a user-authored condition used across one of Datadog's security or observability products where the built-in rule library doesn't already cover a team's own specific requirement.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	custom "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/custom"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own rule response"})

    		ubx.Resource(custom.RuleResponse, "example", custom.RuleResponseConfig{
    			RuleName:    "example-rule-name",
    			RulesetName: "example-ruleset-name",
    		})
    	}))
    }
    ```
  </Tab>
