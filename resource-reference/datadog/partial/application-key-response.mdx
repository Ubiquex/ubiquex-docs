`datadog_partial_application_key_response` is a real, redacted view of a Datadog application key -- shows the key's own metadata (name, owner, scopes) without exposing its real secret value, the safe-to-list counterpart to the one-time-visible full `datadog_application_key_response`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	partial "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/partial"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own application key response"})

    		ubx.Resource(partial.ApplicationKeyResponse, "example", partial.ApplicationKeyResponseConfig{
    			AppKeyId:         "example",
    			Data:             map[string]any{"attributes": map[string]any{"name": "example-name"}, "type": "example"},
    			ServiceAccountId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import { intent, resource, stack } from "@ubx/sdk";

    ```python
    import ubx_sdk as ubx
    from ubx.datadog.partial import ApplicationKeyResponse, ApplicationKeyResponseConfig
    from ubx.datadog.partial.application_key_response import ApplicationKeyResponse_Data, ApplicationKeyResponse_Data_Attributes

    def describe():
        ubx.intent("example own application key response")

        ubx.resource(ApplicationKeyResponse, "example", ApplicationKeyResponseConfig(
            app_key_id="example",
            data=ApplicationKeyResponse_Data(attributes=ApplicationKeyResponse_Data_Attributes(name="example-name"), type="example"),
            service_account_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="app_key_id" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="data" type="object" required>
  Object used to create an application key. Required.
  <Expandable title="properties">
    <ResponseField name="attributes" type="object" required>
      Attributes used to create an application Key. Required.
      <Expandable title="properties">
        <ResponseField name="name" type="string" required>
          Name of the application key. Required.
        </ResponseField>
        <ResponseField name="scopes" type="list(string)">
          Array of scopes to grant the application key. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="type" type="string" required>
      Application Keys resource type. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="service_account_id" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>
