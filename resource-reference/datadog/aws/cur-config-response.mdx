`datadog_aws_cur_config_response` is Datadog's own configured connection to an AWS Cost and Usage Report (CUR) -- which S3 bucket and report Datadog reads to power real, granular AWS cost visibility inside Cloud Cost Management, beyond what the base AWS integration's billing estimate alone provides.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	aws "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/aws"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own cur config response"})

    		ubx.Resource(aws.CurConfigResponse, "example", aws.CurConfigResponseConfig{
    			CloudAccountId: "example",
    			Data:           map[string]any{"type": "example"},
    		})
    	}))
    }
    ```
  </Tab>
