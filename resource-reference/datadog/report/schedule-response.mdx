`datadog_report_schedule_response` is a real, recurring schedule for generating and delivering a Datadog report (usage, cost, or similar) automatically on a real cadence, rather than requiring someone to run it by hand each time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	report "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/report"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own schedule response"})

    		ubx.Resource(report.ScheduleResponse, "example", report.ScheduleResponseConfig{
    			Data:         map[string]any{"attributes": map[string]any{"description": "example"}},
    			ScheduleUuid: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { ScheduleResponse } from "@ubx/sdk-datadog/datadog/report/schedule_response";

  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.datadog.report import ScheduleResponse, ScheduleResponseConfig

    def describe():
        ubx.intent("example own schedule response")

        ubx.resource(ScheduleResponse, "example", ScheduleResponseConfig(
            data={"attributes": {"description": "example"}},
            schedule_uuid="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="data" type="object" required>
  The JSON:API data object for a report schedule creation request. Required.
  <Expandable title="properties">
    <ResponseField name="attributes" type="object" required>
      The configuration of the report schedule to create. Required.
      <Expandable title="properties">
        <ResponseField name="delivery_format" type="string">
          How a PDF-export report is delivered. `pdf` attaches a PDF file, `png` embeds an inline PNG image, and `pdf_and_png` delivers both. Optional.
        </ResponseField>
        <ResponseField name="description" type="string" required>
          A description of the report, up to 4096 characters. Required.
        </ResponseField>
        <ResponseField name="recipients" type="list(string)" required>
          The recipients of the report. Each entry is an email address, a Slack channel reference in the form `slack:&#123;team_id&#125;.&#123;channel_id&#125;.&#123;channel_name&#125;`, or a Microsoft Teams channel reference in the form `teams:&#123;tenant_id&#125;|&#123;team_id&#125;|&#123;channel_id&#125;`. Required.
        </ResponseField>
        <ResponseField name="resource_id" type="string" required>
          The identifier of the dashboard or integration dashboard to render in the report. Required.
        </ResponseField>
        <ResponseField name="resource_type" type="string" required>
          The type of dashboard resource the report schedule targets. Required.
        </ResponseField>
        <ResponseField name="rrule" type="string" required>
          The recurrence rule for the schedule, expressed as an iCalendar `RRULE` string. Required.
        </ResponseField>
        <ResponseField name="tab_id" type="string">
          The identifier of the dashboard tab to render, when the dashboard has tabs. Optional.
        </ResponseField>
        <ResponseField name="template_variables" type="list(object)" required>
          The dashboard template variables applied when rendering the report. Required.
          <Expandable title="properties">
            <ResponseField name="name" type="string">
              Optional; if omitted, computed by Datadog.
            </ResponseField>
            <ResponseField name="values" type="list(string)">
              Optional; if omitted, computed by Datadog.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="timeframe" type="string" required>
          The relative timeframe of data to include in the report. Required.
        </ResponseField>
        <ResponseField name="timezone" type="string" required>
          The IANA time zone identifier the recurrence rule is evaluated in. Required.
        </ResponseField>
        <ResponseField name="title" type="string" required>
          The title of the report, between 1 and 78 characters. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="type" type="string" required>
      JSON:API resource type for report schedules. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="schedule_uuid" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>
