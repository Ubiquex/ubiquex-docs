`datadog_ruleset_resp` is a real, general-purpose named collection of detection or governance rules -- the general form behind the more specific ruleset types (`datadog_custom_ruleset_response`, `datadog_sast_ruleset_response`) elsewhere in this corpus.

## Example

    package main

    import (
    	ruleset "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/ruleset"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resp"})

    		ubx.Resource(ruleset.Resp, "example", ruleset.RespConfig{
    			RulesetId: "example",
    		})
    	}))
    }
