`datadog_opsgenie_service_response` is a real Opsgenie service connected through Datadog's Opsgenie integration -- lets a Datadog monitor page an Opsgenie service directly, for organizations running Opsgenie as their on-call tool alongside Datadog.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	opsgenie "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/opsgenie"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own service response"})

    		ubx.Resource(opsgenie.ServiceResponse, "example", opsgenie.ServiceResponseConfig{
    			Data:                 map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			IntegrationServiceId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { ServiceResponse } from "@ubx/sdk-datadog/datadog/opsgenie/service_response";

    export default stack("example", () => {
      intent({ summary: "example own service response" });

      resource(ServiceResponse, "example", {
        data: { attributes: { name: "example-name" } },
        integrationServiceId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.opsgenie.service_response import ServiceResponse, ServiceResponseConfig

    def describe():
        ubx.intent("example own service response")

        ubx.resource(ServiceResponse, "example", ServiceResponseConfig(
            data={"attributes": {"name": "example-name"}},
            integration_service_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
