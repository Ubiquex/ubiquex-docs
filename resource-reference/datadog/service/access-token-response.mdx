`datadog_service_access_token_response` is a real credential scoped to a specific service (not a human user) -- used by Datadog's Software Catalog / service-to-service auth surface for machine access rather than a `datadog_personal_access_token_response`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	service "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/service"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own access token response"})

    		ubx.Resource(service.AccessTokenResponse, "example", service.AccessTokenResponseConfig{
    			Data:             map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			ServiceAccountId: "example",
    			TokenId:          "example",
    		})
    	}))
    }
    ```
  </Tab>
