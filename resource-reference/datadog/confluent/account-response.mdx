`datadog_confluent_account_response` is a real Confluent Cloud account connected to Datadog's Confluent Cloud integration -- the credentials Datadog uses to collect Kafka cluster metrics and logs from that account.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	confluent "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/confluent"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own account response"})

    		ubx.Resource(confluent.AccountResponse, "example", confluent.AccountResponseConfig{
    			AccountId: "example",
    			Data:      map[string]any{"attributes": map[string]any{"api_key": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
