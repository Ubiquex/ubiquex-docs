`datadog_confluent_resource_response` is one real Confluent Cloud resource (a Kafka cluster, connector, or similar) Datadog has discovered and is monitoring under a connected `datadog_confluent_account_response`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	confluent "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/confluent"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resource response"})

    		ubx.Resource(confluent.ResourceResponse, "example", confluent.ResourceResponseConfig{
    			AccountId:  "example",
    			Data:       map[string]any{"attributes": map[string]any{"resource_type": "example"}},
    			ResourceId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";

    import ubx_sdk as ubx

            account_id="example",

            resource_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
