`datadog_form_response` is a real, structured form definition used inside Datadog Workflow Automation and App Builder -- the real fields, layout, and validation a user-facing intake form (used to kick off a workflow or gather structured input) is built from.

    import ubx_sdk as ubx

    from ubx.datadog.form.response import Response_Data, Response_Data_Attributes, Response_Data_Attributes_DataDefinition, Response_Data_Attributes_UiDefinition

    def describe():
        ubx.intent("example own response")

            data=Response_Data(attributes=Response_Data_Attributes(data_definition=Response_Data_Attributes_DataDefinition(description="example"), name="example-name", ui_definition=Response_Data_Attributes_UiDefinition()), type="example"),
            form_id="example",
