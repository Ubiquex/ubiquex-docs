`datadog_doradeployment_fetch_response` is a real, queried set of DORA (DevOps Research and Assessment) deployment-frequency metrics -- one of the four real DORA metrics Datadog computes from an organization's own CI/CD and deployment data to measure engineering delivery performance.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    		ubx.Resource(doradeployment.FetchResponse, "example", doradeployment.FetchResponseConfig{
