`datadog_widget_response` is a real, standalone Datadog dashboard widget -- one real visualization (a timeseries graph, a query value, a table) addressable on its own, outside the context of the full dashboard it may also appear on.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	widget "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/widget"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own response"})

    		ubx.Resource(widget.Response, "example", widget.ResponseConfig{
    			Data:           map[string]any{"attributes": map[string]any{"definition": map[string]any{"title": "example"}}},
    			ExperienceType: "example",
    			Uuid:           "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    import ubx_sdk as ubx

    def describe():

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="data" type="object" required>
  Data for creating or updating a widget. Required.
  <Expandable title="properties">
    <ResponseField name="attributes" type="object" required>
      Attributes for creating or updating a widget. Required.
      <Expandable title="properties">
        <ResponseField name="definition" type="object" required>
          The definition of a widget, including its type and configuration. Required.
          <Expandable title="properties">
            <ResponseField name="title" type="string" required>
              The display title of the widget. Required.
            </ResponseField>
            <ResponseField name="type" type="string" required>
              Widget types that are allowed to be stored as individual records. This is not a complete list of dashboard and notebook widget types. Required.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="tags" type="list(string)">
          User-defined tags for organizing the widget. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="type" type="string" required>
      Widgets resource type. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="experience_type" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="uuid" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>
