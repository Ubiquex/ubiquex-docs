`datadog_widget_list_response` creates a new widget in Datadog's own reusable widget library (distinct from a dashboard's own embedded widgets), so it can be added to any dashboard without redefining it each time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	widget "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/widget"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own list response"})

    		ubx.Resource(widget.ListResponse, "example", widget.ListResponseConfig{
    			Data:           map[string]any{"attributes": map[string]any{"definition": map[string]any{"title": "example", "type": "example"}}, "type": "example"},
    			ExperienceType: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ListResponse } from "./local-sdk/datadog/sdk/typescript/datadog/widget/list_response.ts";

    export default stack("example", () => {
      intent({ summary: "example own list response" });

      resource(ListResponse, "example", {
        data: {
          attributes: { definition: { title: "example", type: "example" } },
          type: "example",
        },
        experienceType: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.widget import ListResponse, ListResponseConfig
    from ubx.datadog.widget.list_response import ListResponse_Data, ListResponse_Data_Attributes, ListResponse_Data_Attributes_Definition

    def describe():
        ubx.intent("example own list response")

        ubx.resource(ListResponse, "example", ListResponseConfig(
            data=ListResponse_Data(attributes=ListResponse_Data_Attributes(definition=ListResponse_Data_Attributes_Definition(title="example", type="example")), type="example"),
            experience_type="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Input properties

<ResponseField name="data" type="object" required>
  Data for creating or updating a widget. Required.
  <Expandable title="properties">
    <ResponseField name="attributes" type="object" required>
      Attributes for creating or updating a widget. Required.
      <Expandable title="properties">
        <ResponseField name="definition" type="object" required>
          The definition of a widget, including its type and configuration. Required.
          <Expandable title="properties">
            <ResponseField name="title" type="string" required>
              The display title of the widget. Required.
            </ResponseField>
            <ResponseField name="type" type="string" required>
              Widget types that are allowed to be stored as individual records. This is not a complete list of dashboard and notebook widget types. Required.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="tags" type="list(string)">
          User-defined tags for organizing the widget. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="type" type="string" required>
      Widgets resource type. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="experience_type" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>
