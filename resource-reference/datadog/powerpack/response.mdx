`datadog_powerpack_response` is a real, reusable Datadog Powerpack -- a saved, parameterized group of dashboard widgets a team can drop onto multiple dashboards at once, keeping a common visualization (like a service's own golden signals) consistent everywhere it's used.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	powerpack "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/powerpack"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own response"})

    		ubx.Resource(powerpack.Response, "example", powerpack.ResponseConfig{
    			PowerpackId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Response } from "@ubx/sdk-datadog/datadog/powerpack/response";

    export default stack("example", () => {
      intent({ summary: "example own response" });

      resource(Response, "example", {
        powerpackId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.powerpack.response import Response, ResponseConfig

    def describe():
        ubx.intent("example own response")

        ubx.resource(Response, "example", ResponseConfig(
            powerpack_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
  <Tab title="Markdown">
    ```
    Create datadog_powerpack_response, with powerpack id example.
    ```
  </Tab>
</Tabs>
