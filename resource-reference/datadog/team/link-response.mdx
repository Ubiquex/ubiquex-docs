`datadog_team_link_response` is a real, external link attached to a Datadog Team -- a quick-reference URL (to a runbook, a repo, a dashboard) the team keeps associated with itself for others to find easily.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	team "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/team"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own link response"})

    		ubx.Resource(team.LinkResponse, "example", team.LinkResponseConfig{
    			Data:   map[string]any{"attributes": map[string]any{"label": "example"}},
    			LinkId: "example",
    			TeamId: "example",
    		})
    	}))
    }
    ```
