`datadog_team_hierarchy_link_response` is a real, structural link expressing that one Datadog Team is a sub-team of another -- the mechanism Datadog Teams uses to model a real organizational hierarchy rather than a flat list of unrelated teams.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	team "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/team"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own hierarchy link response"})

    		ubx.Resource(team.HierarchyLinkResponse, "example", team.HierarchyLinkResponseConfig{
    			Data:   map[string]any{"relationships": map[string]any{"parent_team": map[string]any{"data": map[string]any{"id": "example"}}}},
    			LinkId: "example",
    		})
    	}))
    }
    ```
