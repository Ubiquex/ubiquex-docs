`datadog_team_response` is v2's own richer representation of a Datadog Team -- a real, named group of users used across Datadog (ownership on services, routing for notifications, on-call) as the organizing unit smaller than the whole org.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	team "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/team"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own response"})

    		ubx.Resource(team.Response, "example", team.ResponseConfig{
    			Data:   map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			TeamId: "example",
    		})
    	}))
    }
    ```
  </Tab>
