`datadog_team_notification_rule_response` is a real rule controlling how a specific Datadog Team is notified -- which channel that team's own alerts route to, scoped to the team rather than to a single monitor.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	team "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/team"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own notification rule response"})

    		ubx.Resource(team.NotificationRuleResponse, "example", team.NotificationRuleResponseConfig{
    			Data:   map[string]any{"attributes": map[string]any{"email": map[string]any{"enabled": true}}},
    			RuleId: "example",
    			TeamId: "example",
    		})
    	}))
    }
    ```
