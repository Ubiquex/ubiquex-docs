`datadog_historical_metrics_configuration_response` is real configuration controlling how far back, and for which metrics, Datadog computes and stores historical rollups -- used by features (like DORA metrics or degradation detection) that need trend data older than a metric's own default retention.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	historical "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/historical"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own metrics configuration response"})

    		ubx.Resource(historical.MetricsConfigurationResponse, "example", historical.MetricsConfigurationResponseConfig{
    			Data:       map[string]any{"id": "example"},
    			MetricName: "example-metric-name",
    		})
    	}))
    }
    ```
