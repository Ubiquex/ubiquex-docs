`datadog_jira_issue_template_response` is a real, saved template controlling how Datadog creates a Jira issue when it integrates with Jira -- the real field mappings (summary, description, project, issue type) applied every time Datadog opens a Jira ticket from an incident, case, or finding.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	jira "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/jira"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own issue template response"})

    		ubx.Resource(jira.IssueTemplateResponse, "example", jira.IssueTemplateResponseConfig{
    			IssueTemplateId: "example",
    		})
    	}))
    }
    ```
  </Tab>

        ubx.intent("example own issue template response")

    if __name__ == "__main__":
