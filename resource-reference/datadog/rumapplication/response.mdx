`datadog_rumapplication_response` is a real Datadog RUM application -- the top-level object representing one real monitored web or mobile application, identified by its own real RUM application ID and client token.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	rumapplication "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/rumapplication"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own response"})

    		ubx.Resource(rumapplication.Response, "example", rumapplication.ResponseConfig{
    			Data: map[string]any{"attributes": map[string]any{"name": "example-name"}, "type": "example"},
    			Id:   "example",
    		})
    	}))
    }
    ```
