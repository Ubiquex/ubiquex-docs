`datadog_ucconfig_pair` is a real, paired configuration entry inside Datadog Cloud Cost Management's usage-and-cost (UC) configuration -- one real key/value association (for example, mapping a cloud account to a cost-attribution tag) the cost pipeline uses.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ucconfig "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/ucconfig"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own pair"})

    		ubx.Resource(ucconfig.Pair, "example", ucconfig.PairConfig{
    			CloudAccountId: "example",
    			Data:           map[string]any{"type": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Pair } from "@ubx/sdk-datadog/datadog/ucconfig/pair";

    export default stack("example", () => {

      resource(Pair, "example", {

  </Tab>
  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.datadog.ucconfig import Pair, PairConfig

    def describe():
        ubx.intent("example own pair")

        ubx.resource(Pair, "example", PairConfig(
            cloud_account_id="example",
            data={"type": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
