`datadog_get_app_response` is a real custom application built with Datadog App Builder -- a real, deployed internal tool (built from Datadog's own low-code components) with its own name, layout, and data connections.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	get "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/get"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own app response"})

    		ubx.Resource(get.AppResponse, "example", get.AppResponseConfig{
    			AppId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { AppResponse } from "@ubx/sdk-datadog/datadog/get/app_response";

    export default stack("example", () => {
      intent({ summary: "example own app response" });

      resource(AppResponse, "example", {
        appId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.get.app_response import AppResponse, AppResponseConfig

    def describe():
        ubx.intent("example own app response")

        ubx.resource(AppResponse, "example", AppResponseConfig(
            app_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
