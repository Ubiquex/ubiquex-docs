`datadog_get_investigation_response` is a real Datadog Bits AI investigation -- an AI-assisted analysis session correlating an organization's own telemetry (logs, traces, metrics) around a specific incident or question, with its own findings and real, tracked state.

    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go

    		})
    	}))
    }

## Input properties

<ResponseField name="data" type="object" required>
  Data for the trigger investigation request. Required.
  <Expandable title="properties">
    <ResponseField name="attributes" type="object" required>
      Attributes for the trigger investigation request. Required.
      <Expandable title="properties">
        <ResponseField name="trigger" type="object" required>
          The trigger definition for starting an investigation. Required.
          <Expandable title="properties">
            <ResponseField name="monitor_alert_trigger" type="object" required>
              Attributes for a monitor alert trigger. Required.
              <Expandable title="properties">
                <ResponseField name="event_id" type="string" required>
                  The event ID associated with the monitor alert. Required.
                </ResponseField>
                <ResponseField name="event_ts" type="number" required>
                  The timestamp of the event in Unix milliseconds. Required.
                </ResponseField>
                <ResponseField name="monitor_id" type="number" required>
                  The monitor ID that triggered the alert. Required.
                </ResponseField>
              </Expandable>
            </ResponseField>
            <ResponseField name="type" type="string" required>
              The type of trigger for the investigation. Required.
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="type" type="string" required>
      The resource type for trigger investigation requests. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="id" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>
