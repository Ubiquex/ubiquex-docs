`datadog_issue_response` is a real, tracked issue inside Datadog's issue-management surface -- a general-purpose trackable problem (used across Datadog's code-quality and security-scanning products) with its own status and assignment, distinct from a `datadog_case_response`.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own response"})

    		ubx.Resource(issue.Response, "example", issue.ResponseConfig{
    			Data:    map[string]any{"id": "example", "type": "example"},
    			IssueId: "example",
    		})
    	}))
    }
