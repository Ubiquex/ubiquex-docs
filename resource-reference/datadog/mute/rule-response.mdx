`datadog_mute_rule_response` is a real, general-purpose rule that suppresses notifications matching a condition -- used by Datadog's security products to mute recurring, already-triaged findings without disabling the underlying detection rule.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	mute "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/mute"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own rule response"})

    		ubx.Resource(mute.RuleResponse, "example", mute.RuleResponseConfig{
    			Data:   map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			RuleId: "example",
    		})
    	}))
    }
    ```
