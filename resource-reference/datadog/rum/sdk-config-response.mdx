`datadog_rum_sdk_config_response` is real, centrally-managed configuration for the Datadog RUM browser/mobile SDK -- letting a team control SDK behavior (sampling rates, feature flags) for a RUM application from Datadog itself rather than only through code-level SDK init options.

## Example

    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	rum "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/rum"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own sdk config response"})

    		ubx.Resource(rum.SdkConfigResponse, "example", rum.SdkConfigResponseConfig{
    			ConfigId: "example",
    			Data:     map[string]any{"attributes": map[string]any{"rum": map[string]any{"default_privacy_level": "example"}}},
    		})
    	}))
    }
    ```
