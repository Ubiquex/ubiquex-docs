`datadog_rum_metric_response` is a real, custom metric generated by aggregating Real User Monitoring (RUM) session data -- the RUM counterpart to `datadog_logs_metric_response`, turning raw session/event data into a cheap, long-retention metric.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    	}))

## Properties

<ResponseField name="data" type="object" required>
  The new RUM-based metric properties. Required.
  <Expandable title="properties">
    <ResponseField name="attributes" type="object" required>
      The object describing the Datadog RUM-based metric to create. Required.
      <Expandable title="properties">
        <ResponseField name="compute" type="object" required>
          The compute rule to compute the RUM-based metric. Required.
          <Expandable title="properties">
            <ResponseField name="aggregation_type" type="string" required>
              The type of aggregation to use. Required.
            </ResponseField>
            <ResponseField name="include_percentiles" type="bool">
              Toggle to include or exclude percentile aggregations for distribution metrics. Only present when `aggregation_type` is `distribution`. Optional.
            </ResponseField>
            <ResponseField name="path" type="string">
              The path to the value the RUM-based metric will aggregate on. Only present when `aggregation_type` is `distribution`. Optional.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="event_type" type="string" required>
          The type of RUM events to filter on. Required.
        </ResponseField>
        <ResponseField name="filter" type="object">
          The RUM-based metric filter. Events matching this filter will be aggregated in this metric. Optional.
          <Expandable title="properties">
            <ResponseField name="query" type="string" required>
              The search query - following the RUM search syntax. Required.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="group_by" type="list(object)">
          The rules for the group by. Optional.
          <Expandable title="properties">
            <ResponseField name="path" type="string">
              Optional; if omitted, computed by Datadog.
            </ResponseField>
            <ResponseField name="tag_name" type="string">
              Optional; if omitted, computed by Datadog.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="uniqueness" type="object">
          The rule to count updatable events. Is only set if `event_type` is `sessions` or `views`. Optional.
          <Expandable title="properties">
            <ResponseField name="when" type="string" required>
              When to count updatable events. `match` when the event is first seen, or `end` when the event is complete. Required.
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="id" type="string" required>
      The name of the RUM-based metric. Required.
    </ResponseField>
    <ResponseField name="type" type="string" required>
      The type of the resource. The value should always be rum_metrics. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="metric_id" type="string" required>
