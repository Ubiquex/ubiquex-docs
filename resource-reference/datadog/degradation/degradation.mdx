`datadog_degradation` is a real, detected performance degradation tracked by Datadog -- a specific service or endpoint whose latency, error rate, or other health signal has measurably worsened, surfaced as its own trackable entity rather than buried in a raw metric.

## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	degradation "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/degradation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own degradation"})

    		ubx.Resource(degradation.Degradation, "example", degradation.DegradationConfig{
    			DegradationId: "example",
    			PageId:        "example",
    			Meta:          map[string]any{"idempotency_key": "example"},
    		})
    	}))
    }
    ```
