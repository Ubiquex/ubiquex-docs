`datadog_degradation_template` is a reusable template defining how Datadog detects and reports a `datadog_degradation` -- the real thresholds and signals a team configures once and applies across many services rather than hand-tuning detection per service.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	degradation "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/degradation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own template"})

    		ubx.Resource(degradation.Template, "example", degradation.TemplateConfig{
    			PageId:     "example",
    			TemplateId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Template } from "./local-sdk/datadog/sdk/typescript/datadog/degradation/template.ts";

    export default stack("example", () => {
      intent({ summary: "example own template" });

      resource(Template, "example", {
        pageId: "example",
        templateId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.degradation import Template, TemplateConfig

    def describe():
        ubx.intent("example own template")

        ubx.resource(Template, "example", TemplateConfig(
            page_id="example",
            template_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
