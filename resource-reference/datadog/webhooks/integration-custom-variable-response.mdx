`datadog_webhooks_integration_custom_variable_response` is a named value you define once in the webhooks integration tile and then reuse across a webhook's own URL, payload, or headers -- useful for a credential or another value several webhook configurations all need but that shouldn't be hardcoded into each one separately.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	webhooks "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/webhooks"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own integration custom variable response"})

    		ubx.Resource(webhooks.IntegrationCustomVariableResponse, "example", webhooks.IntegrationCustomVariableResponseConfig{
    			CustomVariableName: "example-custom-variable-name",
    			IsSecret:           true,
    			Name:               "example",
    			Value:              "example",
    		})
    	}))
    }
    ```
  </Tab>

            custom_variable_name="example-custom-variable-name",
            is_secret=True,
            name="example",
            value="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Properties

<ResponseField name="custom_variable_name" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="is_secret" type="bool" required>
  Make custom variable is secret or not. If the custom variable is secret, the value is not returned in the response payload. Required.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the variable. It corresponds with `&lt;CUSTOM_VARIABLE_NAME&gt;`. Required.
</ResponseField>

<ResponseField name="value" type="string" required>
  Value of the custom variable. Required.
</ResponseField>
