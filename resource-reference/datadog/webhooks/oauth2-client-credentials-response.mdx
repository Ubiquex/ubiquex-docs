`datadog_webhooks_oauth2_client_credentials_response` is real, configured OAuth2 client-credentials authentication for a Datadog outbound webhook -- lets a webhook integration authenticate to the receiving system using OAuth2 client credentials instead of a static secret.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	webhooks "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/webhooks"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own oauth2 client credentials response"})

    		ubx.Resource(webhooks.Oauth2ClientCredentialsResponse, "example", webhooks.Oauth2ClientCredentialsResponseConfig{
    			AuthMethodId: "example",
    			Data:         map[string]any{"attributes": map[string]any{"access_token_url": "example", "client_id": "example", "client_secret": "example", "name": "example-name"}, "type": "example"},
    		})
    	}))
    }
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.datadog.webhooks import Oauth2ClientCredentialsResponse, Oauth2ClientCredentialsResponseConfig
    from ubx.datadog.webhooks.oauth2_client_credentials_response import Oauth2ClientCredentialsResponse_Data, Oauth2ClientCredentialsResponse_Data_Attributes

    def describe():
        ubx.intent("example own oauth2 client credentials response")

        ubx.resource(Oauth2ClientCredentialsResponse, "example", Oauth2ClientCredentialsResponseConfig(
            auth_method_id="example",
            data=Oauth2ClientCredentialsResponse_Data(attributes=Oauth2ClientCredentialsResponse_Data_Attributes(access_token_url="example", client_id="example", client_secret="example", name="example-name"), type="example"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
