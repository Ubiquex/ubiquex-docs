## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	fastly "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/fastly"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own service response"})

    		ubx.Resource(fastly.ServiceResponse, "example", fastly.ServiceResponseConfig{
    			AccountId: "example",
    			Data:      map[string]any{"id": "example"},
    			ServiceId: "example",
    		})
    	}))
    }
    ```
  </Tab>
