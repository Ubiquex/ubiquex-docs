`datadog_fastly_account_response` is a real Fastly account connected to Datadog's own Fastly integration -- the credentials Datadog uses to pull Fastly CDN metrics and logs into Datadog.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	fastly "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/fastly"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own account response"})

    		ubx.Resource(fastly.AccountResponse, "example", fastly.AccountResponseConfig{
    			AccountId: "example",
    			Data:      map[string]any{"attributes": map[string]any{"name": "example-name"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { AccountResponse } from "@ubx/sdk-datadog/datadog/fastly/account_response";

    export default stack("example", () => {
      intent({ summary: "example own account response" });

      resource(AccountResponse, "example", {
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.fastly.account_response import AccountResponse, AccountResponseConfig

    def describe():
        ubx.intent("example own account response")

        ubx.resource(AccountResponse, "example", AccountResponseConfig(
            account_id="example",
            data={"attributes": {"name": "example-name"}},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
