`datadog_monitor_user_template_response` is a real, saved monitor template a user or team can reuse to quickly create new monitors with consistent thresholds and messages, rather than configuring each new monitor from a blank state.

## Example

    ```go
    package main

    import (
    	monitor "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/monitor"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own user template response"})

    		ubx.Resource(monitor.UserTemplateResponse, "example", monitor.UserTemplateResponseConfig{
    			Data:       map[string]any{"attributes": map[string]any{"monitor_definition": map[string]any{"key": "value"}}},
    			TemplateId: "example",
    		})
    	}))
    }
    ```
