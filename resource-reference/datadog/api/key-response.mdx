`datadog_api_key_response` is the credential the Datadog Agent uses to submit metrics, traces, and events to your organization -- every agent-based integration authenticates with one. It's organization-specific, not tied to any one user, and is the primary way infrastructure data actually gets into Datadog in the first place.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	api "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/api"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own key response"})

    		ubx.Resource(api.KeyResponse, "ci-runner", api.KeyResponseConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.datadog.api import KeyResponse, KeyResponseConfig

        ubx.intent("payments own key response")

        ubx.resource(KeyResponse, "ci-runner", KeyResponseConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
