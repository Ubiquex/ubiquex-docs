`datadog_project_response` is a real, tracked project inside a Datadog workflow/app-building context -- a top-level grouping (of workflows, apps, or datastores) used to organize related automation work together.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	project "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/project"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own response"})

    		ubx.Resource(project.Response, "example", project.ResponseConfig{
    			Data:      map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			ProjectId: "example",
    		})
    	}))
    }
    ```
