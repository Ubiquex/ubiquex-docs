`datadog_security_monitoring_integration_config_response` is real configuration controlling how a specific third-party integration's data feeds into Datadog Cloud SIEM's own detection rules.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	security "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/security"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own monitoring integration config response"})

    		ubx.Resource(security.MonitoringIntegrationConfigResponse, "example", security.MonitoringIntegrationConfigResponseConfig{
    			Data:                map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			IntegrationConfigId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { MonitoringIntegrationConfigResponse } from "@ubx/sdk-datadog/datadog/security/monitoring_integration_config_response";

      intent({ summary: "example own monitoring integration config response" });

      resource(MonitoringIntegrationConfigResponse, "example", {
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.datadog.security.monitoring_integration_config_response import MonitoringIntegrationConfigResponse, MonitoringIntegrationConfigResponseConfig

        ubx.run("example", describe)
    ```
