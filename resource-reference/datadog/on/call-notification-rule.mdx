`datadog_on_call_notification_rule` is a real rule controlling how Datadog On-Call notifies a specific responder -- which device or channel (push, SMS, phone call, email) is used and in what order for that person, distinct from the team-level `datadog_escalation_policy`.

    import (
    	on "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/on"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    		ubx.Resource(on.CallNotificationRule, "example", on.CallNotificationRuleConfig{

    			RuleId: "example",
    			UserId: "example",

    	}))
    }
