`datadog_case_view_response` is a saved, filtered view over Datadog Case Management cases -- a real, named query (by status, team, priority) a user or team saves to quickly get back to the specific slice of cases they work from day to day.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	case_ "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/case_"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own view response"})

    		ubx.Resource(case_.ViewResponse, "example", case_.ViewResponseConfig{
    			Data:   map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			ViewId: "example",
    		})
    	}))
    }
    ```
  </Tab>
