`datadog_case_response` is a real case inside Datadog Case Management -- a trackable unit of work (a security incident, an operational issue) with real status, assignee, priority, and linked evidence, the same real primitive that powers Datadog's own Cloud SIEM and App Security investigation workflows.

## Example

    ```go
    package main

    import (
    	case_ "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/case_"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own response"})

    		ubx.Resource(case_.Response, "example", case_.ResponseConfig{
    			CaseId: "example",
    			Data:   map[string]any{"attributes": map[string]any{"title": "example", "type_id": "example"}, "type": "example"},
    		})
    	}))
    }
    ```
