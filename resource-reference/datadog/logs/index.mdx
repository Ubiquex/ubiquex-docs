<ResponseField name="daily_limit" type="number">
  The number of log events you can send in this index per day before you are rate-limited. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="daily_limit_reset" type="object">
  Object containing options to override the default daily limit reset time. Optional; if omitted, computed by Datadog.
  <Expandable title="properties">
    <ResponseField name="reset_time" type="string">
      String in `HH:00` format representing the time of day the daily limit should be reset. The hours must be between 00 and 23 (inclusive). Optional.
    </ResponseField>
    <ResponseField name="reset_utc_offset" type="string">
      String in `(-|+)HH:00` format representing the UTC offset to apply to the given reset time. The hours must be between -12 and +14 (inclusive). Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="daily_limit_warning_threshold_percentage" type="number">
  A percentage threshold of the daily quota at which a Datadog warning event is generated. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="exclusion_filters" type="list(object)">
  An array of exclusion objects. The logs are tested against the query of each filter, following the order of the array. Only the first matching active exclusion matters, others (if any) are ignored. Optional; if omitted, computed by Datadog.
  <Expandable title="properties">
    <ResponseField name="filter" type="object">
      The filter object defines the query criteria that determine which logs are excluded from the index based on the exclusion filter's configuration. Optional; if omitted, computed by Datadog.

      **AI-inferred** -- not sourced from the real provider schema; generated by ubx's own description pipeline and never independently verified against the real provider.
      <Expandable title="properties">
        <ResponseField name="query" type="string">
          The query string to filter logs that should be excluded from the index. It follows the Datadog log search syntax. Optional; if omitted, computed by Datadog.

          **AI-inferred** -- not sourced from the real provider schema; generated by ubx's own description pipeline and never independently verified against the real provider.
        </ResponseField>
        <ResponseField name="sample_attribute" type="string">
          Optional; if omitted, computed by Datadog.
        </ResponseField>
        <ResponseField name="sample_rate" type="number">
          The fraction of logs matching the exclusion filter query that will be excluded. Must be a number between 0 and 1, where 1.0 excludes all matching logs. Optional; if omitted, computed by Datadog.

          **AI-inferred** -- not sourced from the real provider schema; generated by ubx's own description pipeline and never independently verified against the real provider.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="is_enabled" type="bool">
      Whether the exclusion filter is enabled. When true, logs matching the filter's query are excluded from the index. Optional; if omitted, computed by Datadog.

      **AI-inferred** -- not sourced from the real provider schema; generated by ubx's own description pipeline and never independently verified against the real provider.
    </ResponseField>
    <ResponseField name="name" type="string">
      Name of the exclusion filter. This is a user-defined label used to identify the filter within the logs index. Optional; if omitted, computed by Datadog.

      **AI-inferred** -- not sourced from the real provider schema; generated by ubx's own description pipeline and never independently verified against the real provider.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="filter" type="object" required>
  Filter for logs. Required.
  <Expandable title="properties">
    <ResponseField name="query" type="string">
      The filter query. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="is_rate_limited" type="bool">
  A boolean stating if the index is rate limited, meaning more logs than the daily limit have been sent. Rate limit is reset every-day at 2pm UTC. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="name" type="string" required>
  The name of the index. Required.
</ResponseField>

<ResponseField name="num_flex_logs_retention_days" type="number">
  The total number of days logs are stored in Standard and Flex Tier before being deleted from the index. If Standard Tier is enabled on this index, logs are first retained in Standard Tier for the number of days specified through `num_retention_days`, and then stored in Flex Tier until the number of days specified in `num_flex_logs_retention_days` is reached. The available values depend on retention plans specified in your organization's contract/subscriptions. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="num_retention_days" type="number">
  The number of days logs are stored in Standard Tier before aging into the Flex Tier or being deleted from the index. The available values depend on retention plans specified in your organization's contract/subscriptions. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="tags" type="list(string)">
  A list of tags associated with the index. Tags must be in `key:value` format. Optional; if omitted, computed by Datadog.
</ResponseField>
