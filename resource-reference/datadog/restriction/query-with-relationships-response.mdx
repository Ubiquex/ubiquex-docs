`datadog_restriction_query_with_relationships_response` is a real, saved logs query with restricted access, plus its own real relationships to the roles/teams permitted to use it -- lets an organization share a sensitive logs query without exposing the underlying log data to everyone.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	restriction "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/restriction"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own query with relationships response"})

    		ubx.Resource(restriction.QueryWithRelationshipsResponse, "example", restriction.QueryWithRelationshipsResponseConfig{
    			RestrictionQueryId: "example",
    		})
    	}))
    }
    ```
