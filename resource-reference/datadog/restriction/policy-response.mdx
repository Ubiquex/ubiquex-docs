`datadog_restriction_policy_response` is a real, fine-grained access-control policy Datadog applies to a specific resource (a dashboard, a monitor, a notebook) -- who beyond the default org-role permissions may view or edit that one specific object.

## Example

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    		ubx.Resource(restriction.PolicyResponse, "example", restriction.PolicyResponseConfig{

    		})
    	}))
    }

    def describe():

            resource_id="example",
        ))

    if __name__ == "__main__":
