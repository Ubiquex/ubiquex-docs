`datadog_dashboard_list` is a named collection used to organize dashboards so a team can search, browse, and share a related set of them instead of hunting through every dashboard in the organization individually.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	dashboard "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/dashboard"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own list"})

    		ubx.Resource(dashboard.List, "ci-runner", dashboard.ListConfig{
    			ListId: "example",
    			Name:   "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk

    export default stack("payments", () => {
      intent({ summary: "payments own list" });

    });
    ```

  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.dashboard import List, ListConfig

    def describe():
        ubx.intent("payments own list")

        ubx.resource(List, "ci-runner", ListConfig(
            list_id="example",
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
