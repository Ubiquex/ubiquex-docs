`datadog_playlist` is a real, ordered sequence of Datadog dashboards that auto-cycles on a timer -- commonly used to drive a real wall-mounted status display (a NOC or team room TV) through a rotating set of dashboards.

## Example

    package main

    import (
    	playlist "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/playlist"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own playlist"})

    		ubx.Resource(playlist.Playlist, "example", playlist.PlaylistConfig{
    			Data:       map[string]any{"type": "example"},
    			PlaylistId: "example",
    		})
    	}))
    }
