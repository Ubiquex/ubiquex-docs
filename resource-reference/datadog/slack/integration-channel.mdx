`datadog_slack_integration_channel` connects one Slack channel to Datadog so monitor alerts and other notifications can post into it -- set up by inviting the Datadog app into the channel with `/invite @Datadog`, after which the channel becomes a real, addressable notification target and can also run `/datadog` commands directly from Slack.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	slack "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/slack"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own integration channel"})

    		ubx.Resource(slack.IntegrationChannel, "example", slack.IntegrationChannelConfig{
    			AccountName: "example-account-name",
    			ChannelName: "example-channel-name",
    			Name:        "example",
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.slack import IntegrationChannel, IntegrationChannelConfig

    def describe():
        ubx.intent("example own integration channel")

        ubx.resource(IntegrationChannel, "example", IntegrationChannelConfig(
            account_name="example-account-name",
            channel_name="example-channel-name",
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

    </ResponseField>
    <ResponseField name="notified" type="bool">
      Show the list of @-handles in the alert event. Optional.
    </ResponseField>
    <ResponseField name="snapshot" type="bool">
      Show the alert event's snapshot image. Optional.
    </ResponseField>
    <ResponseField name="tags" type="bool">
      Show the scopes on which the monitor alerted. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
