`datadog_synthetics_downtime_response` is a real, scheduled suppression window specifically for Synthetic Monitoring tests -- mutes synthetic test alerts during planned maintenance, the Synthetics-scoped counterpart to `datadog_downtime_response`.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own downtime response"})

    		ubx.Resource(synthetics.DowntimeResponse, "example", synthetics.DowntimeResponseConfig{

    			DowntimeId: "example",
    		})
    	}))
    }

    import ubx_sdk as ubx
    from ubx.datadog.synthetics import DowntimeResponse, DowntimeResponseConfig
    from ubx.datadog.synthetics.downtime_response import DowntimeResponse_Data, DowntimeResponse_Data_Attributes, DowntimeResponse_Data_Attributes_TimeSlots

    def describe():
        ubx.intent("example own downtime response")

        ubx.resource(DowntimeResponse, "example", DowntimeResponseConfig(
            data=DowntimeResponse_Data(attributes=DowntimeResponse_Data_Attributes(is_enabled=True, name="example-name", test_ids=["example"], time_slots=[DowntimeResponse_Data_Attributes_TimeSlots(name="example-name")]), type="example"),
