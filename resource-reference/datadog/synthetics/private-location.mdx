`datadog_synthetics_private_location` is a Docker container or Windows service you run inside your own network so Synthetic tests can reach internal endpoints that the public internet, and Datadog's own managed test locations, can't. It's the real option for testing anything behind a firewall, including internal CI environments before a release ever reaches production.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	synthetics "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/synthetics"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own private location"})

    		ubx.Resource(synthetics.PrivateLocation, "example", synthetics.PrivateLocationConfig{
    			Description: "Managed by ubx.",
    			LocationId:  "example",
    			Name:        "example",
    			Tags:        []string{"example"},
    		})
    	}))
    }
    ```
  </Tab>

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.synthetics.private_location import PrivateLocation, PrivateLocationConfig

    def describe():
        ubx.intent("example own private location")

        ubx.resource(PrivateLocation, "example", PrivateLocationConfig(
            description="Managed by ubx.",
            location_id="example",
            name="example",
            tags=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="description" type="string" required>
  Description of the private location. Required.
</ResponseField>

<ResponseField name="id" type="string">
  Unique identifier of the private location. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="location_id" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="metadata" type="object">
  Object containing metadata about the private location. Optional; if omitted, computed by Datadog.
  <Expandable title="properties">
    <ResponseField name="restricted_roles" type="list(string)">
      A list of role identifiers that can be pulled from the Roles API, for restricting read and write access. This field is deprecated. Use the restriction policies API to manage permissions. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="name" type="string" required>
  Name of the private location. Required.
</ResponseField>

<ResponseField name="secrets" type="object">
  Secrets for the private location. Only present in the response when creating the private location. Optional; if omitted, computed by Datadog.
  <Expandable title="properties">
    <ResponseField name="authentication" type="object">
      Authentication part of the secrets. Optional.
      <Expandable title="properties">
        <ResponseField name="id" type="string">
          Access key for the private location. Computed by Datadog after `ship`.
        </ResponseField>
        <ResponseField name="key" type="string">
          Secret access key for the private location. Computed by Datadog after `ship`.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="config_decryption" type="object">
      Private key for the private location. Optional.
      <Expandable title="properties">
        <ResponseField name="key" type="string">
          Private key for the private location. Computed by Datadog after `ship`.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tags" type="list(string)" required>
  Array of tags attached to the private location. Required.
</ResponseField>
