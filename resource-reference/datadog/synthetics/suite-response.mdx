`datadog_synthetics_suite_response` is a real, named group of Synthetic Monitoring tests run and reported on together -- lets a team organize related synthetic checks (all the tests for one user journey, for example) as one unit.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	synthetics "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/synthetics"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own suite response"})

    		ubx.Resource(synthetics.SuiteResponse, "example", synthetics.SuiteResponseConfig{
    			Data:     map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			PublicId: "example",
    		})
    	}))
    }
    ```
