`datadog_synthetics_network_test_response` is a real Datadog Synthetic Monitoring network test -- checks real TCP/UDP/ICMP-level connectivity and performance to an endpoint, distinct from an HTTP-level API test or a browser test.

## Example

<Tabs>

    ```go

    import (
    	synthetics "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/synthetics"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own network test response"})

    		ubx.Resource(synthetics.NetworkTestResponse, "example", synthetics.NetworkTestResponseConfig{
    			Data:     map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			PublicId: "example",
    		})
    	}))
    }
    ```
