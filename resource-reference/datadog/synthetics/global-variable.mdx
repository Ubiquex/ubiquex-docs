---

`datadog_synthetics_global_variable` is a value shared across every Synthetic test in an organization, rather than duplicated inside each one -- used for things like an auth token pulled from a prior API response or a generated UUID, so tests stay dynamic instead of hardcoding values that go stale.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	synthetics "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/synthetics"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own global variable"})

    		ubx.Resource(synthetics.GlobalVariable, "example", synthetics.GlobalVariableConfig{
    			Description: "Managed by ubx.",
    			Name:        "example",
    			Tags:        []string{"example"},
    			VariableId:  "example",
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.synthetics import GlobalVariable, GlobalVariableConfig

    def describe():
        ubx.intent("example own global variable")

        ubx.resource(GlobalVariable, "example", GlobalVariableConfig(
            description="Managed by ubx.",
            name="example",
            tags=["example"],
            variable_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

<ResponseField name="attributes" type="object">

  <Expandable title="properties">

    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="description" type="string" required>
  Description of the global variable. Required.
</ResponseField>

<ResponseField name="id" type="string">
  Unique identifier of the global variable. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="is_fido" type="bool">
  Determines if the global variable is a FIDO variable. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="is_totp" type="bool">
  Determines if the global variable is a TOTP/MFA variable. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="name" type="string" required>
  Name of the global variable. Unique across Synthetic global variables. Required.
</ResponseField>

<ResponseField name="parse_test_options" type="object">
  Parser options to use for retrieving a Synthetic global variable from a Synthetic test. Used in conjunction with `parse_test_public_id`. Optional; if omitted, computed by Datadog.
  <Expandable title="properties">
    <ResponseField name="field" type="string">
      When type is `http_header`, name of the header to use to extract the value. Optional.
    </ResponseField>
    <ResponseField name="local_variable_name" type="string">
      When type is `local_variable`, name of the local variable to use to extract the value. Optional.
    </ResponseField>
    <ResponseField name="parser" type="object">
      Details of the parser to use for the global variable. Optional.
      <Expandable title="properties">
        <ResponseField name="type" type="string" required>
          Type of parser for a Synthetic global variable from a synthetics test. Required.
        </ResponseField>
        <ResponseField name="value" type="string">
          Regex or JSON path used for the parser. Not used with type `raw`. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="type" type="string" required>
      Type of value to extract from a test for a Synthetic global variable. Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="parse_test_public_id" type="string">
  A Synthetic test ID to use as a test to generate the variable value. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="tags" type="list(string)" required>
  Tags of the global variable. Required.
</ResponseField>

<ResponseField name="value" type="object">
  Value of the global variable. Optional; if omitted, computed by Datadog.
  <Expandable title="properties">
    <ResponseField name="options" type="object">
      Options for the Global Variable for MFA. Optional.
      <Expandable title="properties">

          Parameters for the TOTP/MFA variable Optional.
          <Expandable title="properties">
            <ResponseField name="digits" type="number">
              Number of digits for the OTP code. Optional.
            </ResponseField>
            <ResponseField name="refresh_interval" type="number">
              Interval for which to refresh the token (in seconds). Optional.
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="secure" type="bool">
      Determines if the value of the variable is hidden. Optional.
    </ResponseField>
    <ResponseField name="value" type="string">
      Value of the global variable. When reading a global variable, the value will not be present if the variable is hidden with the `secure` property. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="variable_id" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>
