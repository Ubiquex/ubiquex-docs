`datadog_status_page` is a real, public or private Datadog Status Page -- a real, hosted status/incident-communication page an organization publishes to communicate service health to its own customers or internal stakeholders.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	status "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/status"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own page"})

    		ubx.Resource(status.Page, "example", status.PageConfig{
    			PageId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Page } from "@ubx/sdk-datadog/datadog/status/page";

    export default stack("example", () => {
      intent({ summary: "example own page" });

      resource(Page, "example", {
        pageId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.datadog.status.page import Page, PageConfig

    def describe():
        ubx.intent("example own page")

        ubx.resource(Page, "example", PageConfig(
            page_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>

    ```

    ```
  </Tab>
</Tabs>
