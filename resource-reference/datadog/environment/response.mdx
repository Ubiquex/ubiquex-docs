`datadog_environment_response` is a real, named environment (production, staging, and similar) as modeled by Datadog's App Builder/Software Catalog surface -- a real deployment target an app or service is associated with, distinct from the free-form `env` tag used elsewhere in Datadog.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	environment "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/environment"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own response"})

    		ubx.Resource(environment.Response, "example", environment.ResponseConfig{
    			Data:          map[string]any{"attributes": map[string]any{"name": "example-name"}},
    			EnvironmentId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Response } from "@ubx/sdk-datadog/datadog/environment/response";

    import ubx_sdk as ubx
    from ubx.datadog.environment import Response, ResponseConfig

    def describe():

            data={"attributes": {"name": "example-name"}},
            environment_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
