`datadog_metric_metadata` is the description, unit, and type information attached to one specific metric, editable independently of the metric's own data points. Setting real metadata (what the metric represents, whether it's measured in bytes, requests, or seconds) is what makes a metric self-explanatory to someone who didn't define it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	metric "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/metric"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own metadata"})

    		ubx.Resource(metric.Metadata, "ci-runner", metric.MetadataConfig{
    			MetricName: "example-metric-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Metadata } from "./local-sdk/datadog/sdk/typescript/datadog/metric/metadata.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own metadata" });

      resource(Metadata, "ci-runner", {
        metricName: "example-metric-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.metric import Metadata, MetadataConfig

    def describe():
        ubx.intent("payments own metadata")

        ubx.resource(Metadata, "ci-runner", MetadataConfig(
            metric_name="example-metric-name",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Properties

<ResponseField name="description" type="string">
  Metric description. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="integration" type="string">
  Name of the integration that sent the metric if applicable. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="metric_name" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="per_unit" type="string">
  Per unit of the metric such as `second` in `bytes per second`. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="short_name" type="string">
  A more human-readable and abbreviated version of the metric name. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="statsd_interval" type="number">
  StatsD flush interval of the metric in seconds if applicable. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="type" type="string">
  Metric type such as `gauge` or `rate`. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="unit" type="string">
  Primary unit of the metric such as `byte` or `operation`. Optional; if omitted, computed by Datadog.
</ResponseField>
