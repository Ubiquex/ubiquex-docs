the real Datadog OpenAPI specification, in every SDK language.

enough to save and run exactly as shown.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	host "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/host"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own tags"})

    		ubx.Resource(host.Tags, "ci-runner", host.TagsConfig{
    			HostName: "example-host-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Tags } from "./local-sdk/datadog/sdk/typescript/datadog/host/tags.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own tags" });

      resource(Tags, "ci-runner", {
        hostName: "example-host-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.host import Tags, TagsConfig

    def describe():
        ubx.intent("payments own tags")

        ubx.resource(Tags, "ci-runner", TagsConfig(
            host_name="example-host-name",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
