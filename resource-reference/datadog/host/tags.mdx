---
title: "datadog_host_tags"
description: "`datadog_host_tags` are labels attached to one specific host that every metric, log, trace, and integration reported by that host inherits automatically."
---

Tagging a host is what makes it possible to filter and correlate telemetry across many hosts at once, which matters most in container and cloud environments where hosts are constantly appearing and disappearing.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go
    package main

    import (
    	host "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/host"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own tags"})

    		ubx.Resource(host.Tags, "example", host.TagsConfig{
    			HostName: "example-host-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only datadog --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Tags } from "./local-sdk/datadog/sdk/typescript/datadog/host/tags.ts";

    export default stack("example", () => {
      intent({ summary: "example own tags" });

      resource(Tags, "example", {
        hostName: "example-host-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only datadog --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/datadog/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.datadog.host import Tags, TagsConfig

    def describe():
        ubx.intent("example own tags")

        ubx.resource(Tags, "example", TagsConfig(
            host_name="example-host-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="host" type="string">
  Your host name. Optional; if omitted, computed by Datadog.
</ResponseField>

<ResponseField name="host_name" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="tags" type="list(string)">
  A list of tags associated with a host. Optional; if omitted, computed by Datadog.
</ResponseField>
