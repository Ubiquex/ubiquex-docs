`datadog_llmobs_prompt_version_response` is one real, saved version of a `datadog_llmobs_prompt_sdkresponse` -- the prompt's own real edit history, so a team can compare or roll back to an earlier prompt version.

## Example

    // ubx sdk gen --only datadog --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-datadog/sdk/go => ./local-sdk/datadog/sdk/go

    import (
    	llmobs "github.com/ubiquex/ubx-sdk-datadog/sdk/go/datadog/llmobs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own prompt version response"})

    		ubx.Resource(llmobs.PromptVersionResponse, "example", llmobs.PromptVersionResponseConfig{
    			Data:     map[string]any{"attributes": map[string]any{"template": "example"}},
    			PromptId: "example",
    			Version:  "example",
    		})
    	}))
    }
