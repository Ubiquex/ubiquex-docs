`google_apigee_security_feedback` records a human reviewer's own correction to an Advanced API Security finding (e.g. marking a flagged IP address as a false positive), used to refine future detection.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigee --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigee/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigee/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own security feedback"})

    		ubx.Resource(apigee.ApigeeSecurityFeedback, "ci-runner", apigee.ApigeeSecurityFeedbackConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
