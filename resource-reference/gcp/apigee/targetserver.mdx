`google_apigee_targetserver` defines a named, reusable backend host (address, port, TLS config) an Apigee proxy's own target endpoint routes requests to, letting the actual backend be changed without editing every proxy that references it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigee --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigee/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigee/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own targetserver"})

    		ubx.Resource(apigee.ApigeeTargetserver, "example", apigee.ApigeeTargetserverConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
