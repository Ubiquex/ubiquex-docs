`google_apigee_keystore` holds the TLS certificate and private key entries an Apigee proxy or target endpoint presents for inbound or outbound TLS connections.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigee --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigee/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigee/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own keystore"})

    		ubx.Resource(apigee.ApigeeKeystore, "ci-runner", apigee.ApigeeKeystoreConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
