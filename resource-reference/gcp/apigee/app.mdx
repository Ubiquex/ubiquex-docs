`google_apigee_app` (a Developer App) is the credential-holding registration a developer creates to consume one or more Apigee API products, carrying the API key(s) used to authenticate its own requests.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigee --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigee/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigee/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own app"})

    		ubx.Resource(apigee.ApigeeApp, "ci-runner", apigee.ApigeeAppConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
