## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigee --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigee/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigee/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own entry"})

    		ubx.Resource(apigee.ApigeeEntry, "example", apigee.ApigeeEntryConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_apigee --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ApigeeEntry } from "./local-sdk/google_apigee/sdk/typescript/google_apigee/apigee/apigee_entry.ts";

    export default stack("example", () => {
      intent({ summary: "example own entry" });

      resource(ApigeeEntry, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_apigee --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_apigee/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_apigee.apigee import ApigeeEntry, ApigeeEntryConfig

    def describe():
        ubx.intent("example own entry")

        ubx.resource(ApigeeEntry, "example", ApigeeEntryConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

</ResponseField>
