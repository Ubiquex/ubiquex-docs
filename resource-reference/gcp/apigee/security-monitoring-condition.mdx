`google_apigee_security_monitoring_condition` defines a condition Apigee Advanced API Security continuously monitors traffic against, triggering an alert or action when matched.

## Example

  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigee --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigee/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigee/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own security monitoring condition"})

    		ubx.Resource(apigee.ApigeeSecurityMonitoringCondition, "ci-runner", apigee.ApigeeSecurityMonitoringConditionConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
