`google_apigee_apim_service_extension` wires Apigee's own API key and OAuth token verification into a Google Cloud load balancer via a service extension, enforcing Apigee policy on traffic without routing it through a full Apigee proxy.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigee --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigee/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigee/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own apim service extension"})

    		ubx.Resource(apigee.ApigeeApimServiceExtension, "example", apigee.ApigeeApimServiceExtensionConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
