`google_apigee_aliase` (an Alias) is a single named certificate or key entry stored inside an Apigee keystore or truststore, referenced by TLS configuration on a proxy or target endpoint.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own aliase"})

    		ubx.Resource(apigee.Aliase, "example", apigee.AliaseConfig{
    			ContentType: "example",
    			Data:        "example",
    			Extensions:  []string{"example"},
    		})
    	}))
    }
    ```
  </Tab>
