`google_apigee_override` overrides a specific check within an Apigee Advanced API Security security profile, excluding it from that profile's own assessment.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigee --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigee/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigee/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own override"})

    		ubx.Resource(apigee.ApigeeOverride, "ci-runner", apigee.ApigeeOverrideConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_apigee --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ApigeeOverride } from "./local-sdk/google_apigee/sdk/typescript/google_apigee/apigee/apigee_override.ts";

      intent({ summary: "payments own override" });

      });

    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_apigee --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_apigee/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_apigee.apigee import ApigeeOverride, ApigeeOverrideConfig

    def describe():
        ubx.intent("payments own override")

        ubx.resource(ApigeeOverride, "ci-runner", ApigeeOverrideConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Properties

<ResponseField name="api_proxy" type="string">
  ID of the API proxy that will have its trace configuration overridden. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  ID of the trace configuration override specified as a system-generated UUID. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="sampling_config" type="object">
  TraceSamplingConfig represents the detail settings of distributed tracing. Only the fields that are defined in the distributed trace configuration can be overridden using the distribute trace configuration override APIs. Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="sampler" type="string">
      Sampler of distributed tracing. OFF is the default value. Optional.
    </ResponseField>
    <ResponseField name="sampling_rate" type="number">
      Field sampling rate. This value is only applicable when using the PROBABILITY sampler. The supported values are &gt; 0 and &lt;= 0.5. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
