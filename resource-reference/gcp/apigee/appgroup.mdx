`google_apigee_appgroup` groups multiple developer apps together under a single logical owner (e.g. a company or team) for API access management, an alternative to per-individual-developer app ownership.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigee --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigee/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigee/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own appgroup"})

    		ubx.Resource(apigee.ApigeeAppgroup, "example", apigee.ApigeeAppgroupConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
