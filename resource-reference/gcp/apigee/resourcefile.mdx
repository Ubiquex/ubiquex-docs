`google_apigee_resourcefile` is a supporting file (a JavaScript, XSLT, JSON schema, or similar script/config asset) bundled with an Apigee proxy or shared flow, referenced by its own policy steps.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigee --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigee/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigee/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own resourcefile"})

    		ubx.Resource(apigee.ApigeeResourcefile, "ci-runner", apigee.ApigeeResourcefileConfig{})
    	}))
    }
    ```
