`google_apigee_resourcefile` is a supporting file (a JavaScript, XSLT, JSON schema, or similar script/config asset) bundled with an Apigee proxy or shared flow, referenced by its own policy steps.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resourcefile"})

    		ubx.Resource(apigee.Resourcefile, "example", apigee.ResourcefileConfig{
    			ContentType: "example",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.google.apigee import Resourcefile, ResourcefileConfig

    def describe():
        ubx.intent("example own resourcefile")

        ubx.resource(Resourcefile, "example", ResourcefileConfig(
            content_type="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
