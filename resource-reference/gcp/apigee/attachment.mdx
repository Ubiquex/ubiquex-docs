`google_apigee_attachment` attaches an Apigee environment group to a specific Apigee instance, controlling which runtime instance(s) actually serve traffic for that environment group's own hostnames.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigee --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigee/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigee/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own attachment"})

    		ubx.Resource(apigee.ApigeeAttachment, "ci-runner", apigee.ApigeeAttachmentConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_apigee --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ApigeeAttachment } from "./local-sdk/google_apigee/sdk/typescript/google_apigee/apigee/apigee_attachment.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own attachment" });

      resource(ApigeeAttachment, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_apigee --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_apigee/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_apigee.apigee import ApigeeAttachment, ApigeeAttachmentConfig

    def describe():
        ubx.intent("payments own attachment")

        ubx.resource(ApigeeAttachment, "ci-runner", ApigeeAttachmentConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Properties

<ResponseField name="created_at" type="string">
  Output only. The time at which the environment group attachment was created as milliseconds since epoch. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="environment" type="string">
  Required. ID of the attached environment. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="environment_group_id" type="string">
  Output only. ID of the environment group. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  ID of the environment group attachment. Optional; if omitted, computed by Google Cloud.
</ResponseField>
