`google_apigee_sharedflow` packages a reusable sequence of Apigee policy steps that multiple different API proxies can invoke, avoiding duplicating the same policy logic across proxies.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigee --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigee/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigee/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own sharedflow"})

    		ubx.Resource(apigee.ApigeeSharedflow, "ci-runner", apigee.ApigeeSharedflowConfig{
    			ContentType: "example",
    			Data:        "example",
    			Extensions:  []string{"example"},
    		})
    	}))
    }
    ```
  </Tab>

    ```python
    # ubx sdk gen --only google_apigee --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_apigee/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_apigee.apigee import ApigeeSharedflow, ApigeeSharedflowConfig

    def describe():
        ubx.intent("payments own sharedflow")

        ubx.resource(ApigeeSharedflow, "ci-runner", ApigeeSharedflowConfig(
            content_type="example",
            data="example",
            extensions=["example"],
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
