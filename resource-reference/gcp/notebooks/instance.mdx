`google_notebooks_instance` provisions a legacy Vertex AI Workbench user-managed notebook instance, a preconfigured JupyterLab environment running on a dedicated VM.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_notebooks --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_notebooks/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	notebooks "github.com/ubiquex/ubx-sdk-google/sdk/go/google_notebooks/notebooks"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own instance"})

    		ubx.Resource(notebooks.NotebooksInstance, "example", notebooks.NotebooksInstanceConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_notebooks --lang ts --out ./local-sdk
    ```
