`google_speech_v1p1beta1_custom_class` is a Cloud Speech-to-Text custom class, a real, reusable list of domain-specific terms (like product names) used to bias transcription toward correctly recognizing those terms when they appear in audio.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	speech "github.com/ubiquex/ubx-sdk-google/sdk/go/google/speech"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1p1beta1 custom class"})

    		ubx.Resource(speech.V1p1beta1CustomClass, "example", speech.V1p1beta1CustomClassConfig{
    			CustomClass: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>

        ubx.intent("example own v1p1beta1 custom class")
