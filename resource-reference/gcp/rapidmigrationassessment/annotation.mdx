`google_rapidmigrationassessment_annotation` records real, user-supplied context (such as intent or priority) against assets discovered by a Rapid Migration Assessment `google_rapidmigrationassessment_collector`, informing Google's own first-party migration assessment and planning output.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_rapidmigrationassessment --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_rapidmigrationassessment/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	rapidmigrationassessment "github.com/ubiquex/ubx-sdk-google/sdk/go/google_rapidmigrationassessment/rapidmigrationassessment"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own annotation"})

    		ubx.Resource(rapidmigrationassessment.RapidmigrationassessmentAnnotation, "ci-runner", rapidmigrationassessment.RapidmigrationassessmentAnnotationConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
