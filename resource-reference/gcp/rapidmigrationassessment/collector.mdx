`google_rapidmigrationassessment_collector` is a real Rapid Migration Assessment collector, the agent deployed into an existing environment to discover and inventory the real servers, workloads, and dependencies a migration-planning assessment needs.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_rapidmigrationassessment --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_rapidmigrationassessment/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	rapidmigrationassessment "github.com/ubiquex/ubx-sdk-google/sdk/go/google_rapidmigrationassessment/rapidmigrationassessment"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own collector"})

    		ubx.Resource(rapidmigrationassessment.RapidmigrationassessmentCollector, "ci-runner", rapidmigrationassessment.RapidmigrationassessmentCollectorConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
