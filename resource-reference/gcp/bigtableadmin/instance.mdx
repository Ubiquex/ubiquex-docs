## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	bigtableadmin "github.com/ubiquex/ubx-sdk-google/sdk/go/google/bigtableadmin"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own instance"})

    		ubx.Resource(bigtableadmin.Instance, "example", bigtableadmin.InstanceConfig{
    			Clusters:   map[string]string{"managed-by": "ubx"},
    			Instance:   map[string]any{"name": "example-name"},
    			InstanceId: "example",
    			Parent:     "example",
    		})
    	}))
    }
    ```
