`google_ids_endpoint` provisions a Cloud IDS (Intrusion Detection System) endpoint, mirroring a VPC network's own traffic to a managed Palo Alto Networks threat-detection engine.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_ids --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_ids/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	ids "github.com/ubiquex/ubx-sdk-google/sdk/go/google_ids/ids"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own endpoint"})

    		ubx.Resource(ids.IdsEndpoint, "ci-runner", ids.IdsEndpointConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_ids --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { IdsEndpoint } from "./local-sdk/google_ids/sdk/typescript/google_ids/ids/ids_endpoint.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own endpoint" });

      resource(IdsEndpoint, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_ids --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_ids/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_ids.ids import IdsEndpoint, IdsEndpointConfig

    def describe():
        ubx.intent("payments own endpoint")

        ubx.resource(IdsEndpoint, "ci-runner", IdsEndpointConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
