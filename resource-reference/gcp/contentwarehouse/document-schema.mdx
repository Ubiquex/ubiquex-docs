`google_contentwarehouse_document_schema` defines the real property structure (date, numeric, enum, text fields) a `google_contentwarehouse_document` must conform to in Document AI Warehouse, the schema search and filtering are built against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_contentwarehouse --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_contentwarehouse/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	contentwarehouse "github.com/ubiquex/ubx-sdk-google/sdk/go/google_contentwarehouse/contentwarehouse"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own document schema"})

    		ubx.Resource(contentwarehouse.ContentwarehouseDocumentSchema, "ci-runner", contentwarehouse.ContentwarehouseDocumentSchemaConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
