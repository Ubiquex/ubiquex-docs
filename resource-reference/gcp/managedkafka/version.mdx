## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_managedkafka --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_managedkafka/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	managedkafka "github.com/ubiquex/ubx-sdk-google/sdk/go/google_managedkafka/managedkafka"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own version"})

    		ubx.Resource(managedkafka.ManagedkafkaVersion, "ci-runner", managedkafka.ManagedkafkaVersionConfig{
    			Normalize: true,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_managedkafka --lang ts --out ./local-sdk

    export default stack("payments", () => {
      intent({ summary: "payments own version" });

      resource(ManagedkafkaVersion, "ci-runner", {
        normalize: true,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_managedkafka --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_managedkafka/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_managedkafka.managedkafka import ManagedkafkaVersion, ManagedkafkaVersionConfig

    def describe():
        ubx.intent("payments own version")

        ubx.resource(ManagedkafkaVersion, "ci-runner", ManagedkafkaVersionConfig(
            normalize=True,
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Input properties

<ResponseField name="id" type="number">
  Optional. The schema ID of the schema. If not specified, the schema ID will be generated by the server. If the schema ID is specified, it must not be used by an existing schema that is different from the schema to be created. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="normalize" type="bool">
  Optional. If true, the schema will be normalized before being stored. The default is false. Optional.
</ResponseField>

<ResponseField name="references" type="list(object)">
  Optional. The schema references used by the schema. Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="name" type="string">
      The name of the referenced schema or resource. Optional; if omitted, computed by Google Cloud. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="subject" type="string">
      The schema subject this reference points to. Optional; if omitted, computed by Google Cloud. **(AI-inferred)**
    </ResponseField>
    <ResponseField name="version" type="number">
      The specific version of the referenced schema. Optional; if omitted, computed by Google Cloud. **(AI-inferred)**
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="schema" type="string">
  Required. The schema payload Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="schema_type" type="string">
  Optional. The type of the schema. It is optional. If not specified, the schema type will be AVRO. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="version" type="number">
  Optional. The version to create. It is optional. If not specified, the version will be created with the max version ID of the subject increased by 1. If the version ID is specified, it will be used as the new version ID and must not be used by an existing version of the subject. Optional; if omitted, computed by Google Cloud.
</ResponseField>
