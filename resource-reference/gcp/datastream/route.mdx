`google_datastream_route` defines a network route within a Datastream private connection, letting traffic reach source database(s) on an overlapping or non-default IP range.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_datastream --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_datastream/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	datastream "github.com/ubiquex/ubx-sdk-google/sdk/go/google_datastream/datastream"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own route"})

    		ubx.Resource(datastream.DatastreamRoute, "ci-runner", datastream.DatastreamRouteConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_datastream --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    ```

  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_datastream --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_datastream/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_datastream.datastream import DatastreamRoute, DatastreamRouteConfig

    def describe():
        ubx.intent("payments own route")

        ubx.resource(DatastreamRoute, "ci-runner", DatastreamRouteConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
