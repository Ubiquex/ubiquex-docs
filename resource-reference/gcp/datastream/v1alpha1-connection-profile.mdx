`google_datastream_v1alpha1_connection_profile` defines the connectivity and credentials Datastream uses to reach a specific source or destination database or storage system.

## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	datastream "github.com/ubiquex/ubx-sdk-google/sdk/go/google/datastream"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1alpha1 connection profile"})

    		ubx.Resource(datastream.V1alpha1ConnectionProfile, "example", datastream.V1alpha1ConnectionProfileConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
