## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	appengine "github.com/ubiquex/ubx-sdk-google/sdk/go/google/appengine"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1alpha domain mapping"})

    		ubx.Resource(appengine.V1alphaDomainMapping, "example", appengine.V1alphaDomainMappingConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
