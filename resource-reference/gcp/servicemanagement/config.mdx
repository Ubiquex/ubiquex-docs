`google_servicemanagement_config` is a specific, versioned configuration (a Google Service Configuration) for a managed API, defining its own methods, quota, and monitoring behavior.

## Example

<Tabs>
  <Tab title="Go">

    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	servicemanagement "github.com/ubiquex/ubx-sdk-google/sdk/go/google/servicemanagement"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own config"})

    		ubx.Resource(servicemanagement.Config, "example", servicemanagement.ConfigConfig{
    			Name: "example",
    		})
