`google_composer_user_workloads_secret` is a Kubernetes Secret deployed into a Cloud Composer environment's own underlying GKE cluster, made available to user Airflow task workloads.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_composer --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_composer/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	composer "github.com/ubiquex/ubx-sdk-google/sdk/go/google_composer/composer"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own user workloads secret"})

    		ubx.Resource(composer.ComposerUserWorkloadsSecret, "ci-runner", composer.ComposerUserWorkloadsSecretConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
