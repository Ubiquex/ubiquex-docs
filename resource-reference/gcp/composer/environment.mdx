`google_composer_environment` is a managed Cloud Composer environment -- a fully hosted Apache Airflow deployment -- that DAG(s) are uploaded to and run within.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_composer --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_composer/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	composer "github.com/ubiquex/ubx-sdk-google/sdk/go/google_composer/composer"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own environment"})

    		ubx.Resource(composer.ComposerEnvironment, "ci-runner", composer.ComposerEnvironmentConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
