## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	composer "github.com/ubiquex/ubx-sdk-google/sdk/go/google/composer"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own environment"})

    		ubx.Resource(composer.Environment, "ci-runner", composer.EnvironmentConfig{
    			Name:   "ci-runner",
    			Labels: map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```
