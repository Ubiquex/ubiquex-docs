`google_aiplatform_batch_prediction_job` runs a model against a large, offline batch of input data in one pass and writes the predictions to Cloud Storage or BigQuery, rather than serving them from a live online endpoint.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	aiplatform "github.com/ubiquex/ubx-sdk-google/sdk/go/google/aiplatform"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own batch prediction job"})

    		ubx.Resource(aiplatform.BatchPredictionJob, "example", aiplatform.BatchPredictionJobConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
