`google_aiplatform_index_endpoint` is the deployed, servable target a Vertex AI Vector Search index is attached to, handling nearest-neighbor query traffic against the index's own embeddings.

## Example

    ```go

    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	aiplatform "github.com/ubiquex/ubx-sdk-google/sdk/go/google_aiplatform/aiplatform"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own index endpoint"})

    		ubx.Resource(aiplatform.AiplatformIndexEndpoint, "ci-runner", aiplatform.AiplatformIndexEndpointConfig{

    		})

    }
    ```
