`google_aiplatform_reasoning_engine` deploys an agent (built with a framework like LangChain or the Agent Development Kit) onto Vertex AI Agent Engine's own managed runtime, giving it a servable endpoint without the caller having to manage the underlying infrastructure.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	aiplatform "github.com/ubiquex/ubx-sdk-google/sdk/go/google/aiplatform"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own reasoning engine"})

    		ubx.Resource(aiplatform.ReasoningEngine, "example", aiplatform.ReasoningEngineConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
