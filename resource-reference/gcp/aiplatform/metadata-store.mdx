`google_aiplatform_metadata_store` is the top-level container for a region's own Vertex AI ML Metadata -- every artifact, execution, and context tracking lineage across that region's training and pipeline runs lives inside one.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_aiplatform --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_aiplatform/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	aiplatform "github.com/ubiquex/ubx-sdk-google/sdk/go/google_aiplatform/aiplatform"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own metadata store"})

    		ubx.Resource(aiplatform.AiplatformMetadataStore, "example", aiplatform.AiplatformMetadataStoreConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
