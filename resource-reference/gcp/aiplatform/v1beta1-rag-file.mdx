`google_aiplatform_v1beta1_rag_file` imports a document into a Vertex AI RAG (Retrieval-Augmented Generation) corpus, making its own content retrievable by a grounded model at inference time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	aiplatform "github.com/ubiquex/ubx-sdk-google/sdk/go/google/aiplatform"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta1 rag file"})

    		ubx.Resource(aiplatform.V1beta1RagFile, "example", aiplatform.V1beta1RagFileConfig{
    			ImportRagFilesConfig: map[string]any{"gcs_source": map[string]any{"uris": []string{"example"}}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";

      intent({ summary: "example own v1beta1 rag file" });
    ```

  <Tab title="Python">
    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.aiplatform import V1beta1RagFile, V1beta1RagFileConfig
    from ubx.google.aiplatform.v1beta1_rag_file import V1beta1RagFile_ImportRagFilesConfig, V1beta1RagFile_GcsSource

    def describe():
        ubx.intent("example own v1beta1 rag file")

        ubx.resource(V1beta1RagFile, "example", V1beta1RagFileConfig(
            import_rag_files_config=V1beta1RagFile_ImportRagFilesConfig(gcs_source=V1beta1RagFile_GcsSource(uris=["example"])),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
