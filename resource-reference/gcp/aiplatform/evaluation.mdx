`google_aiplatform_evaluation` imports a completed Vertex AI Gen AI Evaluation Service run's own results, scoring a model or application's outputs against a set of evaluation metrics.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	aiplatform "github.com/ubiquex/ubx-sdk-google/sdk/go/google/aiplatform"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own evaluation"})

    		ubx.Resource(aiplatform.Evaluation, "example", aiplatform.EvaluationConfig{
    			ModelEvaluation: map[string]any{"annotation_schema_uri": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { Evaluation } from "./local-sdk/google/sdk/typescript/google/aiplatform/evaluation.ts";

    export default stack("example", () => {
      intent({ summary: "example own evaluation" });

      resource(Evaluation, "example", {
        modelEvaluation: { annotationSchemaUri: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.aiplatform import Evaluation, EvaluationConfig
    from ubx.google.aiplatform.evaluation import Evaluation_ModelEvaluation

    def describe():
        ubx.intent("example own evaluation")

        ubx.resource(Evaluation, "example", EvaluationConfig(
            model_evaluation=Evaluation_ModelEvaluation(annotation_schema_uri="example"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
