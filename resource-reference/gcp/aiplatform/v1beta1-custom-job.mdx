`google_aiplatform_v1beta1_custom_job` runs caller-supplied training code (a container image or a Python package) on managed Vertex AI compute, for training scenarios a built-in AutoML or pre-built algorithm doesn't cover.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	aiplatform "github.com/ubiquex/ubx-sdk-google/sdk/go/google/aiplatform"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta1 custom job"})

    		ubx.Resource(aiplatform.V1beta1CustomJob, "example", aiplatform.V1beta1CustomJobConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
