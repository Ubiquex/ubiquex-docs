`google_aiplatform_feature_online_store` is the low-latency serving layer in Vertex AI's current Feature Store that feature values are synced into for real-time online lookup during model inference.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_aiplatform --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_aiplatform/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	aiplatform "github.com/ubiquex/ubx-sdk-google/sdk/go/google_aiplatform/aiplatform"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own feature online store"})

    		ubx.Resource(aiplatform.AiplatformFeatureOnlineStore, "ci-runner", aiplatform.AiplatformFeatureOnlineStoreConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
