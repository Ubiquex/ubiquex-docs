`google_aiplatform_persistent_resource` is a long-running, pre-provisioned compute cluster on Vertex AI that multiple training jobs or pipeline runs can reuse, avoiding the provisioning delay of spinning up fresh compute for each one.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_aiplatform --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_aiplatform/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	aiplatform "github.com/ubiquex/ubx-sdk-google/sdk/go/google_aiplatform/aiplatform"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own persistent resource"})

    		ubx.Resource(aiplatform.AiplatformPersistentResource, "ci-runner", aiplatform.AiplatformPersistentResourceConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
