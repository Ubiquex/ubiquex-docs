`google_aiplatform_index` is a Vertex AI Vector Search index built over a set of embedding vectors, enabling fast approximate nearest-neighbor lookup once deployed to an index endpoint.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_aiplatform --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_aiplatform/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	aiplatform "github.com/ubiquex/ubx-sdk-google/sdk/go/google_aiplatform/aiplatform"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own index"})

    		ubx.Resource(aiplatform.AiplatformIndex, "ci-runner", aiplatform.AiplatformIndexConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
