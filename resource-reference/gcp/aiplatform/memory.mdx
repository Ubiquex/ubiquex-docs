`google_aiplatform_memory` is a single stored memory (a durable fact or preference extracted from past conversations) held in Vertex AI Agent Engine's own Memory Bank, retrievable to personalize a future agent interaction.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_aiplatform --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_aiplatform/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	aiplatform "github.com/ubiquex/ubx-sdk-google/sdk/go/google_aiplatform/aiplatform"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own memory"})

    		ubx.Resource(aiplatform.AiplatformMemory, "ci-runner", aiplatform.AiplatformMemoryConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
