## Example

<Tabs>
  <Tab title="Go">

    package main

    import (

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own version"})

    		})
    	}))
    }
