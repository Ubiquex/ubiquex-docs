`google_iam_token` represents a short-lived OAuth access token generated on demand for a service account via impersonation, the real, callable output of the IAM Service Account Credentials API's own `generateAccessToken` method.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_iam --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_iam/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	iam "github.com/ubiquex/ubx-sdk-google/sdk/go/google_iam/iam"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own token"})

    		ubx.Resource(iam.IamToken, "ci-runner", iam.IamTokenConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_iam --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { IamToken } from "./local-sdk/google_iam/sdk/typescript/google_iam/iam/iam_token.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own token" });

      resource(IamToken, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_iam --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_iam/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_iam.iam import IamToken, IamTokenConfig

    def describe():
        ubx.intent("payments own token")

        ubx.resource(IamToken, "ci-runner", IamTokenConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Properties

<ResponseField name="display_name" type="string">
  Optional. Gemini Enterprise only. The display name of the SCIM token. Cannot exceed 32 characters. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  Identifier. Gemini Enterprise only. The resource name of the SCIM Token. Format: `locations/&#123;location&#125;/workforcePools/&#123;workforce_pool&#125;/providers/ &#123;workforce_pool_provider&#125;/scimTenants/&#123;scim_tenant&#125;/tokens/&#123;token&#125;` Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="security_token" type="string">
  Output only. Gemini Enterprise only. The token string. Provide this to the IdP for authentication. Will be set only during creation. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="state" type="string">
  Output only. Gemini Enterprise only. The state of the token. Optional; if omitted, computed by Google Cloud.
</ResponseField>
