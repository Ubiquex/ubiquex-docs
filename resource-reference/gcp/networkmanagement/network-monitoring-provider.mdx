`google_networkmanagement_network_monitoring_provider` registers a monitoring data source (e.g. a specific region's own telemetry) that Network Intelligence Center draws performance data from.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_networkmanagement --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networkmanagement/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkmanagement "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networkmanagement/networkmanagement"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own network monitoring provider"})

    		ubx.Resource(networkmanagement.NetworkmanagementNetworkMonitoringProvider, "ci-runner", networkmanagement.NetworkmanagementNetworkMonitoringProviderConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
