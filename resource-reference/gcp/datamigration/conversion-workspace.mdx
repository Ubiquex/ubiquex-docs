`google_datamigration_conversion_workspace` is a Database Migration Service workspace for converting a source database's own schema to a target engine's dialect (e.g. Oracle to PostgreSQL) ahead of a heterogeneous migration.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_datamigration --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_datamigration/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	datamigration "github.com/ubiquex/ubx-sdk-google/sdk/go/google_datamigration/datamigration"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own conversion workspace"})

    		ubx.Resource(datamigration.DatamigrationConversionWorkspace, "ci-runner", datamigration.DatamigrationConversionWorkspaceConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
