`google_datamigration_private_connection` is a private network path (VPC peering) Database Migration Service uses to reach a source database that isn't reachable over the public internet.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_datamigration --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_datamigration/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	datamigration "github.com/ubiquex/ubx-sdk-google/sdk/go/google_datamigration/datamigration"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own private connection"})

    		ubx.Resource(datamigration.DatamigrationPrivateConnection, "example", datamigration.DatamigrationPrivateConnectionConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
