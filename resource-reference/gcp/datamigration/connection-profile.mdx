`google_datamigration_connection_profile` stores the connection detail (host, credentials, network path) Database Migration Service uses to reach a source or destination database.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	datamigration "github.com/ubiquex/ubx-sdk-google/sdk/go/google/datamigration"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own connection profile"})

    		ubx.Resource(datamigration.ConnectionProfile, "example", datamigration.ConnectionProfileConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
