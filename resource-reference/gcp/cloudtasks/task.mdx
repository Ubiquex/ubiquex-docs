`google_cloudtasks_task` is a single unit of deferred or scheduled work enqueued in a Cloud Tasks queue, dispatched to its own target when its own schedule time arrives.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_cloudtasks --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_cloudtasks/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	cloudtasks "github.com/ubiquex/ubx-sdk-google/sdk/go/google_cloudtasks/cloudtasks"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own task"})

    		ubx.Resource(cloudtasks.CloudtasksTask, "example", cloudtasks.CloudtasksTaskConfig{
    			ResponseView: "example",

    		})
    	}))
    }
    ```
