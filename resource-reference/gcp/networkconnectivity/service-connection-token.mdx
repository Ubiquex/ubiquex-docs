`google_networkconnectivity_service_connection_token` is a one-time credential a service producer issues to authorize a specific consumer network to establish a Private Service Connect connection to its own service.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_networkconnectivity --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networkconnectivity/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkconnectivity "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networkconnectivity/networkconnectivity"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own service connection token"})

    		ubx.Resource(networkconnectivity.NetworkconnectivityServiceConnectionToken, "ci-runner", networkconnectivity.NetworkconnectivityServiceConnectionTokenConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
