`google_networkconnectivity_destination` defines a routable target (e.g. a set of IP prefixes) that Network Connectivity Center's policy-based routing rules can direct traffic toward.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_networkconnectivity --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networkconnectivity/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkconnectivity "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networkconnectivity/networkconnectivity"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own destination"})

    		ubx.Resource(networkconnectivity.NetworkconnectivityDestination, "ci-runner", networkconnectivity.NetworkconnectivityDestinationConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
