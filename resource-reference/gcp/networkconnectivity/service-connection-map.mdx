`google_networkconnectivity_service_connection_map` defines which producer service(s) a consumer network is permitted to connect to via Private Service Connect automation, and which subnet(s) those connections draw addresses from.

## Example

<Tabs>

    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkconnectivity "github.com/ubiquex/ubx-sdk-google/sdk/go/google/networkconnectivity"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own service connection map"})

    		ubx.Resource(networkconnectivity.ServiceConnectionMap, "example", networkconnectivity.ServiceConnectionMapConfig{
    			Name: "example",
    		})
    	}))
    }
