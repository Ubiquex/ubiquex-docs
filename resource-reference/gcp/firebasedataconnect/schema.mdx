`google_firebasedataconnect_schema` defines the GraphQL data model backing a Firebase Data Connect service, mapped onto its own underlying Cloud SQL PostgreSQL database.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_firebasedataconnect --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_firebasedataconnect/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	firebasedataconnect "github.com/ubiquex/ubx-sdk-google/sdk/go/google_firebasedataconnect/firebasedataconnect"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own schema"})

    		ubx.Resource(firebasedataconnect.FirebasedataconnectSchema, "ci-runner", firebasedataconnect.FirebasedataconnectSchemaConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
