## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_parametermanager --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_parametermanager/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	parametermanager "github.com/ubiquex/ubx-sdk-google/sdk/go/google_parametermanager/parametermanager"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own template"})

    		ubx.Resource(parametermanager.ParametermanagerTemplate, "example", parametermanager.ParametermanagerTemplateConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
