`google_parametermanager_parameter` is a named configuration parameter managed by Parameter Manager, with version(s) holding its own structured or plain-text value.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_parametermanager --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_parametermanager/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	parametermanager "github.com/ubiquex/ubx-sdk-google/sdk/go/google_parametermanager/parametermanager"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own parameter"})

    		ubx.Resource(parametermanager.ParametermanagerParameter, "ci-runner", parametermanager.ParametermanagerParameterConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_parametermanager --lang ts --out ./local-sdk
    ```
