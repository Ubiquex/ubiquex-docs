## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	config "github.com/ubiquex/ubx-sdk-google/sdk/go/google/config"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own deployment"})

    		ubx.Resource(config.Deployment, "example", config.DeploymentConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
