`google_cloudresourcemanager_project` is a Google Cloud project, the base-level container that billing, IAM policy, and every other resource in this system are ultimately scoped under.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_cloudresourcemanager --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_cloudresourcemanager/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	cloudresourcemanager "github.com/ubiquex/ubx-sdk-google/sdk/go/google_cloudresourcemanager/cloudresourcemanager"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own project"})

    		ubx.Resource(cloudresourcemanager.CloudresourcemanagerProject, "example", cloudresourcemanager.CloudresourcemanagerProjectConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
