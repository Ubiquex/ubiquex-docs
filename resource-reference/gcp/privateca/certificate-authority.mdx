`google_privateca_certificate_authority` is a private Certificate Authority managed by Certificate Authority Service, issuing X.509 certificate(s) within its own CA pool.

## Example

  <Tab title="Go">

    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	privateca "github.com/ubiquex/ubx-sdk-google/sdk/go/google_privateca/privateca"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own certificate authority"})

    		ubx.Resource(privateca.PrivatecaCertificateAuthority, "example", privateca.PrivatecaCertificateAuthorityConfig{
    			Name: "example",
    		})
    	}))
    }
