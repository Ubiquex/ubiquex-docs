`google_certificatemanager_certificate_map` is the resource a load balancer's target proxy references to serve TLS certificates, resolving the right certificate per incoming hostname from its own map entries.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_certificatemanager --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_certificatemanager/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	certificatemanager "github.com/ubiquex/ubx-sdk-google/sdk/go/google_certificatemanager/certificatemanager"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own certificate map"})

    		ubx.Resource(certificatemanager.CertificatemanagerCertificateMap, "example", certificatemanager.CertificatemanagerCertificateMapConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
