## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dataplex --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dataplex/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataplex "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dataplex/dataplex"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own zone"})

    		ubx.Resource(dataplex.DataplexZone, "ci-runner", dataplex.DataplexZoneConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_dataplex --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { DataplexZone } from "./local-sdk/google_dataplex/sdk/typescript/google_dataplex/dataplex/dataplex_zone.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own zone" });

      resource(DataplexZone, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_dataplex --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_dataplex/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_dataplex.dataplex import DataplexZone, DataplexZoneConfig

    def describe():
        ubx.intent("payments own zone")

            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
