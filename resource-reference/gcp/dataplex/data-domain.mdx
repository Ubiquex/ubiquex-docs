`google_dataplex_data_domain` groups related data product(s) and their own governance ownership under a single business domain (e.g. `Finance` or `Marketing`) within Dataplex's data mesh model.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dataplex --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dataplex/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataplex "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dataplex/dataplex"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own data domain"})

    		ubx.Resource(dataplex.DataplexDataDomain, "example", dataplex.DataplexDataDomainConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
