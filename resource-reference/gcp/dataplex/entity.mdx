`google_dataplex_entity` registers a single structured or semi-structured data entity (a table-like object backed by files or a BigQuery table) discovered or defined within a Dataplex zone.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dataplex --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dataplex/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataplex "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dataplex/dataplex"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own entity"})

    		ubx.Resource(dataplex.DataplexEntity, "example", dataplex.DataplexEntityConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
