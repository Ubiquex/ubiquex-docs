`google_dataplex_entry_group` is a logical container of related catalog entries within Dataplex Universal Catalog, typically grouping entries that share the same source system or namespace.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dataplex --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dataplex/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataplex "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dataplex/dataplex"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own entry group"})

    		ubx.Resource(dataplex.DataplexEntryGroup, "ci-runner", dataplex.DataplexEntryGroupConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
