`google_dataplex_entry_group` is a logical container of related catalog entries within Dataplex Universal Catalog, typically grouping entries that share the same source system or namespace.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataplex "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dataplex"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own entry group"})

    		ubx.Resource(dataplex.EntryGroup, "example", dataplex.EntryGroupConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
