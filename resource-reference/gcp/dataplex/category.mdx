`google_dataplex_category` defines a data classification node within a Dataplex policy tag taxonomy, used to enforce BigQuery column-level access control based on sensitivity.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dataplex --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dataplex/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataplex "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dataplex/dataplex"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own category"})

    		ubx.Resource(dataplex.DataplexCategory, "example", dataplex.DataplexCategoryConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
