`google_dataplex_data_attribute_binding` attaches a data taxonomy attribute (e.g. a sensitivity classification) to a specific data asset, resource, or column within Dataplex.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dataplex --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dataplex/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataplex "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dataplex/dataplex"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own data attribute binding"})

    		ubx.Resource(dataplex.DataplexDataAttributeBinding, "ci-runner", dataplex.DataplexDataAttributeBindingConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
