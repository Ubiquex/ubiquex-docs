`google_dataplex_metadata_job` runs a bulk import or export operation against Dataplex Universal Catalog metadata, e.g. bulk-creating entries, aspects, or glossary term links from an external source.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dataplex --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dataplex/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataplex "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dataplex/dataplex"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own metadata job"})

    		ubx.Resource(dataplex.DataplexMetadataJob, "ci-runner", dataplex.DataplexMetadataJobConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>

</ResponseField>
