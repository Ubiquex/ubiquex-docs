`google_dataplex_entry` is a single cataloged item in Dataplex Universal Catalog (Knowledge Catalog) -- a table, file set, dashboard, or any other real or logical asset -- carrying its own metadata and aspect(s).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dataplex --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dataplex/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataplex "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dataplex/dataplex"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own entry"})

    		ubx.Resource(dataplex.DataplexEntry, "example", dataplex.DataplexEntryConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
