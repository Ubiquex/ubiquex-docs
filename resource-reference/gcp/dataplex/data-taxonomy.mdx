`google_dataplex_data_taxonomy` is a hierarchical classification scheme (a tree of attribute nodes) Dataplex uses to tag data with sensitivity or business categories for governance.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dataplex --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dataplex/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataplex "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dataplex/dataplex"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own data taxonomy"})

    		ubx.Resource(dataplex.DataplexDataTaxonomy, "ci-runner", dataplex.DataplexDataTaxonomyConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```

</ResponseField>

<ResponseField name="name" type="string">
