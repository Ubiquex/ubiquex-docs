`google_developerconnect_insights_config` enables Developer Connect Insights (Application Insights) for a linked repository, automatically discovering the application(s) and runtime deployment(s) built from its own code.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_developerconnect --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_developerconnect/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	developerconnect "github.com/ubiquex/ubx-sdk-google/sdk/go/google_developerconnect/developerconnect"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own insights config"})

    		ubx.Resource(developerconnect.DeveloperconnectInsightsConfig, "ci-runner", developerconnect.DeveloperconnectInsightsConfigConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_developerconnect --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { DeveloperconnectInsightsConfig } from "./local-sdk/google_developerconnect/sdk/typescript/google_developerconnect/developerconnect/developerconnect_insights_config.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own insights config" });

      resource(DeveloperconnectInsightsConfig, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
