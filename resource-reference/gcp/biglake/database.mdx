`google_biglake_database` groups related BigLake table(s) together within a catalog, mirroring the database concept from open-source metastores like Hive or Iceberg.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_biglake --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_biglake/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	biglake "github.com/ubiquex/ubx-sdk-google/sdk/go/google_biglake/biglake"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own database"})

    		ubx.Resource(biglake.BiglakeDatabase, "ci-runner", biglake.BiglakeDatabaseConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_biglake --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { BiglakeDatabase } from "./local-sdk/google_biglake/sdk/typescript/google_biglake/biglake/biglake_database.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own database" });

      resource(BiglakeDatabase, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_biglake --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_biglake/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_biglake.biglake import BiglakeDatabase, BiglakeDatabaseConfig

    def describe():
        ubx.intent("payments own database")

        ubx.resource(BiglakeDatabase, "ci-runner", BiglakeDatabaseConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Properties

<ResponseField name="create_time" type="string">
  Output only. The creation time of the database. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="delete_time" type="string">
  Output only. The deletion time of the database. Only set after the database is deleted. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="expire_time" type="string">
  Output only. The time when this database is considered expired. Only set after the database is deleted. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="hive_options" type="object">
  Options of a Hive database. Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="location_uri" type="string">
      Cloud Storage folder URI where the database data is stored, starting with "gs://". Optional.
    </ResponseField>
    <ResponseField name="parameters" type="map(string)">
      Stores user supplied Hive database parameters. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="name" type="string">
  Output only. The resource name. Format: projects/&#123;project_id_or_number&#125;/locations/&#123;location_id&#125;/catalogs/&#123;catalog_id&#125;/databases/&#123;database_id&#125; Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="type" type="string">
  The database type. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="update_time" type="string">
  Output only. The last modification time of the database. Optional; if omitted, computed by Google Cloud.
</ResponseField>
