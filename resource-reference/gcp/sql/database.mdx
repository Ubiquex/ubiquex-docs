## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	sql "github.com/ubiquex/ubx-sdk-google/sdk/go/google/sql"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own database"})

    		ubx.Resource(sql.Database, "ci-runner", sql.DatabaseConfig{
    			Instance: "example",
    			Name:     "ci-runner",
    		})
    	}))
    }
    ```
