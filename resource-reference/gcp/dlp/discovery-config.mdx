`google_dlp_discovery_config` configures Sensitive Data Protection to automatically and continuously scan data assets across an org, project, or specific resource for sensitive data, rather than requiring a manually triggered job.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dlp --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dlp/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dlp "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dlp/dlp"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own discovery config"})

    		ubx.Resource(dlp.DlpDiscoveryConfig, "ci-runner", dlp.DlpDiscoveryConfigConfig{
    			ConfigId:        "example",
    			DiscoveryConfig: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
