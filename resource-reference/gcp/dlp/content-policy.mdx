`google_dlp_content_policy` defines organization-wide default configuration (inspection templates, discovery settings) automatically applied across Sensitive Data Protection scans unless a scan overrides it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dlp "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dlp"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own content policy"})

    		ubx.Resource(dlp.ContentPolicy, "example", dlp.ContentPolicyConfig{
    			ContentPolicy:   map[string]any{"default_action": map[string]any{"return_verdict": "example"}},
    			ContentPolicyId: "example",
    		})
    	}))
    }
    ```
