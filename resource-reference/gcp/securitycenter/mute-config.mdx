`google_securitycenter_mute_config` automatically mutes matching Security Command Center finding(s), suppressing them from active alerting while keeping them recorded.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_securitycenter --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_securitycenter/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	securitycenter "github.com/ubiquex/ubx-sdk-google/sdk/go/google_securitycenter/securitycenter"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own mute config"})

    		ubx.Resource(securitycenter.SecuritycenterMuteConfig, "ci-runner", securitycenter.SecuritycenterMuteConfigConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
