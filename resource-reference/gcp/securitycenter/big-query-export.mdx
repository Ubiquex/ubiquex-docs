`google_securitycenter_big_query_export` continuously streams Security Command Center finding(s) into a BigQuery dataset for custom analysis, dashboards, or long-term retention.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_securitycenter --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_securitycenter/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	securitycenter "github.com/ubiquex/ubx-sdk-google/sdk/go/google_securitycenter/securitycenter"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own big query export"})

    		ubx.Resource(securitycenter.SecuritycenterBigQueryExport, "example", securitycenter.SecuritycenterBigQueryExportConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
