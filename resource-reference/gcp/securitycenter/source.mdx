`google_securitycenter_source` registers an external or custom detector as a finding source within Security Command Center, letting it publish its own finding(s) alongside Google's built-in detectors.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_securitycenter --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_securitycenter/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	securitycenter "github.com/ubiquex/ubx-sdk-google/sdk/go/google_securitycenter/securitycenter"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own source"})

    		ubx.Resource(securitycenter.SecuritycenterSource, "example", securitycenter.SecuritycenterSourceConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
