`google_securitycenter_resource_value_config` assigns a business-priority value to a scoped set of Google Cloud resources, letting Security Command Center weight findings against them accordingly.

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
