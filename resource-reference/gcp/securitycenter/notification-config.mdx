`google_securitycenter_notification_config` streams matching Security Command Center finding(s) to a Pub/Sub topic in real time as they're created or updated.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_securitycenter --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_securitycenter/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	securitycenter "github.com/ubiquex/ubx-sdk-google/sdk/go/google_securitycenter/securitycenter"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own notification config"})

    		ubx.Resource(securitycenter.SecuritycenterNotificationConfig, "ci-runner", securitycenter.SecuritycenterNotificationConfigConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
