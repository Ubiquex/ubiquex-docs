`google_discoveryengine_license_config` defines a named license tier used to track and assign per-user Discovery Engine / Agent Search entitlements within an organization.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_discoveryengine --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_discoveryengine/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	discoveryengine "github.com/ubiquex/ubx-sdk-google/sdk/go/google_discoveryengine/discoveryengine"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own license config"})

    		ubx.Resource(discoveryengine.DiscoveryengineLicenseConfig, "ci-runner", discoveryengine.DiscoveryengineLicenseConfigConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_discoveryengine --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    ```
