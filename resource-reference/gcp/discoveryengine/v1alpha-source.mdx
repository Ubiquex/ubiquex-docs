`google_discoveryengine_v1alpha_source` connects an external content source (e.g. a website or a third-party connector) to a Vertex AI Search data store, feeding it data to index.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	discoveryengine "github.com/ubiquex/ubx-sdk-google/sdk/go/google/discoveryengine"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1alpha source"})

    		ubx.Resource(discoveryengine.V1alphaSource, "example", discoveryengine.V1alphaSourceConfig{
    			UserContents: []map[string]any{{"agentspace_content": map[string]any{"document_name": "example-document-name"}}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.discoveryengine import V1alphaSource, V1alphaSourceConfig
    from ubx.google.discoveryengine.v1alpha_source import V1alphaSource_UserContents, V1alphaSource_UserContents_AgentspaceContent

    def describe():
        ubx.intent("example own v1alpha source")

        ubx.resource(V1alphaSource, "example", V1alphaSourceConfig(
            user_contents=[V1alphaSource_UserContents(agentspace_content=V1alphaSource_UserContents_AgentspaceContent(document_name="example-document-name"))],
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Output properties

<ResponseField name="metadata" type="object">
  Represents the metadata of a source and some additional information. Computed by Google Cloud after `ship`.
  <Expandable title="properties">
    <ResponseField name="agentspace_metadata" type="object">
      Metadata about an agentspace source. Optional.
      <Expandable title="properties">
        <ResponseField name="document_name" type="string">
          Output only. The full document name in Agentspace. Computed by Google Cloud after `ship`.
        </ResponseField>
        <ResponseField name="document_title" type="string">
          Output only. The title of the document. Computed by Google Cloud after `ship`.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="google_docs_metadata" type="object">
      Metadata about a google doc source. Optional.
      <Expandable title="properties">
        <ResponseField name="document_id" type="string">
          Output only. The document id of the google doc. Computed by Google Cloud after `ship`.
        </ResponseField>
        <ResponseField name="revision_id" type="string">
          Output only. Revision id for the doc. Computed by Google Cloud after `ship`.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="source_added_timestamp" type="string">
      The timestamp the source was added. Optional.
    </ResponseField>
    <ResponseField name="token_count" type="number">
      The number of tokens in the source. Optional.
    </ResponseField>
    <ResponseField name="word_count" type="number">
      The word count of the source. Optional.
    </ResponseField>
    <ResponseField name="youtube_metadata" type="object">
      Metadata about a youtube video source. Optional.
      <Expandable title="properties">
        <ResponseField name="channel_name" type="string">
          Output only. The channel name of the youtube video. Computed by Google Cloud after `ship`.
        </ResponseField>
        <ResponseField name="video_id" type="string">
          Output only. The id of the youtube video. Computed by Google Cloud after `ship`.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="name" type="string">
  Identifier. The full resource name of the source. Format: `projects/&#123;project&#125;/locations/&#123;location&#125;/notebooks/&#123;notebook&#125;/sources/&#123;source_id&#125;`. This field must be a UTF-8 encoded string with a length limit of 1024 characters. Computed by Google Cloud after `ship`.
</ResponseField>
