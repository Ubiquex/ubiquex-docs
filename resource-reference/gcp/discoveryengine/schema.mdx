## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_discoveryengine --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_discoveryengine/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	discoveryengine "github.com/ubiquex/ubx-sdk-google/sdk/go/google_discoveryengine/discoveryengine"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own schema"})

    		ubx.Resource(discoveryengine.DiscoveryengineSchema, "ci-runner", discoveryengine.DiscoveryengineSchemaConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_discoveryengine --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";

    export default stack("payments", () => {
      intent({ summary: "payments own schema" });

      resource(DiscoveryengineSchema, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_discoveryengine --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_discoveryengine/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_discoveryengine.discoveryengine import DiscoveryengineSchema, DiscoveryengineSchemaConfig

    def describe():
        ubx.intent("payments own schema")

        ubx.resource(DiscoveryengineSchema, "ci-runner", DiscoveryengineSchemaConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Properties

<ResponseField name="json_schema" type="string">
  The JSON representation of the schema. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  Immutable. The full resource name of the schema, in the format of `projects/&#123;project&#125;/locations/&#123;location&#125;/collections/&#123;collection&#125;/dataStores/&#123;data_store&#125;/schemas/&#123;schema&#125;`. This field must be a UTF-8 encoded string with a length limit of 1024 characters. Optional; if omitted, computed by Google Cloud.
</ResponseField>

  The structured representation of the schema. Optional; if omitted, computed by Google Cloud.
</ResponseField>
