`google_discoveryengine_push_notification_config` configures an endpoint Discovery Engine sends push notification(s) to when a relevant event (e.g. a new matching document) occurs.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_discoveryengine --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_discoveryengine/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	discoveryengine "github.com/ubiquex/ubx-sdk-google/sdk/go/google_discoveryengine/discoveryengine"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own push notification config"})

    		ubx.Resource(discoveryengine.DiscoveryenginePushNotificationConfig, "example", discoveryengine.DiscoveryenginePushNotificationConfigConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    // ubx sdk gen --only google_discoveryengine --lang ts --out ./local-sdk
