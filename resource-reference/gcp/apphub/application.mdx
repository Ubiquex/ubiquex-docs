`google_apphub_application` groups related service(s) and workload(s) together under a single logical application in App Hub, letting them be reasoned about and managed as one unit rather than individually.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apphub --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apphub/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apphub "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apphub/apphub"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own application"})

    		ubx.Resource(apphub.ApphubApplication, "ci-runner", apphub.ApphubApplicationConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_apphub --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ApphubApplication } from "./local-sdk/google_apphub/sdk/typescript/google_apphub/apphub/apphub_application.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own application" });

      resource(ApphubApplication, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_apphub --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_apphub/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_apphub.apphub import ApphubApplication, ApphubApplicationConfig

    def describe():
        ubx.intent("payments own application")

        ubx.resource(ApphubApplication, "ci-runner", ApphubApplicationConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
