`google_apphub_workload` registers an underlying Google Cloud resource (e.g. a GKE deployment or Compute Engine instance group) as a discoverable compute workload within an App Hub application.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apphub "github.com/ubiquex/ubx-sdk-google/sdk/go/google/apphub"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own workload"})

    		ubx.Resource(apphub.Workload, "example", apphub.WorkloadConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Workload } from "@ubx/sdk-google/google/apphub/workload";

    export default stack("example", () => {
      intent({ summary: "example own workload" });

      resource(Workload, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.apphub import Workload, WorkloadConfig

    def describe():
        ubx.intent("example own workload")

        ubx.resource(Workload, "example", WorkloadConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
