`google_datastore_index` is a composite index for Cloud Datastore, created to support queries Datastore's own automatic single-property indexing can't serve; a newly created index starts in a real `CREATING` state and becomes `READY` once built.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_datastore --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_datastore/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	datastore "github.com/ubiquex/ubx-sdk-google/sdk/go/google_datastore/datastore"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own index"})

    		ubx.Resource(datastore.DatastoreIndex, "ci-runner", datastore.DatastoreIndexConfig{})
    	}))
    }
    ```
