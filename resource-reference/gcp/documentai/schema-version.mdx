## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	documentai "github.com/ubiquex/ubx-sdk-google/sdk/go/google/documentai"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own schema version"})

    		ubx.Resource(documentai.SchemaVersion, "example", documentai.SchemaVersionConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
