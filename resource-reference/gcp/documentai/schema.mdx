## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_documentai --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_documentai/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	documentai "github.com/ubiquex/ubx-sdk-google/sdk/go/google_documentai/documentai"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own schema"})

    		ubx.Resource(documentai.DocumentaiSchema, "example", documentai.DocumentaiSchemaConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>

    # ubx sdk gen --only google_documentai --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_documentai/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_documentai.documentai import DocumentaiSchema, DocumentaiSchemaConfig

    def describe():
        ubx.intent("example own schema")

        ubx.resource(DocumentaiSchema, "example", DocumentaiSchemaConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
