`google_documentai_processor` is a Document AI processor instance -- a specific document-parsing model (e.g. invoice or form parser) -- that document(s) are sent to for extraction.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_documentai --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_documentai/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	documentai "github.com/ubiquex/ubx-sdk-google/sdk/go/google_documentai/documentai"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own processor"})

    		ubx.Resource(documentai.DocumentaiProcessor, "example", documentai.DocumentaiProcessorConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
