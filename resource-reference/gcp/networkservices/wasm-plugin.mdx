`google_networkservices_wasm_plugin` deploys a caller-authored WebAssembly module as a Service Extension, running custom request/response processing logic inline on a load balancer or service mesh's own data path.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_networkservices --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networkservices/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkservices "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networkservices/networkservices"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own wasm plugin"})

    		ubx.Resource(networkservices.NetworkservicesWasmPlugin, "ci-runner", networkservices.NetworkservicesWasmPluginConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
