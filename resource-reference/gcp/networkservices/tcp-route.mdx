`google_networkservices_tcp_route` defines routing rule(s) for raw TCP traffic within a Cloud Service Mesh, matching on port and directing matching connections to a specific backend.

## Example

<Tabs>
  <Tab title="Go">

    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkservices "github.com/ubiquex/ubx-sdk-google/sdk/go/google/networkservices"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own tcp route"})

    		ubx.Resource(networkservices.TcpRoute, "example", networkservices.TcpRouteConfig{
    			Name: "example",
    		})
    	}))
    }
