`google_networkservices_extension_binding` attaches a Service Extension (a WASM plugin or external callout) to a specific load balancer or gateway resource, activating it for that resource's own traffic.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkservices "github.com/ubiquex/ubx-sdk-google/sdk/go/google/networkservices"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own extension binding"})

    		ubx.Resource(networkservices.ExtensionBinding, "example", networkservices.ExtensionBindingConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
