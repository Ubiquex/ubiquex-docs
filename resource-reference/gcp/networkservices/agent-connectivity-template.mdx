`google_networkservices_agent_connectivity_template` defines a reusable connectivity configuration (ingress/egress mode, policy defaults) that Agent Gateway deployments on the Gemini Enterprise Agent Platform are created from.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_networkservices --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networkservices/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkservices "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networkservices/networkservices"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own agent connectivity template"})

    		ubx.Resource(networkservices.NetworkservicesAgentConnectivityTemplate, "ci-runner", networkservices.NetworkservicesAgentConnectivityTemplateConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
