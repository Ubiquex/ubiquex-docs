`google_networkservices_mesh` is the top-level Cloud Service Mesh resource that gRPC and HTTP route(s) are attached to, defining the logical service mesh workloads participate in.

## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networkservices/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkservices "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networkservices/networkservices"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own mesh"})

    		ubx.Resource(networkservices.NetworkservicesMesh, "example", networkservices.NetworkservicesMeshConfig{
    			Name: "example",
    		})
    	}))
    }
