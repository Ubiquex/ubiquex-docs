`google_networkservices_agent_gateway` deploys Agent Gateway, the Gemini Enterprise Agent Platform's own networking component that secures and governs traffic between users and agents, agents and tools, or agents and other agents.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkservices "github.com/ubiquex/ubx-sdk-google/sdk/go/google/networkservices"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own agent gateway"})

    		ubx.Resource(networkservices.AgentGateway, "example", networkservices.AgentGatewayConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
