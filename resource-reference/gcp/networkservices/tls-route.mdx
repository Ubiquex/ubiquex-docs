`google_networkservices_tls_route` defines routing rule(s) for TLS traffic within a Cloud Service Mesh, matching on SNI and directing matching connections to a specific backend without terminating TLS.

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {

    		ubx.Resource(networkservices.NetworkservicesTlsRoute, "ci-runner", networkservices.NetworkservicesTlsRouteConfig{

    }
