`google_networkservices_producer_extension` runs a Service Extension on the producer side of a Private Service Connect-published service, letting the producer inspect or modify traffic before it reaches the underlying service.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_networkservices --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networkservices/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkservices "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networkservices/networkservices"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own producer extension"})

    		ubx.Resource(networkservices.NetworkservicesProducerExtension, "example", networkservices.NetworkservicesProducerExtensionConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
