`google_backupdr_management_server` is the central Backup and DR control-plane instance that manages backup plan(s), backup vault(s), and their own orchestration.

    	backupdr "github.com/ubiquex/ubx-sdk-google/sdk/go/google_backupdr/backupdr"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
