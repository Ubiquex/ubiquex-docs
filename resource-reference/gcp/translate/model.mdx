`google_translate_model` is a real, custom-trained Cloud Translation AutoML model, trained against a `google_translate_dataset` to translate a specific domain's terminology and style more accurately than the general-purpose model.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_translate --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_translate/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	translate "github.com/ubiquex/ubx-sdk-google/sdk/go/google_translate/translate"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own model"})

    		ubx.Resource(translate.TranslateModel, "ci-runner", translate.TranslateModelConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_translate --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { TranslateModel } from "./local-sdk/google_translate/sdk/typescript/google_translate/translate/translate_model.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own model" });

      resource(TranslateModel, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_translate --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_translate/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_translate.translate import TranslateModel, TranslateModelConfig

    def describe():
        ubx.intent("payments own model")

        ubx.resource(TranslateModel, "ci-runner", TranslateModelConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Properties

<ResponseField name="create_time" type="string">
  Output only. Timestamp when the model resource was created, which is also when the training started. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="dataset" type="string">
  Required. The dataset from which the model is trained, in form of `projects/&#123;project-number-or-id&#125;/locations/&#123;location_id&#125;/datasets/&#123;dataset_id&#125;` Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="display_name" type="string">
  The name of the model to show in the interface. The name can be up to 32 characters long and can consist only of ASCII Latin letters A-Z and a-z, underscores (_), and ASCII digits 0-9. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  The resource name of the model, in form of `projects/&#123;project-number-or-id&#125;/locations/&#123;location_id&#125;/models/&#123;model_id&#125;` Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="source_language_code" type="string">
  Output only. The BCP-47 language code of the source language. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="target_language_code" type="string">
  Output only. The BCP-47 language code of the target language. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="test_example_count" type="number">
  Output only. Number of examples (sentence pairs) used to test the model. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="train_example_count" type="number">
  Output only. Number of examples (sentence pairs) used to train the model. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="update_time" type="string">
  Output only. Timestamp when this model was last updated. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="validate_example_count" type="number">
  Output only. Number of examples (sentence pairs) used to validate the model. Optional; if omitted, computed by Google Cloud.
</ResponseField>
