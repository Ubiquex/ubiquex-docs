`google_dataflow_template` is a reusable, parameterized Dataflow pipeline definition that a job can be launched from without providing the full pipeline code each time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dataflow --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dataflow/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dataflow/dataflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own template"})

    		ubx.Resource(dataflow.DataflowTemplate, "example", dataflow.DataflowTemplateConfig{
    			Environment: map[string]any{"additional_experiments": []string{"example"}},
    			GcsPath:     "example",
    			JobName:     "example-job-name",
    			Location:    "example",
    			Parameters:  map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_dataflow --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { DataflowTemplate } from "./local-sdk/google_dataflow/sdk/typescript/google_dataflow/dataflow/dataflow_template.ts";

    export default stack("example", () => {
      intent({ summary: "example own template" });

      resource(DataflowTemplate, "example", {
        environment: { additionalExperiments: ["example"] },
        gcsPath: "example",
        jobName: "example-job-name",
        location: "example",
        parameters: { "managed-by": "ubx" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_dataflow --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_dataflow/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_dataflow.dataflow import DataflowTemplate, DataflowTemplateConfig

    def describe():
        ubx.intent("example own template")

        ubx.resource(DataflowTemplate, "example", DataflowTemplateConfig(
            environment={"additional_experiments": ["example"]},
            gcs_path="example",
            job_name="example-job-name",
            location="example",
            parameters={"managed-by": "ubx"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

    </ResponseField>
    <ResponseField name="network" type="string">
      Optional. Network to which VMs will be assigned. If empty or unspecified, the service will use the network "default". Optional.
    </ResponseField>
    <ResponseField name="num_workers" type="number">
      Optional. The initial number of Google Compute Engine instances for the job. The default value is 11. Optional.
    </ResponseField>
    <ResponseField name="service_account_email" type="string">
      Optional. The email address of the service account to run the job as. Optional.
    </ResponseField>
    <ResponseField name="streaming_mode" type="string">
      Optional. Specifies the Streaming Engine message processing guarantees. Reduces cost and latency but might result in duplicate messages committed to storage. Designed to run simple mapping streaming ETL jobs at the lowest cost. For example, Change Data Capture (CDC) to BigQuery is a canonical use case. For more information, see [Set the pipeline streaming mode](https://cloud.google.com/dataflow/docs/guides/streaming-modes). Optional.
    </ResponseField>
    <ResponseField name="subnetwork" type="string">
      Optional. Subnetwork to which VMs will be assigned, if desired. You can specify a subnetwork using either a complete URL or an abbreviated path. Expected to be of the form "https://www.googleapis.com/compute/v1/projects/HOST_PROJECT_ID/regions/REGION/subnetworks/SUBNETWORK" or "regions/REGION/subnetworks/SUBNETWORK". If the subnetwork is located in a Shared VPC network, you must use the complete URL. Optional.
    </ResponseField>
    <ResponseField name="temp_location" type="string">
      Required. The Cloud Storage path to use for temporary files. Must be a valid Cloud Storage URL, beginning with `gs://`. Optional.
    </ResponseField>
    <ResponseField name="worker_region" type="string">
      Required. The Compute Engine region (https://cloud.google.com/compute/docs/regions-zones/regions-zones) in which worker processing should occur, e.g. "us-west1". Mutually exclusive with worker_zone. If neither worker_region nor worker_zone is specified, default to the control plane's region. Optional.
    </ResponseField>
    <ResponseField name="worker_zone" type="string">
      Optional. The Compute Engine zone (https://cloud.google.com/compute/docs/regions-zones/regions-zones) in which worker processing should occur, e.g. "us-west1-a". Mutually exclusive with worker_region. If neither worker_region nor worker_zone is specified, a zone in the control plane's region is chosen based on available capacity. If both `worker_zone` and `zone` are set, `worker_zone` takes precedence. Optional.
    </ResponseField>
    <ResponseField name="zone" type="string">
      Optional. The Compute Engine [availability zone](https://cloud.google.com/compute/docs/regions-zones/regions-zones) for launching worker instances to run your pipeline. In the future, worker_zone will take precedence. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="gcs_path" type="string">
  Required. A Cloud Storage path to the template from which to create the job. Must be a valid Cloud Storage URL, beginning with `gs://`. Optional.
</ResponseField>

<ResponseField name="job_name" type="string">
  Required. The job name to use for the created job. Optional.
</ResponseField>

<ResponseField name="location" type="string">
  The [regional endpoint] (https://cloud.google.com/dataflow/docs/concepts/regional-endpoints) to which to direct the request. Optional.
</ResponseField>

<ResponseField name="parameters" type="map(string)">
  The runtime parameters to pass to the job. Optional.
</ResponseField>
