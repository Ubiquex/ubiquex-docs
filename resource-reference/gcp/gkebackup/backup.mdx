`google_gkebackup_backup` is a point-in-time backup of a GKE cluster's own workload configuration and (optionally) persistent volume data, created from a backup plan.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_gkebackup --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_gkebackup/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	gkebackup "github.com/ubiquex/ubx-sdk-google/sdk/go/google_gkebackup/gkebackup"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own backup"})

    		ubx.Resource(gkebackup.GkebackupBackup, "ci-runner", gkebackup.GkebackupBackupConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
