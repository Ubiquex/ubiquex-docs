`google_apim_observation_job` runs Apigee's Shadow API Discovery against a Google Cloud project's real API traffic, looking for undocumented (shadow) APIs by observing live activity in the project's own VPC network and comparing what it finds against the APIs already catalogued in API hub.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apim --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apim/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apim "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apim/apim"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own observation job"})

    		ubx.Resource(apim.ApimObservationJob, "example", apim.ApimObservationJobConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
