`google_apim_observation_source` is the configured project, VPC network, and subnetwork that an Apigee Shadow API Discovery `google_apim_observation_job` reads live traffic from in order to detect undocumented APIs.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apim "github.com/ubiquex/ubx-sdk-google/sdk/go/google/apim"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own observation source"})

    		ubx.Resource(apim.ObservationSource, "example", apim.ObservationSourceConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
