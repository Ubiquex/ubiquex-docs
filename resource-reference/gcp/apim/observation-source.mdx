`google_apim_observation_source` is the configured project, VPC network, and subnetwork that an Apigee Shadow API Discovery `google_apim_observation_job` reads live traffic from in order to detect undocumented APIs.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apim --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apim/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apim "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apim/apim"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own observation source"})

    		ubx.Resource(apim.ApimObservationSource, "ci-runner", apim.ApimObservationSourceConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
