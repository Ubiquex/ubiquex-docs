    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	cloudbuildv2 "github.com/ubiquex/ubx-sdk-google/sdk/go/google/cloudbuildv2"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {

    		})
    	}))
    }
