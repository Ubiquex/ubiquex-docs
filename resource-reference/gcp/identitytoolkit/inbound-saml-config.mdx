    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_identitytoolkit/sdk/go

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    		})
    	}))
    }

## Properties

<ResponseField name="display_name" type="string">
  The config's display name set by developers. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="enabled" type="bool">
  True if allows the user to sign in with the provider. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="idp_config" type="object">
  The SAML IdP (Identity Provider) configuration when the project acts as the relying party. Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="idp_certificates" type="list(object)">
      IDP's public keys for verifying signature in the assertions. Optional.
      <Expandable title="properties">
        <ResponseField name="x509_certificate" type="string">
          The X.509 certificate, in PEM format. Optional; if omitted, computed by Google Cloud. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="idp_entity_id" type="string">
      Unique identifier for all SAML entities. Optional.
    </ResponseField>
    <ResponseField name="sign_request" type="bool">
      Indicates if outbounding SAMLRequest should be signed. Optional.
    </ResponseField>
    <ResponseField name="sso_url" type="string">
      URL to send Authentication request to. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="name" type="string">
  The name of the InboundSamlConfig resource, for example: 'projects/my-awesome-project/inboundSamlConfigs/my-config-id'. Ignored during create requests. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="sp_config" type="object">
  The SAML SP (Service Provider) configuration when the project acts as the relying party to receive and accept an authentication assertion issued by a SAML identity provider. Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="callback_uri" type="string">
      Callback URI where responses from IDP are handled. Optional.
    </ResponseField>
    <ResponseField name="sp_certificates" type="list(object)">
      Output only. Public certificates generated by the server to verify the signature in SAMLRequest in the SP-initiated flow. Computed by Google Cloud after `ship`.
      <Expandable title="properties">
        <ResponseField name="expires_at" type="string">
          When this certificate expires. Optional; if omitted, computed by Google Cloud. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="x509_certificate" type="string">
          The X.509 certificate, in PEM format. Optional; if omitted, computed by Google Cloud. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="sp_entity_id" type="string">
      Unique identifier for all SAML entities. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
