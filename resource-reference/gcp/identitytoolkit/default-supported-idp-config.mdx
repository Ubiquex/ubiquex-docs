`google_identitytoolkit_default_supported_idp_config` configures one of Identity Platform's own built-in social identity providers (e.g. Google, Facebook, Apple) for a project or tenant.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	identitytoolkit "github.com/ubiquex/ubx-sdk-google/sdk/go/google/identitytoolkit"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own default supported idp config"})

    		ubx.Resource(identitytoolkit.DefaultSupportedIdpConfig, "example", identitytoolkit.DefaultSupportedIdpConfigConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
