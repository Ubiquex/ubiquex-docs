## Example

    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own datastore"})

    		ubx.Resource(vmwareengine.Datastore, "example", vmwareengine.DatastoreConfig{

    		})
    	}))
    }

## Properties

<ResponseField name="clusters" type="list(string)">
  Output only. Clusters to which the datastore is attached. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="create_time" type="string">
  Output only. Creation time of this resource. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="description" type="string">
  Optional. User-provided description for this datastore Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="etag" type="string">
  Optional. Checksum that may be sent on update and delete requests to ensure that the user-provided value is up to date before the server processes a request. The server computes checksums based on the value of other fields in the request. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  Output only. Identifier. The resource name of this datastore. Resource names are schemeless URIs that follow the conventions in https://cloud.google.com/apis/design/resource_names. For example: `projects/my-project/locations/us-central1/datastores/datastore` Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="nfs_datastore" type="object">
  The NFS datastore configuration. Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="google_file_service" type="object">
      Google service file service configuration Optional.
      <Expandable title="properties">

          Google filestore instance resource name e.g. projects/my-project/locations/me-west1-b/instances/my-instance Optional.
        </ResponseField>
        <ResponseField name="netapp_volume" type="string">
          Google netapp volume resource name e.g. projects/my-project/locations/me-west1-b/volumes/my-volume Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="google_vmware_file_service" type="any">
      Volume message captures user inputs for creation of file services managed by GCVE Optional.
    </ResponseField>
    <ResponseField name="third_party_file_service" type="object">
      Third party file service configuration Optional.
      <Expandable title="properties">
        <ResponseField name="file_share" type="string">
          Required. Required Mount Folder name Optional.
        </ResponseField>
        <ResponseField name="network" type="string">
          Required. Required to identify vpc peering used for NFS access network name of NFS's vpc e.g. projects/project-id/global/networks/my-network_id Optional.
        </ResponseField>
        <ResponseField name="servers" type="list(string)">
          Required. Server IP addresses of the NFS file service. NFS v3, provide a single IP address or DNS name. Multiple servers can be supported in future when NFS 4.1 protocol support is enabled. Optional.

      </Expandable>
    </ResponseField>
  </Expandable>
