`google_vmwareengine_hcx_activation_key` generates a key used to activate a VMware HCX site connecting an on-premises VMware environment to a Google Cloud VMware Engine private cloud, for live workload migration.

## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_vmwareengine/sdk/go
    package main

    import (

    	vmwareengine "github.com/ubiquex/ubx-sdk-google/sdk/go/google_vmwareengine/vmwareengine"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    		})
