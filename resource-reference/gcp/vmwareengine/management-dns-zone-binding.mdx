`google_vmwareengine_management_dns_zone_binding` binds a Google Cloud VMware Engine private cloud's own management DNS zone to a consumer VPC network, letting workloads there resolve vCenter/NSX management hostnames.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	vmwareengine "github.com/ubiquex/ubx-sdk-google/sdk/go/google/vmwareengine"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own management dns zone binding"})

    		ubx.Resource(vmwareengine.ManagementDnsZoneBinding, "example", vmwareengine.ManagementDnsZoneBindingConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
