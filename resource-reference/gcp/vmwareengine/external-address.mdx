`google_vmwareengine_external_address` assigns a public IP address to a specific VM running inside a Google Cloud VMware Engine private cloud, via NAT from the private cloud's own external address range.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_vmwareengine --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_vmwareengine/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	vmwareengine "github.com/ubiquex/ubx-sdk-google/sdk/go/google_vmwareengine/vmwareengine"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own external address"})

    		ubx.Resource(vmwareengine.VmwareengineExternalAddress, "ci-runner", vmwareengine.VmwareengineExternalAddressConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
