`google_artifactregistry_repository` is an Artifact Registry repository, storing container image(s), language package(s), or other artifact(s) of a specific format.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_artifactregistry --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_artifactregistry/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	artifactregistry "github.com/ubiquex/ubx-sdk-google/sdk/go/google_artifactregistry/artifactregistry"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own repository"})

    		ubx.Resource(artifactregistry.ArtifactregistryRepository, "example", artifactregistry.ArtifactregistryRepositoryConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
