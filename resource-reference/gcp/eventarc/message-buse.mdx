`google_eventarc_message_buse` (a Message Bus) is Eventarc's central event routing hub that publishers send event(s) into and enrollment(s) subscribe destination(s) to receive matching event(s) from, the newer alternative to point-to-point triggers.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_eventarc --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_eventarc/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	eventarc "github.com/ubiquex/ubx-sdk-google/sdk/go/google_eventarc/eventarc"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own message buse"})

    		ubx.Resource(eventarc.EventarcMessageBuse, "ci-runner", eventarc.EventarcMessageBuseConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
