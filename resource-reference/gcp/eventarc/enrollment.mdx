## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	eventarc "github.com/ubiquex/ubx-sdk-google/sdk/go/google/eventarc"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own enrollment"})

    		ubx.Resource(eventarc.Enrollment, "example", eventarc.EnrollmentConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
