`google_eventarc_pipeline` transforms and forwards event(s) flowing through Eventarc from a message bus to one or more destination(s), optionally applying a transformation template along the way.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_eventarc --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_eventarc/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	eventarc "github.com/ubiquex/ubx-sdk-google/sdk/go/google_eventarc/eventarc"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own pipeline"})

    		ubx.Resource(eventarc.EventarcPipeline, "ci-runner", eventarc.EventarcPipelineConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
