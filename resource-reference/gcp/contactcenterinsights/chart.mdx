## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	contactcenterinsights "github.com/ubiquex/ubx-sdk-google/sdk/go/google/contactcenterinsights"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own chart"})

    		ubx.Resource(contactcenterinsights.Chart, "example", contactcenterinsights.ChartConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
