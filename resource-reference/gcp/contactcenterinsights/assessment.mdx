`google_contactcenterinsights_assessment` records a completed quality assurance evaluation of a conversation against a scorecard, holding the scored answer(s) to each of the scorecard's own question(s).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	contactcenterinsights "github.com/ubiquex/ubx-sdk-google/sdk/go/google/contactcenterinsights"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own assessment"})

    		ubx.Resource(contactcenterinsights.Assessment, "example", contactcenterinsights.AssessmentConfig{
    			Name: "example",
    		})
    	}))
    }
    ```

<ResponseField name="agent_info" type="object">
  Information about an agent involved in the conversation. Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="agent_id" type="string">
      A user-specified string representing the agent. Optional.
    </ResponseField>
    <ResponseField name="agent_type" type="string">
      The agent type, e.g. HUMAN_AGENT. Optional.
    </ResponseField>
    <ResponseField name="deployment_display_name" type="string">
      The agent's deployment display name. Only applicable to automated agents. Optional.
    </ResponseField>
    <ResponseField name="deployment_id" type="string">
      The agent's deployment ID. Only applicable to automated agents. Optional.
    </ResponseField>
    <ResponseField name="display_name" type="string">
      The agent's name. Optional.
    </ResponseField>
    <ResponseField name="disposition_code" type="string">
      A user-provided string indicating the outcome of the agent's segment of the call. Optional.
    </ResponseField>
    <ResponseField name="entry_subagent_display_name" type="string">
      The entry subagent's display name. Optional.
    </ResponseField>
    <ResponseField name="entry_subagent_id" type="string">
      The entry subagent's ID. Optional.
    </ResponseField>
    <ResponseField name="location" type="string">
      The agent's location. Optional.
    </ResponseField>
    <ResponseField name="team" type="string">
      A user-specified string representing the agent's team. Deprecated in favor of the `teams` field. Optional.

    </ResponseField>

    </ResponseField>

    </ResponseField>
  </Expandable>
