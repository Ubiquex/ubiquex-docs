`google_contactcenterinsights_assessment` records a completed quality assurance evaluation of a conversation against a scorecard, holding the scored answer(s) to each of the scorecard's own question(s).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_contactcenterinsights --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_contactcenterinsights/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	contactcenterinsights "github.com/ubiquex/ubx-sdk-google/sdk/go/google_contactcenterinsights/contactcenterinsights"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own assessment"})

    		ubx.Resource(contactcenterinsights.ContactcenterinsightsAssessment, "ci-runner", contactcenterinsights.ContactcenterinsightsAssessmentConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
