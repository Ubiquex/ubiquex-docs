`google_contactcenterinsights_dataset` is a named collection of conversation(s) grouped together in Contact Center AI Insights, e.g. for training an issue model or running a bulk analysis.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_contactcenterinsights --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_contactcenterinsights/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	contactcenterinsights "github.com/ubiquex/ubx-sdk-google/sdk/go/google_contactcenterinsights/contactcenterinsights"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own dataset"})

    		ubx.Resource(contactcenterinsights.ContactcenterinsightsDataset, "ci-runner", contactcenterinsights.ContactcenterinsightsDatasetConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
