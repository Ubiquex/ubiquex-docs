`google_assuredworkloads_v1beta1_workload` provisions an Assured Workloads folder pre-configured with the data residency, personnel access, and compliance controls required for a specific regulatory regime (e.g. FedRAMP or IL4).

## Example

<Tabs>

    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	assuredworkloads "github.com/ubiquex/ubx-sdk-google/sdk/go/google/assuredworkloads"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta1 workload"})

    		ubx.Resource(assuredworkloads.V1beta1Workload, "example", assuredworkloads.V1beta1WorkloadConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
