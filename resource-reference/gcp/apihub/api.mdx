`google_apihub_api` registers a single API with API hub, Google Cloud's own catalog for discovering and governing every API across an organization, whether it runs on Google Cloud or elsewhere.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apihub --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apihub/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apihub "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apihub/apihub"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own api"})

    		ubx.Resource(apihub.ApihubApi, "example", apihub.ApihubApiConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_apihub --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ApihubApi } from "./local-sdk/google_apihub/sdk/typescript/google_apihub/apihub/apihub_api.ts";

    export default stack("example", () => {
      intent({ summary: "example own api" });

      resource(ApihubApi, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_apihub --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_apihub/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_apihub.apihub import ApihubApi, ApihubApiConfig

    def describe():
        ubx.intent("example own api")

        ubx.resource(ApihubApi, "example", ApihubApiConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
