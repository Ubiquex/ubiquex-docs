`google_apihub_version` is a specific, named version of an API cataloged in API hub, grouping the spec, deployment(s), and operation(s) that belong to that version.

## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apihub/sdk/go

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apihub "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apihub/apihub"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own version"})

    		ubx.Resource(apihub.ApihubVersion, "example", apihub.ApihubVersionConfig{
    			Name: "example",
    		})
    	}))
    }
