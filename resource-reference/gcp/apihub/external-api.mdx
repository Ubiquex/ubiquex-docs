`google_apihub_external_api` catalogs an API that isn't managed by Google Cloud at all (a third-party or partner API), so API hub's own discovery and dependency tracking covers it too.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apihub "github.com/ubiquex/ubx-sdk-google/sdk/go/google/apihub"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own external api"})

    		ubx.Resource(apihub.ExternalApi, "example", apihub.ExternalApiConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { ExternalApi } from "@ubx/sdk-google/google/apihub/external_api";

    export default stack("example", () => {
      intent({ summary: "example own external api" });

      resource(ExternalApi, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.apihub import ExternalApi, ExternalApiConfig

    def describe():
        ubx.intent("example own external api")

        ubx.resource(ExternalApi, "example", ExternalApiConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
