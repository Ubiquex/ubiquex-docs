## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apihub "github.com/ubiquex/ubx-sdk-google/sdk/go/google/apihub"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own spec"})

    		ubx.Resource(apihub.Spec, "example", apihub.SpecConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
