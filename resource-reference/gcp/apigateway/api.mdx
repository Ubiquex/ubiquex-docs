`google_apigateway_api` is the top-level API Gateway resource representing one logical API, the parent every `google_apigateway_config` version is created and deployed under.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigateway --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigateway/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigateway "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigateway/apigateway"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own api"})

    		ubx.Resource(apigateway.ApigatewayApi, "example", apigateway.ApigatewayApiConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_apigateway --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ApigatewayApi } from "./local-sdk/google_apigateway/sdk/typescript/google_apigateway/apigateway/apigateway_api.ts";

    export default stack("example", () => {
      intent({ summary: "example own api" });

      resource(ApigatewayApi, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_apigateway --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_apigateway/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_apigateway.apigateway import ApigatewayApi, ApigatewayApiConfig

    def describe():
        ubx.intent("example own api")

        ubx.resource(ApigatewayApi, "example", ApigatewayApiConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

<ResponseField name="labels" type="map(string)">

</ResponseField>

<ResponseField name="managed_service" type="string">
  Optional. Immutable. The name of a Google Managed Service ( https://cloud.google.com/service-infrastructure/docs/glossary#managed). If not specified, a new Service will automatically be created in the same project as this API. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">

</ResponseField>

</ResponseField>

</ResponseField>
