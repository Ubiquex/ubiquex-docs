`google_apigateway_gateway` is a real, Envoy-based managed proxy that hosts one deployed `google_apigateway_config`, giving that API a public, externally-reachable URL on the `gateway.dev` domain.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_apigateway --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_apigateway/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigateway "github.com/ubiquex/ubx-sdk-google/sdk/go/google_apigateway/apigateway"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own gateway"})

    		ubx.Resource(apigateway.ApigatewayGateway, "example", apigateway.ApigatewayGatewayConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    if __name__ == "__main__":
        ubx.run("example", describe)
