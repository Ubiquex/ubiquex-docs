    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_ml/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	ml "github.com/ubiquex/ubx-sdk-google/sdk/go/google_ml/ml"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own trial"})

    		ubx.Resource(ml.MlTrial, "example", ml.MlTrialConfig{
    			Name: "example",
    		})
    	}))
    }

## Properties

<ResponseField name="client_id" type="string">
  Output only. The identifier of the client that originally requested this trial. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="end_time" type="string">
  Output only. Time at which the trial's status changed to COMPLETED. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="final_measurement" type="object">
  A message representing a measurement. Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="elapsed_time" type="string">
      Output only. Time that the trial has been running at the point of this measurement. Optional.
    </ResponseField>
    <ResponseField name="metrics" type="list(object)">
      Provides a list of metrics that act as inputs into the objective function. Optional.
      <Expandable title="properties">
        <ResponseField name="metric" type="string">
          The name of the metric being reported. Optional; if omitted, computed by Google Cloud. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="value" type="number">
          The literal value of this field. Optional; if omitted, computed by Google Cloud. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="step_count" type="string">
      The number of steps a machine learning model has been trained for. Must be non-negative. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="infeasible_reason" type="string">
  Output only. A human readable string describing why the trial is infeasible. This should only be set if trial_infeasible is true. Optional; if omitted, computed by Google Cloud.
</ResponseField>
