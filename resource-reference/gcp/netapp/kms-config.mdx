`google_netapp_kms_config` configures the Cloud KMS customer-managed encryption key Google Cloud NetApp Volumes uses to encrypt a storage pool's own volume(s).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_netapp --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_netapp/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	netapp "github.com/ubiquex/ubx-sdk-google/sdk/go/google_netapp/netapp"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own kms config"})

    		ubx.Resource(netapp.NetappKmsConfig, "ci-runner", netapp.NetappKmsConfigConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_netapp --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { NetappKmsConfig } from "./local-sdk/google_netapp/sdk/typescript/google_netapp/netapp/netapp_kms_config.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own kms config" });

      resource(NetappKmsConfig, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_netapp --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_netapp/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_netapp.netapp import NetappKmsConfig, NetappKmsConfigConfig

    def describe():
        ubx.intent("payments own kms config")

        ubx.resource(NetappKmsConfig, "ci-runner", NetappKmsConfigConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

  Output only. Create time of the KmsConfig. Optional; if omitted, computed by Google Cloud.

<ResponseField name="crypto_key_name" type="string">

</ResponseField>

<ResponseField name="instructions" type="string">

</ResponseField>
