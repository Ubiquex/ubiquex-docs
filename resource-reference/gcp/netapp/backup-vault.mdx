## Properties

<ResponseField name="backup_region" type="string">
  Optional. Region where the backups are stored. Format: `projects/&#123;project_id&#125;/locations/&#123;location&#125;` Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="backup_retention_policy" type="object">
  Retention policy for backups in the backup vault Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="backup_minimum_enforced_retention_days" type="number">
      Required. Minimum retention duration in days for backups in the backup vault. Optional.
    </ResponseField>
    <ResponseField name="daily_backup_immutable" type="bool">
      Optional. Indicates if the daily backups are immutable. At least one of daily_backup_immutable, weekly_backup_immutable, monthly_backup_immutable and manual_backup_immutable must be true. Optional.
    </ResponseField>
    <ResponseField name="manual_backup_immutable" type="bool">
      Optional. Indicates if the manual backups are immutable. At least one of daily_backup_immutable, weekly_backup_immutable, monthly_backup_immutable and manual_backup_immutable must be true. Optional.
    </ResponseField>
    <ResponseField name="monthly_backup_immutable" type="bool">
      Optional. Indicates if the monthly backups are immutable. At least one of daily_backup_immutable, weekly_backup_immutable, monthly_backup_immutable and manual_backup_immutable must be true. Optional.
    </ResponseField>
    <ResponseField name="weekly_backup_immutable" type="bool">
      Optional. Indicates if the weekly backups are immutable. At least one of daily_backup_immutable, weekly_backup_immutable, monthly_backup_immutable and manual_backup_immutable must be true. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="backup_vault_type" type="string">
  Optional. Type of backup vault to be created. Default is IN_REGION. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="backups_crypto_key_version" type="string">
  Output only. The crypto key version used to encrypt the backup vault. Format: `projects/&#123;project&#125;/locations/&#123;location&#125;/keyRings/&#123;key_ring&#125;/cryptoKeys/&#123;crypto_key&#125;/cryptoKeyVersions/&#123;crypto_key_version&#125;` Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="create_time" type="string">
  Output only. Create time of the backup vault. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="description" type="string">
  Description of the backup vault. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="destination_backup_vault" type="string">
  Output only. Name of the Backup vault created in backup region. Format: `projects/&#123;project_id&#125;/locations/&#123;location&#125;/backupVaults/&#123;backup_vault_id&#125;` Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="encryption_state" type="string">
  Output only. Field indicating encryption state of CMEK backups. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="kms_config" type="string">
  Optional. Specifies the Key Management System (KMS) configuration to be used for backup encryption. Format: `projects/&#123;project&#125;/locations/&#123;location&#125;/kmsConfigs/&#123;kms_config&#125;` Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="labels" type="map(string)">
  Resource labels to represent user provided metadata. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  Identifier. The resource name of the backup vault. Format: `projects/&#123;project_id&#125;/locations/&#123;location&#125;/backupVaults/&#123;backup_vault_id&#125;`. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="source_backup_vault" type="string">
  Output only. Name of the Backup vault created in source region. Format: `projects/&#123;project_id&#125;/locations/&#123;location&#125;/backupVaults/&#123;backup_vault_id&#125;` Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="source_region" type="string">
  Optional. Region in which the backup vault is created. Format: `projects/&#123;project_id&#125;/locations/&#123;location&#125;` Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="state" type="string">
  Output only. The backup vault state. Optional; if omitted, computed by Google Cloud.
</ResponseField>
