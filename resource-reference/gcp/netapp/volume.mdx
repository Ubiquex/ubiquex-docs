`google_netapp_volume` is a managed NFS or SMB file share provisioned from a Google Cloud NetApp Volumes storage pool, mountable by Compute Engine VMs, GKE, or other clients.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_netapp --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_netapp/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	netapp "github.com/ubiquex/ubx-sdk-google/sdk/go/google_netapp/netapp"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own volume"})

    		ubx.Resource(netapp.NetappVolume, "example", netapp.NetappVolumeConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
