`google_connectors_event_subscription` subscribes a connection to real-time event notifications from its own backend system, delivered to a configured destination.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_connectors --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_connectors/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	connectors "github.com/ubiquex/ubx-sdk-google/sdk/go/google_connectors/connectors"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own event subscription"})

    		ubx.Resource(connectors.ConnectorsEventSubscription, "example", connectors.ConnectorsEventSubscriptionConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
