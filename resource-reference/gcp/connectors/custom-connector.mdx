`google_connectors_custom_connector` defines a custom Integration Connectors connector for a system with no built-in connector, built from an OpenAPI specification.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_connectors --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_connectors/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	connectors "github.com/ubiquex/ubx-sdk-google/sdk/go/google_connectors/connectors"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own custom connector"})

    		ubx.Resource(connectors.ConnectorsCustomConnector, "example", connectors.ConnectorsCustomConnectorConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_connectors --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ConnectorsCustomConnector } from "./local-sdk/google_connectors/sdk/typescript/google_connectors/connectors/connectors_custom_connector.ts";

    export default stack("example", () => {
      intent({ summary: "example own custom connector" });

      resource(ConnectorsCustomConnector, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_connectors --lang py --out ./local-sdk

    import ubx_sdk as ubx
    from ubx.google_connectors.connectors import ConnectorsCustomConnector, ConnectorsCustomConnectorConfig

    def describe():
        ubx.intent("example own custom connector")

        ubx.resource(ConnectorsCustomConnector, "example", ConnectorsCustomConnectorConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
