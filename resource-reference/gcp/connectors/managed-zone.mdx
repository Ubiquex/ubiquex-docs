`google_connectors_managed_zone` is a private DNS zone Integration Connectors uses to resolve a backend's own hostname when connecting over a private network path.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_connectors --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_connectors/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	connectors "github.com/ubiquex/ubx-sdk-google/sdk/go/google_connectors/connectors"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own managed zone"})

    		ubx.Resource(connectors.ConnectorsManagedZone, "ci-runner", connectors.ConnectorsManagedZoneConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
