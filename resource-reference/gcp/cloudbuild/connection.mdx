`google_cloudbuild_connection` links a Cloud Build project to a source code platform (GitHub, GitLab, Bitbucket) via a 2nd-gen repository connection, the resource individual repository links are created under.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_cloudbuild --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_cloudbuild/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	cloudbuild "github.com/ubiquex/ubx-sdk-google/sdk/go/google_cloudbuild/cloudbuild"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own connection"})

    		ubx.Resource(cloudbuild.CloudbuildConnection, "example", cloudbuild.CloudbuildConnectionConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_cloudbuild --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { CloudbuildConnection } from "./local-sdk/google_cloudbuild/sdk/typescript/google_cloudbuild/cloudbuild/cloudbuild_connection.ts";

    export default stack("example", () => {
      intent({ summary: "example own connection" });

      resource(CloudbuildConnection, "example", {
        name: "example",
      });
    });
    ```
  </Tab>

    # ubx sdk gen --only google_cloudbuild --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_cloudbuild/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_cloudbuild.cloudbuild import CloudbuildConnection, CloudbuildConnectionConfig

    def describe():
        ubx.intent("example own connection")

            name="example",

    if __name__ == "__main__":
        ubx.run("example", describe)
