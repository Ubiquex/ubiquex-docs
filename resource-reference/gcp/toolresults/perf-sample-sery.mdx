`google_toolresults_perf_sample_sery` is a Tool Results API performance sample series, a real, timestamp-ordered stream of performance samples (such as CPU, memory, or graphics metrics) captured during one `google_toolresults_step`, up to 5,000 samples per series.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	toolresults "github.com/ubiquex/ubx-sdk-google/sdk/go/google/toolresults"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own perf sample sery"})

    		ubx.Resource(toolresults.PerfSampleSery, "example", toolresults.PerfSampleSeryConfig{
    			BasicPerfSampleSeries: map[string]any{"perf_metric_type": "example"},
    		})
    	}))
    }
    ```
