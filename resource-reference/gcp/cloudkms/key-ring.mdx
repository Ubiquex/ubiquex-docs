## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_cloudkms --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_cloudkms/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	cloudkms "github.com/ubiquex/ubx-sdk-google/sdk/go/google_cloudkms/cloudkms"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own key ring"})

    		ubx.Resource(cloudkms.CloudkmsKeyRing, "example", cloudkms.CloudkmsKeyRingConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_cloudkms --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";

    export default stack("example", () => {
      intent({ summary: "example own key ring" });

      resource(CloudkmsKeyRing, "example", {
        name: "example",
      });
    });
    ```

  <Tab title="Python">

    # ubx sdk gen --only google_cloudkms --lang py --out ./local-sdk

    import ubx_sdk as ubx
    from ubx.google_cloudkms.cloudkms import CloudkmsKeyRing, CloudkmsKeyRingConfig

    def describe():
        ubx.intent("example own key ring")

        ubx.resource(CloudkmsKeyRing, "example", CloudkmsKeyRingConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Properties

<ResponseField name="create_time" type="string">
  Output only. The time at which this KeyRing was created. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  Output only. The resource name for the KeyRing in the format `projects/*/locations/*/keyRings/*`. Optional; if omitted, computed by Google Cloud.
</ResponseField>
