`google_oracledatabase_cloud_vm_cluster` provisions a cluster of VM(s) on Oracle Exadata infrastructure within Oracle Database@Google Cloud, that Oracle database instance(s) run on.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	oracledatabase "github.com/ubiquex/ubx-sdk-google/sdk/go/google/oracledatabase"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own cloud vm cluster"})

    		ubx.Resource(oracledatabase.CloudVmCluster, "example", oracledatabase.CloudVmClusterConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
