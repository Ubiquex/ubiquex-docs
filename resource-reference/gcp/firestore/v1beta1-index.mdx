`google_firestore_v1beta1_index` defines a composite or single-field index on a Firestore database, required for certain query shapes (e.g. filtering and sorting on different fields) to run efficiently.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	firestore "github.com/ubiquex/ubx-sdk-google/sdk/go/google/firestore"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta1 index"})

    		ubx.Resource(firestore.V1beta1Index, "example", firestore.V1beta1IndexConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
