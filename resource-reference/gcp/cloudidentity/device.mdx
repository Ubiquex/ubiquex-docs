`google_cloudidentity_device` represents an end-user device (a phone, laptop) enrolled in Cloud Identity's own device management, tracked for compliance and access-control purposes.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_cloudidentity --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_cloudidentity/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	cloudidentity "github.com/ubiquex/ubx-sdk-google/sdk/go/google_cloudidentity/cloudidentity"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own device"})

    		ubx.Resource(cloudidentity.CloudidentityDevice, "example", cloudidentity.CloudidentityDeviceConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
