`google_cloudidentity_v1beta1_device` represents an end-user device (a phone, laptop) enrolled in Cloud Identity's own device management, tracked for compliance and access-control purposes.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	cloudidentity "github.com/ubiquex/ubx-sdk-google/sdk/go/google/cloudidentity"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta1 device"})

    		ubx.Resource(cloudidentity.V1beta1Device, "example", cloudidentity.V1beta1DeviceConfig{
    			Customer: "example",
    			Device:   map[string]any{"android_specific_attributes": map[string]any{"cts_profile_match": true}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { V1beta1Device } from "@ubx/sdk-google/google/cloudidentity/v1beta1_device";

    export default stack("example", () => {
      intent({ summary: "example own v1beta1 device" });

      resource(V1beta1Device, "example", {
        customer: "example",
        device: { androidSpecificAttributes: { ctsProfileMatch: true } },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.cloudidentity import V1beta1Device, V1beta1DeviceConfig
    from ubx.google.cloudidentity.v1beta1_device import V1beta1Device_Device, V1beta1Device_AndroidSpecificAttributes

    def describe():
        ubx.intent("example own v1beta1 device")

        ubx.resource(V1beta1Device, "example", V1beta1DeviceConfig(
            customer="example",
            device=V1beta1Device_Device(android_specific_attributes=V1beta1Device_AndroidSpecificAttributes(cts_profile_match=True)),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
