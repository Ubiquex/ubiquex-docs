`google_healthcare_user_data_mapping` links a specific piece of health data to the patient user it belongs to, within a Cloud Healthcare API consent store, so consent decisions can be evaluated per patient.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_healthcare --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_healthcare/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	healthcare "github.com/ubiquex/ubx-sdk-google/sdk/go/google_healthcare/healthcare"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own user data mapping"})

    		ubx.Resource(healthcare.HealthcareUserDataMapping, "ci-runner", healthcare.HealthcareUserDataMappingConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
