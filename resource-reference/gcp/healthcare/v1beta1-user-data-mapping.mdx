`google_healthcare_v1beta1_user_data_mapping` links a specific piece of health data to the patient user it belongs to, within a Cloud Healthcare API consent store, so consent decisions can be evaluated per patient.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	healthcare "github.com/ubiquex/ubx-sdk-google/sdk/go/google/healthcare"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta1 user data mapping"})

    		ubx.Resource(healthcare.V1beta1UserDataMapping, "example", healthcare.V1beta1UserDataMappingConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
