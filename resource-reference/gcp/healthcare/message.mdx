## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	healthcare "github.com/ubiquex/ubx-sdk-google/sdk/go/google/healthcare"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own message"})

    		ubx.Resource(healthcare.Message, "example", healthcare.MessageConfig{
    			Message: map[string]any{"data": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.healthcare import Message, MessageConfig
    from ubx.google.healthcare.message import Message_Message

    def describe():
        ubx.intent("example own message")

        ubx.resource(Message, "example", MessageConfig(
            message=Message_Message(data="example"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
