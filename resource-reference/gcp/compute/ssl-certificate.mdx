## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	compute "github.com/ubiquex/ubx-sdk-google/sdk/go/google/compute"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own ssl certificate"})

    		ubx.Resource(compute.SslCertificate, "ci-runner", compute.SslCertificateConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google --lang ts --out ./local-sdk

    export default stack("payments", () => {

      resource(SslCertificate, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```

    ```python
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
