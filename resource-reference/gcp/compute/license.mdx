`google_compute_license` tracks the real licensing terms and restrictions attached to an image or disk -- the mechanism behind bring-your-own-license and marketplace image billing, not something a user typically creates directly.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	compute "github.com/ubiquex/ubx-sdk-google/sdk/go/google/compute"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own license"})

    		ubx.Resource(compute.License, "example", compute.LicenseConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
