`google_dns_v1beta2_policy` configures DNS behavior (custom forwarding targets, DNS logging) applied to VPC network(s) attached to it, for private and split-horizon DNS scenarios.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dns "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dns"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta2 policy"})

    		ubx.Resource(dns.V1beta2Policy, "example", dns.V1beta2PolicyConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
