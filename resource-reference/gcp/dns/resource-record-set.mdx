title: "google_dns_resource_record_set"

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dns "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dns"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resource record set"})

    		ubx.Resource(dns.ResourceRecordSet, "example", dns.ResourceRecordSetConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
