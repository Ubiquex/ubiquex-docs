`google_dns_v1beta2_managed_zone` is a Cloud DNS zone -- a container of DNS record(s) for a specific domain, either publicly resolvable or private to a set of VPC networks.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dns "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dns"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta2 managed zone"})

    		ubx.Resource(dns.V1beta2ManagedZone, "example", dns.V1beta2ManagedZoneConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
