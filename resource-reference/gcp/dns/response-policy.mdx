`google_dns_response_policy` overrides standard DNS resolution for VPC network(s) it's attached to, letting queries for specific names return caller-defined responses instead of the real internet answer.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dns --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dns/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dns "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dns/dns"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own response policy"})

    		ubx.Resource(dns.DnsResponsePolicy, "example", dns.DnsResponsePolicyConfig{})
    	}))
    }
    ```
