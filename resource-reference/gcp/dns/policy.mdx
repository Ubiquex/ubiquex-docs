`google_dns_policy` configures DNS behavior (custom forwarding targets, DNS logging) applied to VPC network(s) attached to it, for private and split-horizon DNS scenarios.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dns --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dns/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dns "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dns/dns"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own policy"})

    		ubx.Resource(dns.DnsPolicy, "ci-runner", dns.DnsPolicyConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_dns --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { DnsPolicy } from "./local-sdk/google_dns/sdk/typescript/google_dns/dns/dns_policy.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own policy" });

    ```

  <Tab title="Python">

    from ubx.google_dns.dns import DnsPolicy, DnsPolicyConfig

    if __name__ == "__main__":
        ubx.run("payments", describe)
