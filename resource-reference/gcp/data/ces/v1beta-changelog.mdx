## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	ces "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/ces"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_ces_v1beta_changelog"})

    		ubx.Data(ces.V1betaChangelog, "example", ces.V1betaChangelogConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    export default stack("example", () => {
      intent({ summary: "look up google_ces_v1beta_changelog" });

  <Tab title="Python">
    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.ces import V1betaChangelog, V1betaChangelogConfig

    def describe():
        ubx.intent("look up google_ces_v1beta_changelog")

        ubx.data(V1betaChangelog, "example", V1betaChangelogConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
