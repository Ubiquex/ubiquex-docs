## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	ces "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/ces"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_ces_v1beta_evaluation_run"})

    		ubx.Data(ces.V1betaEvaluationRun, "example", ces.V1betaEvaluationRunConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
