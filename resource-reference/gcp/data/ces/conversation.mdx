`google_ces_conversation` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Google Cloud's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	ces "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/ces"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_ces_conversation"})

    		ubx.Data(ces.Conversation, "example", ces.ConversationConfig{
    			Name: "example",
    			View: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

  </Tab>
  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.google.data.ces import Conversation, ConversationConfig

    def describe():
        ubx.intent("look up google_ces_conversation")

        ubx.data(Conversation, "example", ConversationConfig(
            name="example",
            view="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
