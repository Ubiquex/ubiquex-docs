## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	baremetalsolution "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/baremetalsolution"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_baremetalsolution_os_image"})

    		ubx.Data(baremetalsolution.OsImage, "example", baremetalsolution.OsImageConfig{
    			Name: "example",
    		})
    	}))
    }
    ```

    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.baremetalsolution import OsImage, OsImageConfig

    def describe():
        ubx.intent("look up google_baremetalsolution_os_image")

        ubx.data(OsImage, "example", OsImageConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
