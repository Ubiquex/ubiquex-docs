## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	baremetalsolution "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/baremetalsolution"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_baremetalsolution_volume"})

    		ubx.Data(baremetalsolution.Volume, "example", baremetalsolution.VolumeConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.baremetalsolution import Volume, VolumeConfig

    def describe():
        ubx.intent("look up google_baremetalsolution_volume")

        ubx.data(Volume, "example", VolumeConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="name" type="string" required>
  The real, fully-qualified resource name this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="attached" type="bool">
  Output only. Is the Volume attached at at least one instance. This field is a lightweight counterpart of `instances` field. It is filled in List responses as well. Returned by the lookup.
</ResponseField>

<ResponseField name="auto_grown_size_gib" type="string">
  The size, in GiB, that this storage volume has expanded as a result of an auto grow policy. In the absence of auto-grow, the value is 0. Returned by the lookup.
</ResponseField>

<ResponseField name="boot_volume" type="bool">
  Output only. Whether this volume is a boot volume. A boot volume is one which contains a boot LUN. Returned by the lookup.
</ResponseField>

<ResponseField name="current_size_gib" type="string">
  The current size of this storage volume, in GiB, including space reserved for snapshots. This size might be different than the requested size if the storage volume has been configured with auto grow or auto shrink. Returned by the lookup.
</ResponseField>

<ResponseField name="emergency_size_gib" type="string">
  Additional emergency size that was requested for this Volume, in GiB. current_size_gib includes this value. Returned by the lookup.
</ResponseField>

<ResponseField name="expire_time" type="string">
  Output only. Time after which volume will be fully deleted. It is filled only for volumes in COOLOFF state. Returned by the lookup.
</ResponseField>

<ResponseField name="id" type="string">
  An identifier for the `Volume`, generated by the backend. Returned by the lookup.
</ResponseField>

<ResponseField name="instances" type="list(string)">
  Output only. Instances this Volume is attached to. This field is set only in Get requests. Returned by the lookup.
</ResponseField>

<ResponseField name="labels" type="map(string)">
  Labels as key value pairs. Returned by the lookup.
</ResponseField>

<ResponseField name="max_size_gib" type="string">
  Maximum size volume can be expanded to in case of evergency, in GiB. Returned by the lookup.
</ResponseField>

<ResponseField name="notes" type="string">
  Input only. User-specified notes for new Volume. Used to provision Volumes that require manual intervention. Returned by the lookup.
</ResponseField>

<ResponseField name="originally_requested_size_gib" type="string">
  Originally requested size, in GiB. Returned by the lookup.
</ResponseField>

<ResponseField name="performance_tier" type="string">
  Immutable. Performance tier of the Volume. Default is SHARED. Returned by the lookup.
</ResponseField>

<ResponseField name="pod" type="string">
  Immutable. Pod name. Pod is an independent part of infrastructure. Volume can only be connected to the instances allocated in the same pod. Returned by the lookup.
</ResponseField>

<ResponseField name="protocol" type="string">
  Output only. Storage protocol for the Volume. Returned by the lookup.
</ResponseField>

<ResponseField name="remaining_space_gib" type="string">
  The space remaining in the storage volume for new LUNs, in GiB, excluding space reserved for snapshots. Returned by the lookup.
</ResponseField>

<ResponseField name="requested_size_gib" type="string">
  The requested size of this storage volume, in GiB. Returned by the lookup.
</ResponseField>

<ResponseField name="snapshot_auto_delete_behavior" type="string">
  The behavior to use when snapshot reserved space is full. Returned by the lookup.
</ResponseField>

<ResponseField name="snapshot_enabled" type="bool">
  Whether snapshots are enabled. Returned by the lookup.
</ResponseField>

<ResponseField name="snapshot_reservation_detail" type="object">
  Details about snapshot space reservation and usage on the storage volume. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="reserved_space_gib" type="string">
      The space on this storage volume reserved for snapshots, shown in GiB. Optional lookup argument.
    </ResponseField>
    <ResponseField name="reserved_space_percent" type="number">
      Percent of the total Volume size reserved for snapshot copies. Enabling snapshots requires reserving 20% or more of the storage volume space for snapshots. Maximum reserved space for snapshots is 40%. Setting this field will effectively set snapshot_enabled to true. Optional lookup argument.
    </ResponseField>
    <ResponseField name="reserved_space_remaining_gib" type="string">
      The amount, in GiB, of available space in this storage volume's reserved snapshot space. Optional lookup argument.
    </ResponseField>
    <ResponseField name="reserved_space_used_percent" type="number">
      The percent of snapshot space on this storage volume actually being used by the snapshot copies. This value might be higher than 100% if the snapshot copies have overflowed into the data portion of the storage volume. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="state" type="string">
  The state of this storage volume. Returned by the lookup.
</ResponseField>

<ResponseField name="storage_type" type="string">
  The storage type for this volume. Returned by the lookup.
</ResponseField>

<ResponseField name="workload_profile" type="string">
  The workload profile for the volume. Returned by the lookup.
</ResponseField>
