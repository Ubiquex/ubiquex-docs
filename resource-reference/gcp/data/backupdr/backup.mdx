## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	backupdr "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/backupdr"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_backupdr_backup"})

    		ubx.Data(backupdr.Backup, "example", backupdr.BackupConfig{
    			Name: "example",
    			View: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  </Tab>

    ```python
    import ubx_sdk as ubx

    def describe():
    ```
