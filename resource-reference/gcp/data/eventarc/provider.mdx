`google_eventarc_provider` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Google Cloud's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	eventarc "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/eventarc"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_eventarc_provider"})

    		ubx.Data(eventarc.Provider, "example", eventarc.ProviderConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Provider } from "@ubx/sdk-google/google/data/eventarc/provider";

    export default stack("example", () => {
      intent({ summary: "look up google_eventarc_provider" });

      data(Provider, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.eventarc import Provider, ProviderConfig

    def describe():
        ubx.intent("look up google_eventarc_provider")

        ubx.data(Provider, "example", ProviderConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
