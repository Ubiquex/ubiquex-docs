## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	run "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/run"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_run_revision"})

    		ubx.Data(run.Revision, "example", run.RevisionConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
