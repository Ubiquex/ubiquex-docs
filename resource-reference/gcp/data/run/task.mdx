## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	run "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/run"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_run_task"})

    		ubx.Data(run.Task, "example", run.TaskConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only google --lang ts --out ./local-sdk
    import { Task } from "./local-sdk/google/sdk/typescript/google/data/run/task.ts";

    export default stack("example", () => {
      intent({ summary: "look up google_run_task" });

      data(Task, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.run import Task, TaskConfig

    def describe():
        ubx.intent("look up google_run_task")

        ubx.data(Task, "example", TaskConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
