## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	agentregistry "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/agentregistry"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_agentregistry_mcp_server"})

    		ubx.Data(agentregistry.McpServer, "example", agentregistry.McpServerConfig{
    			Name: "example",
    		})
    	}))
    }
    ```

    from ubx.google.data.agentregistry import McpServer, McpServerConfig

    def describe():
        ubx.intent("look up google_agentregistry_mcp_server")

        ubx.data(McpServer, "example", McpServerConfig(

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
