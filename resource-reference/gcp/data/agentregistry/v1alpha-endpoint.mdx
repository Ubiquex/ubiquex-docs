## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	agentregistry "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/agentregistry"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_agentregistry_v1alpha_endpoint"})

    		ubx.Data(agentregistry.V1alphaEndpoint, "example", agentregistry.V1alphaEndpointConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
