## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	contentwarehouse "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/contentwarehouse"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_contentwarehouse_reference_id"})

    		ubx.Data(contentwarehouse.ReferenceId, "example", contentwarehouse.ReferenceIdConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
