## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	cloudidentity "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/cloudidentity"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_cloudidentity_userinvitation"})

    		ubx.Data(cloudidentity.Userinvitation, "example", cloudidentity.UserinvitationConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
