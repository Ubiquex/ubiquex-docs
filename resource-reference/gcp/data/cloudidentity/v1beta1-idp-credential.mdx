## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	cloudidentity "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/cloudidentity"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_cloudidentity_v1beta1_idp_credential"})

    		ubx.Data(cloudidentity.V1beta1IdpCredential, "example", cloudidentity.V1beta1IdpCredentialConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
