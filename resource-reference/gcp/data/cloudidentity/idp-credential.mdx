`google_cloudidentity_idp_credential` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Google Cloud's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	cloudidentity "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/cloudidentity"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_cloudidentity_idp_credential"})

    		ubx.Data(cloudidentity.IdpCredential, "example", cloudidentity.IdpCredentialConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { IdpCredential } from "@ubx/sdk-google/google/data/cloudidentity/idp_credential";

    export default stack("example", () => {
      intent({ summary: "look up google_cloudidentity_idp_credential" });

      data(IdpCredential, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.cloudidentity import IdpCredential, IdpCredentialConfig

    def describe():
        ubx.intent("look up google_cloudidentity_idp_credential")

        ubx.data(IdpCredential, "example", IdpCredentialConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
