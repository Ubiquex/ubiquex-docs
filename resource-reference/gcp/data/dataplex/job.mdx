## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataplex "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/dataplex"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_dataplex_job"})

    		ubx.Data(dataplex.Job, "example", dataplex.JobConfig{
    			Name: "example",
    			View: "example",
    		})
    	}))
    }
    ```
