## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	chromemanagement "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/chromemanagement"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_chromemanagement_profile"})

    		ubx.Data(chromemanagement.Profile, "example", chromemanagement.ProfileConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Profile } from "@ubx/sdk-google/google/data/chromemanagement/profile";

    export default stack("example", () => {
      intent({ summary: "look up google_chromemanagement_profile" });

      data(Profile, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.chromemanagement import Profile, ProfileConfig

    def describe():
        ubx.intent("look up google_chromemanagement_profile")

        ubx.data(Profile, "example", ProfileConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Result properties

<ResponseField name="affiliation_state" type="string">
  Output only. The specific affiliation state of the profile. Returned by the lookup.
</ResponseField>

<ResponseField name="annotated_location" type="string">
  Optional. Location of the profile annotated by the admin. Returned by the lookup.
</ResponseField>

<ResponseField name="annotated_user" type="string">
  Optional. User of the profile annotated by the admin. Returned by the lookup.
</ResponseField>

<ResponseField name="attestation_credential" type="object">
  Information of public key associated with a Chrome browser profile. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="key_rotation_time" type="string">
      Output only. Latest rotation timestamp of the public key rotation. Returned by the lookup.
    </ResponseField>
    <ResponseField name="key_trust_level" type="string">
      Output only. Trust level of the public key. Returned by the lookup.
    </ResponseField>
    <ResponseField name="key_type" type="string">
      Output only. Type of the public key. Returned by the lookup.
    </ResponseField>
    <ResponseField name="public_key" type="string">
      Output only. Value of the public key. Returned by the lookup.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="browser_channel" type="string">
  Output only. Channel of the browser on which the profile exists. Returned by the lookup.
</ResponseField>

<ResponseField name="browser_version" type="string">
  Output only. Version of the browser on which the profile exists. Returned by the lookup.
</ResponseField>

<ResponseField name="device_info" type="object">
  Information of a device that runs a Chrome browser profile. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="affiliated_device_id" type="string">
      Output only. Device ID that identifies the affiliated device on which the profile exists. If the device type is CHROME_BROWSER, then this represents a unique Directory API ID of the device that can be used in Admin SDK Browsers API. Returned by the lookup.
    </ResponseField>
    <ResponseField name="device_type" type="string">
      Output only. Type of the device on which the profile exists. Returned by the lookup.
    </ResponseField>
    <ResponseField name="hostname" type="string">
      Output only. Hostname of the device on which the profile exists. Returned by the lookup.
    </ResponseField>
    <ResponseField name="machine" type="string">
      Output only. Machine name of the device on which the profile exists. On platforms which do not report the machine name (currently iOS and Android) this is instead set to the browser's device_id - but note that this is a different device_id than the |affiliated_device_id|. Returned by the lookup.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="display_name" type="string">
  Output only. Profile display name set by client. Returned by the lookup.
</ResponseField>

<ResponseField name="etag" type="string">
  Output only. Etag of this ChromeBrowserProfile resource. This etag can be used with UPDATE operation to ensure consistency. Returned by the lookup.
</ResponseField>

<ResponseField name="extension_count" type="string">
  Output only. Number of extensions installed on the profile. Returned by the lookup.
</ResponseField>

<ResponseField name="first_enrollment_time" type="string">
  Output only. Timestamp of the first enrollment of the profile. Returned by the lookup.
</ResponseField>

<ResponseField name="identity_provider" type="string">
  Output only. Identify provider of the profile. Returned by the lookup.
</ResponseField>

<ResponseField name="last_activity_time" type="string">
  Output only. Timestamp of the latest activity by the profile. Returned by the lookup.
</ResponseField>

<ResponseField name="last_policy_fetch_time" type="string">
  Output only. Timestamp of the latest policy fetch by the profile. Returned by the lookup.
</ResponseField>

<ResponseField name="last_policy_sync_time" type="string">
  Output only. Timestamp of the latest policy sync by the profile. Returned by the lookup.
</ResponseField>

<ResponseField name="last_status_report_time" type="string">
  Output only. Timestamp of the latest status report by the profile. Returned by the lookup.
</ResponseField>

<ResponseField name="os_platform_type" type="string">
  Output only. OS platform of the device on which the profile exists. Returned by the lookup.
</ResponseField>

<ResponseField name="os_platform_version" type="string">
  Output only. Major OS platform version of the device on which the profile exists, from profile reporting. Returned by the lookup.
</ResponseField>

<ResponseField name="os_version" type="string">
  Output only. OS version of the device on which the profile exists. Returned by the lookup.
</ResponseField>

<ResponseField name="policy_count" type="string">
  Output only. Number of policies applied on the profile. Returned by the lookup.
</ResponseField>

<ResponseField name="profile_id" type="string">
  Output only. Chrome client side profile ID. Returned by the lookup.
</ResponseField>

<ResponseField name="profile_permanent_id" type="string">
  Output only. Profile permanent ID is the unique identifier of a profile within one customer. Returned by the lookup.
</ResponseField>

<ResponseField name="reporting_data" type="object">
  Reporting data of a Chrome browser profile. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="browser_executable_path" type="string">
      Output only. Executable path of the installed Chrome browser. A valid path is included only in affiliated profiles. Returned by the lookup.
    </ResponseField>
    <ResponseField name="extension_data" type="list(object)">
      Output only. Information of the extensions installed on the profile. Returned by the lookup.
    </ResponseField>
    <ResponseField name="extension_policy_data" type="list(object)">
      Output only. Information of the policies applied on the extensions. Returned by the lookup.
    </ResponseField>
    <ResponseField name="installed_browser_version" type="string">
      Output only. Updated version of a browser, if it is different from the active browser version. Returned by the lookup.
    </ResponseField>
    <ResponseField name="policy_data" type="list(object)">
      Output only. Information of the policies applied on the profile. Returned by the lookup.
    </ResponseField>
    <ResponseField name="profile_path" type="string">
      Output only. Path of the profile. A valid path is included only in affiliated profiles. Returned by the lookup.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="supports_fcm_notifications" type="bool">
  Output only. Whether the profile supports FCM notifications. Returned by the lookup.
</ResponseField>

<ResponseField name="user_email" type="string">
  Output only. Email address of the user to which the profile belongs. Returned by the lookup.
</ResponseField>

<ResponseField name="user_id" type="string">
  Output only. Unique Directory API ID of the user that can be used in Admin SDK Users API. Returned by the lookup.
</ResponseField>
