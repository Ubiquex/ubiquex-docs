## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	chromemanagement "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/chromemanagement"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_chromemanagement_device"})

    		ubx.Data(chromemanagement.Device, "example", chromemanagement.DeviceConfig{
    			Name:     "example",
    			ReadMask: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    import ubx_sdk as ubx
    from ubx.google.data.chromemanagement import Device, DeviceConfig

    def describe():

        ubx.data(Device, "example", DeviceConfig(

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
