## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	bigquerydatatransfer "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/bigquerydatatransfer"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_bigquerydatatransfer_transfer_resource"})

    		ubx.Data(bigquerydatatransfer.TransferResource, "example", bigquerydatatransfer.TransferResourceConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
