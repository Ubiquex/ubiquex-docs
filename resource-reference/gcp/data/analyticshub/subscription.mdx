## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	analyticshub "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/analyticshub"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_analyticshub_subscription"})

    		ubx.Data(analyticshub.Subscription, "example", analyticshub.SubscriptionConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    export default stack("example", () => {

  <Tab title="Python">
    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.analyticshub import Subscription, SubscriptionConfig

    def describe():
        ubx.intent("look up google_analyticshub_subscription")

        ubx.data(Subscription, "example", SubscriptionConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

  <Expandable title="properties">

      Cloud Marketplace commercial metadata for this subscription. Optional lookup argument.
      <Expandable title="properties">
        <ResponseField name="order" type="string">
          Resource name of the Marketplace Order. Optional lookup argument.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="creation_time" type="string">
  Output only. Timestamp when the subscription was created. Returned by the lookup.
</ResponseField>

<ResponseField name="data_exchange" type="string">
  Output only. Resource name of the source Data Exchange. e.g. projects/123/locations/us/dataExchanges/456 Returned by the lookup.
</ResponseField>

<ResponseField name="destination_dataset" type="object">
  Defines the destination bigquery dataset. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="dataset_reference" type="object">
      Optional lookup argument.
      <Expandable title="properties">
        <ResponseField name="dataset_id" type="string">
          Required. A unique ID for this dataset, without the project name. The ID must contain only letters (a-z, A-Z), numbers (0-9), or underscores (_). The maximum length is 1,024 characters. Optional lookup argument.
        </ResponseField>
        <ResponseField name="project_id" type="string">
          Required. The ID of the project containing this dataset. Optional lookup argument.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="description" type="string">
      Optional. A user-friendly description of the dataset. Optional lookup argument.
    </ResponseField>
    <ResponseField name="friendly_name" type="string">
      Optional. A descriptive name for the dataset. Optional lookup argument.
    </ResponseField>
    <ResponseField name="labels" type="map(string)">
      Optional. The labels associated with this dataset. You can use these to organize and group your datasets. You can set this property when inserting or updating a dataset. See https://cloud.google.com/resource-manager/docs/creating-managing-labels for more information. Optional lookup argument.
    </ResponseField>
    <ResponseField name="location" type="string">
      Required. The geographic location where the dataset should reside. See https://cloud.google.com/bigquery/docs/locations for supported locations. Optional lookup argument.
    </ResponseField>
    <ResponseField name="replica_locations" type="list(string)">
      Optional. The geographic locations where the dataset should be replicated. See [BigQuery locations](https://cloud.google.com/bigquery/docs/locations) for supported locations. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="last_modify_time" type="string">
  Output only. Timestamp when the subscription was last modified. Returned by the lookup.
</ResponseField>

<ResponseField name="linked_dataset_map" type="map(object)">
  Output only. Map of listing resource names to associated linked resource, e.g. projects/123/locations/us/dataExchanges/456/listings/789 -&gt; projects/123/datasets/my_dataset For listing-level subscriptions, this is a map of size 1. Only contains values if state == STATE_ACTIVE. Returned by the lookup.
</ResponseField>

<ResponseField name="linked_resources" type="list(object)">
  Output only. Linked resources created in the subscription. Only contains values if state = STATE_ACTIVE. Returned by the lookup.
</ResponseField>

<ResponseField name="listing" type="string">
  Output only. Resource name of the source Listing. e.g. projects/123/locations/us/dataExchanges/456/listings/789 Returned by the lookup.
</ResponseField>

<ResponseField name="log_linked_dataset_query_user_email" type="bool">
  Output only. By default, false. If true, the Subscriber agreed to the email sharing mandate that is enabled for DataExchange/Listing. Returned by the lookup.
</ResponseField>

<ResponseField name="organization_display_name" type="string">
  Output only. Display name of the project of this subscription. Returned by the lookup.
</ResponseField>

<ResponseField name="organization_id" type="string">
  Output only. Organization of the project this subscription belongs to. Returned by the lookup.
</ResponseField>

<ResponseField name="resource_type" type="string">
  Output only. Listing shared asset type. Returned by the lookup.
</ResponseField>

<ResponseField name="state" type="string">
  Output only. Current state of the subscription. Returned by the lookup.
</ResponseField>

<ResponseField name="subscriber_contact" type="string">
  Output only. Email of the subscriber. Returned by the lookup.
</ResponseField>
