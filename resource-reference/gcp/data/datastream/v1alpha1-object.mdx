## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	datastream "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/datastream"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_datastream_v1alpha1_object"})

    		ubx.Data(datastream.V1alpha1Object, "example", datastream.V1alpha1ObjectConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
