## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	aiplatform "github.com/ubiquex/ubx-sdk-google/sdk/go/google/aiplatform"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own rag file"})

    		ubx.Resource(aiplatform.RagFile, "example", aiplatform.RagFileConfig{
    			ImportRagFilesConfig: map[string]any{"gcs_source": map[string]any{"uris": []string{"example"}}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { RagFile } from "@ubx/sdk-google/google/aiplatform/rag_file";

    export default stack("example", () => {
      intent({ summary: "example own rag file" });

      resource(RagFile, "example", {
        importRagFilesConfig: { gcsSource: { uris: ["example"] } },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.aiplatform import RagFile, RagFileConfig
    from ubx.google.aiplatform.rag_file import RagFile_ImportRagFilesConfig, RagFile_GcsSource

    def describe():
        ubx.intent("example own rag file")

        ubx.resource(RagFile, "example", RagFileConfig(
            import_rag_files_config=RagFile_ImportRagFilesConfig(gcs_source=RagFile_GcsSource(uris=["example"])),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="name" type="string" required>
  The real, fully-qualified resource name this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="create_time" type="string">
  Output only. Timestamp when this RagFile was created. Returned by the lookup.
</ResponseField>

<ResponseField name="description" type="string">
  Optional. The description of the RagFile. Returned by the lookup.
</ResponseField>

<ResponseField name="direct_upload_source" type="any">
  The input content is encapsulated and uploaded in the request. Returned by the lookup.
</ResponseField>

<ResponseField name="display_name" type="string">
  Required. The display name of the RagFile. The name can be up to 128 characters long and can consist of any UTF-8 characters. Returned by the lookup.
</ResponseField>

<ResponseField name="file_status" type="object">
  RagFile status. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="error_status" type="string">
      Output only. Only when the `state` field is ERROR. Returned by the lookup.
    </ResponseField>
    <ResponseField name="state" type="string">
      Output only. RagFile state. Returned by the lookup.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="gcs_source" type="object">
  The Google Cloud Storage location for the input content. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="uris" type="list(string)">
      Required. Google Cloud Storage URI(-s) to the input file(s). May contain wildcards. For more information on wildcards, see https://cloud.google.com/storage/docs/wildcards. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="google_drive_source" type="object">
  The Google Drive location for the input content. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="resource_ids" type="list(object)">
      Required. Google Drive resource IDs. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="jira_source" type="object">
  The Jira source for the ImportRagFilesRequest. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="jira_queries" type="list(object)">
      Required. The Jira queries. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="share_point_sources" type="object">
  The SharePointSources to pass to ImportRagFiles. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="share_point_sources" type="list(object)">
      The SharePoint sources. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="slack_source" type="object">
  The Slack source for the ImportRagFilesRequest. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="channels" type="list(object)">
      Required. The Slack channels. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="update_time" type="string">
  Output only. Timestamp when this RagFile was last updated. Returned by the lookup.
</ResponseField>

<ResponseField name="user_metadata" type="string">
  Output only. The metadata for metadata search. The user_metadata Needs to be in JSON format. Returned by the lookup.
</ResponseField>
