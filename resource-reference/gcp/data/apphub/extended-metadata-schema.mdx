`google_apphub_extended_metadata_schema` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Google Cloud's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apphub "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/apphub"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_apphub_extended_metadata_schema"})

    		ubx.Data(apphub.ExtendedMetadataSchema, "example", apphub.ExtendedMetadataSchemaConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

    export default stack("example", () => {

      data(ExtendedMetadataSchema, "example", {

  </Tab>
  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.google.data.apphub import ExtendedMetadataSchema, ExtendedMetadataSchemaConfig

    def describe():

        ubx.data(ExtendedMetadataSchema, "example", ExtendedMetadataSchemaConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
