`google_apigee_data` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Google Cloud's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apigee "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/apigee"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_apigee_data"})

    		ubx.Data(apigee.Data, "example", apigee.DataConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Data } from "@ubx/sdk-google/google/data/apigee/data";

    export default stack("example", () => {
      intent({ summary: "look up google_apigee_data" });

      data(Data, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.apigee import Data, DataConfig

    def describe():
        ubx.intent("look up google_apigee_data")

        ubx.data(Data, "example", DataConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
