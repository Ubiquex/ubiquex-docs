`data_managedidentities_sql_integration` looks up a real integration linking a Managed Service for Microsoft Active Directory domain to a Cloud SQL instance.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	managedidentities "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/managedidentities"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_managedidentities_sql_integration"})

    		ubx.Data(managedidentities.SqlIntegration, "example", managedidentities.SqlIntegrationConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    if __name__ == "__main__":
        ubx.run("example", describe)
