## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	observability "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/observability"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_observability_dataset"})

    		ubx.Data(observability.Dataset, "example", observability.DatasetConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Dataset } from "@ubx/sdk-google/google/data/observability/dataset";

    export default stack("example", () => {
      intent({ summary: "look up google_observability_dataset" });

      data(Dataset, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.observability import Dataset, DatasetConfig

    def describe():
        ubx.intent("look up google_observability_dataset")

        ubx.data(Dataset, "example", DatasetConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
