---
title: "google_securitycenter_mute_config"
description: "`data_securitycenter_mute_config` looks up a real, already-existing `google_securitycenter_mute_config`."
---

It automatically mutes matching Security Command Center finding(s), suppressing them from active alerting while keeping them recorded.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	securitycenter "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/securitycenter"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_securitycenter_mute_config"})

    		ubx.Data(securitycenter.MuteConfig, "example", securitycenter.MuteConfigConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only google --lang ts --out ./local-sdk

    export default stack("example", () => {

    import ubx_sdk as ubx
    from ubx.google.data.securitycenter import MuteConfig, MuteConfigConfig

    def describe():
        ubx.intent("look up google_securitycenter_mute_config")

        ubx.data(MuteConfig, "example", MuteConfigConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
