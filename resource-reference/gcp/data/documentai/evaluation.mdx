## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	documentai "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/documentai"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_documentai_evaluation"})

    		ubx.Data(documentai.Evaluation, "example", documentai.EvaluationConfig{
    			Name: "example",
    		})
    	}))
    }
    ```

        ubx.intent("look up google_documentai_evaluation")

        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="name" type="string" required>
  The real, fully-qualified resource name this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="all_entities_metrics" type="object">
  Metrics across multiple confidence levels. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="auprc" type="number">
      The calculated area under the precision recall curve (AUPRC), computed by integrating over all confidence thresholds. Optional lookup argument.
    </ResponseField>
    <ResponseField name="auprc_exact" type="number">
      The AUPRC for metrics with fuzzy matching disabled, i.e., exact matching only. Optional lookup argument.
    </ResponseField>
    <ResponseField name="confidence_level_metrics" type="list(object)">
      Metrics across confidence levels with fuzzy matching enabled. Optional lookup argument.
    </ResponseField>
    <ResponseField name="confidence_level_metrics_exact" type="list(object)">
      Metrics across confidence levels with only exact matching. Optional lookup argument.
    </ResponseField>
    <ResponseField name="estimated_calibration_error" type="number">
      The Estimated Calibration Error (ECE) of the confidence of the predicted entities. Optional lookup argument.
    </ResponseField>
    <ResponseField name="estimated_calibration_error_exact" type="number">
      The ECE for the predicted entities with fuzzy matching disabled, i.e., exact matching only. Optional lookup argument.
    </ResponseField>
    <ResponseField name="metrics_type" type="string">
      The metrics type for the label. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="create_time" type="string">
  The time that the evaluation was created. Returned by the lookup.
</ResponseField>

<ResponseField name="document_counters" type="object">
  Evaluation counters for the documents that were used. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="evaluated_documents_count" type="number">
      How many documents were used in the evaluation. Optional lookup argument.
    </ResponseField>
    <ResponseField name="failed_documents_count" type="number">
      How many documents were not included in the evaluation as Document AI failed to process them. Optional lookup argument.
    </ResponseField>
    <ResponseField name="input_documents_count" type="number">
      How many documents were sent for evaluation. Optional lookup argument.
    </ResponseField>
    <ResponseField name="invalid_documents_count" type="number">
      How many documents were not included in the evaluation as they didn't pass validation. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="entity_metrics" type="map(object)">
  Metrics across confidence levels, for different entities. Returned by the lookup.
</ResponseField>

<ResponseField name="kms_key_name" type="string">
  The KMS key name used for encryption. Returned by the lookup.
</ResponseField>

<ResponseField name="kms_key_version_name" type="string">
  The KMS key version with which data is encrypted. Returned by the lookup.
</ResponseField>

<ResponseField name="revisions" type="list(object)">
  Contains all revisions of the evaluation, excluding the latest one. Returned by the lookup.
</ResponseField>
