## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	vmwareengine "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/vmwareengine"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_vmwareengine_upgrade"})

    		ubx.Data(vmwareengine.Upgrade, "example", vmwareengine.UpgradeConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
