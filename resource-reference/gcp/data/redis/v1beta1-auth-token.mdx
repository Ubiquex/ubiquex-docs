## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	redis "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/redis"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_redis_v1beta1_auth_token"})

    		ubx.Data(redis.V1beta1AuthToken, "example", redis.V1beta1AuthTokenConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
