`google_storage_service_account` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Google Cloud's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	storage "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/storage"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_storage_service_account"})

    		ubx.Data(storage.ServiceAccount, "example", storage.ServiceAccountConfig{
    			ProjectId:   "example",
    			UserProject: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript

  </Tab>
  <Tab title="Python">

    import ubx_sdk as ubx

    def describe():
        ubx.intent("look up google_storage_service_account")

        ubx.data(ServiceAccount, "example", ServiceAccountConfig(
            project_id="example",
            user_project="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
