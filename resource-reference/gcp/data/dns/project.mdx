`google_dns_project` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Google Cloud's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dns "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/dns"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_dns_project"})

    		ubx.Data(dns.Project, "example", dns.ProjectConfig{
    			Project:           "example",
    			ClientOperationId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Project } from "@ubx/sdk-google/google/data/dns/project";

    export default stack("example", () => {
      intent({ summary: "look up google_dns_project" });

      data(Project, "example", {
        project: "example",
        clientOperationId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.dns import Project, ProjectConfig

    def describe():
        ubx.intent("look up google_dns_project")

        ubx.data(Project, "example", ProjectConfig(
            project="example",
            client_operation_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
