## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	workloadmanager "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/workloadmanager"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_workloadmanager_discoveredprofile"})

    		ubx.Data(workloadmanager.Discoveredprofile, "example", workloadmanager.DiscoveredprofileConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only google --lang ts --out ./local-sdk
    import { Discoveredprofile } from "./local-sdk/google/sdk/typescript/google/data/workloadmanager/discoveredprofile.ts";

    export default stack("example", () => {
      intent({ summary: "look up google_workloadmanager_discoveredprofile" });

      data(Discoveredprofile, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.workloadmanager import Discoveredprofile, DiscoveredprofileConfig

    def describe():
        ubx.intent("look up google_workloadmanager_discoveredprofile")

        ubx.data(Discoveredprofile, "example", DiscoveredprofileConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
