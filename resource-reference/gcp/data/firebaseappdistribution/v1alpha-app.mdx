## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	firebaseappdistribution "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/firebaseappdistribution"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_firebaseappdistribution_v1alpha_app"})

    		ubx.Data(firebaseappdistribution.V1alphaApp, "example", firebaseappdistribution.V1alphaAppConfig{
    			MobilesdkAppId: "example",
    			AppView:        "example",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.google.data.firebaseappdistribution import V1alphaApp, V1alphaAppConfig

    def describe():
        ubx.intent("look up google_firebaseappdistribution_v1alpha_app")

        ubx.data(V1alphaApp, "example", V1alphaAppConfig(
            mobilesdk_app_id="example",
            app_view="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="app_view" type="string">
  The real, requested view (which fields to include) of this app response. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="mobilesdk_app_id" type="string" required>
  The real, unique identifier of the Firebase mobile SDK app this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="aab_certificate" type="object">
  App bundle test certificate Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="certificate_hash_md5" type="string">
      MD5 hash of the certificate used to resign the AAB Optional lookup argument.
    </ResponseField>
    <ResponseField name="certificate_hash_sha1" type="string">
      SHA1 hash of the certificate used to resign the AAB Optional lookup argument.
    </ResponseField>
    <ResponseField name="certificate_hash_sha256" type="string">
      SHA256 hash of the certificate used to resign the AAB Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="aab_state" type="string">
  App bundle state. Only valid for android apps. The app_view field in the request must be set to FULL in order for this to be populated. Returned by the lookup.
</ResponseField>

<ResponseField name="app_id" type="string">
  Firebase gmp app id Returned by the lookup.
</ResponseField>

<ResponseField name="bundle_id" type="string">
  Bundle identifier Returned by the lookup.
</ResponseField>

<ResponseField name="contact_email" type="string">
  Developer contact email for testers to reach out to about privacy or support issues. Returned by the lookup.
</ResponseField>

<ResponseField name="platform" type="string">
  iOS or Android Returned by the lookup.
</ResponseField>

<ResponseField name="project_number" type="string">
  Project number of the Firebase project, for example 300830567303. Returned by the lookup.
</ResponseField>
