## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	connectors "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/connectors"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_connectors_eventtype"})

    		ubx.Data(connectors.Eventtype, "example", connectors.EventtypeConfig{
    			Name: "example",
    		})
    	}))
    }
    ```

    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.connectors import Eventtype, EventtypeConfig

    def describe():
        ubx.intent("look up google_connectors_eventtype")

        ubx.data(Eventtype, "example", EventtypeConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Result properties

<ResponseField name="create_time" type="string">
  Output only. Created time. Returned by the lookup.
</ResponseField>

<ResponseField name="enriched_event_payload_schema" type="string">
  Output only. Schema of the event payload after enriched. Will be null if read before send is not supported. Returned by the lookup.
</ResponseField>

<ResponseField name="entity_type" type="string">
  Output only. Runtime entity type name. Will be null if entity type map is not available. Used for read before send feature. Returned by the lookup.
</ResponseField>

<ResponseField name="event_payload_schema" type="string">
  Output only. Schema of webhook event payload. Returned by the lookup.
</ResponseField>

<ResponseField name="event_type_id" type="string">
  Output only. Event type id. Example: `ticket.created`. Returned by the lookup.
</ResponseField>

<ResponseField name="id_path" type="string">
  Output only. Id path denotes the path of id in webhook payload. Returned by the lookup.
</ResponseField>

<ResponseField name="update_time" type="string">
  Output only. Updated time. Returned by the lookup.
</ResponseField>
