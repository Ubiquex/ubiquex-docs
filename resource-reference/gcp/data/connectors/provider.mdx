title: "google_connectors_provider"
description: "`data_connectors_provider` looks up a real connector provider (such as Salesforce or ServiceNow) supported by Application Integration Connectors."
---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	connectors "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/connectors"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_connectors_provider"})

    		ubx.Data(connectors.Provider, "example", connectors.ProviderConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Provider } from "@ubx/sdk-google/google/data/connectors/provider";

    export default stack("example", () => {
      intent({ summary: "look up google_connectors_provider" });

      data(Provider, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.connectors import Provider, ProviderConfig

    def describe():
        ubx.intent("look up google_connectors_provider")

        ubx.data(Provider, "example", ProviderConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
