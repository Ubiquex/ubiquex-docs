---
title: "google_connectors_version"
description: "`data_connectors_version` looks up a real, available version of a connector supported by Application Integration Connectors."
---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	connectors "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/connectors"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_connectors_version"})

    		ubx.Data(connectors.Version, "example", connectors.VersionConfig{
    			Name: "example",
    			View: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Version } from "@ubx/sdk-google/google/data/connectors/version";

    export default stack("example", () => {
      intent({ summary: "look up google_connectors_version" });

      data(Version, "example", {
        name: "example",
        view: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.connectors import Version, VersionConfig

    def describe():
        ubx.intent("look up google_connectors_version")

        ubx.data(Version, "example", VersionConfig(

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
