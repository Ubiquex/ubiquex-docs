`data_connectors_connector` looks up a real connection instance created via Application Integration Connectors, linking Google Cloud to a third-party or on-premises system.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	connectors "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/connectors"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_connectors_connector"})

    		ubx.Data(connectors.Connector, "example", connectors.ConnectorConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Connector } from "@ubx/sdk-google/google/data/connectors/connector";

    export default stack("example", () => {
      intent({ summary: "look up google_connectors_connector" });

      data(Connector, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.connectors import Connector, ConnectorConfig

    def describe():
        ubx.intent("look up google_connectors_connector")

        ubx.data(Connector, "example", ConnectorConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
