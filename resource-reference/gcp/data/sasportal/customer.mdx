## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	sasportal "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/sasportal"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_sasportal_customer"})

    		ubx.Data(sasportal.Customer, "example", sasportal.CustomerConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.google.data.sasportal import Customer, CustomerConfig

    def describe():
        ubx.intent("look up google_sasportal_customer")

        ubx.data(Customer, "example", CustomerConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
