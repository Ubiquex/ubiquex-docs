## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apim "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/apim"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_apim_api_observation"})

    		ubx.Data(apim.ApiObservation, "example", apim.ApiObservationConfig{
    			Name: "example",
    		})
    	}))
    }
    ```

    from ubx.google.data.apim import ApiObservation, ApiObservationConfig

        ubx.intent("look up google_apim_api_observation")
