## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dlp "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/dlp"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_dlp_project_data_profile"})

    		ubx.Data(dlp.ProjectDataProfile, "example", dlp.ProjectDataProfileConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
