## Example

  <Tab title="Go">

    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	managedkafka "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/managedkafka"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_managedkafka_context"})

    		ubx.Data(managedkafka.Context, "example", managedkafka.ContextConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
