---
title: "google_managedkafka_schema"
description: "`data_managedkafka_schema` looks up a real, already-existing `google_managedkafka_schema_registry`."
---

It is a managed Kafka schema registry, storing and versioning the message schema(s) producers and consumers on a Managed Kafka cluster validate against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	managedkafka "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/managedkafka"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_managedkafka_schema"})

    		ubx.Data(managedkafka.Schema, "example", managedkafka.SchemaConfig{
    			Name:    "example",
    			Subject: "example",
    		})
    	}))
    }
    ```

    from ubx.google.data.managedkafka import Schema, SchemaConfig

    def describe():
        ubx.intent("look up google_managedkafka_schema")

        ubx.data(Schema, "example", SchemaConfig(
            name="example",
            subject="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
