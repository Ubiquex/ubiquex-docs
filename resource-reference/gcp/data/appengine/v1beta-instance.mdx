`google_appengine_v1beta_instance` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Google Cloud's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_appengine_v1beta_instance"})

    		ubx.Data(appengine.V1betaInstance, "example", appengine.V1betaInstanceConfig{
    			AppsId:      "example",

    			VersionsId:  "example",

    	}))
    }
