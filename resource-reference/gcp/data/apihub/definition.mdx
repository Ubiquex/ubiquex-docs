`google_apihub_definition` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Google Cloud's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apihub "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/apihub"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_apihub_definition"})

    		ubx.Data(apihub.Definition, "example", apihub.DefinitionConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Definition } from "@ubx/sdk-google/google/data/apihub/definition";

    export default stack("example", () => {
      intent({ summary: "look up google_apihub_definition" });

      data(Definition, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.apihub import Definition, DefinitionConfig

    def describe():
        ubx.intent("look up google_apihub_definition")

        ubx.data(Definition, "example", DefinitionConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
