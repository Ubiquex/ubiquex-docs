`data_networkconnectivity_multicloud_data_transfer_supported_service` looks up a real destination service eligible for Cross-Cloud Network multicloud data transfer discounted egress.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkconnectivity "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/networkconnectivity"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_networkconnectivity_multicloud_data_transfer_supported_service"})

    		ubx.Data(networkconnectivity.MulticloudDataTransferSupportedService, "example", networkconnectivity.MulticloudDataTransferSupportedServiceConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
