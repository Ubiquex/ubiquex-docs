## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkconnectivity "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/networkconnectivity"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_networkconnectivity_remote_transport_profile"})

    		ubx.Data(networkconnectivity.RemoteTransportProfile, "example", networkconnectivity.RemoteTransportProfileConfig{
    			Name: "example",
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.google.data.networkconnectivity import RemoteTransportProfile, RemoteTransportProfileConfig

    def describe():
        ubx.intent("look up google_networkconnectivity_remote_transport_profile")

        ubx.data(RemoteTransportProfile, "example", RemoteTransportProfileConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="name" type="string" required>
  The real, fully-qualified resource name this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="description" type="string">
  Output only. Description of the profile. Returned by the lookup.
</ResponseField>

<ResponseField name="display_name" type="string">
  Output only. Human readable name of this profile, used to identify this profile in the UI. Returned by the lookup.
</ResponseField>

<ResponseField name="flow" type="string">
  Output only. Type of provisioning flows supported by this profile. Returned by the lookup.
</ResponseField>

<ResponseField name="labels" type="map(string)">
  Output only. Labels as key value pairs. Returned by the lookup.
</ResponseField>

<ResponseField name="order_state" type="string">
  Output only. Order state for this profile. Returned by the lookup.
</ResponseField>

<ResponseField name="provider" type="string">
  Output only. Name of the provider on the other end of this profile. E.g. “Amazon Web Services” or “Microsoft Azure”. Returned by the lookup.
</ResponseField>

<ResponseField name="provider_site" type="string">
  Output only. If the profile is a Cloud Service Provider with compute resources, this is populated with the region where connectivity is being established. If the profile provides facility-level selection, this is an identity of the facility any connections on this profile are going through. Returned by the lookup.
</ResponseField>

<ResponseField name="sla" type="string">
  Output only. Availability class that will be configured for this particular RemoteTransportProfile. Returned by the lookup.
</ResponseField>

<ResponseField name="supported_bandwidths" type="list(string)">
  Output only. List of bandwidth enum values that are supported by this profile. Returned by the lookup.
</ResponseField>
