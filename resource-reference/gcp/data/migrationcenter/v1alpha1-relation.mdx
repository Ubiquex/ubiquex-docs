## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	migrationcenter "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/migrationcenter"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_migrationcenter_v1alpha1_relation"})

    		ubx.Data(migrationcenter.V1alpha1Relation, "example", migrationcenter.V1alpha1RelationConfig{
    			Name: "example",
    		})
    	}))
    }
    ```

</ResponseField>

<ResponseField name="src_asset" type="string">

</ResponseField>
