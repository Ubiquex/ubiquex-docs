## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	containeranalysis "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/containeranalysis"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_containeranalysis_v1alpha1_scan_config"})

    		ubx.Data(containeranalysis.V1alpha1ScanConfig, "example", containeranalysis.V1alpha1ScanConfigConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
