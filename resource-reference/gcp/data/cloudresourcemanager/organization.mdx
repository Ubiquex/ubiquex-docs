    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_cloudresourcemanager_organization"})

    			Name: "example",
    		})
    	}))
    }

## Result properties

<ResponseField name="create_time" type="string">
  Output only. Timestamp when the Organization was created. Returned by the lookup.
</ResponseField>

<ResponseField name="delete_time" type="string">
  Output only. Timestamp when the Organization was requested for deletion. Returned by the lookup.
</ResponseField>

<ResponseField name="directory_customer_id" type="string">
  Immutable. The G Suite / Workspace customer id used in the Directory API. Returned by the lookup.
</ResponseField>

<ResponseField name="display_name" type="string">
  Output only. A human-readable string that refers to the organization in the Google Cloud Console. This string is set by the server and cannot be changed. The string will be set to the primary domain (for example, "google.com") of the Google Workspace customer that owns the organization. Returned by the lookup.
</ResponseField>

<ResponseField name="etag" type="string">
  Output only. A checksum computed by the server based on the current value of the Organization resource. This may be sent on update and delete requests to ensure the client has an up-to-date value before proceeding. Returned by the lookup.
</ResponseField>

<ResponseField name="state" type="string">
  Output only. The organization's current lifecycle state. Returned by the lookup.
</ResponseField>

<ResponseField name="update_time" type="string">
  Output only. Timestamp when the Organization was last modified. Returned by the lookup.
</ResponseField>
