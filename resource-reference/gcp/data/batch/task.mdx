## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	batch "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/batch"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_batch_task"})

    		ubx.Data(batch.Task, "example", batch.TaskConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only google --lang ts --out ./local-sdk
    import { Task } from "./local-sdk/google/sdk/typescript/google/data/batch/task.ts";

    export default stack("example", () => {
      intent({ summary: "look up google_batch_task" });

      data(Task, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.batch import Task, TaskConfig

    def describe():
        ubx.intent("look up google_batch_task")

        ubx.data(Task, "example", TaskConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
