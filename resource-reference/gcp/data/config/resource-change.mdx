`google_config_resource_change` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Google Cloud's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	config "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/config"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_config_resource_change"})

    		ubx.Data(config.ResourceChange, "example", config.ResourceChangeConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { ResourceChange } from "@ubx/sdk-google/google/data/config/resource_change";

    export default stack("example", () => {
      intent({ summary: "look up google_config_resource_change" });

      data(ResourceChange, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.config import ResourceChange, ResourceChangeConfig

    def describe():
        ubx.intent("look up google_config_resource_change")

        ubx.data(ResourceChange, "example", ResourceChangeConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
