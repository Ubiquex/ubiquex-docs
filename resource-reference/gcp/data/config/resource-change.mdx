## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	config "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/config"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_config_resource_change"})

    		ubx.Data(config.ResourceChange, "example", config.ResourceChangeConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only google --lang ts --out ./local-sdk
    import { ResourceChange } from "./local-sdk/google/sdk/typescript/google/data/config/resource_change.ts";

    export default stack("example", () => {
      intent({ summary: "look up google_config_resource_change" });

      data(ResourceChange, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.config import ResourceChange, ResourceChangeConfig

    def describe():
        ubx.intent("look up google_config_resource_change")

        ubx.data(ResourceChange, "example", ResourceChangeConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="name" type="string" required>
  The real, fully-qualified resource name this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
