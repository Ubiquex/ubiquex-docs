## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkmanagement "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/networkmanagement"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_networkmanagement_monitoring_point"})

    		ubx.Data(networkmanagement.MonitoringPoint, "example", networkmanagement.MonitoringPointConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```typescript
    export default stack("example", () => {
    ```

    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.networkmanagement import MonitoringPoint, MonitoringPointConfig

    def describe():
        ubx.intent("look up google_networkmanagement_monitoring_point")

        ubx.data(MonitoringPoint, "example", MonitoringPointConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
