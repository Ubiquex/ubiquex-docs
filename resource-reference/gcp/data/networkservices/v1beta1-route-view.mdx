## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networkservices "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/networkservices"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_networkservices_v1beta1_route_view"})

    		ubx.Data(networkservices.V1beta1RouteView, "example", networkservices.V1beta1RouteViewConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
