`google_artifactregistry_python_package` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Google Cloud's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	artifactregistry "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/artifactregistry"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_artifactregistry_python_package"})

    		ubx.Data(artifactregistry.PythonPackage, "example", artifactregistry.PythonPackageConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.artifactregistry import PythonPackage, PythonPackageConfig

    def describe():
        ubx.intent("look up google_artifactregistry_python_package")

        ubx.data(PythonPackage, "example", PythonPackageConfig(

        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
