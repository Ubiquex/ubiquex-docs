## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	artifactregistry "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/artifactregistry"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_artifactregistry_package"})

    		ubx.Data(artifactregistry.Package, "example", artifactregistry.PackageConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only google --lang ts --out ./local-sdk
    import { Package } from "./local-sdk/google/sdk/typescript/google/data/artifactregistry/package.ts";

    export default stack("example", () => {
      intent({ summary: "look up google_artifactregistry_package" });

      data(Package, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.artifactregistry import Package, PackageConfig

    def describe():
        ubx.intent("look up google_artifactregistry_package")

        ubx.data(Package, "example", PackageConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Result properties

<ResponseField name="annotations" type="map(string)">
  Optional. Client specified annotations. Returned by the lookup.
</ResponseField>

<ResponseField name="create_time" type="string">
  The time when the package was created. Returned by the lookup.
</ResponseField>

<ResponseField name="display_name" type="string">
  The display name of the package. Returned by the lookup.
</ResponseField>

<ResponseField name="update_time" type="string">
  The time when the package was last updated. This includes publishing a new version of the package. Returned by the lookup.
</ResponseField>
