`data_androidmanagement_policy` looks up a real device policy (app installs, restrictions, compliance rules) enforced on devices under Android Management API.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	androidmanagement "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/androidmanagement"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_androidmanagement_policy"})

    		ubx.Data(androidmanagement.Policy, "example", androidmanagement.PolicyConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only google --lang ts --out ./local-sdk
    import { Policy } from "./local-sdk/google/sdk/typescript/google/data/androidmanagement/policy.ts";

    export default stack("example", () => {
      intent({ summary: "look up google_androidmanagement_policy" });

      data(Policy, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.androidmanagement import Policy, PolicyConfig

    def describe():
        ubx.intent("look up google_androidmanagement_policy")

        ubx.data(Policy, "example", PolicyConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
