## Example

    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	compute "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/compute"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_compute_beta_project_view"})

    		ubx.Data(compute.BetaProjectView, "example", compute.BetaProjectViewConfig{
    			Project: "example",
    			Region:  "example",
    		})
    	}))
    }
