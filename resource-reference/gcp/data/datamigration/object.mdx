---
title: "google_datamigration_object"
description: "`data_datamigration_object` looks up a real database object (such as a table or view) tracked by Database Migration Service during a migration."
---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	datamigration "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/datamigration"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_datamigration_object"})

    		ubx.Data(datamigration.Object, "example", datamigration.ObjectConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Object } from "@ubx/sdk-google/google/data/datamigration/object";

    export default stack("example", () => {
      intent({ summary: "look up google_datamigration_object" });

      data(Object, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.data.datamigration import Object, ObjectConfig

    def describe():
        ubx.intent("look up google_datamigration_object")

        ubx.data(Object, "example", ObjectConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
