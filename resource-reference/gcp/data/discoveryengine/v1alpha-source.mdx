## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	discoveryengine "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/discoveryengine"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_discoveryengine_v1alpha_source"})

    		ubx.Data(discoveryengine.V1alphaSource, "example", discoveryengine.V1alphaSourceConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>

## Lookup arguments

  The real, fully-qualified resource name this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="metadata" type="object">
  Represents the metadata of a source and some additional information. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="agentspace_metadata" type="object">
      Metadata about an agentspace source. Optional lookup argument.
      <Expandable title="properties">
        <ResponseField name="document_name" type="string">
          Output only. The full document name in Agentspace. Returned by the lookup.
        </ResponseField>
        <ResponseField name="document_title" type="string">
          Output only. The title of the document. Returned by the lookup.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="google_docs_metadata" type="object">
      Metadata about a google doc source. Optional lookup argument.
      <Expandable title="properties">
        <ResponseField name="document_id" type="string">
          Output only. The document id of the google doc. Returned by the lookup.
        </ResponseField>
        <ResponseField name="revision_id" type="string">
          Output only. Revision id for the doc. Returned by the lookup.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="source_added_timestamp" type="string">
      The timestamp the source was added. Optional lookup argument.
    </ResponseField>
    <ResponseField name="token_count" type="number">
      The number of tokens in the source. Optional lookup argument.
    </ResponseField>
    <ResponseField name="word_count" type="number">
      The word count of the source. Optional lookup argument.
    </ResponseField>
    <ResponseField name="youtube_metadata" type="object">
      Metadata about a youtube video source. Optional lookup argument.
      <Expandable title="properties">
        <ResponseField name="channel_name" type="string">
          Output only. The channel name of the youtube video. Returned by the lookup.
        </ResponseField>
        <ResponseField name="video_id" type="string">
          Output only. The id of the youtube video. Returned by the lookup.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="settings" type="object">
  Allows extension of Source Settings in the BatchCreateSources (Formerly AddSource request). Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="failure_reason" type="object">
      Failure reason containing details about why a source failed to ingest. Optional lookup argument.
      <Expandable title="properties">
        <ResponseField name="audio_transcription_error" type="object">
          An audio file transcription specific error. Optional lookup argument.
          <Expandable title="properties">
            <ResponseField name="language_detection_failed" type="any">
              Could not detect language of the file (it may not be speech). Optional lookup argument.
            </ResponseField>
            <ResponseField name="no_audio_detected" type="any">
              No audio was detected in the input file. Optional lookup argument.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="domain_blocked" type="any">
          Error to indicate that the source was removed because the domain was blocked. Optional lookup argument.
        </ResponseField>
        <ResponseField name="google_drive_error" type="object">
          A google drive specific error. Optional lookup argument.
          <Expandable title="properties">
            <ResponseField name="download_prevented" type="any">
              The user was prevented from downloading the file. Optional lookup argument.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="ingestion_error" type="any">
          Indicates an error occurred while ingesting the source. Optional lookup argument.
        </ResponseField>
        <ResponseField name="mime_type_blocked" type="any">
          Indicates that the source MIME type is blocked. Optional lookup argument.
        </ResponseField>
        <ResponseField name="paywall_error" type="any">
          Indicates that the source is paywalled and cannot be ingested. Optional lookup argument.
        </ResponseField>
        <ResponseField name="policy_check_failed" type="any">
          Indicates that the policy check failed. Optional lookup argument.
        </ResponseField>
        <ResponseField name="source_empty" type="any">
          Indicates that the source is empty. Optional lookup argument.
        </ResponseField>
        <ResponseField name="source_limit_exceeded" type="any">
          Indicates that the user does not have space for this source. Optional lookup argument.
        </ResponseField>
        <ResponseField name="source_too_long" type="object">
          Indicates source word count exceeded the user's limit. Optional lookup argument.
          <Expandable title="properties">
            <ResponseField name="word_count" type="number">
              The number of words in the source. Optional lookup argument.
            </ResponseField>
            <ResponseField name="word_limit" type="number">
              The word count limit for the current user at the time of the upload. Optional lookup argument.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="source_unreachable" type="object">
          Indicates that the source is unreachable. This is primarily used for sources that are added via URL. Optional lookup argument.
          <Expandable title="properties">
            <ResponseField name="error_details" type="string">
              Describes why the source is unreachable. Optional lookup argument.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="unknown" type="any">
          Indicates an unknown error occurred. Optional lookup argument.
        </ResponseField>
        <ResponseField name="upload_error" type="any">
          Indicates an error occurred while uploading the source. Optional lookup argument.
        </ResponseField>
        <ResponseField name="youtube_error" type="object">
          A youtube specific error. Optional lookup argument.
          <Expandable title="properties">
            <ResponseField name="video_deleted" type="any">
              Error to indicate that the source was removed because the video was deleted. Optional lookup argument.
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="status" type="string">
      Status of the source. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>
