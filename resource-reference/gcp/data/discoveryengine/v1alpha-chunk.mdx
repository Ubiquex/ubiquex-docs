## Example

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_discoveryengine_v1alpha_chunk"})

    		ubx.Data(discoveryengine.V1alphaChunk, "example", discoveryengine.V1alphaChunkConfig{
    			Name: "example",
    		})
    	}))
    }

## Lookup arguments

<ResponseField name="name" type="string" required>
  The real, fully-qualified resource name this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="annotation_contents" type="list(string)">
  Output only. Annotation contents if the current chunk contains annotations. Returned by the lookup.
</ResponseField>

<ResponseField name="annotation_metadata" type="list(object)">
  Output only. The annotation metadata includes structured content in the current chunk. Returned by the lookup.
</ResponseField>

<ResponseField name="chunk_metadata" type="object">
  Metadata of the current chunk. This field is only populated on SearchService.Search API. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="next_chunks" type="list(any)">
      The next chunks of the current chunk. The number is controlled by SearchRequest.ContentSearchSpec.ChunkSpec.num_next_chunks. This field is only populated on SearchService.Search API. Optional lookup argument.
    </ResponseField>
    <ResponseField name="previous_chunks" type="list(any)">
      The previous chunks of the current chunk. The number is controlled by SearchRequest.ContentSearchSpec.ChunkSpec.num_previous_chunks. This field is only populated on SearchService.Search API. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="content" type="string">
  Content is a string from a document (parsed content). Returned by the lookup.
</ResponseField>

<ResponseField name="data_urls" type="list(string)">
  Output only. Image Data URLs if the current chunk contains images. Data URLs are composed of four parts: a prefix (data:), a MIME type indicating the type of data, an optional base64 token if non-textual, and the data itself: data:, Returned by the lookup.
</ResponseField>

<ResponseField name="derived_struct_data" type="map(any)">
  Output only. This field is OUTPUT_ONLY. It contains derived data that are not in the original input document. Returned by the lookup.
</ResponseField>

<ResponseField name="document_metadata" type="object">
  Document metadata contains the information of the document of the current chunk. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="mime_type" type="string">
      The mime type of the document. https://www.iana.org/assignments/media-types/media-types.xhtml. Optional lookup argument.
    </ResponseField>
    <ResponseField name="struct_data" type="map(any)">
      Data representation. The structured JSON data for the document. It should conform to the registered Schema or an `INVALID_ARGUMENT` error is thrown. Optional lookup argument.
    </ResponseField>
    <ResponseField name="title" type="string">
      Title of the document. Optional lookup argument.
    </ResponseField>
    <ResponseField name="uri" type="string">
      Uri of the document. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="id" type="string">
  Unique chunk ID of the current chunk. Returned by the lookup.
</ResponseField>

<ResponseField name="page_span" type="object">
  Page span of the chunk. Returned by the lookup.
  <Expandable title="properties">
    <ResponseField name="page_end" type="number">
      The end page of the chunk. Optional lookup argument.
    </ResponseField>
    <ResponseField name="page_start" type="number">
      The start page of the chunk. Optional lookup argument.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="relevance_score" type="number">
  Output only. Represents the relevance score based on similarity. Higher score indicates higher chunk relevance. The score is in range [-1.0, 1.0]. Only populated on SearchResponse. Returned by the lookup.
</ResponseField>
