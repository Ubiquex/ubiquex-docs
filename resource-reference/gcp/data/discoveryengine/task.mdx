## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	discoveryengine "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/discoveryengine"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_discoveryengine_task"})

    		ubx.Data(discoveryengine.Task, "example", discoveryengine.TaskConfig{
    			Name:          "example",
    			Tenant:        "example",
    			HistoryLength: 1,
    		})
    	}))
    }
    ```

    ```python
    # ubx sdk gen --only google --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google.data.discoveryengine import Task, TaskConfig

    def describe():
        ubx.intent("look up google_discoveryengine_task")

        ubx.data(Task, "example", TaskConfig(
            name="example",
            tenant="example",
            history_length=1,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="history_length" type="number">
  The real number of historical entries included in this result. **(AI-inferred)** Optional lookup argument.
</ResponseField>

<ResponseField name="name" type="string" required>
  The real, fully-qualified resource name this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="tenant" type="string" required>
  The real tenant this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
