## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_dialogflow_v3beta1_conversation"})

    		ubx.Data(dialogflow.V3beta1Conversation, "example", dialogflow.V3beta1ConversationConfig{
    			Name: "example",
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.google.data.dialogflow import V3beta1Conversation, V3beta1ConversationConfig

    def describe():
        ubx.intent("look up google_dialogflow_v3beta1_conversation")

        ubx.data(V3beta1Conversation, "example", V3beta1ConversationConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
