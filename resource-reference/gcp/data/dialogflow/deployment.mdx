## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google/data/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up google_dialogflow_deployment"})

    		ubx.Data(dialogflow.Deployment, "example", dialogflow.DeploymentConfig{
    			Name: "example",
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
