## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_alloydb --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_alloydb/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	alloydb "github.com/ubiquex/ubx-sdk-google/sdk/go/google_alloydb/alloydb"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own user"})

    		ubx.Resource(alloydb.AlloydbUser, "ci-runner", alloydb.AlloydbUserConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>

    # ubx sdk gen --only google_alloydb --lang py --out ./local-sdk

    import ubx_sdk as ubx
    from ubx.google_alloydb.alloydb import AlloydbUser, AlloydbUserConfig

    def describe():
        ubx.intent("payments own user")

        ubx.resource(AlloydbUser, "ci-runner", AlloydbUserConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)

## Properties

<ResponseField name="database_roles" type="list(string)">
  Optional. List of database roles this user has. The database role strings are subject to the PostgreSQL naming conventions. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="keep_extra_roles" type="bool">
  Input only. If the user already exists and it has additional roles, keep them granted. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  Output only. Name of the resource in the form of projects/&#123;project&#125;/locations/&#123;location&#125;/cluster/&#123;cluster&#125;/users/&#123;user&#125;. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="password" type="string">
  Input only. Password for the user. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="user_type" type="string">
  Optional. Type of this user. Optional; if omitted, computed by Google Cloud.
</ResponseField>
