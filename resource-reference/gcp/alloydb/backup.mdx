    ```go

    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_alloydb/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	alloydb "github.com/ubiquex/ubx-sdk-google/sdk/go/google_alloydb/alloydb"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own backup"})

    		ubx.Resource(alloydb.AlloydbBackup, "example", alloydb.AlloydbBackupConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
