`google_securityposture_posture` defines a versioned bundle of Organization Policy constraint(s) and Security Health Analytics module setting(s) that Security Posture management can deploy as a unit.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_securityposture --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_securityposture/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	securityposture "github.com/ubiquex/ubx-sdk-google/sdk/go/google_securityposture/securityposture"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own posture"})

    		ubx.Resource(securityposture.SecurityposturePosture, "ci-runner", securityposture.SecurityposturePostureConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
