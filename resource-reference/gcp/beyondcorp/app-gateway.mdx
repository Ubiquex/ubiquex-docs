`google_beyondcorp_app_gateway` is the Google-managed ingress point that routes authorized user traffic from the internet through to a BeyondCorp Enterprise app connection.

## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_beyondcorp/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	beyondcorp "github.com/ubiquex/ubx-sdk-google/sdk/go/google_beyondcorp/beyondcorp"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own app gateway"})

    		ubx.Resource(beyondcorp.BeyondcorpAppGateway, "ci-runner", beyondcorp.BeyondcorpAppGatewayConfig{

    		})
