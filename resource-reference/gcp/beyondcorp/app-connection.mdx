`google_beyondcorp_app_connection` defines a private application endpoint that BeyondCorp Enterprise's zero-trust remote access exposes to authorized users without a traditional VPN.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_beyondcorp --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_beyondcorp/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	beyondcorp "github.com/ubiquex/ubx-sdk-google/sdk/go/google_beyondcorp/beyondcorp"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own app connection"})

    		ubx.Resource(beyondcorp.BeyondcorpAppConnection, "ci-runner", beyondcorp.BeyondcorpAppConnectionConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
