`google_beyondcorp_app_connector` is the agent deployed near a private application that establishes an outbound-only tunnel back to Google Cloud, so BeyondCorp Enterprise can reach it without inbound firewall rules.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_beyondcorp --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_beyondcorp/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	beyondcorp "github.com/ubiquex/ubx-sdk-google/sdk/go/google_beyondcorp/beyondcorp"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own app connector"})

    		ubx.Resource(beyondcorp.BeyondcorpAppConnector, "ci-runner", beyondcorp.BeyondcorpAppConnectorConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
