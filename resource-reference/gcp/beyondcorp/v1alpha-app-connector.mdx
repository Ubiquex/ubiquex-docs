`google_beyondcorp_v1alpha_app_connector` is the agent deployed near a private application that establishes an outbound-only tunnel back to Google Cloud, so BeyondCorp Enterprise can reach it without inbound firewall rules.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	beyondcorp "github.com/ubiquex/ubx-sdk-google/sdk/go/google/beyondcorp"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1alpha app connector"})

    		ubx.Resource(beyondcorp.V1alphaAppConnector, "example", beyondcorp.V1alphaAppConnectorConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
