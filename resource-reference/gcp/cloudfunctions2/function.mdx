## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	cloudfunctions2 "github.com/ubiquex/ubx-sdk-google/sdk/go/google/cloudfunctions2"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own function"})

    		ubx.Resource(cloudfunctions2.Function, "ci-runner", cloudfunctions2.FunctionConfig{
    			Location:    "example",
    			Name:        "ci-runner",
    			Description: "Managed by ubx.",
    			KmsKeyName:  "example-kms-key-name",
    			Labels:      map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.google.cloudfunctions2 import Function, FunctionConfig

    def describe():

        ubx.resource(Function, "ci-runner", FunctionConfig(
            location="example",

    if __name__ == "__main__":
        ubx.run("payments", describe)
