    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	identity "github.com/ubiquex/ubx-sdk-google/sdk/go/google/identity"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own platform config"})

    		ubx.Resource(identity.PlatformConfig, "ci-runner", identity.PlatformConfigConfig{

    		})
    	}))
    }
    ```
