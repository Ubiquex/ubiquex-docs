<Warning>
  This is Google Cloud's own **v3beta1** release channel (beta). Google may change or remove beta endpoints as a feature moves toward general availability -- confirm this endpoint is still live before depending on it in anything long-lived. The stable, generally-available equivalent is `google_dialogflow_generator`.
</Warning>

`google_dialogflow_v3beta1_generator` defines a reusable LLM prompt template a Dialogflow CX agent invokes mid-conversation to generate dynamic content, e.g. a summary or a suggested response.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v3beta1 generator"})

    		ubx.Resource(dialogflow.V3beta1Generator, "example", dialogflow.V3beta1GeneratorConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.google.dialogflow.v3beta1_generator import V3beta1Generator, V3beta1GeneratorConfig

        ubx.resource(V3beta1Generator, "example", V3beta1GeneratorConfig(

        ubx.run("example", describe)
    ```
