`google_dialogflow_playbook` defines a Dialogflow CX agent's own conversational behavior using natural-language instructions and goals for an LLM to follow, the generative alternative to hand-built flow/page state machines.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dialogflow --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dialogflow/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dialogflow/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own playbook"})

    		ubx.Resource(dialogflow.DialogflowPlaybook, "ci-runner", dialogflow.DialogflowPlaybookConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
