`google_dialogflow_intent` maps a category of user input (defined by training phrases) to a specific meaning a Dialogflow agent should recognize and act on.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dialogflow --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dialogflow/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dialogflow/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own intent"})

    		ubx.Resource(dialogflow.DialogflowIntent, "example", dialogflow.DialogflowIntentConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.google_dialogflow.dialogflow import DialogflowIntent, DialogflowIntentConfig

        ubx.intent("example own intent")

            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
