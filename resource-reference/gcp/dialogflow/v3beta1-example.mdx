<Warning>
  This is Google Cloud's own **v3beta1** release channel (beta). Google may change or remove beta endpoints as a feature moves toward general availability -- confirm this endpoint is still live before depending on it in anything long-lived. The stable, generally-available equivalent is `google_dialogflow_example`.
</Warning>

`google_dialogflow_v3beta1_example` is a sample conversation used to few-shot guide a Dialogflow CX generative playbook's own behavior, showing the model an example of the desired input/output pattern.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v3beta1 example"})

    		ubx.Resource(dialogflow.V3beta1Example, "example", dialogflow.V3beta1ExampleConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { V3beta1Example } from "@ubx/sdk-google/google/dialogflow/v3beta1_example";

    export default stack("example", () => {
      intent({ summary: "example own v3beta1 example" });

      resource(V3beta1Example, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.google.dialogflow.v3beta1_example import V3beta1Example, V3beta1ExampleConfig
