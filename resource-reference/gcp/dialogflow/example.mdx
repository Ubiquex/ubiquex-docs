`google_dialogflow_example` is a sample conversation used to few-shot guide a Dialogflow CX generative playbook's own behavior, showing the model an example of the desired input/output pattern.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dialogflow --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dialogflow/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dialogflow/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own example"})

    		ubx.Resource(dialogflow.DialogflowExample, "example", dialogflow.DialogflowExampleConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
