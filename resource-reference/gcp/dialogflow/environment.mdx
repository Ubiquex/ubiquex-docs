`google_dialogflow_environment` is a deployable environment (e.g. `staging` or `production`) that a specific, pinned version of a Dialogflow agent runs in, letting changes be tested before going live.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own environment"})

    		ubx.Resource(dialogflow.Environment, "example", dialogflow.EnvironmentConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Environment } from "@ubx/sdk-google/google/dialogflow/environment";

    export default stack("example", () => {
      intent({ summary: "example own environment" });

      resource(Environment, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.dialogflow.environment import Environment, EnvironmentConfig

    def describe():

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
