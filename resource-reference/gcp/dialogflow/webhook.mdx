`google_dialogflow_webhook` configures an external HTTP endpoint a Dialogflow agent calls during a conversation to run custom business logic or fetch dynamic data.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own webhook"})

    		ubx.Resource(dialogflow.Webhook, "example", dialogflow.WebhookConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Webhook } from "@ubx/sdk-google/google/dialogflow/webhook";

    export default stack("example", () => {
      intent({ summary: "example own webhook" });

      resource(Webhook, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.dialogflow.webhook import Webhook, WebhookConfig

    def describe():
        ubx.intent("example own webhook")

        ubx.resource(Webhook, "example", WebhookConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
