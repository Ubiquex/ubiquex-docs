`google_dialogflow_v2beta1_environment` is a deployable environment (e.g. `staging` or `production`) that a specific, pinned version of a Dialogflow agent runs in, letting changes be tested before going live.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v2beta1 environment"})

    		ubx.Resource(dialogflow.V2beta1Environment, "example", dialogflow.V2beta1EnvironmentConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { V2beta1Environment } from "@ubx/sdk-google/google/dialogflow/v2beta1_environment";

    export default stack("example", () => {
      intent({ summary: "example own v2beta1 environment" });

      resource(V2beta1Environment, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.dialogflow import V2beta1Environment, V2beta1EnvironmentConfig

    def describe():
        ubx.intent("example own v2beta1 environment")

        ubx.resource(V2beta1Environment, "example", V2beta1EnvironmentConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
