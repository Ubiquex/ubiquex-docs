`google_dialogflow_v3beta1_tool` registers an external function, API, or data store a Dialogflow CX playbook can call during a conversation to fetch information or take an action outside the agent itself.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v3beta1 tool"})

    		ubx.Resource(dialogflow.V3beta1Tool, "example", dialogflow.V3beta1ToolConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { V3beta1Tool } from "@ubx/sdk-google/google/dialogflow/v3beta1_tool";

    export default stack("example", () => {
      intent({ summary: "example own v3beta1 tool" });

      resource(V3beta1Tool, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.dialogflow import V3beta1Tool, V3beta1ToolConfig

    def describe():
        ubx.intent("example own v3beta1 tool")

        ubx.resource(V3beta1Tool, "example", V3beta1ToolConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
