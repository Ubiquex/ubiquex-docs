`google_dialogflow_flow` is a self-contained conversational module (a directed graph of page(s) and transition(s)) within a Dialogflow CX agent, typically handling one topic or task.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dialogflow --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dialogflow/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dialogflow/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own flow"})

    		ubx.Resource(dialogflow.DialogflowFlow, "ci-runner", dialogflow.DialogflowFlowConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_dialogflow --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";

      intent({ summary: "payments own flow" });

    ```
  </Tab>

    # ubx sdk gen --only google_dialogflow --lang py --out ./local-sdk
