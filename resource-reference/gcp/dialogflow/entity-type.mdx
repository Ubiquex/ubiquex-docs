`google_dialogflow_entity_type` defines a category of value (e.g. a city name or product SKU) a Dialogflow agent can extract from user input, along with its own synonyms and matching behavior.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dialogflow --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dialogflow/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dialogflow/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own entity type"})

    		ubx.Resource(dialogflow.DialogflowEntityType, "ci-runner", dialogflow.DialogflowEntityTypeConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
