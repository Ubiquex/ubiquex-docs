`google_dialogflow_v3beta1_transition_route_group` is a reusable set of transition rules shared across multiple pages in a Dialogflow CX flow, avoiding duplicating the same routing logic on every page.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v3beta1 transition route group"})

    		ubx.Resource(dialogflow.V3beta1TransitionRouteGroup, "example", dialogflow.V3beta1TransitionRouteGroupConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
