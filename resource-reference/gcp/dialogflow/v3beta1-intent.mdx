<Warning>
  This is Google Cloud's own **v3beta1** release channel (beta). Google may change or remove beta endpoints as a feature moves toward general availability -- confirm this endpoint is still live before depending on it in anything long-lived. The stable, generally-available equivalent is `google_dialogflow_intent`.
</Warning>

`google_dialogflow_v3beta1_intent` maps a category of user input (defined by training phrases) to a specific meaning a Dialogflow agent should recognize and act on.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v3beta1 intent"})

    		ubx.Resource(dialogflow.V3beta1Intent, "example", dialogflow.V3beta1IntentConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { V3beta1Intent } from "@ubx/sdk-google/google/dialogflow/v3beta1_intent";

    export default stack("example", () => {
      intent({ summary: "example own v3beta1 intent" });

      resource(V3beta1Intent, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.dialogflow.v3beta1_intent import V3beta1Intent, V3beta1IntentConfig

    def describe():
        ubx.intent("example own v3beta1 intent")

        ubx.resource(V3beta1Intent, "example", V3beta1IntentConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
