`google_dialogflow_generator` defines a reusable LLM prompt template a Dialogflow CX agent invokes mid-conversation to generate dynamic content, e.g. a summary or a suggested response.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dialogflow --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dialogflow/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dialogflow/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own generator"})

    		ubx.Resource(dialogflow.DialogflowGenerator, "ci-runner", dialogflow.DialogflowGeneratorConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    # ubx sdk gen --only google_dialogflow --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_dialogflow/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_dialogflow.dialogflow import DialogflowGenerator, DialogflowGeneratorConfig

    def describe():
        ubx.intent("payments own generator")

        ubx.resource(DialogflowGenerator, "ci-runner", DialogflowGeneratorConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
