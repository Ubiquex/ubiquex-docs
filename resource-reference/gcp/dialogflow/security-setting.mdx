## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own security setting"})

    		ubx.Resource(dialogflow.SecuritySetting, "example", dialogflow.SecuritySettingConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
