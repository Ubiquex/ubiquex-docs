`google_dialogflow_v3beta1_webhook` configures an external HTTP endpoint a Dialogflow agent calls during a conversation to run custom business logic or fetch dynamic data.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v3beta1 webhook"})

    		ubx.Resource(dialogflow.V3beta1Webhook, "example", dialogflow.V3beta1WebhookConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { V3beta1Webhook } from "@ubx/sdk-google/google/dialogflow/v3beta1_webhook";

    export default stack("example", () => {
      intent({ summary: "example own v3beta1 webhook" });

      resource(V3beta1Webhook, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.google.dialogflow import V3beta1Webhook, V3beta1WebhookConfig

    def describe():
        ubx.intent("example own v3beta1 webhook")

        ubx.resource(V3beta1Webhook, "example", V3beta1WebhookConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
