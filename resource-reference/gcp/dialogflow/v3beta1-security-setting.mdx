<Warning>
  This is Google Cloud's own **v3beta1** release channel (beta). Google may change or remove beta endpoints as a feature moves toward general availability -- confirm this endpoint is still live before depending on it in anything long-lived. The stable, generally-available equivalent is `google_dialogflow_security_setting`.
</Warning>

`google_dialogflow_v3beta1_security_setting` configures data retention, redaction, and storage location policy for a Dialogflow agent's own conversation data.

## Example

  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dialogflow_v3beta1 --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dialogflow_v3beta1/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dialogflow "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dialogflow_v3beta1/dialogflow"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v3beta1 security setting"})

    		ubx.Resource(dialogflow.V3beta1DialogflowV3beta1SecuritySetting, "example", dialogflow.V3beta1DialogflowV3beta1SecuritySettingConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
