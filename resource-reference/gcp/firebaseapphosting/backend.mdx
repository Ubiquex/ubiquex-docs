## Example

    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	firebaseapphosting "github.com/ubiquex/ubx-sdk-google/sdk/go/google/firebaseapphosting"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own backend"})

    		ubx.Resource(firebaseapphosting.Backend, "example", firebaseapphosting.BackendConfig{
    			Name: "example",
    		})
    	}))
    }

Optional.
