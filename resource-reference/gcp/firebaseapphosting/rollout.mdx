## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	firebaseapphosting "github.com/ubiquex/ubx-sdk-google/sdk/go/google/firebaseapphosting"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own rollout"})

    		ubx.Resource(firebaseapphosting.Rollout, "example", firebaseapphosting.RolloutConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
