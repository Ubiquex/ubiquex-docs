## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_firebaseapphosting --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_firebaseapphosting/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	firebaseapphosting "github.com/ubiquex/ubx-sdk-google/sdk/go/google_firebaseapphosting/firebaseapphosting"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own domain"})

    		ubx.Resource(firebaseapphosting.FirebaseapphostingDomain, "ci-runner", firebaseapphosting.FirebaseapphostingDomainConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
