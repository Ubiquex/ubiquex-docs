`google_firebaseapphosting_v1beta_build` is a single compiled build of a Firebase App Hosting backend's own source code, produced from a Git push and deployable via a rollout.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	firebaseapphosting "github.com/ubiquex/ubx-sdk-google/sdk/go/google/firebaseapphosting"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta build"})

    		ubx.Resource(firebaseapphosting.V1betaBuild, "example", firebaseapphosting.V1betaBuildConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
