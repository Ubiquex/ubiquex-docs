`google_observability_bucket` configures cross-project or cross-location routing for Cloud Observability data (logs, metrics), letting telemetry be centralized into a single project's own bucket.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_observability --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_observability/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	observability "github.com/ubiquex/ubx-sdk-google/sdk/go/google_observability/observability"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own bucket"})

    		ubx.Resource(observability.ObservabilityBucket, "ci-runner", observability.ObservabilityBucketConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_observability --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ObservabilityBucket } from "./local-sdk/google_observability/sdk/typescript/google_observability/observability/observability_bucket.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own bucket" });

      resource(ObservabilityBucket, "ci-runner", {
        name: "ci-runner",

    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_observability --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_observability/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_observability.observability import ObservabilityBucket, ObservabilityBucketConfig

    def describe():
        ubx.intent("payments own bucket")

        ubx.resource(ObservabilityBucket, "ci-runner", ObservabilityBucketConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Properties

<ResponseField name="cmek_settings" type="object">
  Settings for configuring CMEK for a bucket. Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="kms_key" type="string">
      Optional. The resource name for the configured Cloud KMS key. The format is: projects/[PROJECT_ID]/locations/[LOCATION]/keyRings/[KEYRING]/cryptoKeys/[KEY] For example: projects/my-project/locations/us-central1/keyRings/my-ring/cryptoKeys/my-key Optional.
    </ResponseField>
    <ResponseField name="kms_key_version" type="string">
      Output only. The CryptoKeyVersion resource name for the configured Cloud KMS key. The format is: projects/[PROJECT_ID]/locations/[LOCATION]/keyRings/[KEYRING]/cryptoKeys/[KEY]/cryptoKeyVersions/[VERSION] For example: projects/my-project/locations/us-central1/keyRings/my-ring/cryptoKeys/my-key/cryptoKeyVersions/1 This read-only field is used to convey the specific configured CryptoKeyVersion of the `kms_key` that has been configured. It is populated when the CMEK settings are bound to a single key version. Computed by Google Cloud after `ship`.
    </ResponseField>
    <ResponseField name="service_account_id" type="string">
      Output only. The service account used to access the key. Computed by Google Cloud after `ship`.
    </ResponseField>
  </Expandable>
</ResponseField>
