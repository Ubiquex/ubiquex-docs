`google_observability_link` links a Cloud Logging bucket to a BigQuery dataset for direct SQL analytics, the Cloud Observability-level resource behind `google_logging_link`.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_observability --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_observability/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	observability "github.com/ubiquex/ubx-sdk-google/sdk/go/google_observability/observability"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own link"})

    		ubx.Resource(observability.ObservabilityLink, "ci-runner", observability.ObservabilityLinkConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_observability --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ObservabilityLink } from "./local-sdk/google_observability/sdk/typescript/google_observability/observability/observability_link.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own link" });

      resource(ObservabilityLink, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_observability --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_observability/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_observability.observability import ObservabilityLink, ObservabilityLinkConfig

    def describe():
        ubx.intent("payments own link")

        ubx.resource(ObservabilityLink, "ci-runner", ObservabilityLinkConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Properties

<ResponseField name="create_time" type="string">
  Output only. Create timestamp. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="description" type="string">
  Optional. Description of the link. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="display_name" type="string">
  Optional. A user friendly display name. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  Identifier. Name of the link. The format is: projects/[PROJECT_ID]/locations/[LOCATION]/buckets/[BUCKET_ID]/datasets/[DATASET_ID]/links/[LINK_ID] Optional; if omitted, computed by Google Cloud.
</ResponseField>
