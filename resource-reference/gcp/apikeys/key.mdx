    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	apikeys "github.com/ubiquex/ubx-sdk-google/sdk/go/google/apikeys"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
