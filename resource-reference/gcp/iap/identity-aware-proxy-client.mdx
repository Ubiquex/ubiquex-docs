`google_iap_identity_aware_proxy_client` is an additional OAuth client registered under an Identity-Aware Proxy brand, letting more than one client application authenticate through the same IAP-protected resource.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_iap --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_iap/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	iap "github.com/ubiquex/ubx-sdk-google/sdk/go/google_iap/iap"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own identity aware proxy client"})

    		ubx.Resource(iap.IapIdentityAwareProxyClient, "ci-runner", iap.IapIdentityAwareProxyClientConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
