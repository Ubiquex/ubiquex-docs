## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_iap --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_iap/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	iap "github.com/ubiquex/ubx-sdk-google/sdk/go/google_iap/iap"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own dest group"})

    		ubx.Resource(iap.IapDestGroup, "ci-runner", iap.IapDestGroupConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    // ubx sdk gen --only google_iap --lang ts --out ./local-sdk

    # ubx sdk gen --only google_iap --lang py --out ./local-sdk

    import ubx_sdk as ubx
    from ubx.google_iap.iap import IapDestGroup, IapDestGroupConfig

    def describe():
        ubx.intent("payments own dest group")

        ubx.resource(IapDestGroup, "ci-runner", IapDestGroupConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)

## Properties

<ResponseField name="cidrs" type="list(string)">
  Optional. Unordered list. List of CIDRs that this group applies to. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="fqdns" type="list(string)">
  Optional. Unordered list. List of FQDNs that this group applies to. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  Identifier. Identifier for the TunnelDestGroup. Must be unique within the project and contain only lower case letters (a-z) and dashes (-). Optional; if omitted, computed by Google Cloud.
</ResponseField>
