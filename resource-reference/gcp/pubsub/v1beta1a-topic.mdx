## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	pubsub "github.com/ubiquex/ubx-sdk-google/sdk/go/google/pubsub"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta1a topic"})

    		ubx.Resource(pubsub.V1beta1aTopic, "example", pubsub.V1beta1aTopicConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
