`google_pubsub_schema` -- real, typed bindings generated directly from the
`hashicorp/google` provider schema, in every SDK language.

Every tab below is a complete, runnable program, not a fragment,
real enough to save and run exactly as shown.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	pubsub "github.com/ubiquex/ubx-sdk-google/sdk/go/google/pubsub"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own schema"})

    		ubx.Resource(pubsub.Schema, "ci-runner", pubsub.SchemaConfig{
    			Name:       "ci-runner",
    			Definition: "example",
    			Type:       "example",
    		})
    	}))
    }
    ```
  </Tab>
