`google_migrationcenter_assets_export_job` exports discovered asset inventory data from Migration Center to an external destination (e.g. BigQuery), for reporting or integration outside the console.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_migrationcenter --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_migrationcenter/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	migrationcenter "github.com/ubiquex/ubx-sdk-google/sdk/go/google_migrationcenter/migrationcenter"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own assets export job"})

    		ubx.Resource(migrationcenter.MigrationcenterAssetsExportJob, "ci-runner", migrationcenter.MigrationcenterAssetsExportJobConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
