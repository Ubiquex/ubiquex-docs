`google_migrationcenter_source` registers the origin (a discovery client, an import job, or a manually entered environment) that Migration Center asset data was collected from.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_migrationcenter --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_migrationcenter/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	migrationcenter "github.com/ubiquex/ubx-sdk-google/sdk/go/google_migrationcenter/migrationcenter"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own source"})

    		ubx.Resource(migrationcenter.MigrationcenterSource, "ci-runner", migrationcenter.MigrationcenterSourceConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_migrationcenter --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { MigrationcenterSource } from "./local-sdk/google_migrationcenter/sdk/typescript/google_migrationcenter/migrationcenter/migrationcenter_source.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own source" });

      resource(MigrationcenterSource, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_migrationcenter --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_migrationcenter/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_migrationcenter.migrationcenter import MigrationcenterSource, MigrationcenterSourceConfig

    def describe():
        ubx.intent("payments own source")

        ubx.resource(MigrationcenterSource, "ci-runner", MigrationcenterSourceConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Properties

<ResponseField name="create_time" type="string">
  Output only. The timestamp when the source was created. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="description" type="string">
  Free-text description. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="display_name" type="string">
  User-friendly display name. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="error_frame_count" type="number">
  Output only. The number of frames that were reported by the source and contained errors. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="managed" type="bool">
  If `true`, the source is managed by other service(s). Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  Output only. The full name of the source. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="pending_frame_count" type="number">
  Output only. Number of frames that are still being processed. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="priority" type="number">
  The information confidence of the source. The higher the value, the higher the confidence. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="state" type="string">
  Output only. The state of the source. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="type" type="string">
  Data source type. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="update_time" type="string">
  Output only. The timestamp when the source was last updated. Optional; if omitted, computed by Google Cloud.
</ResponseField>
