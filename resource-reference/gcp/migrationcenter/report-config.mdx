`google_migrationcenter_report_config` defines the reusable configuration (which asset group and preference set to use) a Migration Center report is generated from.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_migrationcenter --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_migrationcenter/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	migrationcenter "github.com/ubiquex/ubx-sdk-google/sdk/go/google_migrationcenter/migrationcenter"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own report config"})

    		ubx.Resource(migrationcenter.MigrationcenterReportConfig, "example", migrationcenter.MigrationcenterReportConfigConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
