`google_migrationcenter_group` is a named collection of discovered asset(s) within Migration Center, used to scope a sizing report or migration wave to a specific subset of infrastructure.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_migrationcenter --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_migrationcenter/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	migrationcenter "github.com/ubiquex/ubx-sdk-google/sdk/go/google_migrationcenter/migrationcenter"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own group"})

    		ubx.Resource(migrationcenter.MigrationcenterGroup, "example", migrationcenter.MigrationcenterGroupConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
