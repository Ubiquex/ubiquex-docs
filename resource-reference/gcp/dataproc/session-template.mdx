`google_dataproc_session_template` defines a reusable configuration (runtime version, resource sizing) new Dataproc interactive sessions can be created from.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_dataproc --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_dataproc/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	dataproc "github.com/ubiquex/ubx-sdk-google/sdk/go/google_dataproc/dataproc"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own session template"})

    		ubx.Resource(dataproc.DataprocSessionTemplate, "ci-runner", dataproc.DataprocSessionTemplateConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_dataproc --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { DataprocSessionTemplate } from "./local-sdk/google_dataproc/sdk/typescript/google_dataproc/dataproc/dataproc_session_template.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own session template" });

      resource(DataprocSessionTemplate, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```
