`google_redis_v1beta1_acl_policy` defines Redis ACL rule(s) (which command(s) and key pattern(s) a user is permitted to access) applied to a Memorystore for Redis Cluster instance.

## Example

    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	redis "github.com/ubiquex/ubx-sdk-google/sdk/go/google/redis"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta1 acl policy"})

    		ubx.Resource(redis.V1beta1AclPolicy, "example", redis.V1beta1AclPolicyConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
