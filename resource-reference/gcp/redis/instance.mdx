    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_redis/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	redis "github.com/ubiquex/ubx-sdk-google/sdk/go/google_redis/redis"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own instance"})

    		ubx.Resource(redis.RedisInstance, "ci-runner", redis.RedisInstanceConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
