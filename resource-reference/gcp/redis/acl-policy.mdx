`google_redis_acl_policy` defines Redis ACL rule(s) (which command(s) and key pattern(s) a user is permitted to access) applied to a Memorystore for Redis Cluster instance.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_redis --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_redis/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	redis "github.com/ubiquex/ubx-sdk-google/sdk/go/google_redis/redis"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own acl policy"})

    		ubx.Resource(redis.RedisAclPolicy, "example", redis.RedisAclPolicyConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
