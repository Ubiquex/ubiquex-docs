`google_networksecurity_sac_attachment` is a regional Secure Access Connect resource that enables an NCC Gateway to process traffic through a connected security service edge (SSE) provider, e.g. Palo Alto Prisma Access.

## Example

  <Tab title="Go">

    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networksecurity/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networksecurity "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networksecurity/networksecurity"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own sac attachment"})

    		ubx.Resource(networksecurity.NetworksecuritySacAttachment, "ci-runner", networksecurity.NetworksecuritySacAttachmentConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
