`google_networksecurity_v1beta1_url_list` is a named, reusable list of URL pattern(s) that a Secure Web Proxy or Cloud NGFW policy rule can reference by name instead of enumerating URLs inline.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networksecurity "github.com/ubiquex/ubx-sdk-google/sdk/go/google/networksecurity"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta1 url list"})

    		ubx.Resource(networksecurity.V1beta1UrlList, "example", networksecurity.V1beta1UrlListConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
