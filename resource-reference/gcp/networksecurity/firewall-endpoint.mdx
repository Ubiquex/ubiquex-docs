`google_networksecurity_firewall_endpoint` is a zonal, Google-managed data-plane resource that Cloud NGFW's own intrusion prevention service uses to transparently inspect workload traffic against threat signatures.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_networksecurity --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networksecurity/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networksecurity "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networksecurity/networksecurity"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own firewall endpoint"})

    		ubx.Resource(networksecurity.NetworksecurityFirewallEndpoint, "ci-runner", networksecurity.NetworksecurityFirewallEndpointConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
