`google_networksecurity_url_list` is a named, reusable list of URL pattern(s) that a Secure Web Proxy or Cloud NGFW policy rule can reference by name instead of enumerating URLs inline.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_networksecurity --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networksecurity/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networksecurity "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networksecurity/networksecurity"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own url list"})

    		ubx.Resource(networksecurity.NetworksecurityUrlList, "ci-runner", networksecurity.NetworksecurityUrlListConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_networksecurity --lang ts --out ./local-sdk
    ```
