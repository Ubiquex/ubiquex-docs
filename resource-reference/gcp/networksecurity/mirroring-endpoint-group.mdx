`google_networksecurity_mirroring_endpoint_group` is the packet-mirroring entry point associated with a mirroring deployment group, that VPC network traffic copies are routed through.

    	networksecurity "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networksecurity/networksecurity"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
