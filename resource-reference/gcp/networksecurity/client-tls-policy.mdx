`google_networksecurity_client_tls_policy` configures the TLS settings (client certificate, trusted server CA) a Google Cloud service uses when it acts as the TLS client connecting outbound to another service.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networksecurity "github.com/ubiquex/ubx-sdk-google/sdk/go/google/networksecurity"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own client tls policy"})

    		ubx.Resource(networksecurity.ClientTlsPolicy, "example", networksecurity.ClientTlsPolicyConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
