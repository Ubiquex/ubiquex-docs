`google_networksecurity_mirroring_deployment` is a zonal instance of a third-party security or monitoring appliance under Google Cloud's Network Security Integration framework, that mirrored packet copies are sent to for out-of-band analysis.

## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networksecurity/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networksecurity "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networksecurity/networksecurity"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own mirroring deployment"})

    		ubx.Resource(networksecurity.NetworksecurityMirroringDeployment, "ci-runner", networksecurity.NetworksecurityMirroringDeploymentConfig{

    		})
    	}))
    }
