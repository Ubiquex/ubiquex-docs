`google_networksecurity_address_group` is a named, reusable list of IP address(es) or CIDR range(s) that Cloud Firewall or Cloud Armor policy rules can reference by name instead of enumerating addresses inline.

## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networksecurity/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networksecurity "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networksecurity/networksecurity"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own address group"})

    		ubx.Resource(networksecurity.NetworksecurityAddressGroup, "example", networksecurity.NetworksecurityAddressGroupConfig{
    			Name: "example",
    		})
    	}))
    }
