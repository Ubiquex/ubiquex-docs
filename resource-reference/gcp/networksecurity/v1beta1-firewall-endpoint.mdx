`google_networksecurity_v1beta1_firewall_endpoint` is a zonal, Google-managed data-plane resource that Cloud NGFW's own intrusion prevention service uses to transparently inspect workload traffic against threat signatures.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networksecurity "github.com/ubiquex/ubx-sdk-google/sdk/go/google/networksecurity"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta1 firewall endpoint"})

    		ubx.Resource(networksecurity.V1beta1FirewallEndpoint, "example", networksecurity.V1beta1FirewallEndpointConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
