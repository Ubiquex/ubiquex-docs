`google_networksecurity_security_profile` defines the inspection behavior (threat prevention settings, custom signatures) applied by a Cloud NGFW intrusion prevention or Secure Web Proxy security profile group.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_networksecurity --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networksecurity/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networksecurity "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networksecurity/networksecurity"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own security profile"})

    		ubx.Resource(networksecurity.NetworksecuritySecurityProfile, "ci-runner", networksecurity.NetworksecuritySecurityProfileConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
