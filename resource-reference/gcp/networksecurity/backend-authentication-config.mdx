`google_networksecurity_backend_authentication_config` configures a load balancer to authenticate itself to backend services via mutual TLS, referencing the client certificate presented on outbound connections.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_networksecurity --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_networksecurity/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	networksecurity "github.com/ubiquex/ubx-sdk-google/sdk/go/google_networksecurity/networksecurity"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own backend authentication config"})

    		ubx.Resource(networksecurity.NetworksecurityBackendAuthenticationConfig, "ci-runner", networksecurity.NetworksecurityBackendAuthenticationConfigConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
