## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_sqladmin --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_sqladmin/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	sqladmin "github.com/ubiquex/ubx-sdk-google/sdk/go/google_sqladmin/sqladmin"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own database"})

    		ubx.Resource(sqladmin.SqladminDatabase, "example", sqladmin.SqladminDatabaseConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_sqladmin --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { SqladminDatabase } from "./local-sdk/google_sqladmin/sdk/typescript/google_sqladmin/sqladmin/sqladmin_database.ts";

    export default stack("example", () => {
      intent({ summary: "example own database" });

      resource(SqladminDatabase, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_sqladmin --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_sqladmin/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_sqladmin.sqladmin import SqladminDatabase, SqladminDatabaseConfig

    def describe():
        ubx.intent("example own database")

        ubx.resource(SqladminDatabase, "example", SqladminDatabaseConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="charset" type="string">
  The Cloud SQL charset value. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="collation" type="string">
  The Cloud SQL collation value. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="etag" type="string">
  This field is deprecated and will be removed from a future version of the API. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="instance" type="string">
  The name of the Cloud SQL instance. This does not include the project ID. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="kind" type="string">
  This is always `sql#database`. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  The name of the database in the Cloud SQL instance. This does not include the project ID or instance name. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="project" type="string">
  The project ID of the project containing the Cloud SQL database. The Google apps domain is prefixed if applicable. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="self_link" type="string">
  The URI of this resource. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="sqlserver_database_details" type="object">

</ResponseField>
