## Example

    ```go

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta4 user"})

    		})
    ```

## Properties

<ResponseField name="database_roles" type="list(string)">
  Optional. Role memberships of the user Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="dual_password_type" type="string">
  Dual password status for the user. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="etag" type="string">
  This field is deprecated and will be removed from a future version of the API. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="host" type="string">
  Optional. The host from which the user can connect. For `insert` operations, host defaults to an empty string. For `update` operations, host is specified as part of the request URL. The host name cannot be updated after insertion. For a MySQL instance, it's required; for a PostgreSQL or SQL Server instance, it's optional. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="iam_email" type="string">
  Optional. The full email for an IAM user. For normal database users, this will not be filled. Only applicable to MySQL database users. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="iam_status" type="string">
  Indicates if a group is active or inactive for IAM database authentication. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="instance" type="string">
  The name of the Cloud SQL instance. This does not include the project ID. Can be omitted for *update* because it is already specified on the URL. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="kind" type="string">
  This is always `sql#user`. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  The name of the user in the Cloud SQL instance. Can be omitted for `update` because it is already specified in the URL. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="password" type="string">
  The password for the user. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="password_policy" type="object">
  User level password validation policy. Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="allowed_failed_attempts" type="number">
      Number of failed login attempts allowed before user get locked. Optional.
    </ResponseField>
    <ResponseField name="enable_failed_attempts_check" type="bool">
      If true, failed login attempts check will be enabled. Optional.
    </ResponseField>
    <ResponseField name="enable_password_verification" type="bool">
      If true, the user must specify the current password before changing the password. This flag is supported only for MySQL. Optional.
    </ResponseField>
    <ResponseField name="password_expiration_duration" type="string">
      Expiration duration after password is updated. Optional.
    </ResponseField>
    <ResponseField name="status" type="object">
      Read-only password status. Optional.
      <Expandable title="properties">
        <ResponseField name="locked" type="bool">
          If true, user does not have login privileges. Optional.
        </ResponseField>
        <ResponseField name="password_expiration_time" type="string">
          The expiration time of the current password. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="project" type="string">
  The project ID of the project containing the Cloud SQL database. The Google apps domain is prefixed if applicable. Can be omitted for *update* because it is already specified on the URL. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="server_roles" type="list(string)">
  Optional. The server roles for the SQL Server login. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="sqlserver_user_details" type="object">
  Represents a Sql Server user on the Cloud SQL instance. Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="disabled" type="bool">
      If the user has been disabled Optional.
    </ResponseField>
    <ResponseField name="server_roles" type="list(string)">
      The server roles for this user Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="type" type="string">
  The user type. It determines the method to authenticate the user during login. The default is the database's built-in user type. Optional; if omitted, computed by Google Cloud.
</ResponseField>
