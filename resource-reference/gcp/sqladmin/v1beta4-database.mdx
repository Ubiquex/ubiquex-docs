## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	sqladmin "github.com/ubiquex/ubx-sdk-google/sdk/go/google/sqladmin"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own v1beta4 database"})

    		ubx.Resource(sqladmin.V1beta4Database, "example", sqladmin.V1beta4DatabaseConfig{
    			Name: "example",
    		})
    	}))
    }
    ```

## Properties

<ResponseField name="charset" type="string">
  The Cloud SQL charset value. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="collation" type="string">
  The Cloud SQL collation value. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="etag" type="string">
  This field is deprecated and will be removed from a future version of the API. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="instance" type="string">
  The name of the Cloud SQL instance. This does not include the project ID. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="kind" type="string">
  This is always `sql#database`. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  The name of the database in the Cloud SQL instance. This does not include the project ID or instance name. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="project" type="string">
  The project ID of the project containing the Cloud SQL database. The Google apps domain is prefixed if applicable. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="self_link" type="string">
  The URI of this resource. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="sqlserver_database_details" type="object">
  Represents a Sql Server database on the Cloud SQL instance. Optional; if omitted, computed by Google Cloud.
  <Expandable title="properties">
    <ResponseField name="compatibility_level" type="number">
      The version of SQL Server with which the database is to be made compatible Optional.
    </ResponseField>
    <ResponseField name="recovery_model" type="string">
      The recovery model of a SQL Server database Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
