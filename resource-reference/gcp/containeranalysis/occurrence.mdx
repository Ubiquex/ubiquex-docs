`google_containeranalysis_occurrence` records a single instance of metadata (e.g. one detected vulnerability) about a specific container image resource, conforming to a Container Analysis note's own schema.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_containeranalysis --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_containeranalysis/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	containeranalysis "github.com/ubiquex/ubx-sdk-google/sdk/go/google_containeranalysis/containeranalysis"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own occurrence"})

    		ubx.Resource(containeranalysis.ContaineranalysisOccurrence, "example", containeranalysis.ContaineranalysisOccurrenceConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
