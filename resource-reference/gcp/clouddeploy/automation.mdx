`google_clouddeploy_automation` configures Cloud Deploy to automatically run action(s) (e.g. promoting a release to the next stage, or repeating a rollout) when a defined event occurs in a delivery pipeline, without a human triggering it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_clouddeploy --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_clouddeploy/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	clouddeploy "github.com/ubiquex/ubx-sdk-google/sdk/go/google_clouddeploy/clouddeploy"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own automation"})

    		ubx.Resource(clouddeploy.ClouddeployAutomation, "ci-runner", clouddeploy.ClouddeployAutomationConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
