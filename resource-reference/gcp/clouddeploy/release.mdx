`google_clouddeploy_release` is a specific, immutable set of application artifacts (rendered from a delivery pipeline's own configuration) that Cloud Deploy progressively rolls out across a pipeline's own target(s).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_clouddeploy --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_clouddeploy/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	clouddeploy "github.com/ubiquex/ubx-sdk-google/sdk/go/google_clouddeploy/clouddeploy"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own release"})

    		ubx.Resource(clouddeploy.ClouddeployRelease, "ci-runner", clouddeploy.ClouddeployReleaseConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
