`google_clouddeploy_target` defines a single deployment destination (a GKE cluster, Cloud Run service, or custom target) a Cloud Deploy delivery pipeline can roll out a release to.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_clouddeploy --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_clouddeploy/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	clouddeploy "github.com/ubiquex/ubx-sdk-google/sdk/go/google_clouddeploy/clouddeploy"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own target"})

    		ubx.Resource(clouddeploy.ClouddeployTarget, "ci-runner", clouddeploy.ClouddeployTargetConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_clouddeploy --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ClouddeployTarget } from "./local-sdk/google_clouddeploy/sdk/typescript/google_clouddeploy/clouddeploy/clouddeploy_target.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own target" });

      resource(ClouddeployTarget, "ci-runner", {
        name: "ci-runner",

    ```
