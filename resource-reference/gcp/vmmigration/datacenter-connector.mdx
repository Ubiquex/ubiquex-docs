`google_vmmigration_datacenter_connector` is the real, on-premises or source-cloud agent VM Migration uses to discover VMs and replicate their data to Google Cloud ahead of a clone or cutover.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_vmmigration --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_vmmigration/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	vmmigration "github.com/ubiquex/ubx-sdk-google/sdk/go/google_vmmigration/vmmigration"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own datacenter connector"})

    		ubx.Resource(vmmigration.VmmigrationDatacenterConnector, "ci-runner", vmmigration.VmmigrationDatacenterConnectorConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
