`google_vmmigration_disk_migration_job` migrates a real, standalone disk (not an entire running VM) to Google Cloud Persistent Disk, for cases where only the data, not the source machine's live state, needs to move.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	vmmigration "github.com/ubiquex/ubx-sdk-google/sdk/go/google/vmmigration"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own disk migration job"})

    		ubx.Resource(vmmigration.DiskMigrationJob, "example", vmmigration.DiskMigrationJobConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
