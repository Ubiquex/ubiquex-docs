`google_vmmigration_utilization_report` is a real, generated report of a source environment's own resource utilization (CPU, memory, disk), used to right-size the migrated VM's machine type in Google Cloud before cutover.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_vmmigration --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_vmmigration/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	vmmigration "github.com/ubiquex/ubx-sdk-google/sdk/go/google_vmmigration/vmmigration"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own utilization report"})

    		ubx.Resource(vmmigration.VmmigrationUtilizationReport, "ci-runner", vmmigration.VmmigrationUtilizationReportConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
