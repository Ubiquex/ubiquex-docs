`google_vmmigration_group` is a real VM Migration group, letting related `google_vmmigration_migrating_vm` resources be managed together (for example migrated or cut over as one coordinated batch) rather than one at a time.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_vmmigration --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_vmmigration/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	vmmigration "github.com/ubiquex/ubx-sdk-google/sdk/go/google_vmmigration/vmmigration"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own group"})

    		ubx.Resource(vmmigration.VmmigrationGroup, "example", vmmigration.VmmigrationGroupConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only google_vmmigration --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";

    export default stack("example", () => {
      intent({ summary: "example own group" });

    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only google_vmmigration --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/google_vmmigration/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.google_vmmigration.vmmigration import VmmigrationGroup, VmmigrationGroupConfig

    def describe():
        ubx.intent("example own group")

        ubx.resource(VmmigrationGroup, "example", VmmigrationGroupConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Properties

<ResponseField name="create_time" type="string">
  Output only. The create time timestamp. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="description" type="string">
  User-provided description of the group. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="display_name" type="string">
  Display name is a user defined name for this group which can be updated. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="migration_target_type" type="string">
  Immutable. The target type of this group. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="name" type="string">
  Output only. The Group name. Optional; if omitted, computed by Google Cloud.
</ResponseField>

<ResponseField name="update_time" type="string">
  Output only. The update time timestamp. Optional; if omitted, computed by Google Cloud.
</ResponseField>
