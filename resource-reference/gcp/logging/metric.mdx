    package main

    import (
    	logging "github.com/ubiquex/ubx-sdk-aws/sdk/go/google/logging"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own metric"})

    		ubx.Resource(logging.Metric, "ci-runner", logging.MetricConfig{
    			Filter:          "example",
    			Name:            "ci-runner",

    			Disabled:        true,
    			LabelExtractors: map[string]string{"managed-by": "ubx"},
    			ValueExtractor:  "example",
    		})
    	}))
    }

    import ubx_sdk as ubx
    from ubx.google.logging import Metric, MetricConfig

    def describe():
        ubx.intent("payments own metric")

        ubx.resource(Metric, "ci-runner", MetricConfig(
            filter="example",
            name="ci-runner",

            label_extractors={"managed-by": "ubx"},
