## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only google_memcache --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-google/sdk/go => ./local-sdk/google_memcache/sdk/go
    package main

    import (
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    	memcache "github.com/ubiquex/ubx-sdk-google/sdk/go/google_memcache/memcache"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own instance"})

    		ubx.Resource(memcache.MemcacheInstance, "example", memcache.MemcacheInstanceConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
