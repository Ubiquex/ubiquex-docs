## Properties

<ResponseField name="etag" type="string">
  A unique read-only string that changes whenever the resource is updated. Optional; if omitted, computed by Microsoft Azure.
</ResponseField>

<ResponseField name="properties" type="object">
  Route resource. Optional; if omitted, computed by Microsoft Azure.
  <Expandable title="properties">
    <ResponseField name="address_prefix" type="string">
      The destination CIDR to which the route applies. Optional.
    </ResponseField>
    <ResponseField name="has_bgp_override" type="bool">
      A value indicating whether this route overrides overlapping BGP routes regardless of LPM. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="next_hop" type="object">
      The next hop definition for ECMP routes containing multiple next hop IP addresses. Optional.
      <Expandable title="properties">
        <ResponseField name="next_hop_ip_addresses" type="list(string)" required>
          List of next hop IP addresses for ECMP routing. Must contain between 2 and 64 IP addresses. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="next_hop_ip_address" type="string">
      The IP address packets should be forwarded to. Next hop values are only allowed in routes where the next hop type is VirtualAppliance. Optional.
    </ResponseField>
    <ResponseField name="next_hop_type" type="string" required>
      The type of Azure hop the packet should be sent to. Required.
    </ResponseField>
    <ResponseField name="provisioning_state" type="string">
      Provisioning states of a resource. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
