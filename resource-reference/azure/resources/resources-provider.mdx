`azure_resources_resources_provider` reports a registered Azure resource provider (e.g. `Microsoft.Compute`) and the resource type(s) it exposes within a subscription.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	resources "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/resources"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own resources provider"})

    		ubx.Resource(resources.Provider, "example", resources.ProviderConfig{
    			ThirdPartyProviderConsent: map[string]any{"consent_to_authorization": true},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Provider } from "@ubx/sdk-azure/azure/resources/provider";

    export default stack("example", () => {
      intent({ summary: "example own resources provider" });

      resource(Provider, "example", {
        thirdPartyProviderConsent: { consentToAuthorization: true },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.resources import Provider, ProviderConfig

    def describe():
        ubx.intent("example own resources provider")

        ubx.resource(Provider, "example", ProviderConfig(
            third_party_provider_consent={"consent_to_authorization": True},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
  <Tab title="Markdown">
    ```
    Create azure_resources_provider, with third party provider
    consent a set value.
    ```
  </Tab>
</Tabs>
