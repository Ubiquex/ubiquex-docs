`azure_resources_deploymentscripts_deployment_script` runs a PowerShell or Azure CLI script as part of an ARM template deployment, for imperative logic a declarative template alone can't express.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_resources_deploymentscripts --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_resources_deploymentscripts/sdk/go
    package main

    import (
    	resources "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_resources_deploymentscripts/resources"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own deploymentscripts deployment script"})

    		ubx.Resource(resources.DeploymentscriptsDeploymentScript, "ci-runner", resources.DeploymentscriptsDeploymentScriptConfig{
    			Kind: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_resources_deploymentscripts --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { DeploymentscriptsDeploymentScript } from "./local-sdk/azure_resources_deploymentscripts/sdk/typescript/azure_resources_deploymentscripts/resources/deploymentscripts_deployment_script.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own deploymentscripts deployment script" });

      resource(DeploymentscriptsDeploymentScript, "ci-runner", {
        kind: "example",
      });
    });
    ```

  <Tab title="Python">

    # ubx sdk gen --only azure_resources_deploymentscripts --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_resources_deploymentscripts/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx

        ubx.intent("payments own deploymentscripts deployment script")

        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
