`azure_resources_resources_resource_group` is an Azure resource group, the logical container that related resource(s) are deployed into and managed as a unit.

    package main

    import (
    	resources "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_resources_resources/resources"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own resources resource group"})

    	}))
