## Properties

<ResponseField name="location" type="string" required>
  The location of the Template Spec Version. It must match the location of the parent Template Spec. Required.
</ResponseField>

<ResponseField name="properties" type="object" required>
  Template Spec Version properties. Required.
  <Expandable title="properties">
    <ResponseField name="description" type="string">
      Template Spec version description. Optional.
    </ResponseField>
    <ResponseField name="linked_templates" type="list(object)">
      An array of linked template artifacts. Optional.
      <Expandable title="properties">
        <ResponseField name="path" type="string">
          The relative path identifying this linked template within the parent template's own deployment. Optional; if omitted, computed by Microsoft Azure. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="template" type="any">
          The linked ARM template's own JSON content. Optional; if omitted, computed by Microsoft Azure. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="main_template" type="any">
      The main Azure Resource Manager template content. Optional.
    </ResponseField>
    <ResponseField name="metadata" type="any">
      The version metadata. Metadata is an open-ended object and is typically a collection of key-value pairs. Optional.
    </ResponseField>
    <ResponseField name="ui_form_definition" type="any">
      The Azure Resource Manager template UI definition content. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tags" type="map(string)">
  Resource tags. Optional; if omitted, computed by Microsoft Azure.
</ResponseField>
