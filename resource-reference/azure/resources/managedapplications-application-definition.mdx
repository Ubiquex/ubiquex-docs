`azure_resources_managedapplications_application_definition` publishes a reusable Managed Application template to the Azure Marketplace or a private catalog, that customer(s) can deploy instances from.

## Example

    package main

    import (
    	resources "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_resources_managedapplications/resources"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own managedapplications application definition"})

    		ubx.Resource(resources.ManagedapplicationsApplicationDefinition, "ci-runner", resources.ManagedapplicationsApplicationDefinitionConfig{
