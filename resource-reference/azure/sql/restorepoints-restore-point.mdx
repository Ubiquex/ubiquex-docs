---
title: "azure_sql_restorepoints_restore_point"
description: "`azure_sql_restorepoints_restore_point` is a caller-created, named restore point for an Azure SQL Database, letting it be restored to that exact moment."
---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_sql_restorepoints --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_sql_restorepoints/sdk/go
    package main

    import (
    	sql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_sql_restorepoints/sql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own restorepoints restore point"})

    		ubx.Resource(sql.RestorepointsRestorePoint, "ci-runner", sql.RestorepointsRestorePointConfig{
    			RestorePointLabel: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_sql_restorepoints --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { RestorepointsRestorePoint } from "./local-sdk/azure_sql_restorepoints/sdk/typescript/azure_sql_restorepoints/sql/restorepoints_restore_point.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own restorepoints restore point" });

      resource(RestorepointsRestorePoint, "ci-runner", {
        restorePointLabel: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_sql_restorepoints --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_sql_restorepoints/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_sql_restorepoints.sql import RestorepointsRestorePoint, RestorepointsRestorePointConfig

    def describe():
        ubx.intent("payments own restorepoints restore point")

        ubx.resource(RestorepointsRestorePoint, "ci-runner", RestorepointsRestorePointConfig(
            restore_point_label="example",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Input properties

<ResponseField name="restore_point_label" type="string" required>
  The restore point label to apply Required.
</ResponseField>

## Output properties

<ResponseField name="location" type="string">
  Resource location. Computed by Microsoft Azure after `ship`.
</ResponseField>

<ResponseField name="properties" type="object">
  Properties of a database restore point Computed by Microsoft Azure after `ship`.
  <Expandable title="properties">
    <ResponseField name="earliest_restore_date" type="string">
      The earliest time to which this database can be restored Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="restore_point_creation_date" type="string">
      The time the backup was taken Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="restore_point_label" type="string">
      The label of restore point for backup request by user Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="restore_point_type" type="string">
      The type of restore point Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
