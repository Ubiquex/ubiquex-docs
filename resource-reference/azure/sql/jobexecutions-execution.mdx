## Output properties

<ResponseField name="properties" type="object">
  Properties for an Azure SQL Database Elastic job execution. Computed by Microsoft Azure after `ship`.
  <Expandable title="properties">
    <ResponseField name="create_time" type="string">
      The time that the job execution was created. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="current_attempt_start_time" type="string">
      Start time of the current attempt. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="current_attempts" type="number">
      Number of times the job execution has been attempted. Optional.
    </ResponseField>
    <ResponseField name="end_time" type="string">
      The time that the job execution completed. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="job_execution_id" type="string">
      Universally Unique Identifier Optional.
    </ResponseField>
    <ResponseField name="job_version" type="number">
      The job version number. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="last_message" type="string">
      The last status or error message. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="lifecycle" type="string">
      The detailed state of the job execution. Optional.
    </ResponseField>
    <ResponseField name="provisioning_state" type="string">
      The ARM provisioning state of the job execution. Optional.
    </ResponseField>
    <ResponseField name="start_time" type="string">
      The time that the job execution started. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="step_id" type="number">
      The job step id. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="step_name" type="string">
      The job step name. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="target" type="object">
      The target that a job execution is executed on. Optional.
      <Expandable title="properties">
        <ResponseField name="database_name" type="string">
          The database name. Computed by Microsoft Azure after `ship`.
        </ResponseField>
        <ResponseField name="server_name" type="string">
          The server name. Computed by Microsoft Azure after `ship`.
        </ResponseField>
        <ResponseField name="type" type="string">
          The type of the target. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
