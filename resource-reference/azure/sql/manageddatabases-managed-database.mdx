## Properties

<ResponseField name="properties" type="object">
  The managed database's properties. Optional; if omitted, computed by Microsoft Azure.
  <Expandable title="properties">
    <ResponseField name="auto_complete_restore" type="bool">
      Whether to auto complete restore of this managed database. Optional.
    </ResponseField>
    <ResponseField name="catalog_collation" type="string">
      Collation of the metadata catalog. Optional.
    </ResponseField>
    <ResponseField name="collation" type="string">
      Collation of the managed database. Optional.
    </ResponseField>
    <ResponseField name="create_mode" type="string">
      Managed database create mode. PointInTimeRestore: Create a database by restoring a point in time backup of an existing database. SourceDatabaseName, SourceManagedInstanceName and PointInTime must be specified. RestoreExternalBackup: Create a database by restoring from external backup files. Collation, StorageContainerUri and StorageContainerSasToken must be specified. Recovery: Creates a database by restoring a geo-replicated backup. RecoverableDatabaseId must be specified as the recoverable database resource ID to restore. RestoreLongTermRetentionBackup: Create a database by restoring from a long term retention backup (longTermRetentionBackupResourceId required). Optional.
    </ResponseField>
    <ResponseField name="creation_date" type="string">
      Creation date of the database. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="cross_subscription_restorable_dropped_database_id" type="string">
      The restorable cross-subscription dropped database resource id to restore when creating this database. Optional.
    </ResponseField>
    <ResponseField name="cross_subscription_source_database_id" type="string">
      The resource identifier of the cross-subscription source database associated with create operation of this database. Optional.
    </ResponseField>
    <ResponseField name="cross_subscription_target_managed_instance_id" type="string">
      Target managed instance id used in cross-subscription restore. Optional.
    </ResponseField>
    <ResponseField name="default_secondary_location" type="string">
      Geo paired region. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="earliest_restore_point" type="string">
      Earliest restore point in time for point in time restore. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="extended_accessibility_info" type="object">
      Managed Database Extended Accessibility Information Optional.
      <Expandable title="properties">
        <ResponseField name="inaccessibility_reason_description" type="string">
          Root cause explanation and mitigation action. Computed by Microsoft Azure after `ship`.
        </ResponseField>
        <ResponseField name="inaccessibility_reason_error_code" type="string">
          SQL Server error code connected to the inaccessibility root cause. Computed by Microsoft Azure after `ship`.
        </ResponseField>
        <ResponseField name="inaccessibility_reason_kind" type="string" required>
          Root cause kind. Allowed values are “TransparentDataEncryption”, “DatabaseReplication”, and “Unknown”. Required.
        </ResponseField>
        <ResponseField name="inaccessibility_reason_tde_key_uri" type="string">
          For the root cause kind “TransparentDataEncryption”, the CMK URI. Computed by Microsoft Azure after `ship`.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="failover_group_id" type="string">
      Instance Failover Group resource identifier that this managed database belongs to. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="is_ledger_on" type="bool">
      Whether or not this database is a ledger database, which means all tables in the database are ledger tables. Note: the value of this property cannot be changed after the database has been created. Optional.
    </ResponseField>
    <ResponseField name="last_backup_name" type="string">
      Last backup file name for restore of this managed database. Optional.
    </ResponseField>
    <ResponseField name="long_term_retention_backup_resource_id" type="string">
      The name of the Long Term Retention backup to be used for restore of this managed database. Optional.
    </ResponseField>
    <ResponseField name="recoverable_database_id" type="string">
      The resource identifier of the recoverable database associated with create operation of this database. Optional.
    </ResponseField>
    <ResponseField name="restorable_dropped_database_id" type="string">
      The restorable dropped database resource id to restore when creating this database. Optional.
    </ResponseField>
    <ResponseField name="restore_point_in_time" type="string">
      Conditional. If createMode is PointInTimeRestore, this value is required. Specifies the point in time (ISO8601 format) of the source database that will be restored to create the new database. Optional.
    </ResponseField>
    <ResponseField name="source_database_id" type="string">
      The resource identifier of the source database associated with create operation of this database. Optional.
    </ResponseField>
    <ResponseField name="status" type="string">
      Status of the database. Optional.
    </ResponseField>
    <ResponseField name="storage_container_identity" type="string">
      Conditional. If createMode is RestoreExternalBackup, this value is used. Specifies the identity used for storage container authentication. Can be 'SharedAccessSignature' or 'ManagedIdentity'; if not specified 'SharedAccessSignature' is assumed. Optional.
    </ResponseField>
    <ResponseField name="storage_container_sas_token" type="string">
      Conditional. If createMode is RestoreExternalBackup and storageContainerIdentity is not ManagedIdentity, this value is required. Specifies the storage container sas token. Optional.
    </ResponseField>
    <ResponseField name="storage_container_uri" type="string">
      Conditional. If createMode is RestoreExternalBackup, this value is required. Specifies the uri of the storage container where backups for this restore are stored. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
