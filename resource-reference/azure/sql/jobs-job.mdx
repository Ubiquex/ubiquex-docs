## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_sql_jobs --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_sql_jobs/sdk/go
    package main

    import (
    	sql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_sql_jobs/sql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own jobs job"})

    		ubx.Resource(sql.JobsJob, "ci-runner", sql.JobsJobConfig{})
    	}))
    }
    ```
