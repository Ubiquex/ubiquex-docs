## Properties

<ResponseField name="properties" type="object">
  Properties of a security alert policy. Optional; if omitted, computed by Microsoft Azure.
  <Expandable title="properties">
    <ResponseField name="creation_time" type="string">
      Specifies the UTC creation time of the policy. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="disabled_alerts" type="list(string)">
      Specifies an array of alerts that are disabled. Allowed values are: Sql_Injection, Sql_Injection_Vulnerability, Access_Anomaly, Data_Exfiltration, Unsafe_Action, Brute_Force Optional.
    </ResponseField>
    <ResponseField name="email_account_admins" type="bool">
      Specifies that the alert is sent to the account administrators. Optional.
    </ResponseField>
    <ResponseField name="email_addresses" type="list(string)">
      Specifies an array of e-mail addresses to which the alert is sent. Optional.
    </ResponseField>
    <ResponseField name="retention_days" type="number">
      Specifies the number of days to keep in the Threat Detection audit logs. Optional.
    </ResponseField>
    <ResponseField name="state" type="string" required>
      Specifies the state of the policy, whether it is enabled or disabled or a policy has not been applied yet on the specific database. Required.
    </ResponseField>
    <ResponseField name="storage_account_access_key" type="string">
      Specifies the identifier key of the Threat Detection audit storage account. Optional.
    </ResponseField>
    <ResponseField name="storage_endpoint" type="string">
      Specifies the blob storage endpoint (e.g. https://MyAccount.blob.core.windows.net). This blob storage will hold all Threat Detection audit logs. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
