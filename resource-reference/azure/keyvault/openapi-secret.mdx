## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_keyvault_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_keyvault_openapi/sdk/go
    package main

    import (
    	keyvault "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_keyvault_openapi/keyvault"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi secret"})

    		ubx.Resource(keyvault.OpenapiSecret, "example", keyvault.OpenapiSecretConfig{
    			Properties: map[string]any{"attributes": map[string]any{"created": 1}},
    		})
    	}))
    }
    ```
