`azure_notificationhubs_namespace_resource` is a Notification Hubs namespace, the management container that notification hub(s) are created within.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	notificationhubs "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/notificationhubs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own namespace resource"})

    		ubx.Resource(notificationhubs.NamespaceResource, "example", notificationhubs.NamespaceResourceConfig{
    			Sku: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { NamespaceResource } from "@ubx/sdk-azure/azure/notificationhubs/namespace_resource";

    export default stack("example", () => {
      intent({ summary: "example own namespace resource" });

      resource(NamespaceResource, "example", {
        sku: { name: "example-name" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.notificationhubs import NamespaceResource, NamespaceResourceConfig
    from ubx.azure.notificationhubs.namespace_resource import NamespaceResource_Sku

    def describe():
        ubx.intent("example own namespace resource")

        ubx.resource(NamespaceResource, "example", NamespaceResourceConfig(
            sku=NamespaceResource_Sku(name="example-name"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
