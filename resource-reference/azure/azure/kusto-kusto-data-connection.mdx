`azure_azure_kusto_kusto_data_connection` continuously ingests data from an external source (Event Hub, IoT Hub, Event Grid) into an Azure Data Explorer database table.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_azure_kusto_kusto --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_azure_kusto_kusto/sdk/go
    package main

    import (
    	azure "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_azure_kusto_kusto/azure"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own kusto kusto data connection"})

    		ubx.Resource(azure.KustoKustoDataConnection, "ci-runner", azure.KustoKustoDataConnectionConfig{
    			Kind: "example",
    		})
    	}))
    }
    ```
