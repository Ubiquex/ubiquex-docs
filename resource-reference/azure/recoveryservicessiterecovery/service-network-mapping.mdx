`azure_recoveryservicessiterecovery_service_network_mapping` maps a source network to a target network in Azure Site Recovery, so failed-over VM(s) automatically land on the correct target network.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_recoveryservicessiterecovery_service --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_recoveryservicessiterecovery_service/sdk/go
    package main

    import (
    	recoveryservicessiterecovery "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_recoveryservicessiterecovery_service/recoveryservicessiterecovery"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own service network mapping"})

    		ubx.Resource(recoveryservicessiterecovery.ServiceNetworkMapping, "ci-runner", recoveryservicessiterecovery.ServiceNetworkMappingConfig{
    			Properties: map[string]any{"recovery_network_id": "example"},
    		})
    	}))
    }
    ```
  </Tab>
