`azure_recoveryservicessiterecovery_service_network_mapping` maps a source network to a target network in Azure Site Recovery, so failed-over VM(s) automatically land on the correct target network.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	recoveryservicessiterecovery "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/recoveryservicessiterecovery"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own service network mapping"})

    		ubx.Resource(recoveryservicessiterecovery.ServiceNetworkMapping, "example", recoveryservicessiterecovery.ServiceNetworkMappingConfig{
    			Properties: map[string]any{"recovery_network_id": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { ServiceNetworkMapping } from "@ubx/sdk-azure/azure/recoveryservicessiterecovery/service_network_mapping";

    export default stack("example", () => {
      intent({ summary: "example own service network mapping" });

      resource(ServiceNetworkMapping, "example", {
        properties: { recoveryNetworkId: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.recoveryservicessiterecovery import ServiceNetworkMapping, ServiceNetworkMappingConfig
    from ubx.azure.recoveryservicessiterecovery.service_network_mapping import ServiceNetworkMapping_Properties

    def describe():
        ubx.intent("example own service network mapping")

        ubx.resource(ServiceNetworkMapping, "example", ServiceNetworkMappingConfig(
            properties=ServiceNetworkMapping_Properties(recovery_network_id="example"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
