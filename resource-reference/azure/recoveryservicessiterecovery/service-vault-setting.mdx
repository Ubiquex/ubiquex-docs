`azure_recoveryservicessiterecovery_service_vault_setting` configures Recovery Services vault-level settings (e.g. migration solution details) used by Azure Site Recovery.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	recoveryservicessiterecovery "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/recoveryservicessiterecovery"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own service vault setting"})

    		ubx.Resource(recoveryservicessiterecovery.ServiceVaultSetting, "example", recoveryservicessiterecovery.ServiceVaultSettingConfig{
    			Properties: map[string]any{"migration_solution_id": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import { intent, resource, stack } from "@ubx/sdk";
