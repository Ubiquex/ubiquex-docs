`azure_recoveryservicessiterecovery_service_vault_setting` configures Recovery Services vault-level settings (e.g. migration solution details) used by Azure Site Recovery.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_recoveryservicessiterecovery_service --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_recoveryservicessiterecovery_service/sdk/go
    package main

    import (
    	recoveryservicessiterecovery "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_recoveryservicessiterecovery_service/recoveryservicessiterecovery"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own service vault setting"})

    		ubx.Resource(recoveryservicessiterecovery.ServiceVaultSetting, "ci-runner", recoveryservicessiterecovery.ServiceVaultSettingConfig{
    			Properties: map[string]any{"migration_solution_id": "example"},
    		})
    	}))
    }
    ```
