`azure_recoveryservicessiterecovery_service_migration_item` tracks a single machine undergoing agentless migration (rather than ongoing disaster recovery replication) into Azure through Azure Site Recovery's own migration capability.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	recoveryservicessiterecovery "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/recoveryservicessiterecovery"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own service migration item"})

    		ubx.Resource(recoveryservicessiterecovery.ServiceMigrationItem, "example", recoveryservicessiterecovery.ServiceMigrationItemConfig{
    			Properties: map[string]any{"policy_id": "example", "provider_specific_details": map[string]any{"instance_type": "example"}},
    		})
    	}))
    }
    ```
