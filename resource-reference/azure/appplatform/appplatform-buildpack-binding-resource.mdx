`azure_appplatform_appplatform_buildpack_binding_resource` supplies configuration or credentials (e.g. an APM key) that a specific buildpack needs at build time within Azure Spring Apps.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_appplatform_appplatform --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_appplatform_appplatform/sdk/go
    package main

    import (
    	appplatform "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_appplatform_appplatform/appplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own appplatform buildpack binding resource"})

    		ubx.Resource(appplatform.AppplatformBuildpackBindingResource, "ci-runner", appplatform.AppplatformBuildpackBindingResourceConfig{})
    	}))
    }
    ```
