`azure_appplatform_appplatform_gateway_custom_domain_resource` maps a custom domain to the Spring Cloud Gateway front door of an Azure Spring Apps instance.

    package main

    import (
    	appplatform "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_appplatform_appplatform/appplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
