`azure_appplatform_gateway_custom_domain_resource` maps a custom domain to the Spring Cloud Gateway front door of an Azure Spring Apps instance.

## Example

<Tabs>
  <Tab title="Go">

    package main

    import (
    	appplatform "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/appplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own appplatform gateway custom domain resource"})

    		ubx.Resource(appplatform.GatewayCustomDomainResource, "example", appplatform.GatewayCustomDomainResourceConfig{})
    	}))
    }
