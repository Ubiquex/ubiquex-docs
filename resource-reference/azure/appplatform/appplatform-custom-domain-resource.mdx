## Properties

<ResponseField name="properties" type="object">
  Custom domain of app resource payload. Optional; if omitted, computed by Microsoft Azure.
  <Expandable title="properties">
    <ResponseField name="app_name" type="string">
      The app name of domain. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="cert_name" type="string">
      The bound certificate name of domain. Optional.
    </ResponseField>
    <ResponseField name="provisioning_state" type="string">
      Provisioning state of the Domain Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="thumbprint" type="string">
      The thumbprint of bound certificate. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
