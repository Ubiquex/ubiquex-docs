`azure_appplatform_appplatform_api_portal_custom_domain_resource` maps a custom domain to an Azure Spring Apps API portal, the self-service API documentation and testing UI for services in the app.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_appplatform_appplatform --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_appplatform_appplatform/sdk/go
    package main

    import (
    	appplatform "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_appplatform_appplatform/appplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own appplatform api portal custom domain resource"})

    		ubx.Resource(appplatform.AppplatformApiPortalCustomDomainResource, "ci-runner", appplatform.AppplatformApiPortalCustomDomainResourceConfig{})
    	}))
    }
    ```
