`azure_appplatform_api_portal_custom_domain_resource` maps a custom domain to an Azure Spring Apps API portal, the self-service API documentation and testing UI for services in the app.

## Example

<Tabs>

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own appplatform api portal custom domain resource"})

    		ubx.Resource(appplatform.ApiPortalCustomDomainResource, "example", appplatform.ApiPortalCustomDomainResourceConfig{})
    	}))
    }
