`azure_loadtestservice_playwright_playwright_workspace` is a Playwright Workspace under Azure App Testing (the GA successor to the now-retired Microsoft Playwright Testing service, retired 2026-03-08), providing managed cloud browsers for running Playwright end-to-end tests at scale.

    package main

    import (
    	loadtestservice "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_loadtestservice_playwright/loadtestservice"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own playwright playwright workspace"})

    		ubx.Resource(loadtestservice.PlaywrightPlaywrightWorkspace, "ci-runner", loadtestservice.PlaywrightPlaywrightWorkspaceConfig{})

    }

  Playwright workspace resource properties. Optional; if omitted, computed by Microsoft Azure.
  <Expandable title="properties">
    <ResponseField name="dataplane_uri" type="string">
      The workspace data plane service API URI. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="local_auth" type="string">
      Enables the workspace to use local authentication through service access tokens for operations. Optional.
    </ResponseField>
    <ResponseField name="provisioning_state" type="string">
      The status of the last resource operation. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="regional_affinity" type="string">
      Controls the connection region for client workers to cloud-hosted browsers. When enabled, workers connect to browsers in the closest Azure region for lower latency. When disabled, workers connect to browsers in the Azure region where the workspace was created. Optional.
    </ResponseField>
    <ResponseField name="workspace_id" type="string">
      The workspace ID in GUID format. Computed by Microsoft Azure after `ship`.
    </ResponseField>
