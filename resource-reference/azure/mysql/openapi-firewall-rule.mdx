`azure_mysql_openapi_firewall_rule` allows a specific IP address range to connect to an Azure Database for MySQL flexible server that hasn't been placed inside a virtual network.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	mysql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/mysql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi firewall rule"})

    		ubx.Resource(mysql.OpenapiFirewallRule, "example", mysql.OpenapiFirewallRuleConfig{
    			Properties: map[string]any{"end_ip_address": "example", "start_ip_address": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiFirewallRule } from "@ubx/sdk-azure/azure/mysql/openapi_firewall_rule";

    export default stack("example", () => {
      intent({ summary: "example own openapi firewall rule" });

      resource(OpenapiFirewallRule, "example", {

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.mysql import OpenapiFirewallRule, OpenapiFirewallRuleConfig
    from ubx.azure.mysql.openapi_firewall_rule import OpenapiFirewallRule_Properties

    def describe():
        ubx.intent("example own openapi firewall rule")

        ubx.resource(OpenapiFirewallRule, "example", OpenapiFirewallRuleConfig(
            properties=OpenapiFirewallRule_Properties(end_ip_address="example", start_ip_address="example"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

  <Expandable title="properties">

    <ResponseField name="start_ip_address" type="string" required>
