`azure_mysql_openapi_firewall_rule` allows a specific IP address range to connect to an Azure Database for MySQL flexible server that hasn't been placed inside a virtual network.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_mysql_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_mysql_openapi/sdk/go
    package main

    import (
    	mysql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_mysql_openapi/mysql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi firewall rule"})

    		ubx.Resource(mysql.OpenapiFirewallRule, "example", mysql.OpenapiFirewallRuleConfig{
    			Properties: map[string]any{"end_ip_address": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
