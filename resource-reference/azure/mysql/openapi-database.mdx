## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_mysql_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_mysql_openapi/sdk/go
    package main

    import (
    	mysql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_mysql_openapi/mysql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi database"})

    		ubx.Resource(mysql.OpenapiDatabase, "ci-runner", mysql.OpenapiDatabaseConfig{})
    	}))
    }
    ```
