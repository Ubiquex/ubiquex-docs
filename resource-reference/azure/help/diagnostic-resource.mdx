## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	help "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/help"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own diagnostic resource"})

    		ubx.Resource(help.DiagnosticResource, "example", help.DiagnosticResourceConfig{
    			Scope: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { DiagnosticResource } from "@ubx/sdk-azure/azure/help/diagnostic_resource";

    export default stack("example", () => {
      intent({ summary: "example own diagnostic resource" });

      resource(DiagnosticResource, "example", {
        scope: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.help import DiagnosticResource, DiagnosticResourceConfig

    def describe():
        ubx.intent("example own diagnostic resource")

        ubx.resource(DiagnosticResource, "example", DiagnosticResourceConfig(
            scope="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
  <Tab title="Markdown">
    ```
    Create azure_help_diagnostic_resource, with scope example.
    ```
  </Tab>
</Tabs>
