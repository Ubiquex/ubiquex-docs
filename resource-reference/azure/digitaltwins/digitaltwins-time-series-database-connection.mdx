`azure_digitaltwins_digitaltwins_time_series_database_connection` continuously streams Azure Digital Twins twin property change(s) into a connected Azure Data Explorer database for historical time-series analysis.

    package main

    import (
    	digitaltwins "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_digitaltwins_digitaltwins/digitaltwins"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
