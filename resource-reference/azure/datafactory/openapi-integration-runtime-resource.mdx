`azure_datafactory_openapi_integration_runtime_resource` is the compute infrastructure (Azure-hosted, self-hosted, or Azure-SSIS) a Data Factory pipeline's own activities actually execute on.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_datafactory_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_datafactory_openapi/sdk/go
    package main

    import (
    	datafactory "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_datafactory_openapi/datafactory"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi integration runtime resource"})

    		ubx.Resource(datafactory.OpenapiIntegrationRuntimeResource, "example", datafactory.OpenapiIntegrationRuntimeResourceConfig{
    			Properties: map[string]any{"type": "example"},
    		})
    	}))
    }
    ```
