`azure_datafactory_openapi_trigger_resource` defines when a Data Factory pipeline runs automatically -- on a schedule, in response to a storage event, or when a tumbling window elapses.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_datafactory_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_datafactory_openapi/sdk/go
    package main

    import (
    	datafactory "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_datafactory_openapi/datafactory"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi trigger resource"})

    		ubx.Resource(datafactory.OpenapiTriggerResource, "example", datafactory.OpenapiTriggerResourceConfig{
    			Properties: map[string]any{"type": "example"},
    		})
    	}))
    }
    ```
