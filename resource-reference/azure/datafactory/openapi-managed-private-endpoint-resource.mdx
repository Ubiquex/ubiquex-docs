`azure_datafactory_openapi_managed_private_endpoint_resource` creates a Private Endpoint within a Data Factory's own managed virtual network, letting it privately reach a data source without traversing the public internet.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	datafactory "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/datafactory"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi managed private endpoint resource"})

    		ubx.Resource(datafactory.OpenapiManagedPrivateEndpointResource, "example", datafactory.OpenapiManagedPrivateEndpointResourceConfig{
    			Properties: map[string]any{"connection_state": map[string]any{"actions_required": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
