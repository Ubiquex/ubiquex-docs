`azure_datafactory_openapi_global_parameter_resource` defines a factory-wide parameter value referenceable from any pipeline within a Data Factory, without redeclaring it per pipeline.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_datafactory_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_datafactory_openapi/sdk/go
    package main

    import (
    	datafactory "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_datafactory_openapi/datafactory"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi global parameter resource"})

    		ubx.Resource(datafactory.OpenapiGlobalParameterResource, "ci-runner", datafactory.OpenapiGlobalParameterResourceConfig{
    			Properties: map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```
