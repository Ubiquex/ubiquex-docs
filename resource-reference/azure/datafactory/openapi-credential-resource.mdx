`azure_datafactory_openapi_credential_resource` stores a reusable credential (a managed identity or service principal) within a Data Factory, referenced by linked service(s) needing to authenticate.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	datafactory "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/datafactory"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi credential resource"})

    		ubx.Resource(datafactory.OpenapiCredentialResource, "example", datafactory.OpenapiCredentialResourceConfig{
    			Properties: map[string]any{"type": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiCredentialResource } from "@ubx/sdk-azure/azure/datafactory/openapi_credential_resource";

    export default stack("example", () => {
      intent({ summary: "example own openapi credential resource" });

      resource(OpenapiCredentialResource, "example", {
        properties: { type: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.datafactory import OpenapiCredentialResource, OpenapiCredentialResourceConfig

    def describe():
        ubx.intent("example own openapi credential resource")

        ubx.resource(OpenapiCredentialResource, "example", OpenapiCredentialResourceConfig(
            properties={"type": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
