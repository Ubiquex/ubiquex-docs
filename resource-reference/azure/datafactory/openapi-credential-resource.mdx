`azure_datafactory_openapi_credential_resource` stores a reusable credential (a managed identity or service principal) within a Data Factory, referenced by linked service(s) needing to authenticate.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_datafactory_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_datafactory_openapi/sdk/go
    package main

    import (
    	datafactory "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_datafactory_openapi/datafactory"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi credential resource"})

    		ubx.Resource(datafactory.OpenapiCredentialResource, "ci-runner", datafactory.OpenapiCredentialResourceConfig{
    			Properties: map[string]any{"type": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_datafactory_openapi --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiCredentialResource } from "./local-sdk/azure_datafactory_openapi/sdk/typescript/azure_datafactory_openapi/datafactory/openapi_credential_resource.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own openapi credential resource" });

      resource(OpenapiCredentialResource, "ci-runner", {
        properties: { type: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_datafactory_openapi --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_datafactory_openapi/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_datafactory_openapi.datafactory import OpenapiCredentialResource, OpenapiCredentialResourceConfig

    def describe():
        ubx.intent("payments own openapi credential resource")

        ubx.resource(OpenapiCredentialResource, "ci-runner", OpenapiCredentialResourceConfig(
            properties={"type": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
