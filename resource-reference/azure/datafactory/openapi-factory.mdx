<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_datafactory_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_datafactory_openapi/sdk/go
    package main

    import (
    	datafactory "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_datafactory_openapi/datafactory"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi factory"})

    		ubx.Resource(datafactory.OpenapiFactory, "example", datafactory.OpenapiFactoryConfig{
    			FactoryResourceId: "example",
    			RepoConfiguration: map[string]any{"account_name": "example-account-name"},
    		})
    	}))
    }
    ```
