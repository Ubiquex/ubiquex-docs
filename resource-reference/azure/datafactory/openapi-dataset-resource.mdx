`azure_datafactory_openapi_dataset_resource` defines a named reference to data (a table, file, or folder) at a specific linked service, that pipeline activities read from or write to.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_datafactory_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_datafactory_openapi/sdk/go
    package main

    import (
    	datafactory "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_datafactory_openapi/datafactory"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi dataset resource"})

    		ubx.Resource(datafactory.OpenapiDatasetResource, "ci-runner", datafactory.OpenapiDatasetResourceConfig{
    			Properties: map[string]any{"linked_service_name": map[string]any{"reference_name": "example-reference-name"}},
    		})
    	}))
    }
    ```
