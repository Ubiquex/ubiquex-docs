`azure_graphservicesprod_account_resource` provisions a Microsoft Graph Data Connect account, enabling bulk export of Microsoft 365 data into Azure data stores for analytics.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	graphservicesprod "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/graphservicesprod"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own account resource"})

    		ubx.Resource(graphservicesprod.AccountResource, "example", graphservicesprod.AccountResourceConfig{
    			Properties: map[string]any{"app_id": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { AccountResource } from "@ubx/sdk-azure/azure/graphservicesprod/account_resource";

    export default stack("example", () => {
      intent({ summary: "example own account resource" });

      resource(AccountResource, "example", {
        properties: { appId: "example" },

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.graphservicesprod import AccountResource, AccountResourceConfig
    from ubx.azure.graphservicesprod.account_resource import AccountResource_Properties

    def describe():
        ubx.intent("example own account resource")

        ubx.resource(AccountResource, "example", AccountResourceConfig(
            properties=AccountResource_Properties(app_id="example"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
