    // ubx sdk gen --only azure_fabric --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_fabric/sdk/go
    package main

    import (
    	fabric "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_fabric/fabric"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own capacity"})

    			Properties: map[string]any{"administration": map[string]any{"members": []string{"example"}}},

    		})
    	}))
    }
