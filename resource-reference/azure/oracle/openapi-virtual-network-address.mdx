`azure_oracle_openapi_virtual_network_address` reserves an IP address from an Azure virtual network's own address space for use by an Oracle Database@Azure resource.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_oracle_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_oracle_openapi/sdk/go
    package main

    import (
    	oracle "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_oracle_openapi/oracle"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi virtual network address"})

    		ubx.Resource(oracle.OpenapiVirtualNetworkAddress, "ci-runner", oracle.OpenapiVirtualNetworkAddressConfig{
    			Cloudvmclustername:        "example",
    			Virtualnetworkaddressname: "example",
    		})
    	}))
    }
    ```
  </Tab>
