`azure_oracle_openapi_cloud_vm_cluster` provisions a cluster of VM(s) on Oracle Exadata infrastructure within Oracle Database@Azure, that Oracle database instance(s) run on.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_oracle_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_oracle_openapi/sdk/go
    package main

    import (
    	oracle "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_oracle_openapi/oracle"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi cloud vm cluster"})

    		ubx.Resource(oracle.OpenapiCloudVmCluster, "ci-runner", oracle.OpenapiCloudVmClusterConfig{
    			Cloudvmclustername: "example",
    		})
    	}))
    }
    ```
  </Tab>
