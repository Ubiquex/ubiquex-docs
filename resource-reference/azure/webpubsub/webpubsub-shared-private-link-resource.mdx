`azure_webpubsub_webpubsub_shared_private_link_resource` creates a Private Endpoint from a Web PubSub resource to another Private Link-enabled Azure service, so outbound event handler calls can stay private.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_webpubsub_webpubsub --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_webpubsub_webpubsub/sdk/go
    package main

    import (
    	webpubsub "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_webpubsub_webpubsub/webpubsub"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own webpubsub shared private link resource"})

    		ubx.Resource(webpubsub.WebpubsubSharedPrivateLinkResource, "ci-runner", webpubsub.WebpubsubSharedPrivateLinkResourceConfig{})
    	}))
    }
    ```
