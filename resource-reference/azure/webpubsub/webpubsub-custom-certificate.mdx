---
title: "azure_webpubsub_custom_certificate"
description: "`azure_webpubsub_custom_certificate` uploads a TLS certificate (referenced from Key Vault) to a Web PubSub resource, bindable to a custom domain."
---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_webpubsub_webpubsub --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_webpubsub_webpubsub/sdk/go
    package main

    import (
    	webpubsub "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_webpubsub_webpubsub/webpubsub"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own webpubsub custom certificate"})

    		ubx.Resource(webpubsub.WebpubsubCustomCertificate, "example", webpubsub.WebpubsubCustomCertificateConfig{
    			Properties: map[string]any{"key_vault_base_uri": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_webpubsub_webpubsub --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { WebpubsubCustomCertificate } from "./local-sdk/azure_webpubsub_webpubsub/sdk/typescript/azure_webpubsub_webpubsub/webpubsub/webpubsub_custom_certificate.ts";

    export default stack("example", () => {
      intent({ summary: "example own webpubsub custom certificate" });

      resource(WebpubsubCustomCertificate, "example", {
        properties: { keyVaultBaseUri: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_webpubsub_webpubsub --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_webpubsub_webpubsub/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_webpubsub_webpubsub.webpubsub import WebpubsubCustomCertificate, WebpubsubCustomCertificateConfig

    def describe():
        ubx.intent("example own webpubsub custom certificate")

        ubx.resource(WebpubsubCustomCertificate, "example", WebpubsubCustomCertificateConfig(
            properties={"key_vault_base_uri": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
