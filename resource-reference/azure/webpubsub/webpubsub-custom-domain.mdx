## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_webpubsub_webpubsub --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_webpubsub_webpubsub/sdk/go
    package main

    import (
    	webpubsub "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_webpubsub_webpubsub/webpubsub"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own webpubsub custom domain"})

    		ubx.Resource(webpubsub.WebpubsubCustomDomain, "ci-runner", webpubsub.WebpubsubCustomDomainConfig{
    			Name:       "ci-runner",
    			Properties: map[string]any{"custom_certificate": map[string]any{"id": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_webpubsub_webpubsub --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { WebpubsubCustomDomain } from "./local-sdk/azure_webpubsub_webpubsub/sdk/typescript/azure_webpubsub_webpubsub/webpubsub/webpubsub_custom_domain.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own webpubsub custom domain" });

      resource(WebpubsubCustomDomain, "ci-runner", {
        name: "ci-runner",
        properties: { customCertificate: { id: "example" } },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_webpubsub_webpubsub --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_webpubsub_webpubsub/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_webpubsub_webpubsub.webpubsub import WebpubsubCustomDomain, WebpubsubCustomDomainConfig

    def describe():
        ubx.intent("payments own webpubsub custom domain")

        ubx.resource(WebpubsubCustomDomain, "ci-runner", WebpubsubCustomDomainConfig(
            name="ci-runner",
            properties={"custom_certificate": {"id": "example"}},
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
