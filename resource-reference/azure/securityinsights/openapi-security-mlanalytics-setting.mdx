`azure_securityinsights_openapi_security_mlanalytics_setting` configures a Microsoft Sentinel built-in machine learning analytics detection (e.g. anomalous sign-in detection).

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi security mlanalytics setting"})

    		ubx.Resource(securityinsights.OpenapiSecurityMlanalyticsSetting, "example", securityinsights.OpenapiSecurityMlanalyticsSettingConfig{
