`azure_securityinsights_openapi_security_mlanalytics_setting` configures a Microsoft Sentinel built-in machine learning analytics detection (e.g. anomalous sign-in detection).

    import (

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi security mlanalytics setting"})

    		ubx.Resource(securityinsights.OpenapiSecurityMlanalyticsSetting, "ci-runner", securityinsights.OpenapiSecurityMlanalyticsSettingConfig{

    		})

    }
