`azure_securityinsights_openapi_action_response` defines a playbook (Logic App) that an alert rule or automation rule triggers to run automatically in Microsoft Sentinel.

    package main

    import (
    	securityinsights "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_securityinsights_openapi/securityinsights"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi action response"})

    	}))
    }
