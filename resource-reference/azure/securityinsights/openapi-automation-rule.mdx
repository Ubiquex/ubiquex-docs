`azure_securityinsights_openapi_automation_rule` automatically applies action(s) (assign an owner, run a playbook, close as false-positive) to Microsoft Sentinel incident(s) matching a defined condition.

## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_securityinsights_openapi/sdk/go
    package main

    import (
    	securityinsights "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_securityinsights_openapi/securityinsights"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi automation rule"})

    		ubx.Resource(securityinsights.OpenapiAutomationRule, "ci-runner", securityinsights.OpenapiAutomationRuleConfig{
    			Properties: map[string]any{"actions": []map[string]any{{"action_type": "example"}}},
    		})
    	}))
    }
