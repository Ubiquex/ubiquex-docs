`azure_securityinsights_openapi_data_connector_definition` defines the connection UI and configuration schema for a Microsoft Sentinel data connector, used when publishing a connector as reusable content.

## Example

<Tabs>

    ```go
    package main

    import (
    	securityinsights "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/securityinsights"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi data connector definition"})

    		ubx.Resource(securityinsights.OpenapiDataConnectorDefinition, "example", securityinsights.OpenapiDataConnectorDefinitionConfig{
    			Kind: "example",
    		})
    	}))
    }
    ```
