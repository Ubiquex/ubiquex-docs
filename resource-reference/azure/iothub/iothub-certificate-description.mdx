`azure_iothub_iothub_certificate_description` uploads a CA certificate to an IoT Hub, used to establish trust when verifying connecting device certificates.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_iothub_iothub --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_iothub_iothub/sdk/go
    package main

    import (
    	iothub "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_iothub_iothub/iothub"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own iothub certificate description"})

    		ubx.Resource(iothub.IothubCertificateDescription, "example", iothub.IothubCertificateDescriptionConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
