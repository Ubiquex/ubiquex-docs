`azure_servicefabric_application_application_type_resource` registers an application type with a Service Fabric cluster, the versioned template application instance(s) are created from.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_servicefabric_application --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_servicefabric_application/sdk/go
    package main

    import (
    	servicefabric "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_servicefabric_application/servicefabric"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own application application type resource"})

    		ubx.Resource(servicefabric.ApplicationApplicationTypeResource, "ci-runner", servicefabric.ApplicationApplicationTypeResourceConfig{})
    	}))
    }
    ```
