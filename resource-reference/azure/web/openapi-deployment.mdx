## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi deployment"})

    		ubx.Resource(web.OpenapiDeployment, "example", web.OpenapiDeploymentConfig{
    			Id:   "example",
    			Name: "example",
    		})
    	}))
    }
    ```
