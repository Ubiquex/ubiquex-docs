## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_web_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_web_openapi/sdk/go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_web_openapi/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi deployment"})

    		ubx.Resource(web.OpenapiDeployment, "ci-runner", web.OpenapiDeploymentConfig{
    			Id:   "example",
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_web_openapi --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiDeployment } from "./local-sdk/azure_web_openapi/sdk/typescript/azure_web_openapi/web/openapi_deployment.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own openapi deployment" });

      resource(OpenapiDeployment, "ci-runner", {
        id: "example",
        name: "ci-runner",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_web_openapi --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_web_openapi/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_web_openapi.web import OpenapiDeployment, OpenapiDeploymentConfig

    def describe():
        ubx.intent("payments own openapi deployment")

        ubx.resource(OpenapiDeployment, "ci-runner", OpenapiDeploymentConfig(
            id="example",
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
