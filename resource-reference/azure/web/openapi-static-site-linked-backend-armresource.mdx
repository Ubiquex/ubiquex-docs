`azure_web_openapi_static_site_linked_backend_armresource` links an external API backend (e.g. an App Service or Container App) to a Static Web App, exposed through its own integrated `/api` routing.

## Example

    package main

    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_web_openapi/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi static site linked backend armresource"})

    		ubx.Resource(web.OpenapiStaticSiteLinkedBackendArmresource, "ci-runner", web.OpenapiStaticSiteLinkedBackendArmresourceConfig{

    }
