## Input properties

<ResponseField name="name" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="properties" type="object">
  BackupRequest resource specific properties Optional; if omitted, computed by Microsoft Azure.
  <Expandable title="properties">
    <ResponseField name="backup_name" type="string">
      Name of the backup. Optional.
    </ResponseField>
    <ResponseField name="backup_schedule" type="object">
      Description of a backup schedule. Describes how often should be the backup performed and what should be the retention policy. Optional.
      <Expandable title="properties">
        <ResponseField name="frequency_interval" type="number" required>
          How often the backup should be executed (e.g. for weekly backup, this should be set to 7 and FrequencyUnit should be set to Day) Required.
        </ResponseField>
        <ResponseField name="frequency_unit" type="string" required>
          The unit of time for how often the backup should be executed (e.g. for weekly backup, this should be set to Day and FrequencyInterval should be set to 7) Required.
        </ResponseField>
        <ResponseField name="keep_at_least_one_backup" type="bool" required>
          True if the retention policy should always keep at least one backup in the storage account, regardless how old it is; false otherwise. Required.
        </ResponseField>
        <ResponseField name="last_execution_time" type="string">
          Last time when this schedule was triggered. Computed by Microsoft Azure after `ship`.
        </ResponseField>
        <ResponseField name="retention_period_in_days" type="number" required>
          After how many days backups should be deleted. Required.
        </ResponseField>
        <ResponseField name="start_time" type="string">
          When the schedule should start working. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="databases" type="list(object)">
      Databases included in the backup. Optional.
      <Expandable title="properties">
        <ResponseField name="connection_string" type="string">
          The connection string used to reach this resource. Optional; if omitted, computed by Microsoft Azure. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="connection_string_name" type="string">
          The name of the app setting or connection string this applies to. Optional; if omitted, computed by Microsoft Azure. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="database_type" type="string">
          The kind of database this connection targets, e.g. `SQLAzure` or `MySql`. Optional; if omitted, computed by Microsoft Azure. **(AI-inferred)**
        </ResponseField>
        <ResponseField name="name" type="string">
          The resource's own name, unique within its parent scope. Optional; if omitted, computed by Microsoft Azure. **(AI-inferred)**
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="enabled" type="bool">
      True if the backup schedule is enabled (must be included in that case), false if the backup schedule should be disabled. Optional.
    </ResponseField>
    <ResponseField name="storage_account_url" type="string" required>
      SAS URL to the container. Required.
    </ResponseField>
  </Expandable>
</ResponseField>
