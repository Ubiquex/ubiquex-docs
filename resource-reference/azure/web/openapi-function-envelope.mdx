`azure_web_openapi_function_envelope` describes a single function within an Azure Functions app hosted on App Service infrastructure, including its own trigger, bindings, and configuration.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_web_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_web_openapi/sdk/go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_web_openapi/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi function envelope"})

    		ubx.Resource(web.OpenapiFunctionEnvelope, "example", web.OpenapiFunctionEnvelopeConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
