`azure_web_openapi_ai_gateway` configures the AI Gateway tier of Azure API Management (public preview, 2026), a dedicated gateway for AI traffic that fronts model and MCP server endpoints behind one governed entry point with policies like semantic caching and token rate limiting.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_web_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_web_openapi/sdk/go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_web_openapi/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi ai gateway"})

    		ubx.Resource(web.OpenapiAiGateway, "ci-runner", web.OpenapiAiGatewayConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
