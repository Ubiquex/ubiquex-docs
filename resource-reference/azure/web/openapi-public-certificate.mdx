`azure_web_openapi_public_certificate` uploads a public certificate (no private key) to an App Service web app, made available to the app's own code for validating outbound connections.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_web_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_web_openapi/sdk/go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_web_openapi/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi public certificate"})

    		ubx.Resource(web.OpenapiPublicCertificate, "example", web.OpenapiPublicCertificateConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
