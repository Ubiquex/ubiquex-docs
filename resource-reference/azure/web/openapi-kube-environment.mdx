`azure_web_openapi_kube_environment` is the legacy App Service Kubernetes Environment resource, an earlier Azure Arc-enabled Kubernetes hosting environment for App Service and Functions, since superseded by newer Container Apps-based hosting.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_web_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_web_openapi/sdk/go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_web_openapi/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi kube environment"})

    		ubx.Resource(web.OpenapiKubeEnvironment, "example", web.OpenapiKubeEnvironmentConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
