## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_web_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_web_openapi/sdk/go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_web_openapi/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi vnet gateway"})

    		ubx.Resource(web.OpenapiVnetGateway, "ci-runner", web.OpenapiVnetGatewayConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_web_openapi --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiVnetGateway } from "./local-sdk/azure_web_openapi/sdk/typescript/azure_web_openapi/web/openapi_vnet_gateway.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own openapi vnet gateway" });

      resource(OpenapiVnetGateway, "ci-runner", {
        name: "ci-runner",
      });
    });
    ```

    ```python
    # ubx sdk gen --only azure_web_openapi --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_web_openapi/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_web_openapi.web import OpenapiVnetGateway, OpenapiVnetGatewayConfig

    def describe():
        ubx.intent("payments own openapi vnet gateway")

        ubx.resource(OpenapiVnetGateway, "ci-runner", OpenapiVnetGatewayConfig(
            name="ci-runner",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
