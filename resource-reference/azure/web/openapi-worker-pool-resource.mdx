`azure_web_openapi_worker_pool_resource` defines a named pool of worker compute within an App Service Environment, that web app(s) can be scaled onto independently of the environment's own front-end capacity.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_web_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_web_openapi/sdk/go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_web_openapi/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi worker pool resource"})

    		ubx.Resource(web.OpenapiWorkerPoolResource, "ci-runner", web.OpenapiWorkerPoolResourceConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
