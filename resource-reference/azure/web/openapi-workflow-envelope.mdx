`azure_web_openapi_workflow_envelope` describes a single Logic Apps (Standard) workflow hosted within an App Service-based Logic App resource, including its own definition and run history metadata.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_web_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_web_openapi/sdk/go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_web_openapi/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi workflow envelope"})

    		ubx.Resource(web.OpenapiWorkflowEnvelope, "example", web.OpenapiWorkflowEnvelopeConfig{
    			Name: "example",
    			Slot: "example",
    		})
    	}))
    }
    ```
