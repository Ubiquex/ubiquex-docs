## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_web_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_web_openapi/sdk/go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_web_openapi/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi database connection"})

    		ubx.Resource(web.OpenapiDatabaseConnection, "ci-runner", web.OpenapiDatabaseConnectionConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.azure_web_openapi.web import OpenapiDatabaseConnection, OpenapiDatabaseConnectionConfig

    if __name__ == "__main__":
