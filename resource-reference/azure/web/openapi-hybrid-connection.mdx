`azure_web_openapi_hybrid_connection` establishes outbound-only connectivity from an App Service web app to an on-premises or other-network TCP endpoint, without requiring inbound firewall changes.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_web_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_web_openapi/sdk/go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_web_openapi/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi hybrid connection"})

    		ubx.Resource(web.OpenapiHybridConnection, "example", web.OpenapiHybridConnectionConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
