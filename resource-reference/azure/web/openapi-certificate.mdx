## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_web_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_web_openapi/sdk/go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_web_openapi/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi certificate"})

    		ubx.Resource(web.OpenapiCertificate, "example", web.OpenapiCertificateConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_web_openapi --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiCertificate } from "./local-sdk/azure_web_openapi/sdk/typescript/azure_web_openapi/web/openapi_certificate.ts";

    export default stack("example", () => {
      intent({ summary: "example own openapi certificate" });

      resource(OpenapiCertificate, "example", {
        name: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_web_openapi --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_web_openapi/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_web_openapi.web import OpenapiCertificate, OpenapiCertificateConfig

    def describe():
        ubx.intent("example own openapi certificate")

        ubx.resource(OpenapiCertificate, "example", OpenapiCertificateConfig(
            name="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
