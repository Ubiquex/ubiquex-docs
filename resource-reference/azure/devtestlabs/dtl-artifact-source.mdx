`azure_devtestlabs_dtl_artifact_source` is a source repository (e.g. a Git repo) DevTest Labs pulls reusable artifact definitions from, for installation onto lab virtual machines.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_devtestlabs_dtl --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_devtestlabs_dtl/sdk/go
    package main

    import (
    	devtestlabs "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_devtestlabs_dtl/devtestlabs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dtl artifact source"})

    		ubx.Resource(devtestlabs.DtlArtifactSource, "example", devtestlabs.DtlArtifactSourceConfig{
    			Name:       "example",
    			Properties: map[string]any{"arm_template_folder_path": "example"},
    		})
    	}))
    }
    ```
