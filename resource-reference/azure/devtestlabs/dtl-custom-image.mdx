`azure_devtestlabs_dtl_custom_image` is a reusable VM image captured from a configured lab virtual machine, usable to create new lab VMs pre-loaded with the same software and configuration.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_devtestlabs_dtl --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_devtestlabs_dtl/sdk/go
    package main

    import (
    	devtestlabs "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_devtestlabs_dtl/devtestlabs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own dtl custom image"})

    		ubx.Resource(devtestlabs.DtlCustomImage, "ci-runner", devtestlabs.DtlCustomImageConfig{
    			Name:       "ci-runner",
    			Properties: map[string]any{"author": "example"},
    		})
    	}))
    }
    ```
