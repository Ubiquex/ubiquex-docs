`azure_devtestlabs_dtl_secret` is a per-user secret value (e.g. a credential) stored within a lab, scoped to that individual lab user rather than shared across the lab.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_devtestlabs_dtl --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_devtestlabs_dtl/sdk/go
    package main

    import (
    	devtestlabs "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_devtestlabs_dtl/devtestlabs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dtl secret"})

    		ubx.Resource(devtestlabs.DtlSecret, "example", devtestlabs.DtlSecretConfig{
    			Name:       "example",
    			Properties: map[string]any{"provisioning_state": "example"},
    		})
    	}))
    }
    ```
