## Example

    ```go
    package main

    import (
    	devtestlabs "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/devtestlabs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dtl user"})

    		ubx.Resource(devtestlabs.DtlUser, "example", devtestlabs.DtlUserConfig{
    			Name:       "example",
    			Properties: map[string]any{"created_date": "example"},
    		})
    	}))
    }
    ```
