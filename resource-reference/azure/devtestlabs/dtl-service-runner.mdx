`azure_devtestlabs_dtl_service_runner` is a managed identity-bearing container DevTest Labs uses to run lab automation tasks (like artifact installation or autoshutdown) against a lab virtual machine.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_devtestlabs_dtl --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_devtestlabs_dtl/sdk/go
    package main

    import (
    	devtestlabs "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_devtestlabs_dtl/devtestlabs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own dtl service runner"})

    		ubx.Resource(devtestlabs.DtlServiceRunner, "ci-runner", devtestlabs.DtlServiceRunnerConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
