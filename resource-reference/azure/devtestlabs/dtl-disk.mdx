`azure_devtestlabs_dtl_disk` is a data disk attached to a lab virtual machine, managed as its own DevTest Labs resource independent of the VM's own OS disk.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_devtestlabs_dtl --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_devtestlabs_dtl/sdk/go
    package main

    import (
    	devtestlabs "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_devtestlabs_dtl/devtestlabs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dtl disk"})

    		ubx.Resource(devtestlabs.DtlDisk, "example", devtestlabs.DtlDiskConfig{
    			Name:       "example",
    			Properties: map[string]any{"created_date": "example"},
    		})
    	}))
    }
    ```
