`azure_devtestlabs_dtl_lab` is a DevTest Lab, the top-level container for a team's own set of development/test virtual machines, environments, artifacts, and cost/policy controls.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	devtestlabs "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/devtestlabs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dtl lab"})

    		ubx.Resource(devtestlabs.DtlLab, "example", devtestlabs.DtlLabConfig{
    			Name:       "example",
    			Properties: map[string]any{"announcement": map[string]any{"enabled": "example"}},
    		})
    	}))
    }
    ```
