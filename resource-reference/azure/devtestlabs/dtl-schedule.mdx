`azure_devtestlabs_dtl_schedule` is an automated schedule within a lab, most commonly an auto-shutdown or auto-start time applied to a lab virtual machine or the whole lab.

## Example

    import (

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dtl schedule"})

    		ubx.Resource(devtestlabs.DtlSchedule, "example", devtestlabs.DtlScheduleConfig{
    			Name:       "example",
    			Properties: map[string]any{"created_date": "example"},
    		})
    	}))
    }
