`azure_devtestlabs_dtl_lab_virtual_machine` is a virtual machine created within a DevTest Lab, distinct from a plain Azure VM in that it's subject to the lab's own cost, schedule, and access policies.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	devtestlabs "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/devtestlabs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dtl lab virtual machine"})

    		ubx.Resource(devtestlabs.DtlLabVirtualMachine, "example", devtestlabs.DtlLabVirtualMachineConfig{
    			Name:       "example",
    			Properties: map[string]any{"allow_claim": true},
    		})
    	}))
    }
    ```
  </Tab>
