---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_chaos_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_chaos_openapi/sdk/go
    package main

    import (
    	chaos "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_chaos_openapi/chaos"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi target"})

    		ubx.Resource(chaos.OpenapiTarget, "ci-runner", chaos.OpenapiTargetConfig{
    			Properties: map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_chaos_openapi --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiTarget } from "./local-sdk/azure_chaos_openapi/sdk/typescript/azure_chaos_openapi/chaos/openapi_target.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own openapi target" });

      resource(OpenapiTarget, "ci-runner", {
        properties: { "managed-by": "ubx" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_chaos_openapi --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_chaos_openapi/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_chaos_openapi.chaos import OpenapiTarget, OpenapiTargetConfig

    def describe():
        ubx.intent("payments own openapi target")

        ubx.resource(OpenapiTarget, "ci-runner", OpenapiTargetConfig(
            properties={"managed-by": "ubx"},
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
