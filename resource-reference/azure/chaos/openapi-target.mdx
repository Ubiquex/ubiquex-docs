---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	chaos "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/chaos"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi target"})

    		ubx.Resource(chaos.OpenapiTarget, "example", chaos.OpenapiTargetConfig{
    			Properties: map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiTarget } from "@ubx/sdk-azure/azure/chaos/openapi_target";

    export default stack("example", () => {
      intent({ summary: "example own openapi target" });

      resource(OpenapiTarget, "example", {
        properties: { "managed-by": "ubx" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.chaos import OpenapiTarget, OpenapiTargetConfig

    def describe():
        ubx.intent("example own openapi target")

        ubx.resource(OpenapiTarget, "example", OpenapiTargetConfig(
            properties={"managed-by": "ubx"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
