`azure_chaos_openapi_capability` enables a specific fault-injection action type (e.g. a network latency or CPU pressure fault) on an Azure Chaos Studio target.

    package main

    import (
    	chaos "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_chaos_openapi/chaos"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi capability"})

    		ubx.Resource(chaos.OpenapiCapability, "example", chaos.OpenapiCapabilityConfig{})
    	}))
    }
