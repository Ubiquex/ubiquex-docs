`azure_healthbot_health_bot` provisions a Microsoft Healthcare Bot instance, a conversational AI service purpose-built for healthcare organizations to build compliant symptom-checking and triage bots.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_healthbot_healthbot --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_healthbot_healthbot/sdk/go
    package main

    import (
    	healthbot "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_healthbot_healthbot/healthbot"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own healthbot health bot"})

    		ubx.Resource(healthbot.HealthbotHealthBot, "example", healthbot.HealthbotHealthBotConfig{
    			Sku: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_healthbot_healthbot --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";

    export default stack("example", () => {

    });
    ```
  </Tab>

    # ubx sdk gen --only azure_healthbot_healthbot --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_healthbot_healthbot/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx

    if __name__ == "__main__":
        ubx.run("example", describe)
