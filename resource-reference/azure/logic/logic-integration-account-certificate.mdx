`azure_logic_integration_account_certificate` stores a certificate within a Logic Apps integration account, used to sign or encrypt B2B message exchanges.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	logic "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/logic"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own integration account certificate"})

    		ubx.Resource(logic.IntegrationAccountCertificate, "example", logic.IntegrationAccountCertificateConfig{
    			Properties: map[string]any{"key": map[string]any{"key_name": "example-key-name", "key_vault": map[string]any{"id": "example"}}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { IntegrationAccountCertificate } from "@ubx/sdk-azure/azure/logic/integration_account_certificate";

    export default stack("example", () => {
      intent({ summary: "example own integration account certificate" });

      resource(IntegrationAccountCertificate, "example", {
        properties: {
          key: { keyName: "example-key-name", keyVault: { id: "example" } },
        },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.logic import IntegrationAccountCertificate, IntegrationAccountCertificateConfig
    from ubx.azure.logic.integration_account_certificate import IntegrationAccountCertificate_Properties, IntegrationAccountCertificate_Properties_Key, IntegrationAccountCertificate_Properties_Key_KeyVault

    def describe():
        ubx.intent("example own integration account certificate")

        ubx.resource(IntegrationAccountCertificate, "example", IntegrationAccountCertificateConfig(
            properties=IntegrationAccountCertificate_Properties(key=IntegrationAccountCertificate_Properties_Key(key_name="example-key-name", key_vault=IntegrationAccountCertificate_Properties_Key_KeyVault(id="example"))),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
