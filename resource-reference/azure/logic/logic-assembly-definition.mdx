`azure_logic_logic_assembly_definition` uploads a .NET assembly to a Logic Apps integration account, made available to custom code steps within its own workflow(s).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	logic "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/logic"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own logic assembly definition"})

    		ubx.Resource(logic.AssemblyDefinition, "example", logic.AssemblyDefinitionConfig{
    			Properties: map[string]any{"assembly_name": "example-assembly-name"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { AssemblyDefinition } from "@ubx/sdk-azure/azure/logic/assembly_definition";

    export default stack("example", () => {
      intent({ summary: "example own logic assembly definition" });

      resource(AssemblyDefinition, "example", {
        properties: { assemblyName: "example-assembly-name" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.logic import AssemblyDefinition, AssemblyDefinitionConfig

    def describe():
        ubx.intent("example own logic assembly definition")

        ubx.resource(AssemblyDefinition, "example", AssemblyDefinitionConfig(
            properties={"assembly_name": "example-assembly-name"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
