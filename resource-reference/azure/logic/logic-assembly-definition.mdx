`azure_logic_logic_assembly_definition` uploads a .NET assembly to a Logic Apps integration account, made available to custom code steps within its own workflow(s).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_logic_logic --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_logic_logic/sdk/go
    package main

    import (
    	logic "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_logic_logic/logic"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own logic assembly definition"})

    		ubx.Resource(logic.LogicAssemblyDefinition, "ci-runner", logic.LogicAssemblyDefinitionConfig{
    			Properties: map[string]any{"assembly_name": "example-assembly-name"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_logic_logic --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { LogicAssemblyDefinition } from "./local-sdk/azure_logic_logic/sdk/typescript/azure_logic_logic/logic/logic_assembly_definition.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own logic assembly definition" });

      resource(LogicAssemblyDefinition, "ci-runner", {
        properties: { assemblyName: "example-assembly-name" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_logic_logic --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_logic_logic/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_logic_logic.logic import LogicAssemblyDefinition, LogicAssemblyDefinitionConfig

    def describe():
        ubx.intent("payments own logic assembly definition")

        ubx.resource(LogicAssemblyDefinition, "ci-runner", LogicAssemblyDefinitionConfig(
            properties={"assembly_name": "example-assembly-name"},
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
