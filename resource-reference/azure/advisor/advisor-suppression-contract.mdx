`azure_advisor_advisor_suppression_contract` suppresses a specific Azure Advisor recommendation for a resource, either permanently or for a defined time window.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_advisor_advisor --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_advisor_advisor/sdk/go
    package main

    import (
    	advisor "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_advisor_advisor/advisor"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own advisor suppression contract"})

    		ubx.Resource(advisor.AdvisorSuppressionContract, "ci-runner", advisor.AdvisorSuppressionContractConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
