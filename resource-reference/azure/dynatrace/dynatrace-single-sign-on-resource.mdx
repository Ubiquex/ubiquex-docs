`azure_dynatrace_dynatrace_dynatrace_single_sign_on_resource` configures single sign-on from Microsoft Entra ID into the linked Dynatrace environment for a Dynatrace resource on Azure.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_dynatrace_dynatrace --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_dynatrace_dynatrace/sdk/go
    package main

    import (
    	dynatrace "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_dynatrace_dynatrace/dynatrace"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own dynatrace single sign on resource"})

    		ubx.Resource(dynatrace.DynatraceDynatraceSingleSignOnResource, "ci-runner", dynatrace.DynatraceDynatraceSingleSignOnResourceConfig{
    			Properties: map[string]any{"aad_domains": []string{"example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_dynatrace_dynatrace --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { DynatraceDynatraceSingleSignOnResource } from "./local-sdk/azure_dynatrace_dynatrace/sdk/typescript/azure_dynatrace_dynatrace/dynatrace/dynatrace_dynatrace_single_sign_on_resource.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own dynatrace single sign on resource" });

      resource(DynatraceDynatraceSingleSignOnResource, "ci-runner", {
        properties: { aadDomains: ["example"] },
      });
    });
    ```
