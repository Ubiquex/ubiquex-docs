`azure_cognitiveservices_agent_application` registers an AI agent application built on Azure AI Foundry's own Agent Service, the deployable unit an agent's tools and instructions are packaged into.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_cognitiveservices --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_cognitiveservices/sdk/go
    package main

    import (
    	cognitiveservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_cognitiveservices/cognitiveservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own cognitiveservices agent application"})

    		ubx.Resource(cognitiveservices.CognitiveservicesAgentApplication, "example", cognitiveservices.CognitiveservicesAgentApplicationConfig{
    			Name:       "example",
    			Properties: map[string]any{"agent_identity_blueprint": map[string]any{"client_id": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
