`azure_cognitiveservices_agent_application` registers an AI agent application built on Azure AI Foundry's own Agent Service, the deployable unit an agent's tools and instructions are packaged into.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cognitiveservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/cognitiveservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own agent application"})

    		ubx.Resource(cognitiveservices.AgentApplication, "example", cognitiveservices.AgentApplicationConfig{
    			Name:       "example",
    			Properties: map[string]any{"agent_identity_blueprint": map[string]any{"client_id": "example", "kind": "example", "principal_id": "example", "tenant_id": "example", "type": "example"}},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx
    from ubx.azure.cognitiveservices import AgentApplication, AgentApplicationConfig
    from ubx.azure.cognitiveservices.agent_application import AgentApplication_Properties, AgentApplication_Properties_AgentIdentityBlueprint

    def describe():
        ubx.intent("example own agent application")

        ubx.resource(AgentApplication, "example", AgentApplicationConfig(
            name="example",
            properties=AgentApplication_Properties(agent_identity_blueprint=AgentApplication_Properties_AgentIdentityBlueprint(client_id="example", kind="example", principal_id="example", tenant_id="example", type="example")),
