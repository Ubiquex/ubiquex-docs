`azure_domainservices_oucontainer_ou_container` is an Organizational Unit within a Microsoft Entra Domain Services managed domain, letting delegated administrators be granted control over just their own OU.

## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_domainservices_oucontainer/sdk/go
    package main

    import (
    	domainservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_domainservices_oucontainer/domainservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own oucontainer ou container"})

    		ubx.Resource(domainservices.OucontainerOuContainer, "example", domainservices.OucontainerOuContainerConfig{
    			AccountName: "example-account-name",
    			Password:    "example",
    			Spn:         "example",
    		})
    	}))
    }
