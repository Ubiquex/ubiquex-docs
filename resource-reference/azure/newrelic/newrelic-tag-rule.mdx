## Example

    ```go
    package main

    import (
    	newrelic "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/newrelic"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own tag rule"})

    		ubx.Resource(newrelic.TagRule, "example", newrelic.TagRuleConfig{
    			Properties: map[string]any{"log_rules": map[string]any{"filtering_tags": []map[string]any{{"name": "example-name"}}}},
    		})
    	}))
    }
    ```

  <Tab title="TypeScript">

    import { intent, resource, stack } from "@ubx/sdk";

    import ubx_sdk as ubx
    from ubx.azure.newrelic import TagRule, TagRuleConfig
    from ubx.azure.newrelic.tag_rule import TagRule_Properties, TagRule_Properties_LogRules, TagRule_Properties_LogRules_FilteringTags

    def describe():
        ubx.intent("example own tag rule")

        ubx.resource(TagRule, "example", TagRuleConfig(
            properties=TagRule_Properties(log_rules=TagRule_Properties_LogRules(filtering_tags=[TagRule_Properties_LogRules_FilteringTags(name="example-name")])),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
