## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	botservice "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/botservice"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own bot"})

    		ubx.Resource(botservice.Bot, "example", botservice.BotConfig{
    			Properties: map[string]any{"display_name": "example-display-name", "endpoint": "example", "msa_app_id": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Bot } from "@ubx/sdk-azure/azure/botservice/bot";

    export default stack("example", () => {
      intent({ summary: "example own bot" });

      resource(Bot, "example", {
        properties: {
          displayName: "example-display-name",
          endpoint: "example",
          msaAppId: "example",
        },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.botservice import Bot, BotConfig
    from ubx.azure.botservice.bot import Bot_Properties

    def describe():
        ubx.intent("example own bot")

        ubx.resource(Bot, "example", BotConfig(
            properties=Bot_Properties(display_name="example-display-name", endpoint="example", msa_app_id="example"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
