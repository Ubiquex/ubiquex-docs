`azure_apimanagement_apimopenidconnectproviders_openid_connect_provider` registers an OpenID Connect identity provider with Azure API Management, referenced by API(s) requiring OIDC-based caller authentication.

    <ResponseField name="client_id" type="string" required>

    <ResponseField name="client_secret" type="string">

    <ResponseField name="description" type="string">

    </ResponseField>
    <ResponseField name="display_name" type="string" required>

    </ResponseField>

      Metadata endpoint URI. Required.
    </ResponseField>
    <ResponseField name="use_in_api_documentation" type="bool">

    </ResponseField>
    <ResponseField name="use_in_test_console" type="bool">
