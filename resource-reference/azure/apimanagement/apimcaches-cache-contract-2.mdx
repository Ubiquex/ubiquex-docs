`azure_apimanagement_apimcaches_cache_contract_2` looks up a real external cache (such as Azure Cache for Redis) registered with an API Management instance.

## Example

    ```go
    	apimanagement "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/apimanagement"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own apimcaches cache contract 2"})

    		ubx.Resource(apimanagement.ApimcachesCacheContract2, "example", apimanagement.ApimcachesCacheContract2Config{
    			Properties: map[string]any{"connection_string": "example", "use_from_location": "example"},
    		})
    	}))
    }
    ```
