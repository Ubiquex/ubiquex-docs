`azure_apimanagement_apimopenidconnectproviders_openid_connect_provider_contract_2` looks up a real OpenID Connect provider registered with an API Management instance.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

<ResponseField name="opid" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="properties" type="object">
  OpenID Connect Providers Contract. Optional; if omitted, computed by Azure.
  <Expandable title="properties">
    <ResponseField name="client_id" type="string" required>
      Client ID of developer console which is the client application. Required.
    </ResponseField>
    <ResponseField name="client_secret" type="string">
      Client Secret of developer console which is the client application. Optional.
    </ResponseField>
    <ResponseField name="description" type="string">
      User-friendly description of OpenID Connect Provider. Optional.
    </ResponseField>
    <ResponseField name="display_name" type="string" required>
      User-friendly OpenID Connect Provider name. Required.
    </ResponseField>
    <ResponseField name="metadata_endpoint" type="string" required>
      Metadata endpoint URI. Required.
    </ResponseField>
    <ResponseField name="use_in_api_documentation" type="bool">
      If true, the Open ID Connect provider will be used in the API documentation in the developer portal. False by default if no value is provided. Optional.
    </ResponseField>
    <ResponseField name="use_in_test_console" type="bool">
      If true, the Open ID Connect provider may be used in the developer portal test console. True by default if no value is provided. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
