`azure_apimanagement_apimgateways_gateway` registers a self-hosted or workspace gateway with Azure API Management, that specific API(s) can be deployed to independently of the managed cloud gateway.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_apimanagement_apimgateways --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_apimanagement_apimgateways/sdk/go
    package main

    import (
    	apimanagement "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_apimanagement_apimgateways/apimanagement"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own apimgateways gateway"})

    		ubx.Resource(apimanagement.ApimgatewaysGateway, "ci-runner", apimanagement.ApimgatewaysGatewayConfig{})
    	}))
    }
    ```
