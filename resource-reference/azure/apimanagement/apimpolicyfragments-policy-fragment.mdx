`azure_apimanagement_apimpolicyfragments_policy_fragment` is a reusable snippet of policy XML that can be included from multiple policy definitions in Azure API Management.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_apimanagement_apimpolicyfragments --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_apimanagement_apimpolicyfragments/sdk/go
    package main

    import (
    	apimanagement "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_apimanagement_apimpolicyfragments/apimanagement"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own apimpolicyfragments policy fragment"})

    		ubx.Resource(apimanagement.ApimpolicyfragmentsPolicyFragment, "ci-runner", apimanagement.ApimpolicyfragmentsPolicyFragmentConfig{
    			Id: "example",
    		})
    	}))
    }
    ```
  </Tab>
