`azure_apimanagement_apimusers_subscription` grants a specific developer access to product(s) in Azure API Management, carrying the subscription key(s) used to authenticate.

## Example

  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_apimanagement_apimusers --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_apimanagement_apimusers/sdk/go
    package main

    import (
    	apimanagement "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_apimanagement_apimusers/apimanagement"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own apimusers subscription"})

    		ubx.Resource(apimanagement.ApimusersSubscription, "ci-runner", apimanagement.ApimusersSubscriptionConfig{
    			Sid: "example",
    		})
    	}))
    }
    ```
