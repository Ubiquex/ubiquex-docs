`azure_apimanagement_apimworkspacediagnostics_diagnostic_contract_2` looks up a real diagnostic logging configuration scoped to an API Management workspace.

    ```go
    package main

    import (
    	apimanagement "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/apimanagement"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own apimworkspacediagnostics diagnostic contract 2"})

    		ubx.Resource(apimanagement.ApimworkspacediagnosticsDiagnosticContract2, "example", apimanagement.ApimworkspacediagnosticsDiagnosticContract2Config{
    			Properties: map[string]any{"logger_id": "example"},
    		})
    	}))
    }
    ```
  </Tab>
