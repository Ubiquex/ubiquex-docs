`azure_apimanagement_apimpolicyfragments_policy_fragment_contract_2` looks up a real, reusable policy fragment that can be included from multiple API Management policies.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own apimpolicyfragments policy fragment contract 2"})

    		ubx.Resource(apimanagement.ApimpolicyfragmentsPolicyFragmentContract2, "example", apimanagement.ApimpolicyfragmentsPolicyFragmentContract2Config{
