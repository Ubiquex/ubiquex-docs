`azure_apimanagement_apimworkspaces_policy_fragment` is a reusable snippet of policy XML that can be included from multiple policy definitions within an API Management workspace.

## Example

  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_apimanagement_apimworkspaces --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_apimanagement_apimworkspaces/sdk/go
    package main

    import (
    	apimanagement "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_apimanagement_apimworkspaces/apimanagement"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own apimworkspaces policy fragment"})

    		ubx.Resource(apimanagement.ApimworkspacesPolicyFragment, "example", apimanagement.ApimworkspacesPolicyFragmentConfig{
    			Id: "example",
    		})
    	}))
    }
    ```
