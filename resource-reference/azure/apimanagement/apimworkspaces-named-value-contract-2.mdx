      Unique name of NamedValue. It may contain only letters, digits, period, dash, and underscore characters. Required.
    </ResponseField>
    <ResponseField name="key_vault" type="object">
      Create keyVault contract details. Optional.
      <Expandable title="properties">
        <ResponseField name="identity_client_id" type="string">
          Null for SystemAssignedIdentity or Client Id for UserAssignedIdentity , which will be used to access key vault secret. Optional.
        </ResponseField>
        <ResponseField name="secret_identifier" type="string">
          Key vault secret identifier for fetching secret. Providing a versioned secret will prevent auto-refresh. This requires API Management service to be configured with aka.ms/apimmsi Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="value" type="string">
      Value of the NamedValue. Can contain policy expressions. It may not be empty or consist only of whitespace. This property will not be filled on 'GET' operations! Use '/listSecrets' POST request to get the value. Optional.
    </ResponseField>
