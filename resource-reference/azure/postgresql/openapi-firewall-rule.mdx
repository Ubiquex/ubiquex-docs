## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	postgresql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/postgresql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi firewall rule"})

    		ubx.Resource(postgresql.OpenapiFirewallRule, "example", postgresql.OpenapiFirewallRuleConfig{
    			Properties: map[string]any{"end_ip_address": "example"},
    		})
    	}))
    }
    ```
  </Tab>

    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.postgresql import OpenapiFirewallRule, OpenapiFirewallRuleConfig

    def describe():
        ubx.intent("example own openapi firewall rule")

        ubx.resource(OpenapiFirewallRule, "example", OpenapiFirewallRuleConfig(
            properties={"end_ip_address": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Properties

<ResponseField name="properties" type="object" required>
  Properties of a firewall rule. Required.
  <Expandable title="properties">
    <ResponseField name="end_ip_address" type="string" required>
      IP address defining the end of the range of addresses of a firewall rule. Must be expressed in IPv4 format. Required.
    </ResponseField>
    <ResponseField name="start_ip_address" type="string" required>
      IP address defining the start of the range of addresses of a firewall rule. Must be expressed in IPv4 format. Required.
    </ResponseField>
  </Expandable>
</ResponseField>
