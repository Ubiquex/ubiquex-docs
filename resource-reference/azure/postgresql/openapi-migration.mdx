`azure_postgresql_openapi_migration` migrates a PostgreSQL database (from on-premises, another cloud, or a different Azure PostgreSQL offering) into an Azure Database for PostgreSQL flexible server.

## Example

    ```go

    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_postgresql_openapi/sdk/go
    package main

    import (
    	postgresql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_postgresql_openapi/postgresql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi migration"})

    		ubx.Resource(postgresql.OpenapiMigration, "example", postgresql.OpenapiMigrationConfig{})
    	}))
    }
    ```
