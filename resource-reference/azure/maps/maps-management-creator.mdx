`azure_maps_management_creator` provisions Azure Maps Creator, letting custom indoor map data be uploaded and converted into a usable indoor mapping dataset.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	maps "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/maps"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own maps management creator"})

    		ubx.Resource(maps.ManagementCreator, "example", maps.ManagementCreatorConfig{
    			Properties: map[string]any{"storage_units": 1},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { ManagementCreator } from "@ubx/sdk-azure/azure/maps/management_creator";

    export default stack("example", () => {
      intent({ summary: "example own maps management creator" });

      resource(ManagementCreator, "example", {
        properties: { storageUnits: 1 },

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.maps.management_creator import ManagementCreator, ManagementCreatorConfig

        ubx.run("example", describe)
    ```
