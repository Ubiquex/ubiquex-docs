  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_dns_dns --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_dns_dns/sdk/go
    package main

    import (
    	dns "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_dns_dns/dns"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own dns record set"})

    		ubx.Resource(dns.DnsRecordSet, "example", dns.DnsRecordSetConfig{
    			Name: "example",
    		})
    	}))
    }
    ```
