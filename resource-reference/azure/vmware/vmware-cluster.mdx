`azure_vmware_vmware_cluster` is a cluster of node(s) within an Azure VMware Solution private cloud, the unit VMware ESXi/vSAN compute and storage capacity is added in.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_vmware_vmware --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_vmware_vmware/sdk/go
    package main

    import (
    	vmware "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_vmware_vmware/vmware"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own vmware cluster"})

    		ubx.Resource(vmware.VmwareCluster, "example", vmware.VmwareClusterConfig{
    			Sku: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
