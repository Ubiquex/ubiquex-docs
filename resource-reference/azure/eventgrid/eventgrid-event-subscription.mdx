`azure_eventgrid_event_subscription` subscribes a destination (a webhook, Azure Function, or queue) to matching event(s) published on an Event Grid topic or Azure resource.

## Example

    package main

    import (

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own eventgrid event subscription"})

    		ubx.Resource(eventgrid.EventSubscription, "example", eventgrid.EventSubscriptionConfig{
    			Scope: "example",
    		})
    	}))
    }
