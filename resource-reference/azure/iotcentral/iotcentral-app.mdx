`azure_iotcentral_app` is an IoT Central application, a fully managed SaaS IoT solution builder requiring no separate backend service management.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	iotcentral "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/iotcentral"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own app"})

    		ubx.Resource(iotcentral.App, "example", iotcentral.AppConfig{
    			Sku: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { App } from "@ubx/sdk-azure/azure/iotcentral/app";

    export default stack("example", () => {
      intent({ summary: "example own app" });

      resource(App, "example", {
        sku: { name: "example-name" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.iotcentral import App, AppConfig
    from ubx.azure.iotcentral.app import App_Sku

    def describe():
        ubx.intent("example own app")

        ubx.resource(App, "example", AppConfig(
            sku=App_Sku(name="example-name"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>

## Properties

<ResponseField name="identity" type="object">
  Managed service identity (either system assigned, or none) Optional; if omitted, computed by Microsoft Azure.
  <Expandable title="properties">
    <ResponseField name="principal_id" type="string">
      The service principal ID of the system assigned identity. This property will only be provided for a system assigned identity. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="tenant_id" type="string">
      The tenant ID of the system assigned identity. This property will only be provided for a system assigned identity. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="type" type="string" required>
      Type of managed service identity (either system assigned, or none). Required.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="properties" type="object">
  The properties of an IoT Central application. Optional; if omitted, computed by Microsoft Azure.
  <Expandable title="properties">
    <ResponseField name="application_id" type="string">
      The ID of the application. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="display_name" type="string">
      The display name of the application. Optional.
    </ResponseField>
    <ResponseField name="state" type="string">
      The current state of the application. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="subdomain" type="string">
      The subdomain of the application. Optional.
    </ResponseField>
    <ResponseField name="template" type="string">
      The ID of the application template, which is a blueprint that defines the characteristics and behaviors of an application. Optional; if not specified, defaults to a blank blueprint and allows the application to be defined from scratch. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="sku" type="object" required>
  Information about the SKU of the IoT Central application. Required.
  <Expandable title="properties">
    <ResponseField name="name" type="string" required>
      The name of the SKU. Required.
    </ResponseField>
  </Expandable>
</ResponseField>
