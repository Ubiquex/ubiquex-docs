`azure_azurestackhci_hci_edge_device` represents a single physical node's own Azure resource within an Azure Stack HCI deployment, prior to or during cluster formation.

## Example

  <Tab title="Go">

    // ubx sdk gen --only azure_azurestackhci_hci --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_azurestackhci_hci/sdk/go
    package main

    import (
    	azurestackhci "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_azurestackhci_hci/azurestackhci"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own hci edge device"})

    		ubx.Resource(azurestackhci.HciEdgeDevice, "ci-runner", azurestackhci.HciEdgeDeviceConfig{
    			Kind: "example",
    		})
    	}))
    }
