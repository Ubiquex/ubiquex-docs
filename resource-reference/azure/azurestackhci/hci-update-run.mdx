## Properties

<ResponseField name="location" type="string">
  The geo-location where the resource lives Optional; if omitted, computed by Microsoft Azure.
</ResponseField>

<ResponseField name="properties" type="object">
  Details of an Update run Optional; if omitted, computed by Microsoft Azure.
  <Expandable title="properties">
    <ResponseField name="duration" type="string">
      Duration of the update run. Optional.
    </ResponseField>
    <ResponseField name="last_updated_time" type="string">
      Timestamp of the most recently completed step in the update run. Optional.
    </ResponseField>
    <ResponseField name="progress" type="object">
      Progress representation of the update run steps. Optional.
      <Expandable title="properties">
        <ResponseField name="description" type="string">
          More detailed description of the step. Optional.
        </ResponseField>
        <ResponseField name="end_time_utc" type="string">
          When the step reached a terminal state. Optional.
        </ResponseField>
        <ResponseField name="error_message" type="string">
          Error message, specified if the step is in a failed state. Optional.
        </ResponseField>
        <ResponseField name="expected_execution_time" type="string">
          Expected execution time of a given step. This is optionally authored in the update action plan and can be empty. Optional.
        </ResponseField>
        <ResponseField name="last_updated_time_utc" type="string">
          Completion time of this step or the last completed sub-step. Optional.
        </ResponseField>
        <ResponseField name="name" type="string">
          Name of the step. Optional.
        </ResponseField>
        <ResponseField name="start_time_utc" type="string">
          When the step started, or empty if it has not started executing. Optional.
        </ResponseField>
        <ResponseField name="status" type="string">
          Status of the step, bubbled up from the ECE action plan for installation attempts. Values are: 'Success', 'Error', 'InProgress', and 'Unknown status'. Optional.
        </ResponseField>
        <ResponseField name="steps" type="list(object)">
          Recursive model for child steps of this step. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="provisioning_state" type="string">
      Provisioning state of the ArcSetting proxy resource. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="state" type="string">
      Represents the current state of the update run. Indicates whether the update is in progress, has completed successfully, failed, or is in an unknown state. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="time_started" type="string">
      Timestamp of the update run was started. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
