`azure_app_openapi_managed_environment_storage` mounts an external storage volume (Azure Files) accessible to container app(s) running in a Container Apps environment.

    package main

    import (
    	app "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_app_openapi/app"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
