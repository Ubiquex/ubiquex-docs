`azure_app_sreagent_agent_space_connector` connects an external tool or data source at the agent space level, made available to every SRE agent within that space.

## Example

  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_app_sreagent --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_app_sreagent/sdk/go
    package main

    import (
    	app "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_app_sreagent/app"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own sreagent agent space connector"})

    		ubx.Resource(app.SreagentAgentSpaceConnector, "ci-runner", app.SreagentAgentSpaceConnectorConfig{})
    	}))
    }
    ```
