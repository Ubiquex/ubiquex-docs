`azure_app_openapi_java_component` enables a managed Java runtime add-on (e.g. Config Server, Eureka service registry, Admin dashboard) for Java-based container app(s) in a Container Apps environment.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_app_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_app_openapi/sdk/go
    package main

    import (
    	app "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_app_openapi/app"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi java component"})

    		ubx.Resource(app.OpenapiJavaComponent, "ci-runner", app.OpenapiJavaComponentConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_app_openapi --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.azure_app_openapi.app import OpenapiJavaComponent, OpenapiJavaComponentConfig

        ubx.intent("payments own openapi java component")

        ubx.resource(OpenapiJavaComponent, "ci-runner", OpenapiJavaComponentConfig(
            name="ci-runner",

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
