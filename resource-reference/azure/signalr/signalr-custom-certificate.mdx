`azure_signalr_custom_certificate` uploads a TLS certificate (referenced from Key Vault) to a SignalR Service resource, bindable to a custom domain.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_signalr_signalr --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_signalr_signalr/sdk/go
    package main

    import (
    	signalr "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_signalr_signalr/signalr"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own signalr custom certificate"})

    		ubx.Resource(signalr.SignalrCustomCertificate, "example", signalr.SignalrCustomCertificateConfig{
    			Properties: map[string]any{"key_vault_base_uri": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_signalr_signalr --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { SignalrCustomCertificate } from "./local-sdk/azure_signalr_signalr/sdk/typescript/azure_signalr_signalr/signalr/signalr_custom_certificate.ts";

    export default stack("example", () => {
      intent({ summary: "example own signalr custom certificate" });

      resource(SignalrCustomCertificate, "example", {
        properties: { keyVaultBaseUri: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_signalr_signalr --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_signalr_signalr/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_signalr_signalr.signalr import SignalrCustomCertificate, SignalrCustomCertificateConfig

    def describe():
        ubx.intent("example own signalr custom certificate")

        ubx.resource(SignalrCustomCertificate, "example", SignalrCustomCertificateConfig(
            properties={"key_vault_base_uri": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

  <Expandable title="properties">

    <ResponseField name="key_vault_secret_name" type="string" required>
