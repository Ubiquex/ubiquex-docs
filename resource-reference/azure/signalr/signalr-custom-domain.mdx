## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	signalr "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/signalr"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own signalr custom domain"})

    		ubx.Resource(signalr.CustomDomain, "example", signalr.CustomDomainConfig{
    			Name:       "example",
    			Properties: map[string]any{"custom_certificate": map[string]any{"id": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { CustomDomain } from "@ubx/sdk-azure/azure/signalr/custom_domain";

    export default stack("example", () => {
      intent({ summary: "example own signalr custom domain" });

      resource(CustomDomain, "example", {
        name: "example",
        properties: { customCertificate: { id: "example" } },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.signalr import CustomDomain, CustomDomainConfig

    def describe():
        ubx.intent("example own signalr custom domain")

        ubx.resource(CustomDomain, "example", CustomDomainConfig(
            name="example",
            properties={"custom_certificate": {"id": "example"}},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
  <Tab title="Markdown">
    ```
    Create azure_signalr_custom_domain called "example", with
    properties a set value.
    ```
  </Tab>
</Tabs>
