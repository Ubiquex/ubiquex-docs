`azure_management_management_management_group` groups subscription(s) and other management group(s) together above the subscription level, letting policy and access control be applied hierarchically across an entire organization.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_management_management --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_management_management/sdk/go
    package main

    import (
    	management "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_management_management/management"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own management management group"})

    		ubx.Resource(management.ManagementManagementGroup, "ci-runner", management.ManagementManagementGroupConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
