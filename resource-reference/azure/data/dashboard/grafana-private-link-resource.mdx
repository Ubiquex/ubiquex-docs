`azure_dashboard_grafana_private_link_resource` reads a real Private Link resource exposed by an Azure Managed Grafana workspace, that a Private Endpoint can connect to.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dashboard "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/dashboard"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_dashboard_grafana_private_link_resource"})

    		ubx.Data(dashboard.GrafanaPrivateLinkResource, "example", dashboard.GrafanaPrivateLinkResourceConfig{
    			WorkspaceName: "example-workspace-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { GrafanaPrivateLinkResource } from "@ubx/sdk-azure/azure/data/dashboard/grafana_private_link_resource";

    export default stack("example", () => {
      intent({ summary: "look up azure_dashboard_grafana_private_link_resource" });

      data(GrafanaPrivateLinkResource, "example", {
        workspaceName: "example-workspace-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.dashboard import GrafanaPrivateLinkResource, GrafanaPrivateLinkResourceConfig

    def describe():
        ubx.intent("look up azure_dashboard_grafana_private_link_resource")

        ubx.data(GrafanaPrivateLinkResource, "example", GrafanaPrivateLinkResourceConfig(
            workspace_name="example-workspace-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
