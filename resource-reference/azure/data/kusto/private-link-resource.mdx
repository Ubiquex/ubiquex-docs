## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	kusto "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/kusto"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_kusto_private_link_resource"})

    		ubx.Data(kusto.PrivateLinkResource, "example", kusto.PrivateLinkResourceConfig{
    			ClusterName: "example-cluster-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk

    export default stack("example", () => {
      intent({ summary: "look up azure_kusto_private_link_resource" });

      data(PrivateLinkResource, "example", {
        clusterName: "example-cluster-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.kusto import PrivateLinkResource, PrivateLinkResourceConfig

    def describe():
        ubx.intent("look up azure_kusto_private_link_resource")

        ubx.data(PrivateLinkResource, "example", PrivateLinkResourceConfig(
            cluster_name="example-cluster-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
