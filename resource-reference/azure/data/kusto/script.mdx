---
title: "azure_kusto_script"
description: "`data_azure_kusto_script` looks up a real, already-existing `azure_kusto_script`."
---

It runs a batch of KQL control commands against an Azure Data Explorer database as part of resource deployment, e.g. to create tables or ingestion mappings declaratively.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	kusto "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/kusto"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_kusto_script"})

    		ubx.Data(kusto.Script, "example", kusto.ScriptConfig{
    			ClusterName:  "example-cluster-name",
    			DatabaseName: "example-database-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { Script } from "./local-sdk/azure/sdk/typescript/azure/data/kusto/script.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_kusto_script" });

      data(Script, "example", {
        clusterName: "example-cluster-name",
        databaseName: "example-database-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.kusto import Script, ScriptConfig

    def describe():
        ubx.intent("look up azure_kusto_script")

        ubx.data(Script, "example", ScriptConfig(
            cluster_name="example-cluster-name",
            database_name="example-database-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Lookup arguments

<ResponseField name="cluster_name" type="string" required>
  The real name of the cluster this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="database_name" type="string" required>
  The real name of the database this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="next_link" type="string">
  Link to the next page of results. Returned by the lookup.
</ResponseField>

<ResponseField name="value" type="list(object)">
  The list of Kusto scripts. Returned by the lookup.
</ResponseField>
