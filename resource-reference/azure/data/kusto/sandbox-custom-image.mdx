---
title: "azure_kusto_sandbox_custom_image"
description: "`data_azure_kusto_sandbox_custom_image` looks up a real, already-existing `azure_kusto_sandbox_custom_image`."
---

It defines a custom Python runtime image (base image, package requirements) available to the `python()` plugin's own isolated sandbox execution in Azure Data Explorer queries.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	kusto "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/kusto"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_kusto_sandbox_custom_image"})

    		ubx.Data(kusto.SandboxCustomImage, "example", kusto.SandboxCustomImageConfig{
    			ClusterName: "example-cluster-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { SandboxCustomImage } from "./local-sdk/azure/sdk/typescript/azure/data/kusto/sandbox_custom_image.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_kusto_sandbox_custom_image" });

      data(SandboxCustomImage, "example", {
        clusterName: "example-cluster-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.kusto import SandboxCustomImage, SandboxCustomImageConfig

    def describe():
        ubx.intent("look up azure_kusto_sandbox_custom_image")

        ubx.data(SandboxCustomImage, "example", SandboxCustomImageConfig(
            cluster_name="example-cluster-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
