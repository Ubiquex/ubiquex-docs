## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	storage "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/storage"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_storage_openapi_file_service_properties"})

    		ubx.Data(storage.OpenapiFileServiceProperties, "example", storage.OpenapiFileServicePropertiesConfig{
    			AccountName: "example-account-name",
    		})
    	}))
    }
    ```
  </Tab>
