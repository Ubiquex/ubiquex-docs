## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	scvmm "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/scvmm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_scvmm_virtual_machine_instance"})

    		ubx.Data(scvmm.VirtualMachineInstance, "example", scvmm.VirtualMachineInstanceConfig{
    			ResourceUri: "example",
    		})
    	}))
    }
    ```
