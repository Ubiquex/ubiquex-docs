## Example

    package main

    import (
    	sql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/sql"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_sql_sensitivitylabels_sensitivity_label"})

    		ubx.Data(sql.SensitivitylabelsSensitivityLabel, "example", sql.SensitivitylabelsSensitivityLabelConfig{
    			DatabaseName: "example-database-name",
    			ServerName:   "example-server-name",
    		})
    	}))
    }
