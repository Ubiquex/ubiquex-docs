`data_azure_sql_serverconfigurationoptions_server_configuration_option` looks up a real, already-existing `azure_sql_serverconfigurationoptions_server_configuration_option`. It sets a server-wide configuration option (e.g. minimal TLS version enforcement mode) on an Azure SQL logical server.

## Example

    import (
    	sql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/sql"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_sql_serverconfigurationoptions_server_configuration_option"})

    		ubx.Data(sql.ServerconfigurationoptionsServerConfigurationOption, "example", sql.ServerconfigurationoptionsServerConfigurationOptionConfig{
    			ManagedInstanceName: "example-managed-instance-name",
    		})
    	}))
    }
