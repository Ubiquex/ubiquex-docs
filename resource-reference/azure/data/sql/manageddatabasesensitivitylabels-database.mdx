`azure_sql_manageddatabasesensitivitylabels_database` reads a real sensitivity label (data classification tag) applied to a column in a SQL Managed Instance database.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_sql_manageddatabasesensitivitylabels_database"})

    		ubx.Data(sql.ManageddatabasesensitivitylabelsDatabase, "example", sql.ManageddatabasesensitivitylabelsDatabaseConfig{
    			DatabaseName:        "example-database-name",

    		})
    	}))
    }
