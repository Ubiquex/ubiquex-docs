It is an Azure SQL Database, a fully managed, single or elastic-pool-hosted relational database.

## Example

    // ubx sdk gen --only azure --lang go --out ./local-sdk

    import (
    	sql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/sql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_sql_databases_database"})

    		})
    	}))
    }
