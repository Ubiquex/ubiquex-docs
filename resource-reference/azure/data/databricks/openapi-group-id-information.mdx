## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	databricks "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/databricks"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_databricks_openapi_group_id_information"})

    		ubx.Data(databricks.OpenapiGroupIdInformation, "example", databricks.OpenapiGroupIdInformationConfig{
    			WorkspaceName: "example-workspace-name",
    		})
    	}))
    }
    ```
