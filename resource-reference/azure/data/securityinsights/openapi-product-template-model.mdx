`azure_securityinsights_openapi_product_template_model` looks up a real template (an analytics rule, playbook, or workbook) within a Microsoft Sentinel content package.

## Example

    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go

    import (
    	securityinsights "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/securityinsights"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_securityinsights_openapi_product_template_model"})

    		ubx.Data(securityinsights.OpenapiProductTemplateModel, "example", securityinsights.OpenapiProductTemplateModelConfig{
    			WorkspaceName: "example-workspace-name",
    		})
    	}))
    }
