`data_azure_securityinsights_openapi_sentinel_onboarding_state` looks up a real, already-existing `azure_securityinsights_openapi_sentinel_onboarding_state`. It tracks whether Microsoft Sentinel has been onboarded (activated) onto a specific Log Analytics workspace.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	securityinsights "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/securityinsights"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_securityinsights_openapi_sentinel_onboarding_state"})

    		ubx.Data(securityinsights.OpenapiSentinelOnboardingState, "example", securityinsights.OpenapiSentinelOnboardingStateConfig{
    			WorkspaceName: "example-workspace-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.securityinsights import OpenapiSentinelOnboardingState, OpenapiSentinelOnboardingStateConfig

    def describe():
        ubx.intent("look up azure_securityinsights_openapi_sentinel_onboarding_state")

        ubx.data(OpenapiSentinelOnboardingState, "example", OpenapiSentinelOnboardingStateConfig(
            workspace_name="example-workspace-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
