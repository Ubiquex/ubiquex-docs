---
title: "azure_securityinsights_openapi_automation_rule"
description: "`data_azure_securityinsights_openapi_automation_rule` looks up a real, already-existing `azure_securityinsights_openapi_automation_rule`."
---

It automatically applies action(s) (assign an owner, run a playbook, close as false-positive) to Microsoft Sentinel incident(s) matching a defined condition.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	securityinsights "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/securityinsights"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_securityinsights_openapi_automation_rule"})

    		ubx.Data(securityinsights.OpenapiAutomationRule, "example", securityinsights.OpenapiAutomationRuleConfig{
    			WorkspaceName: "example-workspace-name",
    		})
    	}))
    }
    ```
  </Tab>

    from ubx.azure.data.securityinsights import OpenapiAutomationRule, OpenapiAutomationRuleConfig

    def describe():
        ubx.intent("look up azure_securityinsights_openapi_automation_rule")

        ubx.data(OpenapiAutomationRule, "example", OpenapiAutomationRuleConfig(
            workspace_name="example-workspace-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
