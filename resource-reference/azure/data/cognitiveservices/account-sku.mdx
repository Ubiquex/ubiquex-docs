## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cognitiveservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/cognitiveservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_cognitiveservices_account_sku"})

    		ubx.Data(cognitiveservices.AccountSku, "example", cognitiveservices.AccountSkuConfig{
    			AccountName: "example-account-name",
    		})
    	}))
    }
    ```
