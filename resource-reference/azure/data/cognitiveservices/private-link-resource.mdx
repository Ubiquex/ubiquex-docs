`azure_cognitiveservices_private_link_resource` reads a real Private Link resource exposed by a Cognitive Services account, that a Private Endpoint can connect to.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	cognitiveservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/cognitiveservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_cognitiveservices_private_link_resource"})

    		ubx.Data(cognitiveservices.PrivateLinkResource, "example", cognitiveservices.PrivateLinkResourceConfig{
    			AccountName: "example-account-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { PrivateLinkResource } from "./local-sdk/azure/sdk/typescript/azure/data/cognitiveservices/private_link_resource.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_cognitiveservices_private_link_resource" });

      data(PrivateLinkResource, "example", {
        accountName: "example-account-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.cognitiveservices import PrivateLinkResource, PrivateLinkResourceConfig

    def describe():
        ubx.intent("look up azure_cognitiveservices_private_link_resource")

        ubx.data(PrivateLinkResource, "example", PrivateLinkResourceConfig(
            account_name="example-account-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
