`data_azure_cognitiveservices_sku_resource` is a real, live lookup against Microsoft Azure, generated from azure's own real schema -- read-only, never created or destroyed.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cognitiveservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/cognitiveservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_cognitiveservices_sku_resource"})

    		ubx.Data(cognitiveservices.SkuResource, "example", cognitiveservices.SkuResourceConfig{
    			AccountName:    "example-account-name",
    			DeploymentName: "example-deployment-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

    import ubx_sdk as ubx
    from ubx.azure.data.cognitiveservices import SkuResource, SkuResourceConfig

    def describe():
        ubx.intent("look up azure_cognitiveservices_sku_resource")

        ubx.data(SkuResource, "example", SkuResourceConfig(
            account_name="example-account-name",
            deployment_name="example-deployment-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
