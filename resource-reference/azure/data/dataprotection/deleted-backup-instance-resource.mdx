## Example

    package main

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_dataprotection_deleted_backup_instance_resource"})

    		ubx.Data(dataprotection.DeletedBackupInstanceResource, "example", dataprotection.DeletedBackupInstanceResourceConfig{
    			VaultName: "example-vault-name",
    		})
    	}))
    }
