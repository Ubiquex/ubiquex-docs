`azure_recoveryservicessiterecovery_service_logical_network` looks up a real logical network registered with Azure Site Recovery for VMM-to-Azure replication.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	recoveryservicessiterecovery "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/recoveryservicessiterecovery"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_recoveryservicessiterecovery_service_logical_network"})

    		ubx.Data(recoveryservicessiterecovery.ServiceLogicalNetwork, "example", recoveryservicessiterecovery.ServiceLogicalNetworkConfig{
    			FabricName:   "example-fabric-name",
    			ResourceName: "example-resource-name",
    		})
    	}))
    }
    ```
