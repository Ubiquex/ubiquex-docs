`azure_recoveryservicessiterecovery_service_replication_appliance` looks up a real, on-premises Azure Site Recovery replication appliance registered with a Recovery Services vault.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_recoveryservicessiterecovery_service_replication_appliance"})

    		ubx.Data(recoveryservicessiterecovery.ServiceReplicationAppliance, "example", recoveryservicessiterecovery.ServiceReplicationApplianceConfig{

    }
