`azure_eventgrid_extension_topic` looks up a real extension topic -- the implicit Event Grid topic a supported Azure resource publishes its own events through.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	eventgrid "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/eventgrid"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_eventgrid_extension_topic"})

    		ubx.Data(eventgrid.ExtensionTopic, "example", eventgrid.ExtensionTopicConfig{
    			ApiVersion: "example",
    			Scope:      "example",
    		})
    	}))
    }
    ```
