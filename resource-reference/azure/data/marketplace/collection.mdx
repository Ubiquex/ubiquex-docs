`azure_marketplace_collection` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	marketplace "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/marketplace"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_marketplace_collection"})

    		ubx.Data(marketplace.Collection, "example", marketplace.CollectionConfig{
    			PrivateStoreId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Collection } from "@ubx/sdk-azure/azure/data/marketplace/collection";

    export default stack("example", () => {
      intent({ summary: "look up azure_marketplace_collection" });

      data(Collection, "example", {
        privateStoreId: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.marketplace import Collection, CollectionConfig

    def describe():
        ubx.intent("look up azure_marketplace_collection")

        ubx.data(Collection, "example", CollectionConfig(
            private_store_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
