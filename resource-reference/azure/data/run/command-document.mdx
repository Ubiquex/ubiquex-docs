`azure_run_command_document` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	run "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/run"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_run_command_document"})

    		ubx.Data(run.CommandDocument, "example", run.CommandDocumentConfig{
    			CommandId: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.run import CommandDocument, CommandDocumentConfig

    def describe():
        ubx.intent("look up azure_run_command_document")

        ubx.data(CommandDocument, "example", CommandDocumentConfig(
            command_id="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
