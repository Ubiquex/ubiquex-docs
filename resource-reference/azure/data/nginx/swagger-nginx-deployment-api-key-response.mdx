It is an API key issued for authenticating dataplane API calls to an NGINX for Azure deployment.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_nginx_swagger_nginx_deployment_api_key_response"})
