`data_azure_cost_management_openapi_export_run` is a real, live lookup against Microsoft Azure, generated from azure's own real schema -- read-only, never created or destroyed.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cost "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/cost"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_cost_management_openapi_export_run"})

    		ubx.Data(cost.ManagementOpenapiExportRun, "example", cost.ManagementOpenapiExportRunConfig{
    			ExportName: "example-export-name",
    			Scope:      "example",
    		})
    	}))
    }
    ```
  </Tab>
