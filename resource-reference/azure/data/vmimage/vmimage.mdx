## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	vmimage "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/vmimage"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_vmimage"})

    		ubx.Data(vmimage.Vmimage, "example", vmimage.VmimageConfig{
    			EdgeZone:      "example",
    			PublisherName: "example-publisher-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.azure.data.vmimage import Vmimage, VmimageConfig

    def describe():
        ubx.intent("look up azure_vmimage")

        ubx.data(Vmimage, "example", VmimageConfig(
            edge_zone="example",
            publisher_name="example-publisher-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
