---
title: "azure_azurestackhci_hci_update_run"
description: "`data_azure_azurestackhci_hci_update_run` looks up a real, already-existing `azure_azurestackhci_hci_update_run`."
---

It is a single execution of applying an update to an Azure Stack HCI cluster, tracked step by step.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	azurestackhci "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/azurestackhci"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_azurestackhci_hci_update_run"})

    		ubx.Data(azurestackhci.HciUpdateRun, "example", azurestackhci.HciUpdateRunConfig{
    			ClusterName: "example-cluster-name",
    			UpdateName:  "example-update-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.azurestackhci import HciUpdateRun, HciUpdateRunConfig

    def describe():
        ubx.intent("look up azure_azurestackhci_hci_update_run")

        ubx.data(HciUpdateRun, "example", HciUpdateRunConfig(
            cluster_name="example-cluster-name",
            update_name="example-update-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
