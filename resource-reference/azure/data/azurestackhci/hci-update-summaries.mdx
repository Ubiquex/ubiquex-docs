## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	azurestackhci "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/azurestackhci"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_azurestackhci_hci_update_summaries"})

    		ubx.Data(azurestackhci.HciUpdateSummaries, "example", azurestackhci.HciUpdateSummariesConfig{
    			ClusterName: "example-cluster-name",
    		})
    	}))
    }
    ```
  </Tab>
