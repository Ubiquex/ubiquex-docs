---
title: "azure_azurestackhci_hci_extension"
description: "`data_azure_azurestackhci_hci_extension` looks up a real, already-existing `azure_azurestackhci_hci_extension`."
---

It installs an Azure Arc extension onto an Azure Stack HCI cluster's own node(s), e.g. for monitoring or Kubernetes.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	azurestackhci "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/azurestackhci"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_azurestackhci_hci_extension"})

    		ubx.Data(azurestackhci.HciExtension, "example", azurestackhci.HciExtensionConfig{
    			ArcSettingName: "example-arc-setting-name",
    			ClusterName:    "example-cluster-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { HciExtension } from "./local-sdk/azure/sdk/typescript/azure/data/azurestackhci/hci_extension.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_azurestackhci_hci_extension" });

        clusterName: "example-cluster-name",
      });
    });
    ```

  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.azurestackhci import HciExtension, HciExtensionConfig

    def describe():
        ubx.intent("look up azure_azurestackhci_hci_extension")

        ubx.data(HciExtension, "example", HciExtensionConfig(
            arc_setting_name="example-arc-setting-name",
            cluster_name="example-cluster-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

## Lookup arguments

<ResponseField name="arc_setting_name" type="string" required>
  The real name of the Azure Arc setting this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="cluster_name" type="string" required>
  The real name of the cluster this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
