It tracks an asynchronous operation (e.g. a firmware update) running against an Azure Stack HCI edge device.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	azurestackhci "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/azurestackhci"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_azurestackhci_hci_edge_device_job"})

    		ubx.Data(azurestackhci.HciEdgeDeviceJob, "example", azurestackhci.HciEdgeDeviceJobConfig{
    			EdgeDeviceName: "example-edge-device-name",
    			ResourceUri:    "example",
    		})
    	}))
    }
    ```
