## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	synapse "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/synapse"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_synapse_sqlpool_sql_pool_schema"})

    		ubx.Data(synapse.SqlpoolSqlPoolSchema, "example", synapse.SqlpoolSqlPoolSchemaConfig{
    			SqlPoolName: "example-sql-pool-name",
    		})
    	}))
    }
    ```
