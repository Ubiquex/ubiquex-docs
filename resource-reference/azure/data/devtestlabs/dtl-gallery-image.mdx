## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	devtestlabs "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/devtestlabs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_devtestlabs_dtl_gallery_image"})

    		ubx.Data(devtestlabs.DtlGalleryImage, "example", devtestlabs.DtlGalleryImageConfig{
    			LabName: "example-lab-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
