## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	chaos "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/chaos"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_chaos_openapi_experiment"})

    		ubx.Data(chaos.OpenapiExperiment, "example", chaos.OpenapiExperimentConfig{
    			ContinuationToken: "example",
    			Running:           true,
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { OpenapiExperiment } from "@ubx/sdk-azure/azure/data/chaos/openapi_experiment";

    export default stack("example", () => {
      intent({ summary: "look up azure_chaos_openapi_experiment" });

      data(OpenapiExperiment, "example", {
        continuationToken: "example",
        running: true,
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.chaos import OpenapiExperiment, OpenapiExperimentConfig

    def describe():
        ubx.intent("look up azure_chaos_openapi_experiment")

        ubx.data(OpenapiExperiment, "example", OpenapiExperimentConfig(
            continuation_token="example",
            running=True,
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
