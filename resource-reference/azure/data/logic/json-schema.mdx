## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	logic "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/logic"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_logic_json_schema"})

    		ubx.Data(logic.JsonSchema, "example", logic.JsonSchemaConfig{
    			ApiVersion:        "example",
    			ResourceGroupName: "example-resource-group-name",
    			SubscriptionId:    "example",
    			TriggerName:       "example-trigger-name",
    			WorkflowName:      "example-workflow-name",
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.azure.data.logic import JsonSchema, JsonSchemaConfig

    def describe():
        ubx.intent("look up azure_logic_json_schema")

        ubx.data(JsonSchema, "example", JsonSchemaConfig(
            api_version="example",
            resource_group_name="example-resource-group-name",
            subscription_id="example",
            trigger_name="example-trigger-name",
            workflow_name="example-workflow-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

## Lookup arguments

<ResponseField name="api_version" type="string" required>
  The real Azure REST API version this request targets. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="resource_group_name" type="string" required>
  The real name of the resource group this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="subscription_id" type="string" required>
  The real, unique identifier of the Azure subscription this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="trigger_name" type="string" required>
  The real name of the trigger this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="workflow_name" type="string" required>
  The real name of the workflow this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="content" type="string">
