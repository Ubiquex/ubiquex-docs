It is a dedicated, isolated deployment of the Logic Apps runtime running inside the caller's own virtual network, for workflows needing private connectivity or dedicated throughput.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_logic_integration_service_environment"})
