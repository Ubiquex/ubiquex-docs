---
title: "azure_virtual_machine_extension"
description: "`data_azure_virtual_machine_extension` looks up a real, already-existing `azure_virtual_machine_extension`."
---

It installs a small, publisher-provided application onto a running VM after deployment -- monitoring agents, security tooling, and custom scripts are all shipped this way. Extensions are managed entirely through the Azure platform, never by logging into the VM directly, and their own lifecycle (install, upgrade, remove) is tracked outside the guest OS.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	virtual "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/virtual"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_virtual_machine_extension"})

    		ubx.Data(virtual.MachineExtension, "example", virtual.MachineExtensionConfig{
    			VmName: "example-vm-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { MachineExtension } from "./local-sdk/azure/sdk/typescript/azure/data/virtual/machine_extension.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_virtual_machine_extension" });

      data(MachineExtension, "example", {
        vmName: "example-vm-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.virtual import MachineExtension, MachineExtensionConfig

    def describe():
        ubx.intent("look up azure_virtual_machine_extension")

        ubx.data(MachineExtension, "example", MachineExtensionConfig(
            vm_name="example-vm-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Lookup arguments

<ResponseField name="vm_name" type="string" required>
  The real name of the virtual machine this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
