`azure_dedicated_host` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dedicated "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/dedicated"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_dedicated_host"})

    		ubx.Data(dedicated.Host, "example", dedicated.HostConfig{
    			HostGroupName: "example-host-group-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Host } from "@ubx/sdk-azure/azure/data/dedicated/host";

    export default stack("example", () => {
      intent({ summary: "look up azure_dedicated_host" });

      data(Host, "example", {
        hostGroupName: "example-host-group-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.dedicated import Host, HostConfig

    def describe():
        ubx.intent("look up azure_dedicated_host")

        ubx.data(Host, "example", HostConfig(
            host_group_name="example-host-group-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
