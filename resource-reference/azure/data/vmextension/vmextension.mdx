`azure_vmextension` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	vmextension "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/vmextension"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_vmextension"})

    		ubx.Data(vmextension.Vmextension, "example", vmextension.VmextensionConfig{
    			PublisherName: "example-publisher-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Vmextension } from "@ubx/sdk-azure/azure/data/vmextension/vmextension";

    export default stack("example", () => {
      intent({ summary: "look up azure_vmextension" });

      data(Vmextension, "example", {
        publisherName: "example-publisher-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.vmextension import Vmextension, VmextensionConfig

    def describe():
        ubx.intent("look up azure_vmextension")

        ubx.data(Vmextension, "example", VmextensionConfig(
            publisher_name="example-publisher-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
