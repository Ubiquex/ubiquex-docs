`azure_keyvault_openapi_secret` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	keyvault "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/keyvault"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_keyvault_openapi_secret"})

    		ubx.Data(keyvault.OpenapiSecret, "example", keyvault.OpenapiSecretConfig{
    			VaultName: "example-vault-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { OpenapiSecret } from "@ubx/sdk-azure/azure/data/keyvault/openapi_secret";

    export default stack("example", () => {
      intent({ summary: "look up azure_keyvault_openapi_secret" });

      data(OpenapiSecret, "example", {
        vaultName: "example-vault-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.keyvault import OpenapiSecret, OpenapiSecretConfig

    def describe():
        ubx.intent("look up azure_keyvault_openapi_secret")

        ubx.data(OpenapiSecret, "example", OpenapiSecretConfig(
            vault_name="example-vault-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
