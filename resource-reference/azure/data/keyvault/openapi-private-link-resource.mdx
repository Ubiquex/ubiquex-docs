## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	keyvault "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/keyvault"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_keyvault_openapi_private_link_resource"})

    		ubx.Data(keyvault.OpenapiPrivateLinkResource, "example", keyvault.OpenapiPrivateLinkResourceConfig{
    			VaultName: "example-vault-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { OpenapiPrivateLinkResource } from "@ubx/sdk-azure/azure/data/keyvault/openapi_private_link_resource";

    export default stack("example", () => {
      intent({ summary: "look up azure_keyvault_openapi_private_link_resource" });

      data(OpenapiPrivateLinkResource, "example", {
        vaultName: "example-vault-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.keyvault import OpenapiPrivateLinkResource, OpenapiPrivateLinkResourceConfig

    def describe():
        ubx.intent("look up azure_keyvault_openapi_private_link_resource")

        ubx.data(OpenapiPrivateLinkResource, "example", OpenapiPrivateLinkResourceConfig(
            vault_name="example-vault-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
