## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	dynatrace "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/dynatrace"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_dynatrace_monitored_subscription_properties"})

    		ubx.Data(dynatrace.MonitoredSubscriptionProperties, "example", dynatrace.MonitoredSubscriptionPropertiesConfig{
    			MonitorName: "example-monitor-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

    export default stack("example", () => {
      intent({
        summary: "look up azure_dynatrace_monitored_subscription_properties",

      data(MonitoredSubscriptionProperties, "example", {

      });

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx

            monitor_name="example-monitor-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
