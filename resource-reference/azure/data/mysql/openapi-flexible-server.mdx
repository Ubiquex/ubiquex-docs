## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	mysql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/mysql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_mysql_openapi_flexible_server"})

    		ubx.Data(mysql.OpenapiFlexibleServer, "example", mysql.OpenapiFlexibleServerConfig{
    			ServerName: "example-server-name",
    		})
    	}))
    }
    ```
  </Tab>

    export default stack("example", () => {
      intent({ summary: "look up azure_mysql_openapi_flexible_server" });

    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.mysql import OpenapiFlexibleServer, OpenapiFlexibleServerConfig

    def describe():
        ubx.intent("look up azure_mysql_openapi_flexible_server")

        ubx.data(OpenapiFlexibleServer, "example", OpenapiFlexibleServerConfig(
            server_name="example-server-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
