title: "azure_mysql_openapi_log_file"

---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	mysql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/mysql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_mysql_openapi_log_file"})

    		ubx.Data(mysql.OpenapiLogFile, "example", mysql.OpenapiLogFileConfig{
    			ServerName: "example-server-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { OpenapiLogFile } from "@ubx/sdk-azure/azure/data/mysql/openapi_log_file";

    export default stack("example", () => {
      intent({ summary: "look up azure_mysql_openapi_log_file" });

      data(OpenapiLogFile, "example", {
        serverName: "example-server-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.mysql import OpenapiLogFile, OpenapiLogFileConfig

    def describe():
        ubx.intent("look up azure_mysql_openapi_log_file")

        ubx.data(OpenapiLogFile, "example", OpenapiLogFileConfig(
            server_name="example-server-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
