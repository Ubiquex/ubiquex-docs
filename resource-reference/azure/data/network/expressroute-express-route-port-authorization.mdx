`data_azure_network_expressroute_express_route_port_authorization` looks up a real, already-existing `azure_network_expressroute_express_route_port_authorization`. It generates an authorization key letting another subscription create an ExpressRoute circuit on this ExpressRoute Direct port.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	network "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/network"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_network_expressroute_express_route_port_authorization"})

    		ubx.Data(network.ExpressrouteExpressRoutePortAuthorization, "example", network.ExpressrouteExpressRoutePortAuthorizationConfig{
    			ExpressRoutePortName: "example-express-route-port-name",
    		})
    	}))
    }
    ```
