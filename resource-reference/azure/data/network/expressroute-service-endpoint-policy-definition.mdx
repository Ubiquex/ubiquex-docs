`data_azure_network_expressroute_service_endpoint_policy_definition` looks up a real, already-existing `azure_network_expressroute_service_endpoint_policy_definition`. It is a single rule within a service endpoint policy, granting access to a specific Azure service resource.

## Example

    ```go

    import (
    	network "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/network"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_network_expressroute_service_endpoint_policy_definition"})

    		ubx.Data(network.ExpressrouteServiceEndpointPolicyDefinition, "example", network.ExpressrouteServiceEndpointPolicyDefinitionConfig{
    			ServiceEndpointPolicyName: "example-service-endpoint-policy-name",
    		})
    	}))
    }
    ```
