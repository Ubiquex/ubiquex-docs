`data_azure_network_virtualwan_network_virtual_appliance_connection` looks up a real, already-existing `azure_network_virtualwan_network_virtual_appliance_connection`. It connects a network virtual appliance deployed in a Virtual WAN hub to the hub's own routing infrastructure.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	network "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/network"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_network_virtualwan_network_virtual_appliance_connection"})

    		ubx.Data(network.VirtualwanNetworkVirtualApplianceConnection, "example", network.VirtualwanNetworkVirtualApplianceConnectionConfig{
    			NetworkVirtualApplianceName: "example-network-virtual-appliance-name",
    		})
    	}))
    }
    ```
