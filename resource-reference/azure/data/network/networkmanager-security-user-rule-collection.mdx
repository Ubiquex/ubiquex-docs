`data_azure_network_networkmanager_security_user_rule_collection` looks up a real, already-existing `azure_network_networkmanager_security_user_rule_collection`. It groups related security user rule(s) together within a Virtual Network Manager security user configuration.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	network "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/network"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_network_networkmanager_security_user_rule_collection"})

    		ubx.Data(network.NetworkmanagerSecurityUserRuleCollection, "example", network.NetworkmanagerSecurityUserRuleCollectionConfig{
    			ConfigurationName:  "example-configuration-name",
    			NetworkManagerName: "example-network-manager-name",
    		})
    	}))
    }
    ```
  </Tab>
