`azure_network_virtualnetwork_dscp_configuration` looks up a real DSCP (Differentiated Services Code Point) marking configuration applied to network traffic.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_network_virtualnetwork_dscp_configuration"})
