`data_azure_network_expressroute_express_route_circuit_authorization` looks up a real, already-existing `azure_network_expressroute_express_route_circuit_authorization`. It generates an authorization key letting another subscription's own virtual network gateway connect to this ExpressRoute circuit.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	network "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/network"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_network_expressroute_express_route_circuit_authorization"})

    		ubx.Data(network.ExpressrouteExpressRouteCircuitAuthorization, "example", network.ExpressrouteExpressRouteCircuitAuthorizationConfig{
    			CircuitName: "example-circuit-name",
    		})
    	}))
    }
    ```
