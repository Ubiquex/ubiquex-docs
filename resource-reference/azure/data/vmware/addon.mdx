---
title: "azure_vmware_addon"
description: "`data_azure_vmware_addon` looks up a real, already-existing `azure_vmware_addon`."
---

It enables an optional integration (e.g. HCX, Site Recovery, or Arc) on an Azure VMware Solution private cloud.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	vmware "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/vmware"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_vmware_addon"})

    		ubx.Data(vmware.Addon, "example", vmware.AddonConfig{
    			PrivateCloudName: "example-private-cloud-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { Addon } from "./local-sdk/azure/sdk/typescript/azure/data/vmware/addon.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_vmware_addon" });

      data(Addon, "example", {
        privateCloudName: "example-private-cloud-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.vmware import Addon, AddonConfig

    def describe():
        ubx.intent("look up azure_vmware_addon")

        ubx.data(Addon, "example", AddonConfig(
            private_cloud_name="example-private-cloud-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
