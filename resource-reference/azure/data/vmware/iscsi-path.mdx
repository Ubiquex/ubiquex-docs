## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	vmware "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/vmware"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_vmware_iscsi_path"})

    		ubx.Data(vmware.IscsiPath, "example", vmware.IscsiPathConfig{
    			PrivateCloudName: "example-private-cloud-name",
    		})
    	}))
    }
    ```

    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.vmware import IscsiPath, IscsiPathConfig

    def describe():
        ubx.intent("look up azure_vmware_iscsi_path")

        ubx.data(IscsiPath, "example", IscsiPathConfig(
            private_cloud_name="example-private-cloud-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
