title: "azure_vmware_script_cmdlet"

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	vmware "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/vmware"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_vmware_script_cmdlet"})

    		ubx.Data(vmware.ScriptCmdlet, "example", vmware.ScriptCmdletConfig{
    			PrivateCloudName:  "example-private-cloud-name",
    			ScriptPackageName: "example-script-package-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { ScriptCmdlet } from "@ubx/sdk-azure/azure/data/vmware/script_cmdlet";

    export default stack("example", () => {
      intent({ summary: "look up azure_vmware_script_cmdlet" });

      data(ScriptCmdlet, "example", {
        privateCloudName: "example-private-cloud-name",
        scriptPackageName: "example-script-package-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.vmware import ScriptCmdlet, ScriptCmdletConfig

    def describe():
        ubx.intent("look up azure_vmware_script_cmdlet")

        ubx.data(ScriptCmdlet, "example", ScriptCmdletConfig(
            private_cloud_name="example-private-cloud-name",
            script_package_name="example-script-package-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Lookup arguments

<ResponseField name="private_cloud_name" type="string" required>
  The real name of the Azure VMware Solution private cloud this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="script_package_name" type="string" required>
  The real name of the script package this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="next_link" type="string">
  The link to the next page of items Returned by the lookup.
</ResponseField>

<ResponseField name="value" type="list(object)">
  The ScriptCmdlet items on this page Returned by the lookup.
</ResponseField>
