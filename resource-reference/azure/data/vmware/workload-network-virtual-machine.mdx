`azure_vmware_workload_network_virtual_machine` looks up a real virtual machine as seen from an Azure VMware Solution private cloud's own NSX-T workload network.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	vmware "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/vmware"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_vmware_workload_network_virtual_machine"})

    		ubx.Data(vmware.WorkloadNetworkVirtualMachine, "example", vmware.WorkloadNetworkVirtualMachineConfig{
    			PrivateCloudName: "example-private-cloud-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
