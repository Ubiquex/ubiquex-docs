`data_azure_postgresql_openapi_object_recommendation` is a real, live lookup against Microsoft Azure, generated from azure's own real schema -- read-only, never created or destroyed.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	postgresql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/postgresql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_postgresql_openapi_object_recommendation"})

    		ubx.Data(postgresql.OpenapiObjectRecommendation, "example", postgresql.OpenapiObjectRecommendationConfig{
    			ServerName:         "example-server-name",
    			TuningOption:       "example",
    			RecommendationType: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
