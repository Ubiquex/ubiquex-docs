`azure_postgresql_openapi_private_endpoint_connection_list` lists the real Private Endpoint connections attached to an Azure Database for PostgreSQL flexible server.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	postgresql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/postgresql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_postgresql_openapi_private_endpoint_connection_list"})

    		ubx.Data(postgresql.OpenapiPrivateEndpointConnectionList, "example", postgresql.OpenapiPrivateEndpointConnectionListConfig{
    			ServerName: "example-server-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

    export default stack("example", () => {

          "look up azure_postgresql_openapi_private_endpoint_connection_list",
    ```

  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.postgresql import OpenapiPrivateEndpointConnectionList, OpenapiPrivateEndpointConnectionListConfig

    def describe():

        ubx.data(OpenapiPrivateEndpointConnectionList, "example", OpenapiPrivateEndpointConnectionListConfig(
            server_name="example-server-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
