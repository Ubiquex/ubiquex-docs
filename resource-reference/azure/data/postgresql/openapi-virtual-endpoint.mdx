`azure_postgresql_openapi_virtual_endpoint` looks up a real virtual endpoint providing read/write or read-only routing for an Azure Database for PostgreSQL flexible server with high availability.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	postgresql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/postgresql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_postgresql_openapi_virtual_endpoint"})

    		ubx.Data(postgresql.OpenapiVirtualEndpoint, "example", postgresql.OpenapiVirtualEndpointConfig{
    			ServerName: "example-server-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.azure.data.postgresql import OpenapiVirtualEndpoint, OpenapiVirtualEndpointConfig

    def describe():
        ubx.intent("look up azure_postgresql_openapi_virtual_endpoint")

        ubx.data(OpenapiVirtualEndpoint, "example", OpenapiVirtualEndpointConfig(
            server_name="example-server-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
