---
title: "azure_postgresql_openapi_migration"
description: "`data_azure_postgresql_openapi_migration` looks up a real, already-existing `azure_postgresql_openapi_migration`."
---

It migrates a PostgreSQL database (from on-premises, another cloud, or a different Azure PostgreSQL offering) into an Azure Database for PostgreSQL flexible server.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	postgresql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/postgresql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_postgresql_openapi_migration"})

    		ubx.Data(postgresql.OpenapiMigration, "example", postgresql.OpenapiMigrationConfig{
    			ServerName:          "example-server-name",
    			MigrationListFilter: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.azure.data.postgresql import OpenapiMigration, OpenapiMigrationConfig

    def describe():
        ubx.intent("look up azure_postgresql_openapi_migration")

        ubx.data(OpenapiMigration, "example", OpenapiMigrationConfig(
            server_name="example-server-name",
            migration_list_filter="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
