`azure_postgresql_openapi_advanced_threat_protection_settings_model` reads the real Advanced Threat Protection settings configured for an Azure Database for PostgreSQL flexible server.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	postgresql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/postgresql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_postgresql_openapi_advanced_threat_protection_settings_model"})

    		ubx.Data(postgresql.OpenapiAdvancedThreatProtectionSettingsModel, "example", postgresql.OpenapiAdvancedThreatProtectionSettingsModelConfig{
    			ServerName: "example-server-name",
    		})
    	}))
    }
    ```
