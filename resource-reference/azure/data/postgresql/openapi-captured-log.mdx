`azure_postgresql_openapi_captured_log` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	postgresql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/postgresql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_postgresql_openapi_captured_log"})

    		ubx.Data(postgresql.OpenapiCapturedLog, "example", postgresql.OpenapiCapturedLogConfig{
    			ServerName: "example-server-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { OpenapiCapturedLog } from "@ubx/sdk-azure/azure/data/postgresql/openapi_captured_log";

    export default stack("example", () => {
      intent({ summary: "look up azure_postgresql_openapi_captured_log" });

    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.postgresql import OpenapiCapturedLog, OpenapiCapturedLogConfig

    def describe():
        ubx.intent("look up azure_postgresql_openapi_captured_log")

        ubx.data(OpenapiCapturedLog, "example", OpenapiCapturedLogConfig(
            server_name="example-server-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
