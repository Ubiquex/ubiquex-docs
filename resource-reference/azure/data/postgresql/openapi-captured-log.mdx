## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	postgresql "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/postgresql"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_postgresql_openapi_captured_log"})

    		ubx.Data(postgresql.OpenapiCapturedLog, "example", postgresql.OpenapiCapturedLogConfig{
    			ServerName: "example-server-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { OpenapiCapturedLog } from "./local-sdk/azure/sdk/typescript/azure/data/postgresql/openapi_captured_log.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_postgresql_openapi_captured_log" });

      data(OpenapiCapturedLog, "example", {
        serverName: "example-server-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.postgresql import OpenapiCapturedLog, OpenapiCapturedLogConfig

    def describe():
        ubx.intent("look up azure_postgresql_openapi_captured_log")

        ubx.data(OpenapiCapturedLog, "example", OpenapiCapturedLogConfig(
            server_name="example-server-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
