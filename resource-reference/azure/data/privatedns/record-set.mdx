---
title: "azure_privatedns_record_set"
description: "`data_azure_privatedns_record_set` looks up a real, already-existing `azure_privatedns_record_set`."
---

It is a single DNS record set within an Azure Private DNS zone.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	privatedns "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/privatedns"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_privatedns_record_set"})

    		ubx.Data(privatedns.RecordSet, "example", privatedns.RecordSetConfig{
    			PrivateZoneName: "example-private-zone-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { RecordSet } from "./local-sdk/azure/sdk/typescript/azure/data/privatedns/record_set.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_privatedns_record_set" });

      data(RecordSet, "example", {
        privateZoneName: "example-private-zone-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.privatedns import RecordSet, RecordSetConfig

    def describe():
        ubx.intent("look up azure_privatedns_record_set")

        ubx.data(RecordSet, "example", RecordSetConfig(
            private_zone_name="example-private-zone-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
