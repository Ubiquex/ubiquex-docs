## Example

<Tabs>

    ```go
    package main

    import (
    	signalr "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/signalr"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_signalr_signal_rusage"})

    		ubx.Data(signalr.SignalRusage, "example", signalr.SignalRusageConfig{
    			Location: "example",
    		})
    	}))
    }
    ```
