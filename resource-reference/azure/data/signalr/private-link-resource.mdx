## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	signalr "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/signalr"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_signalr_private_link_resource"})

    		ubx.Data(signalr.PrivateLinkResource, "example", signalr.PrivateLinkResourceConfig{
    			ResourceName: "example-resource-name",
    		})
    	}))
    }
    ```

    from ubx.azure.data.signalr import PrivateLinkResource, PrivateLinkResourceConfig

    def describe():
        ubx.intent("look up azure_signalr_private_link_resource")

        ubx.data(PrivateLinkResource, "example", PrivateLinkResourceConfig(

        ))

    if __name__ == "__main__":
