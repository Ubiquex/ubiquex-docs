`azure_signalr_custom_domain` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	signalr "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/signalr"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_signalr_custom_domain"})

    		ubx.Data(signalr.CustomDomain, "example", signalr.CustomDomainConfig{
    			ResourceName: "example-resource-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { CustomDomain } from "@ubx/sdk-azure/azure/data/signalr/custom_domain";

    export default stack("example", () => {
      intent({ summary: "look up azure_signalr_custom_domain" });

      data(CustomDomain, "example", {
        resourceName: "example-resource-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.signalr import CustomDomain, CustomDomainConfig

    def describe():
        ubx.intent("look up azure_signalr_custom_domain")

        ubx.data(CustomDomain, "example", CustomDomainConfig(
            resource_name="example-resource-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
