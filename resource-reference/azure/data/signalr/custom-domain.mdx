---
title: "azure_signalr_custom_domain"
description: "`data_azure_signalr_custom_domain` looks up a real, already-existing `azure_signalr_custom_domain`."
---

It maps a custom domain to a SignalR Service resource.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	signalr "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/signalr"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_signalr_custom_domain"})

    		ubx.Data(signalr.CustomDomain, "example", signalr.CustomDomainConfig{
    			ResourceName: "example-resource-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { CustomDomain } from "./local-sdk/azure/sdk/typescript/azure/data/signalr/custom_domain.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_signalr_custom_domain" });

      data(CustomDomain, "example", {
        resourceName: "example-resource-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.signalr import CustomDomain, CustomDomainConfig

    def describe():
        ubx.intent("look up azure_signalr_custom_domain")

        ubx.data(CustomDomain, "example", CustomDomainConfig(
            resource_name="example-resource-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Lookup arguments

<ResponseField name="resource_name" type="string" required>
  The real name of the resource this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

## Result properties

<ResponseField name="next_link" type="string">
  The URL the client should use to fetch the next page (per server side paging). It's null for now, added for future use. Returned by the lookup.
</ResponseField>

<ResponseField name="value" type="list(object)">
  List of custom domains that bind to this resource. Returned by the lookup.
</ResponseField>
