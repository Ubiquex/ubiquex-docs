## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	batch "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/batch"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_batch_openapi_detector_response"})

    		ubx.Data(batch.OpenapiDetectorResponse, "example", batch.OpenapiDetectorResponseConfig{
    			AccountName: "example-account-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.batch import OpenapiDetectorResponse, OpenapiDetectorResponseConfig

    def describe():
        ubx.intent("look up azure_batch_openapi_detector_response")

        ubx.data(OpenapiDetectorResponse, "example", OpenapiDetectorResponseConfig(
            account_name="example-account-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
