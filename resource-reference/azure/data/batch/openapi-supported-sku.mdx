`data_azure_batch_openapi_supported_sku` is a real, live lookup against Microsoft Azure, generated from azure's own real schema -- read-only, never created or destroyed.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	batch "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/batch"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_batch_openapi_supported_sku"})

    		ubx.Data(batch.OpenapiSupportedSku, "example", batch.OpenapiSupportedSkuConfig{
    			LocationName: "example-location-name",
    			Maxresults:   1,
    		})
    	}))
    }
    ```
  </Tab>
