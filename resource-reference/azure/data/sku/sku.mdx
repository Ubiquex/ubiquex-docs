    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_sku"})

    		ubx.Data(sku.Sku, "example", sku.SkuConfig{
    			EdgeZone:      "example",
    			Offer:         "example",
    			PublisherName: "example-publisher-name",
    			Skus:          "example",
    		})
    	}))
    }

## Lookup arguments

<ResponseField name="edge_zone" type="string" required>
  The real Azure Edge Zone this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="offer" type="string" required>
  The real Marketplace offer this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="publisher_name" type="string" required>
  The real name of the publisher this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="skus" type="string" required>
  The real, available pricing tiers (SKUs) included in this result. **(AI-inferred)** Required lookup argument.
</ResponseField>
