## Example

<Tabs>

    package main

    import (
    	hybridcompute "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/hybridcompute"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_hybridcompute_openapi_network_interface"})

    		ubx.Data(hybridcompute.OpenapiNetworkInterface, "example", hybridcompute.OpenapiNetworkInterfaceConfig{
    			MachineName: "example-machine-name",
    		})
    	}))
    }
