`data_azure_communication_openapi_suppression_list_address_resource` looks up a real, already-existing `azure_communication_openapi_suppression_list_address_resource`. It is a single email address added to an Azure Communication Services suppression list, blocking further email delivery to it.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_communication_openapi_suppression_list_address_resource"})

    		ubx.Data(communication.OpenapiSuppressionListAddressResource, "example", communication.OpenapiSuppressionListAddressResourceConfig{
    			DomainName:          "example-domain-name",

    		})
