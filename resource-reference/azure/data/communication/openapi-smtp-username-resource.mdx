It is a SMTP-specific credential username for authenticating to Azure Communication Services Email over SMTP.

## Example

    import (

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_communication_openapi_smtp_username_resource"})

    		ubx.Data(communication.OpenapiSmtpUsernameResource, "example", communication.OpenapiSmtpUsernameResourceConfig{
    			CommunicationServiceName: "example-communication-service-name",
    		})
    	}))
    }
