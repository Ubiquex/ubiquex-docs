`azure_servicefabric_application_type_resource` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_servicefabric_application_type_resource"})

    		ubx.Data(servicefabric.ApplicationTypeResource, "example", servicefabric.ApplicationTypeResourceConfig{
    			ApiVersion:        "example",
    			ClusterName:       "example-cluster-name",

    			SubscriptionId:    "example",
    		})
    	}))
    }
