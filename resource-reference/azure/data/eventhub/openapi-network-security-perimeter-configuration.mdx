`azure_eventhub_openapi_network_security_perimeter_configuration` reads the real Network Security Perimeter configuration applied to an Event Hubs namespace.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	eventhub "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/eventhub"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_eventhub_openapi_network_security_perimeter_configuration"})

    		ubx.Data(eventhub.OpenapiNetworkSecurityPerimeterConfiguration, "example", eventhub.OpenapiNetworkSecurityPerimeterConfigurationConfig{
    			NamespaceName: "example-namespace-name",
    		})
    	}))
    }
    ```
  </Tab>
