`data_azure_datafactory_openapi_private_endpoint_connection_resource` looks up a real, already-existing `azure_datafactory_openapi_private_endpoint_connection_resource`. It represents the connection between a Private Endpoint and a Data Factory, tracking its own approval state.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	datafactory "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/datafactory"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_datafactory_openapi_private_endpoint_connection_resource"})

    		ubx.Data(datafactory.OpenapiPrivateEndpointConnectionResource, "example", datafactory.OpenapiPrivateEndpointConnectionResourceConfig{
    			FactoryName: "example-factory-name",
    		})
    	}))
    }
    ```
  </Tab>
