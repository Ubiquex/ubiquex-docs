---
title: "azure_datafactory_openapi_global_parameter_resource"
description: "`data_azure_datafactory_openapi_global_parameter_resource` looks up a real, already-existing `azure_datafactory_openapi_global_parameter_resource`."
---

It defines a factory-wide parameter value referenceable from any pipeline within a Data Factory, without redeclaring it per pipeline.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	datafactory "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/datafactory"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_datafactory_openapi_global_parameter_resource"})

    		ubx.Data(datafactory.OpenapiGlobalParameterResource, "example", datafactory.OpenapiGlobalParameterResourceConfig{
    			FactoryName: "example-factory-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { OpenapiGlobalParameterResource } from "./local-sdk/azure/sdk/typescript/azure/data/datafactory/openapi_global_parameter_resource.ts";

    export default stack("example", () => {
      intent({
        summary: "look up azure_datafactory_openapi_global_parameter_resource",
      });

      data(OpenapiGlobalParameterResource, "example", {
        factoryName: "example-factory-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.datafactory import OpenapiGlobalParameterResource, OpenapiGlobalParameterResourceConfig

    def describe():
        ubx.intent("look up azure_datafactory_openapi_global_parameter_resource")

        ubx.data(OpenapiGlobalParameterResource, "example", OpenapiGlobalParameterResourceConfig(
            factory_name="example-factory-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
