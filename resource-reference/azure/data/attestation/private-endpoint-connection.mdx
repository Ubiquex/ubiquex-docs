## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	attestation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/attestation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_attestation_private_endpoint_connection"})

    		ubx.Data(attestation.PrivateEndpointConnection, "example", attestation.PrivateEndpointConnectionConfig{
    			ProviderName: "example-provider-name",
    		})
    	}))
    }
    ```
  </Tab>
