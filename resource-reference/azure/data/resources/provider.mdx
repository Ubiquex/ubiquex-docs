title: "azure_resources_provider"

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	resources "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/resources"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_resources_provider"})

    		ubx.Data(resources.Provider, "example", resources.ProviderConfig{})
    	}))
    }
    ```
