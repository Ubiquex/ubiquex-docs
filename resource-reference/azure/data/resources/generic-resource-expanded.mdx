## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	resources "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/resources"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_resources_generic_resource_expanded"})

    		ubx.Data(resources.GenericResourceExpanded, "example", resources.GenericResourceExpandedConfig{
    			ResourceGroupName: "example-resource-group-name",
    		})
    	}))
    }
    ```

    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.resources import GenericResourceExpanded, GenericResourceExpandedConfig

    def describe():
        ubx.intent("look up azure_resources_generic_resource_expanded")

        ubx.data(GenericResourceExpanded, "example", GenericResourceExpandedConfig(
            resource_group_name="example-resource-group-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
