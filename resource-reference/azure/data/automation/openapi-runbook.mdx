---
title: "azure_automation_openapi_runbook"
description: "`data_azure_automation_openapi_runbook` looks up a real, already-existing `azure_automation_openapi_runbook`."
---

It is the executable script (PowerShell, Python, or a graphical workflow) that Azure Automation runs, either on a schedule, on demand, or via webhook.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi runbook"})

    		ubx.Resource(automation.OpenapiRunbook, "example", automation.OpenapiRunbookConfig{
    			Properties: map[string]any{"runbook_type": "example"},
    			Name:       "example",
    			Location:   "example",
    			Tags:       map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```

    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.automation import OpenapiRunbook, OpenapiRunbookConfig
    from ubx.azure.automation.openapi_runbook import OpenapiRunbook_Properties

    def describe():
        ubx.intent("example own openapi runbook")

        ubx.resource(OpenapiRunbook, "example", OpenapiRunbookConfig(
            properties=OpenapiRunbook_Properties(runbook_type="example"),
            name="example",
            location="example",
            tags={"managed-by": "ubx"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
