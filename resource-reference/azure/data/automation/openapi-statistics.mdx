## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_automation_openapi_statistics"})

    		ubx.Data(automation.OpenapiStatistics, "example", automation.OpenapiStatisticsConfig{
    			AutomationAccountName: "example-automation-account-name",
    		})
    	}))
    }
    ```
