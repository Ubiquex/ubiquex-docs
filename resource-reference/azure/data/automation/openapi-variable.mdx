---
title: "azure_automation_openapi_variable"
description: "`data_azure_automation_openapi_variable` looks up a real, already-existing `azure_automation_openapi_variable`."
---

It stores a named configuration value (plain or encrypted) within an Azure Automation account, referenced by runbook(s).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_automation_openapi_variable"})

    		ubx.Data(automation.OpenapiVariable, "example", automation.OpenapiVariableConfig{
    			AutomationAccountName: "example-automation-account-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { OpenapiVariable } from "./local-sdk/azure/sdk/typescript/azure/data/automation/openapi_variable.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_automation_openapi_variable" });

      data(OpenapiVariable, "example", {
        automationAccountName: "example-automation-account-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.automation import OpenapiVariable, OpenapiVariableConfig

    def describe():
        ubx.intent("look up azure_automation_openapi_variable")

        ubx.data(OpenapiVariable, "example", OpenapiVariableConfig(
            automation_account_name="example-automation-account-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
