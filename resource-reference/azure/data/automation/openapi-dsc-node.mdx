## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_automation_openapi_dsc_node"})

    		ubx.Data(automation.OpenapiDscNode, "example", automation.OpenapiDscNodeConfig{
    			AutomationAccountName: "example-automation-account-name",
    		})
    	}))
    }
    ```

    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.automation import OpenapiDscNode, OpenapiDscNodeConfig

    def describe():
        ubx.intent("look up azure_automation_openapi_dsc_node")

        ubx.data(OpenapiDscNode, "example", OpenapiDscNodeConfig(
            automation_account_name="example-automation-account-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
