## Example

    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/automation"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_automation_openapi_job_collection_item"})

    		ubx.Data(automation.OpenapiJobCollectionItem, "example", automation.OpenapiJobCollectionItemConfig{
    			AutomationAccountName: "example-automation-account-name",
    		})
    	}))
    }
    ```
