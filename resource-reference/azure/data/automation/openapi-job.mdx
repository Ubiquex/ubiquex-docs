---
title: "azure_automation_openapi_job"
description: "`data_azure_automation_openapi_job` looks up a real, already-existing `azure_automation_openapi_job`."
---

It is a single execution of an Azure Automation runbook.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi job"})

    		ubx.Resource(automation.OpenapiJob, "example", automation.OpenapiJobConfig{
    			Properties: map[string]any{"parameters": map[string]string{"key": "value"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiJob } from "@ubx/sdk-azure/azure/automation/openapi_job";

    export default stack("example", () => {
      intent({ summary: "example own openapi job" });

      resource(OpenapiJob, "example", {
        properties: { parameters: { key: "value" } },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.automation import OpenapiJob, OpenapiJobConfig
    from ubx.azure.automation.openapi_job import OpenapiJob_Properties

    def describe():
        ubx.intent("example own openapi job")

        ubx.resource(OpenapiJob, "example", OpenapiJobConfig(
            properties=OpenapiJob_Properties(parameters={"key": "value"}),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
