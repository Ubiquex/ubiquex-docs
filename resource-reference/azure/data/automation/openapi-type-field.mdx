`data_azure_automation_openapi_type_field` is a real, live lookup against Microsoft Azure, generated from azure's own real schema -- read-only, never created or destroyed.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_automation_openapi_type_field"})

    		ubx.Data(automation.OpenapiTypeField, "example", automation.OpenapiTypeFieldConfig{
    			AutomationAccountName: "example-automation-account-name",
    			ModuleName:            "example-module-name",
    			TypeName:              "example-type-name",
    		})
    	}))
    }
    ```
  </Tab>
