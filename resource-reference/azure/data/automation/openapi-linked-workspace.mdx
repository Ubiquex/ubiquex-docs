## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_automation_openapi_linked_workspace"})

    		ubx.Data(automation.OpenapiLinkedWorkspace, "example", automation.OpenapiLinkedWorkspaceConfig{
    			AutomationAccountName: "example-automation-account-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { OpenapiLinkedWorkspace } from "@ubx/sdk-azure/azure/data/automation/openapi_linked_workspace";

    export default stack("example", () => {
      intent({ summary: "look up azure_automation_openapi_linked_workspace" });

      data(OpenapiLinkedWorkspace, "example", {

      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.automation import OpenapiLinkedWorkspace, OpenapiLinkedWorkspaceConfig

    def describe():
        ubx.intent("look up azure_automation_openapi_linked_workspace")

        ubx.data(OpenapiLinkedWorkspace, "example", OpenapiLinkedWorkspaceConfig(
            automation_account_name="example-automation-account-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
