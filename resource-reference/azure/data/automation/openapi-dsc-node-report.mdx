    		})
    	}))
    }

## Lookup arguments

<ResponseField name="automation_account_name" type="string" required>
  The real name of the Azure Automation account this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="node_id" type="string" required>
  The real, unique identifier of the node this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
