---
title: "azure_automation_openapi_package"
description: "`data_azure_automation_openapi_package` looks up a real, already-existing `azure_automation_openapi_package`."
---

It imports a Python package into an Azure Automation account, making it available to Python runbook(s).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_automation_openapi_package"})

    		ubx.Data(automation.OpenapiPackage, "example", automation.OpenapiPackageConfig{
    			AutomationAccountName:  "example-automation-account-name",
    			RuntimeEnvironmentName: "example-runtime-environment-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { OpenapiPackage } from "./local-sdk/azure/sdk/typescript/azure/data/automation/openapi_package.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_automation_openapi_package" });

      data(OpenapiPackage, "example", {
        automationAccountName: "example-automation-account-name",
        runtimeEnvironmentName: "example-runtime-environment-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.automation import OpenapiPackage, OpenapiPackageConfig

    def describe():
        ubx.intent("look up azure_automation_openapi_package")

        ubx.data(OpenapiPackage, "example", OpenapiPackageConfig(
            automation_account_name="example-automation-account-name",
            runtime_environment_name="example-runtime-environment-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Lookup arguments

<ResponseField name="automation_account_name" type="string" required>
  The real name of the Azure Automation account this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="runtime_environment_name" type="string" required>
  The real name of the runtime environment this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>
