`azure_automation_openapi_package` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_automation_openapi_package"})

    		ubx.Data(automation.OpenapiPackage, "example", automation.OpenapiPackageConfig{
    			AutomationAccountName:  "example-automation-account-name",
    			RuntimeEnvironmentName: "example-runtime-environment-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.automation import OpenapiPackage, OpenapiPackageConfig

    def describe():

        ubx.data(OpenapiPackage, "example", OpenapiPackageConfig(
            automation_account_name="example-automation-account-name",
            runtime_environment_name="example-runtime-environment-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
