---
title: "azure_automation_openapi_source_control"
description: "`data_azure_automation_openapi_source_control` looks up a real, already-existing `azure_automation_openapi_source_control`."
---

It links an Azure Automation account to an external Git repository, syncing runbook source code automatically on push.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_automation_openapi_source_control"})

    		ubx.Data(automation.OpenapiSourceControl, "example", automation.OpenapiSourceControlConfig{
    			AutomationAccountName: "example-automation-account-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { OpenapiSourceControl } from "./local-sdk/azure/sdk/typescript/azure/data/automation/openapi_source_control.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_automation_openapi_source_control" });

      data(OpenapiSourceControl, "example", {
        automationAccountName: "example-automation-account-name",
      });
    });
    ```

  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.automation import OpenapiSourceControl, OpenapiSourceControlConfig

    def describe():
        ubx.intent("look up azure_automation_openapi_source_control")

        ubx.data(OpenapiSourceControl, "example", OpenapiSourceControlConfig(
            automation_account_name="example-automation-account-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
