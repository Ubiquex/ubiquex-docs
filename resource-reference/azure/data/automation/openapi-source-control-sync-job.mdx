`azure_automation_openapi_source_control_sync_job` reads the real, current status of a sync between an Azure Automation account and its own connected source control repository.

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_automation_openapi_source_control_sync_job"})

    		ubx.Data(automation.OpenapiSourceControlSyncJob, "example", automation.OpenapiSourceControlSyncJobConfig{

    		})
    	}))
    }
