`azure_automation_openapi_agent_registration` reads the real registration information (endpoint and key) a Hybrid Runbook Worker uses to connect to an Azure Automation account.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_automation_openapi_agent_registration"})

    		ubx.Data(automation.OpenapiAgentRegistration, "example", automation.OpenapiAgentRegistrationConfig{
    			AutomationAccountName: "example-automation-account-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { OpenapiAgentRegistration } from "./local-sdk/azure/sdk/typescript/azure/data/automation/openapi_agent_registration.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_automation_openapi_agent_registration" });
    ```

  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.automation import OpenapiAgentRegistration, OpenapiAgentRegistrationConfig

    def describe():
        ubx.intent("look up azure_automation_openapi_agent_registration")

        ubx.data(OpenapiAgentRegistration, "example", OpenapiAgentRegistrationConfig(
            automation_account_name="example-automation-account-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
