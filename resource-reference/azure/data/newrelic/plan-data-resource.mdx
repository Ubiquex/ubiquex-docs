`data_azure_newrelic_plan_data_resource` is a real, live lookup against Microsoft Azure, generated from azure's own real schema -- read-only, never created or destroyed.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	newrelic "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/newrelic"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_newrelic_plan_data_resource"})

    		ubx.Data(newrelic.PlanDataResource, "example", newrelic.PlanDataResourceConfig{
    			AccountId:      "example",
    			OrganizationId: "example",
    		})
    	}))
    }
    ```
