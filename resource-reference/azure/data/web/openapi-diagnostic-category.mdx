## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_web_openapi_diagnostic_category"})

    		ubx.Data(web.OpenapiDiagnosticCategory, "example", web.OpenapiDiagnosticCategoryConfig{
    			SiteName: "example-site-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript

    export default stack("example", () => {
    ```

  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.web import OpenapiDiagnosticCategory, OpenapiDiagnosticCategoryConfig

    def describe():
        ubx.intent("look up azure_web_openapi_diagnostic_category")

        ubx.data(OpenapiDiagnosticCategory, "example", OpenapiDiagnosticCategoryConfig(
            site_name="example-site-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
