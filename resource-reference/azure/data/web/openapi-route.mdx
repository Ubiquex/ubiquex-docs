    ```python
    import ubx_sdk as ubx
    from ubx.azure.web import OpenapiRoute, OpenapiRouteConfig
    from ubx.azure.web.openapi_route import OpenapiRoute_Properties

    def describe():
        ubx.intent("example own openapi route")

        ubx.resource(OpenapiRoute, "example", OpenapiRouteConfig(
            name="example",
            kind="example",
            properties=OpenapiRoute_Properties(end_address="example"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
