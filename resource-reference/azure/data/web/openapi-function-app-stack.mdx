## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_web_openapi_function_app_stack"})

    		ubx.Data(web.OpenapiFunctionAppStack, "example", web.OpenapiFunctionAppStackConfig{
    			StackOsType: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { OpenapiFunctionAppStack } from "@ubx/sdk-azure/azure/data/web/openapi_function_app_stack";

    export default stack("example", () => {
      intent({ summary: "look up azure_web_openapi_function_app_stack" });

      data(OpenapiFunctionAppStack, "example", {
        stackOsType: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.web import OpenapiFunctionAppStack, OpenapiFunctionAppStackConfig

    def describe():
        ubx.intent("look up azure_web_openapi_function_app_stack")

        ubx.data(OpenapiFunctionAppStack, "example", OpenapiFunctionAppStackConfig(
            stack_os_type="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>

## Lookup arguments

<ResponseField name="stack_os_type" type="string">

</ResponseField>
