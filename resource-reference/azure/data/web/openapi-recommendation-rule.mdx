`data_azure_web_openapi_recommendation_rule` is a real, live lookup against Microsoft Azure, generated from azure's own real schema -- read-only, never created or destroyed.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	web "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/web"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_web_openapi_recommendation_rule"})

    		ubx.Data(web.OpenapiRecommendationRule, "example", web.OpenapiRecommendationRuleConfig{
    			HostingEnvironmentName: "example-hosting-environment-name",
    			Name:                   "example",
    			RecommendationId:       "example",
    			UpdateSeen:             true,
    		})
    	}))
    }
    ```
