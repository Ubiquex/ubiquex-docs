`data_azure_machinelearningservices_openapi_serverless_endpoint` looks up a real, already-existing `azure_machinelearningservices_openapi_serverless_endpoint`. It deploys a Models as a Service (MaaS) foundation model from the Azure AI Foundry Model Catalog to a pay-per-token serverless endpoint, without provisioning dedicated compute.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_machinelearningservices_openapi_serverless_endpoint"})

    		ubx.Data(machinelearningservices.OpenapiServerlessEndpoint, "example", machinelearningservices.OpenapiServerlessEndpointConfig{
    			WorkspaceName: "example-workspace-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.machinelearningservices import OpenapiServerlessEndpoint, OpenapiServerlessEndpointConfig

    def describe():
        ubx.intent("look up azure_machinelearningservices_openapi_serverless_endpoint")

        ubx.data(OpenapiServerlessEndpoint, "example", OpenapiServerlessEndpointConfig(
            workspace_name="example-workspace-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
