`azure_machinelearningservices_openapi_datastore` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    package main

    import (

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_machinelearningservices_openapi_datastore"})

    		ubx.Data(machinelearningservices.OpenapiDatastore, "example", machinelearningservices.OpenapiDatastoreConfig{
    			WorkspaceName: "example-workspace-name",
    			Count:         1,
    			IsDefault:     true,
    			Names:         []string{"example"},
    			OrderBy:       "example",
    			OrderByAsc:    true,
    			SearchText:    "example",
    		})
    	}))
    }
