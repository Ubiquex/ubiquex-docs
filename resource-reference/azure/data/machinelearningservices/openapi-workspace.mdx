## Example

  <Tab title="Go">
    ```go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_machinelearningservices_openapi_workspace"})

    		ubx.Data(machinelearningservices.OpenapiWorkspace, "example", machinelearningservices.OpenapiWorkspaceConfig{
    			AiCapabilities: "example",
    			Kind:           "example",
    		})
    	}))
    }
    ```
