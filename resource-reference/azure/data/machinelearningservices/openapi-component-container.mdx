`data_azure_machinelearningservices_openapi_component_container` looks up a real, already-existing `azure_machinelearningservices_openapi_component_container`. It is the named container in an Azure Machine Learning workspace that version(s) of a reusable pipeline component are registered under.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi component container"})

    		ubx.Resource(machinelearningservices.OpenapiComponentContainer, "example", machinelearningservices.OpenapiComponentContainerConfig{
    			Properties: map[string]any{"provisioning_state": "example"},
    		})
    	}))
    }
    ```

        ubx.intent("example own openapi component container")

    if __name__ == "__main__":
