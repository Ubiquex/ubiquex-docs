`data_azure_machinelearningservices_openapi_component_container` looks up a real, already-existing `azure_machinelearningservices_openapi_component_container`. It is the named container in an Azure Machine Learning workspace that version(s) of a reusable pipeline component are registered under.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_machinelearningservices_openapi_component_container"})

    		ubx.Data(machinelearningservices.OpenapiComponentContainer, "example", machinelearningservices.OpenapiComponentContainerConfig{
    			RegistryName: "example-registry-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.machinelearningservices import OpenapiComponentContainer, OpenapiComponentContainerConfig

    def describe():
        ubx.intent("look up azure_machinelearningservices_openapi_component_container")

        ubx.data(OpenapiComponentContainer, "example", OpenapiComponentContainerConfig(
            registry_name="example-registry-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
