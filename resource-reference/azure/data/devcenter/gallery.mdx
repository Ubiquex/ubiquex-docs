---
title: "azure_devcenter_gallery"
description: "`data_azure_devcenter_gallery` looks up a real, already-existing `azure_devcenter_gallery`."
---

It attaches an Azure Compute Gallery to a dev center, making its own custom VM images available for dev box definitions.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	devcenter "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/devcenter"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_devcenter_gallery"})

    		ubx.Data(devcenter.Gallery, "example", devcenter.GalleryConfig{
    			DevCenterName: "example-dev-center-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { Gallery } from "@ubx/sdk-azure/azure/data/devcenter/gallery";

    export default stack("example", () => {
      intent({ summary: "look up azure_devcenter_gallery" });

      data(Gallery, "example", {
        devCenterName: "example-dev-center-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.devcenter import Gallery, GalleryConfig

    def describe():
        ubx.intent("look up azure_devcenter_gallery")

        ubx.data(Gallery, "example", GalleryConfig(
            dev_center_name="example-dev-center-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
