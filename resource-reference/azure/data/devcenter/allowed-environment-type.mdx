## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	devcenter "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/devcenter"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_devcenter_allowed_environment_type"})

    		ubx.Data(devcenter.AllowedEnvironmentType, "example", devcenter.AllowedEnvironmentTypeConfig{
    			ProjectName: "example-project-name",
    		})
    	}))
    }
    ```
