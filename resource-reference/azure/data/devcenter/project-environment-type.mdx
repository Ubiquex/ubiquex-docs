---
title: "azure_devcenter_project_environment_type"
description: "`data_azure_devcenter_project_environment_type` looks up a real, already-existing `azure_devcenter_project_environment_type`."
---

It enables a dev center-level environment type within a specific project, with its own project-scoped deployment identity and target subscription.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	devcenter "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/devcenter"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_devcenter_project_environment_type"})

    		ubx.Data(devcenter.ProjectEnvironmentType, "example", devcenter.ProjectEnvironmentTypeConfig{
    			ProjectName: "example-project-name",
    		})
    	}))
    }
    ```

    from ubx.azure.data.devcenter import ProjectEnvironmentType, ProjectEnvironmentTypeConfig

    def describe():
        ubx.intent("look up azure_devcenter_project_environment_type")

        ubx.data(ProjectEnvironmentType, "example", ProjectEnvironmentTypeConfig(
            project_name="example-project-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
