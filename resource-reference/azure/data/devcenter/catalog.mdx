---
title: "azure_devcenter_catalog"
description: "`data_azure_devcenter_catalog` looks up a real, already-existing `azure_devcenter_catalog`."
---

It links a Git repository of Environment Definitions or image templates to a dev center, making its own reusable infrastructure templates available for self-service provisioning.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	devcenter "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/devcenter"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_devcenter_catalog"})

    		ubx.Data(devcenter.Catalog, "example", devcenter.CatalogConfig{
    			DevCenterName: "example-dev-center-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { Catalog } from "./local-sdk/azure/sdk/typescript/azure/data/devcenter/catalog.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_devcenter_catalog" });

      data(Catalog, "example", {
        devCenterName: "example-dev-center-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.devcenter import Catalog, CatalogConfig

    def describe():
        ubx.intent("look up azure_devcenter_catalog")

        ubx.data(Catalog, "example", CatalogConfig(
            dev_center_name="example-dev-center-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
</Tabs>
