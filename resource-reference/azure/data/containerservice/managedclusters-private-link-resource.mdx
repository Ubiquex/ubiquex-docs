`azure_containerservice_managedclusters_private_link_resource` reads a real Private Link resource exposed by an AKS cluster, that a Private Endpoint can connect to.

## Example

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_containerservice_managedclusters_private_link_resource"})

    		ubx.Data(containerservice.ManagedclustersPrivateLinkResource, "example", containerservice.ManagedclustersPrivateLinkResourceConfig{

    		})
    	}))
    }
