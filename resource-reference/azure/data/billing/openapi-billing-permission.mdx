## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	billing "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/billing"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_billing_openapi_billing_permission"})

    		ubx.Data(billing.OpenapiBillingPermission, "example", billing.OpenapiBillingPermissionConfig{
    			BillingAccountName: "example-billing-account-name",
    		})
    	}))
    }
    ```
