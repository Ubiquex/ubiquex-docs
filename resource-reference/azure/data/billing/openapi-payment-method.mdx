    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	billing "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/billing"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_billing_openapi_payment_method"})

    		ubx.Data(billing.OpenapiPaymentMethod, "example", billing.OpenapiPaymentMethodConfig{
    			BillingAccountName: "example-billing-account-name",
    		})
    	}))
    }
    ```

    ```python
    from ubx.azure.data.billing import OpenapiPaymentMethod, OpenapiPaymentMethodConfig

    def describe():
        ubx.intent("look up azure_billing_openapi_payment_method")

        ubx.data(OpenapiPaymentMethod, "example", OpenapiPaymentMethodConfig(

        ))

    if __name__ == "__main__":
    ```
