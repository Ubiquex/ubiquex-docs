    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	devopsinfrastructure "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/devopsinfrastructure"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_devopsinfrastructure_image_version"})

    		ubx.Data(devopsinfrastructure.ImageVersion, "example", devopsinfrastructure.ImageVersionConfig{
    			ImageName: "example-image-name",
    		})
    	}))
    }
