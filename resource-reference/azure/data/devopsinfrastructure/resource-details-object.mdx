    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {

    		ubx.Data(devopsinfrastructure.ResourceDetailsObject, "example", devopsinfrastructure.ResourceDetailsObjectConfig{

    		})
    	}))
    }
