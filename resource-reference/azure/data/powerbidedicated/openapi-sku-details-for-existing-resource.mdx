`azure_powerbidedicated_openapi_sku_details_for_existing_resource` reads a real, available pricing tier (SKU) an existing Power BI Embedded capacity can be resized to.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	powerbidedicated "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/powerbidedicated"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_powerbidedicated_openapi_sku_details_for_existing_resource"})

    		ubx.Data(powerbidedicated.OpenapiSkuDetailsForExistingResource, "example", powerbidedicated.OpenapiSkuDetailsForExistingResourceConfig{
    			DedicatedCapacityName: "example-dedicated-capacity-name",
    		})
    	}))
    }
    ```
