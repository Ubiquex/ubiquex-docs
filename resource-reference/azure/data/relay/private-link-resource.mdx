## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	relay "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/relay"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_relay_private_link_resource"})

    		ubx.Data(relay.PrivateLinkResource, "example", relay.PrivateLinkResourceConfig{
    			NamespaceName: "example-namespace-name",
    		})
    	}))
    }
    ```

    from ubx.azure.data.relay import PrivateLinkResource, PrivateLinkResourceConfig

        ubx.intent("look up azure_relay_private_link_resource")

        ubx.data(PrivateLinkResource, "example", PrivateLinkResourceConfig(

        ))

    if __name__ == "__main__":
