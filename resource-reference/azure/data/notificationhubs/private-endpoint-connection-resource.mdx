`data_azure_notificationhubs_private_endpoint_connection_resource` looks up a real, already-existing `azure_notificationhubs_private_endpoint_connection_resource`. It represents the connection between a Private Endpoint and a Notification Hubs namespace, tracking its own approval state.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	notificationhubs "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/notificationhubs"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_notificationhubs_private_endpoint_connection_resource"})

    		ubx.Data(notificationhubs.PrivateEndpointConnectionResource, "example", notificationhubs.PrivateEndpointConnectionResourceConfig{
    			NamespaceName: "example-namespace-name",
    		})
    	}))
    }
    ```
