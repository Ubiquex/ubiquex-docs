---
title: "azure_cdn_openapi_afddomain"
description: "`data_azure_cdn_openapi_afddomain` looks up a real, already-existing `azure_cdn_openapi_afddomain`."
---

It maps a custom domain to an Azure Front Door endpoint, the current-generation (AFD Standard/Premium) equivalent of a classic CDN custom domain.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	cdn "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/cdn"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_cdn_openapi_afddomain"})

    		ubx.Data(cdn.OpenapiAfddomain, "example", cdn.OpenapiAfddomainConfig{
    			ProfileName: "example-profile-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { OpenapiAfddomain } from "./local-sdk/azure/sdk/typescript/azure/data/cdn/openapi_afddomain.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_cdn_openapi_afddomain" });

      data(OpenapiAfddomain, "example", {
        profileName: "example-profile-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.cdn import OpenapiAfddomain, OpenapiAfddomainConfig

    def describe():
        ubx.intent("look up azure_cdn_openapi_afddomain")

        ubx.data(OpenapiAfddomain, "example", OpenapiAfddomainConfig(
            profile_name="example-profile-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
