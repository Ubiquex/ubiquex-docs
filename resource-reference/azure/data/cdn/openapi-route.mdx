`azure_cdn_openapi_route` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	cdn "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/cdn"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_cdn_openapi_route"})

    		ubx.Data(cdn.OpenapiRoute, "example", cdn.OpenapiRouteConfig{
    			EndpointName: "example-endpoint-name",
    			ProfileName:  "example-profile-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    import { OpenapiRoute } from "@ubx/sdk-azure/azure/data/cdn/openapi_route";

    export default stack("example", () => {
      intent({ summary: "look up azure_cdn_openapi_route" });

      data(OpenapiRoute, "example", {

        profileName: "example-profile-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.data.cdn import OpenapiRoute, OpenapiRouteConfig

    def describe():
        ubx.intent("look up azure_cdn_openapi_route")

        ubx.data(OpenapiRoute, "example", OpenapiRouteConfig(
            endpoint_name="example-endpoint-name",
            profile_name="example-profile-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
