`azure_appplatform_predefined_accelerator_resource` looks up a real, Microsoft-provided project template available in Azure Spring Apps' own Application Accelerator.

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_appplatform_predefined_accelerator_resource"})
