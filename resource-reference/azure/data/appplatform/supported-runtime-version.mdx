## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	appplatform "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/appplatform"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_appplatform_supported_runtime_version"})

    		ubx.Data(appplatform.SupportedRuntimeVersion, "example", appplatform.SupportedRuntimeVersionConfig{
    			ApiVersion: "example",
    		})
    	}))
    }
    ```
  </Tab>
