## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	app "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/app"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_app_openapi_revision"})

    		ubx.Data(app.OpenapiRevision, "example", app.OpenapiRevisionConfig{
    			ContainerAppName: "example-container-app-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { OpenapiRevision } from "./local-sdk/azure/sdk/typescript/azure/data/app/openapi_revision.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_app_openapi_revision" });

      data(OpenapiRevision, "example", {
        containerAppName: "example-container-app-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.app import OpenapiRevision, OpenapiRevisionConfig

    def describe():
        ubx.intent("look up azure_app_openapi_revision")

        ubx.data(OpenapiRevision, "example", OpenapiRevisionConfig(
            container_app_name="example-container-app-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
