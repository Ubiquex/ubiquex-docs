`azure_app_openapi_private_link_resource` reads a real Private Link resource exposed by a Container Apps environment, that a Private Endpoint can connect to.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	app "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/app"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_app_openapi_private_link_resource"})

    		ubx.Data(app.OpenapiPrivateLinkResource, "example", app.OpenapiPrivateLinkResourceConfig{
    			EnvironmentName: "example-environment-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    from ubx.azure.data.app import OpenapiPrivateLinkResource, OpenapiPrivateLinkResourceConfig

    def describe():
        ubx.intent("look up azure_app_openapi_private_link_resource")

        ubx.data(OpenapiPrivateLinkResource, "example", OpenapiPrivateLinkResourceConfig(
            environment_name="example-environment-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
