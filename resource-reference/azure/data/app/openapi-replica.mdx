    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go

    import (
    	app "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/app"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_app_openapi_replica"})

    		ubx.Data(app.OpenapiReplica, "example", app.OpenapiReplicaConfig{
    			ContainerAppName: "example-container-app-name",
    			RevisionName:     "example-revision-name",
    		})
    	}))
    }
