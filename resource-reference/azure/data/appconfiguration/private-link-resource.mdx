`azure_appconfiguration_private_link_resource` reads a real Private Link resource exposed by an App Configuration store, that a Private Endpoint can connect to.

## Example

    	appconfiguration "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/appconfiguration"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_appconfiguration_private_link_resource"})

    		ubx.Data(appconfiguration.PrivateLinkResource, "example", appconfiguration.PrivateLinkResourceConfig{
    			ApiVersion:        "example",
    			ConfigStoreName:   "example-config-store-name",
    			ResourceGroupName: "example-resource-group-name",
    			SubscriptionId:    "example",
    		})
    	}))
    }

<ResponseField name="resource_group_name" type="string" required>
  The real name of the resource group this lookup is scoped to. **(AI-inferred)** Required lookup argument.
</ResponseField>

<ResponseField name="subscription_id" type="string" required>
  The real, unique identifier of the Azure subscription this lookup is scoped to. **(AI-inferred)** Required lookup argument.
