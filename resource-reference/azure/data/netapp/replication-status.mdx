`azure_netapp_replication_status` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    	netapp "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/netapp"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_netapp_replication_status"})

    		ubx.Data(netapp.ReplicationStatus, "example", netapp.ReplicationStatusConfig{
    			AccountName: "example-account-name",

    		})
    	}))
    }
