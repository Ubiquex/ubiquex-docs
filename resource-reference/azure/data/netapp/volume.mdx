---
title: "azure_netapp_volume"
description: "`data_azure_netapp_volume` looks up a real, already-existing `azure_netapp_volume`."
---

It is a managed NFS or SMB file share provisioned from an Azure NetApp Files capacity pool.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	netapp "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/netapp"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_netapp_volume"})

    		ubx.Data(netapp.Volume, "example", netapp.VolumeConfig{
    			AccountName: "example-account-name",
    			PoolName:    "example-pool-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    from ubx.azure.data.netapp import Volume, VolumeConfig

    def describe():
        ubx.intent("look up azure_netapp_volume")

        ubx.data(Volume, "example", VolumeConfig(
            account_name="example-account-name",
            pool_name="example-pool-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
