`azure_reservations_catalog` is a real, live data source: `ubx.Data`/`data`/`ubx.data` (per language) executes this lookup at resolve time, against Microsoft Azure's own real API, and returns whatever it finds. It is never created, modified, or destroyed by ubx -- see [docs/schema.md's own "Amendment: data sources"] for the real wire shape this compiles to (`data_sources[]`, never `resources[]`).

    	reservations "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/reservations"

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_reservations_catalog"})

    		ubx.Data(reservations.Catalog, "example", reservations.CatalogConfig{
    			Location:             "example",
    			OfferId:              "example",
    			PlanId:               "example",
    			PublisherId:          "example",
    			ReservedResourceType: "example",
    		})
    	}))
    }
