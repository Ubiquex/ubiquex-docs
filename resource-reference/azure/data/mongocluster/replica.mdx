## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure/sdk/go
    package main

    import (
    	mongocluster "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/mongocluster"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_mongocluster_replica"})

    		ubx.Data(mongocluster.Replica, "example", mongocluster.ReplicaConfig{
    			MongoClusterName: "example-mongo-cluster-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";
    // ubx sdk gen --only azure --lang ts --out ./local-sdk
    import { Replica } from "./local-sdk/azure/sdk/typescript/azure/data/mongocluster/replica.ts";

    export default stack("example", () => {
      intent({ summary: "look up azure_mongocluster_replica" });

      data(Replica, "example", {
        mongoClusterName: "example-mongo-cluster-name",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure.data.mongocluster import Replica, ReplicaConfig

    def describe():
        ubx.intent("look up azure_mongocluster_replica")

        ubx.data(Replica, "example", ReplicaConfig(
            mongo_cluster_name="example-mongo-cluster-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
