---
title: "azure_mongocluster_user"
description: "`data_azure_mongocluster_user` looks up a real, already-existing `azure_mongocluster_user`."
---

It is a database user on an Azure Cosmos DB for MongoDB (vCore) cluster.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	mongocluster "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/data/mongocluster"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "look up azure_mongocluster_user"})

    		ubx.Data(mongocluster.User, "example", mongocluster.UserConfig{
    			MongoClusterName: "example-mongo-cluster-name",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { data, intent, stack } from "@ubx/sdk";

  <Tab title="Python">

    import ubx_sdk as ubx
    from ubx.azure.data.mongocluster import User, UserConfig

    def describe():
        ubx.intent("look up azure_mongocluster_user")

        ubx.data(User, "example", UserConfig(
            mongo_cluster_name="example-mongo-cluster-name",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
