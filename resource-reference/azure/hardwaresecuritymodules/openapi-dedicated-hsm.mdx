`azure_hardwaresecuritymodules_openapi_dedicated_hsm` provisions Azure Dedicated HSM, dedicated Thales Luna Network HSM 7 appliances directly connected to a customer's own virtual network -- now retiring in favor of Azure Cloud HSM, supported through July 2028.

## Example

    ```go
    // ubx sdk gen --only azure_hardwaresecuritymodules_openapi --lang go --out ./local-sdk

    package main

    import (

    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi dedicated hsm"})

    		ubx.Resource(hardwaresecuritymodules.OpenapiDedicatedHsm, "ci-runner", hardwaresecuritymodules.OpenapiDedicatedHsmConfig{
    			Name:       "ci-runner",

    		})
    	}))
    ```
