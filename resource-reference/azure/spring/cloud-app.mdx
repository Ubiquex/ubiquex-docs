## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	spring "github.com/ubiquex/ubx-sdk-azure/sdk/go/azurerm/spring"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own cloud app"})

    		ubx.Resource(spring.CloudApp, "ci-runner", spring.CloudAppConfig{
    			Name:                  "ci-runner",
    			ResourceGroupName:     "example-resource-group-name",
    			ServiceName:           "example-service-name",
    			HttpsOnly:             true,
    			IsPublic:              true,
    			PublicEndpointEnabled: true,
    			TlsEnabled:            true,
    		})
    	}))
    }
    ```

    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { CloudApp } from "jsr:@ubx/sdk-azure/azurerm/spring/cloud_app";

    export default stack("payments", () => {
      intent({ summary: "payments own cloud app" });

      resource(CloudApp, "ci-runner", {
        name: "ci-runner",
        resourceGroupName: "example-resource-group-name",
        serviceName: "example-service-name",
        httpsOnly: true,
        isPublic: true,
        publicEndpointEnabled: true,
        tlsEnabled: true,
      });
    });
    ```

    ```python
    import ubx_sdk as ubx
    from ubx.azurerm.spring import CloudApp, CloudAppConfig

    def describe():
        ubx.intent("payments own cloud app")

        ubx.resource(CloudApp, "ci-runner", CloudAppConfig(
            name="ci-runner",
            resource_group_name="example-resource-group-name",
            service_name="example-service-name",
            https_only=True,
            is_public=True,
            public_endpoint_enabled=True,
            tls_enabled=True,
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
