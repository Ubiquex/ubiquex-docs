`azure_automation_openapi_runbook` is the executable script (PowerShell, Python, or a graphical workflow) that Azure Automation runs, either on a schedule, on demand, or via webhook.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi runbook"})

    		ubx.Resource(automation.OpenapiRunbook, "example", automation.OpenapiRunbookConfig{
    			Properties: map[string]any{"runbook_type": "example"},
    			Name:       "example",
    			Location:   "example",
    			Tags:       map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```

    import ubx_sdk as ubx

    def describe():

        ubx.resource(OpenapiRunbook, "example", OpenapiRunbookConfig(

            location="example",

    if __name__ == "__main__":
        ubx.run("example", describe)
