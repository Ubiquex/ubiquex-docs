`azure_automation_openapi_watcher` runs a long-lived Azure Automation script that watches for a defined condition and triggers a follow-up runbook when it's detected.

    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_automation_openapi/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
