`azure_automation_openapi_certificate` stores a certificate securely within an Azure Automation account, referenced by runbook(s) needing it for authentication.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_automation_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_automation_openapi/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_automation_openapi/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi certificate"})

    		ubx.Resource(automation.OpenapiCertificate, "ci-runner", automation.OpenapiCertificateConfig{
    			Name:       "ci-runner",
    			Properties: map[string]any{"base64_value": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_automation_openapi --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiCertificate } from "./local-sdk/azure_automation_openapi/sdk/typescript/azure_automation_openapi/automation/openapi_certificate.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own openapi certificate" });

      resource(OpenapiCertificate, "ci-runner", {
        name: "ci-runner",
        properties: { base64Value: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_automation_openapi --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_automation_openapi/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_automation_openapi.automation import OpenapiCertificate, OpenapiCertificateConfig

    def describe():
        ubx.intent("payments own openapi certificate")

        ubx.resource(OpenapiCertificate, "ci-runner", OpenapiCertificateConfig(
            name="ci-runner",
            properties={"base64_value": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
