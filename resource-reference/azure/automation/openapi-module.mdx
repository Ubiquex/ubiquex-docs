## Example

    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi module"})

    		ubx.Resource(automation.OpenapiModule, "example", automation.OpenapiModuleConfig{
    			Properties: map[string]any{"content_link": map[string]any{"content_hash": map[string]any{"algorithm": "example", "value": "example"}}},
    			Name:       "example",
    			Location:   "example",
    			Tags:       map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }

  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.automation import OpenapiModule, OpenapiModuleConfig
    from ubx.azure.automation.openapi_module import OpenapiModule_Properties, OpenapiModule_Properties_ContentLink, OpenapiModule_Properties_ContentLink_ContentHash

    def describe():
        ubx.intent("example own openapi module")

        ubx.resource(OpenapiModule, "example", OpenapiModuleConfig(
            properties=OpenapiModule_Properties(content_link=OpenapiModule_Properties_ContentLink(content_hash=OpenapiModule_Properties_ContentLink_ContentHash(algorithm="example", value="example"))),
            name="example",
            location="example",
            tags={"managed-by": "ubx"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
