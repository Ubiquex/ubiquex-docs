`azure_automation_openapi_runtime_environment` defines a custom language version and package set that PowerShell or Python runbook(s) execute within, beyond Azure Automation's own default runtime.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_automation_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_automation_openapi/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_automation_openapi/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi runtime environment"})

    		ubx.Resource(automation.OpenapiRuntimeEnvironment, "ci-runner", automation.OpenapiRuntimeEnvironmentConfig{})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_automation_openapi --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiRuntimeEnvironment } from "./local-sdk/azure_automation_openapi/sdk/typescript/azure_automation_openapi/automation/openapi_runtime_environment.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own openapi runtime environment" });

      resource(OpenapiRuntimeEnvironment, "ci-runner", {});
    });
    ```
