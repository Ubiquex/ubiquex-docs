`azure_automation_openapi_automation_account` is the top-level Azure Automation resource that runbook(s), schedule(s), and other automation asset(s) are organized within.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_automation_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_automation_openapi/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_automation_openapi/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi automation account"})

    		ubx.Resource(automation.OpenapiAutomationAccount, "ci-runner", automation.OpenapiAutomationAccountConfig{
    			Name:     "ci-runner",
    			Location: "example",
    			Tags:     map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_automation_openapi --lang ts --out ./local-sdk

      resource(OpenapiAutomationAccount, "ci-runner", {
        name: "ci-runner",

        tags: { "managed-by": "ubx" },
    ```

    ```python
    # ubx sdk gen --only azure_automation_openapi --lang py --out ./local-sdk
    ```
