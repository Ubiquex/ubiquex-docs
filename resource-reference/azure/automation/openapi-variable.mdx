---
title: "azure_automation_openapi_variable"
description: "`azure_automation_openapi_variable` stores a named configuration value (plain or encrypted) within an Azure Automation account, referenced by runbook(s)."
---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_automation_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_automation_openapi/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_automation_openapi/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi variable"})

    		ubx.Resource(automation.OpenapiVariable, "ci-runner", automation.OpenapiVariableConfig{
    			Name:       "ci-runner",
    			Properties: map[string]any{"description": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_automation_openapi --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiVariable } from "./local-sdk/azure_automation_openapi/sdk/typescript/azure_automation_openapi/automation/openapi_variable.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own openapi variable" });

      resource(OpenapiVariable, "ci-runner", {
        name: "ci-runner",
        properties: { description: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_automation_openapi --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_automation_openapi/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_automation_openapi.automation import OpenapiVariable, OpenapiVariableConfig

    def describe():
        ubx.intent("payments own openapi variable")

        ubx.resource(OpenapiVariable, "ci-runner", OpenapiVariableConfig(
            name="ci-runner",
            properties={"description": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
  </Tab>

