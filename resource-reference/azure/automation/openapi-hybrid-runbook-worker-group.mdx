`azure_automation_openapi_hybrid_runbook_worker_group` groups related hybrid runbook worker(s) together, that a runbook job can be targeted to run on any member of.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_automation_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_automation_openapi/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_automation_openapi/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi hybrid runbook worker group"})

    		ubx.Resource(automation.OpenapiHybridRunbookWorkerGroup, "ci-runner", automation.OpenapiHybridRunbookWorkerGroupConfig{
    			Name: "ci-runner",
    		})
    	}))
    }
    ```
