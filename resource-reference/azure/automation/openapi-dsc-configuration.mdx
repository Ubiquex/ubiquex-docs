`azure_automation_openapi_dsc_configuration` defines a PowerShell Desired State Configuration script managed by Azure Automation State Configuration, describing the target state for registered node(s).

## Example

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi dsc configuration"})

    		ubx.Resource(automation.OpenapiDscConfiguration, "example", automation.OpenapiDscConfigurationConfig{
    			Properties: map[string]any{"source": map[string]any{"hash": map[string]any{"algorithm": "example"}}},
    			Name:       "example",
    			Location:   "example",
    			Tags:       map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
