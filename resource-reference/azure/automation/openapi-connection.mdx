`azure_automation_openapi_connection` stores connection details for an external service (of a specific connection type) that Azure Automation runbook(s) authenticate against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi connection"})

    		ubx.Resource(automation.OpenapiConnection, "example", automation.OpenapiConnectionConfig{
    			Name:       "example",
    			Properties: map[string]any{"connection_type": map[string]any{"name": "example-name"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiConnection } from "@ubx/sdk-azure/azure/automation/openapi_connection";

    export default stack("example", () => {
      intent({ summary: "example own openapi connection" });

      resource(OpenapiConnection, "example", {
        name: "example",
        properties: { connectionType: { name: "example-name" } },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.automation.openapi_connection import OpenapiConnection, OpenapiConnectionConfig

    def describe():
        ubx.intent("example own openapi connection")

        ubx.resource(OpenapiConnection, "example", OpenapiConnectionConfig(
            name="example",
            properties={"connection_type": {"name": "example-name"}},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
