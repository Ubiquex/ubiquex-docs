---
title: "azure_automation_openapi_credential"
description: "`azure_automation_openapi_credential` stores a username/password credential securely within an Azure Automation account, referenced by runbook(s)."
---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi credential"})

    		ubx.Resource(automation.OpenapiCredential, "example", automation.OpenapiCredentialConfig{
    			Name:       "example",
    			Properties: map[string]any{"password": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiCredential } from "@ubx/sdk-azure/azure/automation/openapi_credential";

    export default stack("example", () => {
      intent({ summary: "example own openapi credential" });

      resource(OpenapiCredential, "example", {
        name: "example",
        properties: { password: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.automation.openapi_credential import OpenapiCredential, OpenapiCredentialConfig

    def describe():
        ubx.intent("example own openapi credential")

        ubx.resource(OpenapiCredential, "example", OpenapiCredentialConfig(
            name="example",
            properties={"password": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
  <Tab title="Markdown">
    ```
    Create azure_automation_openapi_credential called "example",
    with properties a set value.
    ```

</Tabs>
