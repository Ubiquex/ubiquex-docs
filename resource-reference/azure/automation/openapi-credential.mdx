---
title: "azure_automation_openapi_credential"
description: "`azure_automation_openapi_credential` stores a username/password credential securely within an Azure Automation account, referenced by runbook(s)."
---

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_automation_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_automation_openapi/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_automation_openapi/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi credential"})

    		ubx.Resource(automation.OpenapiCredential, "ci-runner", automation.OpenapiCredentialConfig{
    			Name:       "ci-runner",
    			Properties: map[string]any{"password": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_automation_openapi --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiCredential } from "./local-sdk/azure_automation_openapi/sdk/typescript/azure_automation_openapi/automation/openapi_credential.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own openapi credential" });

      resource(OpenapiCredential, "ci-runner", {
        name: "ci-runner",
        properties: { password: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_automation_openapi --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_automation_openapi/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_automation_openapi.automation import OpenapiCredential, OpenapiCredentialConfig

    def describe():
        ubx.intent("payments own openapi credential")

        ubx.resource(OpenapiCredential, "ci-runner", OpenapiCredentialConfig(
            name="ci-runner",
            properties={"password": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
  </Tab>
