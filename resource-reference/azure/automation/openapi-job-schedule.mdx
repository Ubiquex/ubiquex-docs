`azure_automation_openapi_job_schedule` links an Azure Automation runbook to a schedule, triggering it to run automatically at the schedule's own defined times.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_automation_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_automation_openapi/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_automation_openapi/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi job schedule"})

    		ubx.Resource(automation.OpenapiJobSchedule, "ci-runner", automation.OpenapiJobScheduleConfig{
    			Properties: map[string]any{"runbook": map[string]any{"name": "example-name"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_automation_openapi --lang ts --out ./local-sdk
    ```
