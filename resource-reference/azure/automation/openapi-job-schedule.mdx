`azure_automation_openapi_job_schedule` links an Azure Automation runbook to a schedule, triggering it to run automatically at the schedule's own defined times.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi job schedule"})

    		ubx.Resource(automation.OpenapiJobSchedule, "example", automation.OpenapiJobScheduleConfig{
    			Properties: map[string]any{"runbook": map[string]any{"name": "example-name"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiJobSchedule } from "@ubx/sdk-azure/azure/automation/openapi_job_schedule";

    export default stack("example", () => {
      intent({ summary: "example own openapi job schedule" });
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.automation.openapi_job_schedule import OpenapiJobSchedule, OpenapiJobScheduleConfig

    def describe():
        ubx.intent("example own openapi job schedule")

        ubx.resource(OpenapiJobSchedule, "example", OpenapiJobScheduleConfig(
            properties={"runbook": {"name": "example-name"}},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
