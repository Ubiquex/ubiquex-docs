## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_automation_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_automation_openapi/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_automation_openapi/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi schedule"})

    		ubx.Resource(automation.OpenapiSchedule, "ci-runner", automation.OpenapiScheduleConfig{
    			Name:       "ci-runner",
    			Properties: map[string]any{"frequency": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    ```python
    # ubx sdk gen --only azure_automation_openapi --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_automation_openapi/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_automation_openapi.automation import OpenapiSchedule, OpenapiScheduleConfig

    def describe():
        ubx.intent("payments own openapi schedule")

        ubx.resource(OpenapiSchedule, "ci-runner", OpenapiScheduleConfig(
            name="ci-runner",
            properties={"frequency": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
