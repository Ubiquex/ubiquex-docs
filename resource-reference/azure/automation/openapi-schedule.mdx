## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi schedule"})

    		ubx.Resource(automation.OpenapiSchedule, "example", automation.OpenapiScheduleConfig{
    			Name:       "example",
    			Properties: map[string]any{"frequency": "example"},
    		})
    	}))
    }
    ```
  </Tab>

## Properties

<ResponseField name="name" type="string" required>
  Gets or sets the name of the Schedule. Required.
</ResponseField>

<ResponseField name="properties" type="object" required>
  The parameters supplied to the create or update schedule operation. Required.
  <Expandable title="properties">
    <ResponseField name="advanced_schedule" type="object">
      The properties of the create Advanced Schedule. Optional.
      <Expandable title="properties">
        <ResponseField name="month_days" type="list(number)">
          Days of the month that the job should execute on. Must be between 1 and 31. Optional.
        </ResponseField>
        <ResponseField name="monthly_occurrences" type="list(object)">
          Occurrences of days within a month. Optional.
          <Expandable title="properties">
            <ResponseField name="day" type="string">
              The day-of-week or day-of-month component of this schedule. Optional; if omitted, computed by Microsoft Azure. **(AI-inferred)**
            </ResponseField>
            <ResponseField name="occurrence" type="number">
              Which specific occurrence of the matching day within the schedule's own recurrence this applies to. Optional; if omitted, computed by Microsoft Azure. **(AI-inferred)**
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="week_days" type="list(string)">
          Days of the week that the job should execute on. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="description" type="string">
      Gets or sets the description of the schedule. Optional.
    </ResponseField>
    <ResponseField name="expiry_time" type="string">
      Gets or sets the end time of the schedule. Optional.
    </ResponseField>
    <ResponseField name="frequency" type="string" required>
      Gets or sets the frequency of the schedule. Required.
    </ResponseField>
    <ResponseField name="interval" type="any">
      Gets or sets the interval of the schedule. Optional.
    </ResponseField>
    <ResponseField name="start_time" type="string" required>
      Gets or sets the start time of the schedule. Required.
    </ResponseField>
    <ResponseField name="time_zone" type="string">
      Gets or sets the time zone of the schedule. Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
