## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi webhook"})

    		ubx.Resource(automation.OpenapiWebhook, "example", automation.OpenapiWebhookConfig{
    			Name:       "example",
    			Properties: map[string]any{"expiry_time": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";

    export default stack("example", () => {
      intent({ summary: "example own openapi webhook" });

      resource(OpenapiWebhook, "example", {
        name: "example",
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.automation.openapi_webhook import OpenapiWebhook, OpenapiWebhookConfig

    def describe():
        ubx.intent("example own openapi webhook")

        ubx.resource(OpenapiWebhook, "example", OpenapiWebhookConfig(
            name="example",
            properties={"expiry_time": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```

</ResponseField>

<ResponseField name="properties" type="object" required>

  <Expandable title="properties">
    <ResponseField name="expiry_time" type="string">
      Gets or sets the expiry time. Optional.
    </ResponseField>
    <ResponseField name="is_enabled" type="bool">
      Gets or sets the value of the enabled flag of webhook. Optional.
    </ResponseField>
    <ResponseField name="parameters" type="map(string)">
      Gets or sets the parameters of the job. Optional.
    </ResponseField>
    <ResponseField name="run_on" type="string">
      Gets or sets the name of the hybrid worker group the webhook job will run on. Optional.
    </ResponseField>
    <ResponseField name="runbook" type="object">
      The runbook property associated with the entity. Optional.
      <Expandable title="properties">
        <ResponseField name="name" type="string">
          Gets or sets the name of the runbook. Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
