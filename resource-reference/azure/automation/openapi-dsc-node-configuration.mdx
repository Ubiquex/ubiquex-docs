`azure_automation_openapi_dsc_node_configuration` is a compiled DSC configuration assigned to specific registered node(s), the actual configuration those nodes are enforced against.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_automation_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_automation_openapi/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_automation_openapi/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi dsc node configuration"})

    		ubx.Resource(automation.OpenapiDscNodeConfiguration, "example", automation.OpenapiDscNodeConfigurationConfig{
    			Name: "example",
    			Tags: map[string]string{"managed-by": "ubx"},
    		})
    	}))
    }
    ```
