## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi package"})

    		ubx.Resource(automation.OpenapiPackage, "example", automation.OpenapiPackageConfig{
    			Properties: map[string]any{"content_link": map[string]any{"content_hash": map[string]any{"algorithm": "example"}}},
    			AllOf:      map[string]any{"location": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiPackage } from "@ubx/sdk-azure/azure/automation/openapi_package";

    export default stack("example", () => {
      intent({ summary: "example own openapi package" });

      resource(OpenapiPackage, "example", {
        properties: { contentLink: { contentHash: { algorithm: "example" } } },
        allOf: { location: "example" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.automation.openapi_package import OpenapiPackage, OpenapiPackageConfig

    def describe():
        ubx.intent("example own openapi package")

        ubx.resource(OpenapiPackage, "example", OpenapiPackageConfig(
            properties={"content_link": {"content_hash": {"algorithm": "example"}}},
            all_of={"location": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
  </Tab>
