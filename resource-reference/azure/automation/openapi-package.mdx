## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_automation_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_automation_openapi/sdk/go
    package main

    import (
    	automation "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_automation_openapi/automation"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi package"})

    		ubx.Resource(automation.OpenapiPackage, "example", automation.OpenapiPackageConfig{
    			Properties: map[string]any{"content_link": map[string]any{"content_hash": map[string]any{"algorithm": "example"}}},
    			AllOf:      map[string]any{"location": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_automation_openapi --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { OpenapiPackage } from "./local-sdk/azure_automation_openapi/sdk/typescript/azure_automation_openapi/automation/openapi_package.ts";

    export default stack("example", () => {
      intent({ summary: "example own openapi package" });

      resource(OpenapiPackage, "example", {
        properties: { contentLink: { contentHash: { algorithm: "example" } } },
        allOf: { location: "example" },
      });
    ```

  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_automation_openapi --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_automation_openapi/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_automation_openapi.automation import OpenapiPackage, OpenapiPackageConfig

    def describe():
        ubx.intent("example own openapi package")

        ubx.resource(OpenapiPackage, "example", OpenapiPackageConfig(
            properties={"content_link": {"content_hash": {"algorithm": "example"}}},
            all_of={"location": "example"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
