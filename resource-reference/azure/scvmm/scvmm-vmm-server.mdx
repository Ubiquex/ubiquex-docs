`azure_scvmm_vmm_server` registers an on-premises System Center Virtual Machine Manager management server with Azure Arc, making its own managed VM(s), cloud(s), and template(s) manageable through Azure.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	scvmm "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/scvmm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own scvmm vmm server"})

    		ubx.Resource(scvmm.VmmServer, "example", scvmm.VmmServerConfig{
    			ExtendedLocation: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    ```

  <Tab title="Python">

    ```python
    import ubx_sdk as ubx
    from ubx.azure.scvmm.vmm_server import VmmServer, VmmServerConfig

            extended_location={"name": "example-name"},
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
