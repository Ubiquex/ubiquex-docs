## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_scvmm_scvmm --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_scvmm_scvmm/sdk/go
    package main

    import (
    	scvmm "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_scvmm_scvmm/scvmm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own scvmm virtual network"})

    		ubx.Resource(scvmm.ScvmmVirtualNetwork, "ci-runner", scvmm.ScvmmVirtualNetworkConfig{
    			ExtendedLocation: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_scvmm_scvmm --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { ScvmmVirtualNetwork } from "./local-sdk/azure_scvmm_scvmm/sdk/typescript/azure_scvmm_scvmm/scvmm/scvmm_virtual_network.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own scvmm virtual network" });

      resource(ScvmmVirtualNetwork, "ci-runner", {
        extendedLocation: { name: "example-name" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_scvmm_scvmm --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_scvmm_scvmm/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_scvmm_scvmm.scvmm import ScvmmVirtualNetwork, ScvmmVirtualNetworkConfig

    def describe():
        ubx.intent("payments own scvmm virtual network")

        ubx.resource(ScvmmVirtualNetwork, "ci-runner", ScvmmVirtualNetworkConfig(
            extended_location={"name": "example-name"},
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```
