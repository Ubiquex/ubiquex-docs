## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	scvmm "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/scvmm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own virtual network"})

    		ubx.Resource(scvmm.VirtualNetwork, "example", scvmm.VirtualNetworkConfig{
    			ExtendedLocation: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { VirtualNetwork } from "@ubx/sdk-azure/azure/scvmm/virtual_network";

    export default stack("example", () => {
      intent({ summary: "example own virtual network" });

      resource(VirtualNetwork, "example", {
        extendedLocation: { name: "example-name" },
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.scvmm import VirtualNetwork, VirtualNetworkConfig
    from ubx.azure.scvmm.virtual_network import VirtualNetwork_ExtendedLocation

    def describe():
        ubx.intent("example own virtual network")

        ubx.resource(VirtualNetwork, "example", VirtualNetworkConfig(
            extended_location=VirtualNetwork_ExtendedLocation(name="example-name"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
