`azure_scvmm_cloud` represents an SCVMM private cloud resource pool, exposed as an Azure Arc resource that VM(s) can be provisioned onto through Azure tooling.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_scvmm_scvmm --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_scvmm_scvmm/sdk/go
    package main

    import (
    	scvmm "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_scvmm_scvmm/scvmm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own scvmm cloud"})

    		ubx.Resource(scvmm.ScvmmCloud, "example", scvmm.ScvmmCloudConfig{
    			ExtendedLocation: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
