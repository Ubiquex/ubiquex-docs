`azure_scvmm_scvmm_virtual_machine_template` is an SCVMM VM template, exposed as an Azure Arc resource that new Arc-enabled SCVMM VM(s) can be provisioned from.

## Example

<Tabs>

    ```go
    // ubx sdk gen --only azure_scvmm_scvmm --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_scvmm_scvmm/sdk/go
    package main

    import (
    	scvmm "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_scvmm_scvmm/scvmm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own scvmm virtual machine template"})

    		ubx.Resource(scvmm.ScvmmVirtualMachineTemplate, "example", scvmm.ScvmmVirtualMachineTemplateConfig{
    			ExtendedLocation: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
