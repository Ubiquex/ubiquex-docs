`azure_scvmm_availability_set` groups VM(s) together on an Azure Arc-enabled System Center Virtual Machine Manager private cloud so the underlying hypervisor keeps them on separate hosts for fault tolerance.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	scvmm "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/scvmm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own availability set"})

    		ubx.Resource(scvmm.AvailabilitySet, "example", scvmm.AvailabilitySetConfig{
    			ExtendedLocation: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>

    import ubx_sdk as ubx
    from ubx.azure.scvmm import AvailabilitySet, AvailabilitySetConfig
