`azure_scvmm_availability_set` groups VM(s) together on an Azure Arc-enabled System Center Virtual Machine Manager private cloud so the underlying hypervisor keeps them on separate hosts for fault tolerance.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_scvmm_scvmm --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_scvmm_scvmm/sdk/go
    package main

    import (
    	scvmm "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_scvmm_scvmm/scvmm"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own scvmm availability set"})

    		ubx.Resource(scvmm.ScvmmAvailabilitySet, "example", scvmm.ScvmmAvailabilitySetConfig{
    			ExtendedLocation: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
