`azure_operationalinsights_openapi_data_source` configures a legacy Log Analytics data collection source (e.g. Windows event logs) for a connected agent-based machine.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_operationalinsights_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_operationalinsights_openapi/sdk/go
    package main

    import (
    	operationalinsights "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_operationalinsights_openapi/operationalinsights"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi data source"})

    		ubx.Resource(operationalinsights.OpenapiDataSource, "ci-runner", operationalinsights.OpenapiDataSourceConfig{
    			Kind:       "example",
    			Properties: "example",
    		})
    	}))
    }
    ```
