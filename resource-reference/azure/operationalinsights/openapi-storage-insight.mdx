## Properties

<ResponseField name="e_tag" type="string">
  The ETag of the storage insight. Optional; if omitted, computed by Microsoft Azure.
</ResponseField>

<ResponseField name="properties" type="object">
  Storage insight properties. Optional; if omitted, computed by Microsoft Azure.
  <Expandable title="properties">
    <ResponseField name="containers" type="list(string)">
      The names of the blob containers that the workspace should read Optional.
    </ResponseField>
    <ResponseField name="status" type="object">
      The status of the storage insight. Optional.
      <Expandable title="properties">
        <ResponseField name="description" type="string">
          Description of the state of the storage insight. Optional.
        </ResponseField>
        <ResponseField name="state" type="string" required>
          The state of the storage insight connection to the workspace Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="storage_account" type="object" required>
      Describes a storage account connection. Required.
      <Expandable title="properties">
        <ResponseField name="id" type="string" required>
          The Azure Resource Manager ID of the storage account resource. Required.
        </ResponseField>
        <ResponseField name="key" type="string" required>
          The storage account key. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="tables" type="list(string)">
      The names of the Azure tables that the workspace should read Optional.
    </ResponseField>
  </Expandable>
</ResponseField>
