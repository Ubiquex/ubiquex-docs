`azure_operationalinsights_openapi_data_export` continuously exports matching Log Analytics workspace data (specific table(s)) to an external destination (a Storage account or Event Hub).

## Example

<Tabs>
  <Tab title="Go">

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi data export"})

    		ubx.Resource(operationalinsights.OpenapiDataExport, "example", operationalinsights.OpenapiDataExportConfig{})
    	}))
    }

## Properties

<ResponseField name="properties" type="object">
  Data Export properties. Optional; if omitted, computed by Microsoft Azure.
  <Expandable title="properties">
    <ResponseField name="created_date" type="string">
      The latest data export rule modification time. Optional.
    </ResponseField>
    <ResponseField name="data_export_id" type="string">
      The data export rule ID. Optional.
    </ResponseField>
    <ResponseField name="destination" type="object">
      Destination properties. Optional.
      <Expandable title="properties">
        <ResponseField name="meta_data" type="object">
          Destination meta data. Optional.
          <Expandable title="properties">
            <ResponseField name="event_hub_name" type="string">
              Optional. Allows to define an Event Hub name. Not applicable when destination is Storage Account. Optional.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="resource_id" type="string" required>
          The destination resource ID. This can be copied from the Properties entry of the destination resource in Azure. Required.
        </ResponseField>
        <ResponseField name="type" type="string">
          The type of the destination resource Optional.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="enable" type="bool">
      Active when enabled. Optional.
    </ResponseField>
    <ResponseField name="last_modified_date" type="string">
      Date and time when the export was last modified. Optional.
    </ResponseField>
    <ResponseField name="table_names" type="list(string)" required>
      An array of tables to export, for example: [“Heartbeat, SecurityEvent”]. Required.
    </ResponseField>
  </Expandable>
</ResponseField>
