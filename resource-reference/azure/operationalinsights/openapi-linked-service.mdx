`azure_operationalinsights_openapi_linked_service` links a Log Analytics workspace to another Azure resource (e.g. an Automation account) for integrated functionality.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	operationalinsights "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/operationalinsights"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi linked service"})

    		ubx.Resource(operationalinsights.OpenapiLinkedService, "example", operationalinsights.OpenapiLinkedServiceConfig{
    			Properties: map[string]any{"provisioning_state": "example"},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">

    import ubx_sdk as ubx
    from ubx.azure.operationalinsights import OpenapiLinkedService, OpenapiLinkedServiceConfig
    from ubx.azure.operationalinsights.openapi_linked_service import OpenapiLinkedService_Properties

    def describe():
        ubx.intent("example own openapi linked service")

        ubx.resource(OpenapiLinkedService, "example", OpenapiLinkedServiceConfig(
            properties=OpenapiLinkedService_Properties(provisioning_state="example"),
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)

<ResponseField name="properties" type="object" required>
  Linked service properties. Required.
  <Expandable title="properties">
    <ResponseField name="provisioning_state" type="string">
      The provisioning state of the linked service. Optional.
    </ResponseField>
    <ResponseField name="resource_id" type="string">
      The resource id of the resource that will be linked to the workspace. This should be used for linking resources which require read access Optional.
    </ResponseField>
    <ResponseField name="write_access_resource_id" type="string">
