    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi connector"})

    			Properties: map[string]any{"data_source_type": "example"},
    		})
    	}))
    }

## Properties

<ResponseField name="properties" type="object" required>
  The storage connector properties Required.
  <Expandable title="properties">
    <ResponseField name="creation_time" type="string">
      System-generated creation time of the Storage Connector in ISO 8601 date-time format (YYYY-MM-DDTHH:mm:ssZ). Not a valid input parameter during creating. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="data_source_type" type="string" required>
      The type of the backing data source for storage connector Required.
    </ResponseField>
    <ResponseField name="description" type="string">
      Arbitrary description of this Storage Connector. Max 250 characters. Optional.
    </ResponseField>
    <ResponseField name="provisioning_state" type="string">
      Provisioning state of the resource at the time the operation was called. Computed by Microsoft Azure after `ship`.
    </ResponseField>
    <ResponseField name="source" type="object" required>
      The storage connector backing data source information Required.
      <Expandable title="properties">
        <ResponseField name="type" type="string" required>
          The type of the backing data source for storage connector Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="state" type="string">
      State - Active or Inactive. Whether or not the Storage Connector should start as active (default: Active) (While set to false on the Storage Connector, all data plane requests using this Storage Connector fail, and this Storage Connector is not billed if it would be otherwise. Optional.
    </ResponseField>
    <ResponseField name="test_connection" type="bool">
      Test connection to backing data source before creating the storage connector. Optional.
    </ResponseField>
    <ResponseField name="unique_id" type="string">
      System-generated GUID identifier for the Storage Connector. Not a valid input parameter when creating. Computed by Microsoft Azure after `ship`.
    </ResponseField>
  </Expandable>
</ResponseField>
