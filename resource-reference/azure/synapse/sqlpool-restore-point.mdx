`azure_synapse_sqlpool_restore_point` is a caller-created, named restore point for a Synapse dedicated SQL pool, letting it be restored to that exact moment.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_synapse_sqlpool --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_synapse_sqlpool/sdk/go
    package main

    import (
    	synapse "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_synapse_sqlpool/synapse"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own sqlpool restore point"})

    		ubx.Resource(synapse.SqlpoolRestorePoint, "ci-runner", synapse.SqlpoolRestorePointConfig{
    			RestorePointLabel: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // ubx sdk gen --only azure_synapse_sqlpool --lang ts --out ./local-sdk
    import { intent, resource, stack } from "@ubx/sdk";
    import { SqlpoolRestorePoint } from "./local-sdk/azure_synapse_sqlpool/sdk/typescript/azure_synapse_sqlpool/synapse/sqlpool_restore_point.ts";

    export default stack("payments", () => {
      intent({ summary: "payments own sqlpool restore point" });

      resource(SqlpoolRestorePoint, "ci-runner", {
        restorePointLabel: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    # ubx sdk gen --only azure_synapse_sqlpool --lang py --out ./local-sdk
    # export PYTHONPATH=./local-sdk/azure_synapse_sqlpool/sdk/python:$PYTHONPATH
    import ubx_sdk as ubx
    from ubx.azure_synapse_sqlpool.synapse import SqlpoolRestorePoint, SqlpoolRestorePointConfig

    def describe():
        ubx.intent("payments own sqlpool restore point")

        ubx.resource(SqlpoolRestorePoint, "ci-runner", SqlpoolRestorePointConfig(
            restore_point_label="example",
        ))

    if __name__ == "__main__":
        ubx.run("payments", describe)
    ```

## Input properties

<ResponseField name="restore_point_label" type="string" required>
  The restore point label to apply Required.
</ResponseField>

## Output properties

  Resource location. Computed by Microsoft Azure after `ship`.
</ResponseField>
