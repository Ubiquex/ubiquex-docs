`azure_databricks_openapi_private_endpoint_connection` represents the connection between a Private Endpoint and an Azure Databricks workspace, tracking its own approval state.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	databricks "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/databricks"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi private endpoint connection"})

    		ubx.Resource(databricks.OpenapiPrivateEndpointConnection, "example", databricks.OpenapiPrivateEndpointConnectionConfig{
    			Properties: map[string]any{"private_link_service_connection_state": map[string]any{"status": "example"}},
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
