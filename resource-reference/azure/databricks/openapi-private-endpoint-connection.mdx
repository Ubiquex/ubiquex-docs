`azure_databricks_openapi_private_endpoint_connection` represents the connection between a Private Endpoint and an Azure Databricks workspace, tracking its own approval state.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_databricks_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_databricks_openapi/sdk/go
    package main

    import (
    	databricks "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_databricks_openapi/databricks"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi private endpoint connection"})

    		ubx.Resource(databricks.OpenapiPrivateEndpointConnection, "ci-runner", databricks.OpenapiPrivateEndpointConnectionConfig{
    			Properties: map[string]any{"private_link_service_connection_state": map[string]any{"status": "example"}},
    		})
    	}))
    }
    ```
