`azure_datamigration_sqlmigration_sql_migration_service` is the SQL-specific Azure Database Migration Service resource coordinating migration(s) into Azure SQL targets, working with a self-hosted integration runtime.

## Example

    package main

    import (
    	datamigration "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_datamigration_sqlmigration/datamigration"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own sqlmigration sql migration service"})

    		ubx.Resource(datamigration.SqlmigrationSqlMigrationService, "ci-runner", datamigration.SqlmigrationSqlMigrationServiceConfig{})
    	}))
    }
