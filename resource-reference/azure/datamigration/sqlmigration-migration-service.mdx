`azure_datamigration_sqlmigration_migration_service` is the top-level Azure Database Migration Service resource that migration project(s) and their own tracked database migration(s) run through.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_datamigration_sqlmigration --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_datamigration_sqlmigration/sdk/go
    package main

    import (
    	datamigration "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_datamigration_sqlmigration/datamigration"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own sqlmigration migration service"})

    		ubx.Resource(datamigration.SqlmigrationMigrationService, "example", datamigration.SqlmigrationMigrationServiceConfig{})
    	}))
    }
    ```
