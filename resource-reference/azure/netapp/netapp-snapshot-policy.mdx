`azure_netapp_netapp_snapshot_policy` defines the automated snapshot schedule (hourly, daily, weekly, monthly, with its own retention) applied to Azure NetApp Files volume(s) it's attached to.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	netapp "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/netapp"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own netapp snapshot policy"})

    		ubx.Resource(netapp.SnapshotPolicy, "example", netapp.SnapshotPolicyConfig{
    			Properties: map[string]any{"daily_schedule": map[string]any{"hour": 1}},
    		})
    	}))
    }
    ```
  </Tab>
