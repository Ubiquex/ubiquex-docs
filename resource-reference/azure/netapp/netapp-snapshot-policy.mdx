`azure_netapp_netapp_snapshot_policy` defines the automated snapshot schedule (hourly, daily, weekly, monthly, with its own retention) applied to Azure NetApp Files volume(s) it's attached to.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_netapp_netapp --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_netapp_netapp/sdk/go
    package main

    import (
    	netapp "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_netapp_netapp/netapp"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own netapp snapshot policy"})

    		ubx.Resource(netapp.NetappSnapshotPolicy, "ci-runner", netapp.NetappSnapshotPolicyConfig{
    			Properties: map[string]any{"daily_schedule": map[string]any{"hour": 1}},
    		})
    	}))
    }
    ```
  </Tab>
