`azure_netapp_netapp_capacity_pool` is the underlying provisioned capacity and performance tier in Azure NetApp Files that individual volume(s) are carved out of.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_netapp_netapp --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_netapp_netapp/sdk/go
    package main

    import (
    	netapp "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_netapp_netapp/netapp"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own netapp capacity pool"})

    		ubx.Resource(netapp.NetappCapacityPool, "ci-runner", netapp.NetappCapacityPoolConfig{
    			Properties: map[string]any{"service_level": "example"},
    		})
    	}))
    }
    ```
