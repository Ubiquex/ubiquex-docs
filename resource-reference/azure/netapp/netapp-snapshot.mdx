`azure_netapp_netapp_snapshot` is a point-in-time, storage-efficient snapshot of an Azure NetApp Files volume, restorable in place or usable as the source for a new volume clone.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_netapp_netapp --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_netapp_netapp/sdk/go
    package main

    import (
    	netapp "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_netapp_netapp/netapp"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own netapp snapshot"})

    		ubx.Resource(netapp.NetappSnapshot, "ci-runner", netapp.NetappSnapshotConfig{
    			Location: "example",
    		})
    	}))
    }
    ```
  </Tab>
