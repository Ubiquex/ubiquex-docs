`azure_powerbidedicated_openapi_dedicated_capacity` provisions Power BI Embedded dedicated capacity, letting an application embed Power BI reports and dashboards without requiring end users to have their own Power BI license.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_powerbidedicated_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_powerbidedicated_openapi/sdk/go
    package main

    import (
    	powerbidedicated "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_powerbidedicated_openapi/powerbidedicated"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi dedicated capacity"})

    		ubx.Resource(powerbidedicated.OpenapiDedicatedCapacity, "example", powerbidedicated.OpenapiDedicatedCapacityConfig{
    			Sku: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>
