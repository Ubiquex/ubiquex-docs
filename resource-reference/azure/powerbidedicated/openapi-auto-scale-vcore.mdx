`azure_powerbidedicated_openapi_auto_scale_vcore` provisions additional, automatically scaling v-core capacity for a Power BI Embedded capacity during load spikes.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_powerbidedicated_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_powerbidedicated_openapi/sdk/go
    package main

    import (
    	powerbidedicated "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_powerbidedicated_openapi/powerbidedicated"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi auto scale vcore"})

    		ubx.Resource(powerbidedicated.OpenapiAutoScaleVcore, "example", powerbidedicated.OpenapiAutoScaleVcoreConfig{
    			Sku: map[string]any{"name": "example-name"},
    		})
    	}))
    }
    ```
  </Tab>
