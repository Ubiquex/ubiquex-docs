## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_kusto --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_kusto/sdk/go
    package main

    import (
    	azure "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_kusto/azure"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own kusto database"})

    		ubx.Resource(azure.KustoKustoDatabase, "example", azure.KustoKustoDatabaseConfig{
    			Kind: "example",
    		})
    	}))
    }
    ```
