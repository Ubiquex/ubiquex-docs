## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	kusto "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/kusto"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own kusto database"})

    		ubx.Resource(kusto.Database, "example", kusto.DatabaseConfig{
    			Kind: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    import { Database } from "@ubx/sdk-azure/azure/kusto/database";

    export default stack("example", () => {
      intent({ summary: "example own kusto database" });

      resource(Database, "example", {
        kind: "example",
      });
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.kusto.database import Database, DatabaseConfig

    def describe():
        ubx.intent("example own kusto database")

        ubx.resource(Database, "example", DatabaseConfig(
            kind="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
