`azure_kusto_data_connection` continuously ingests data from an external source (Event Hub, IoT Hub, Event Grid) into an Azure Data Explorer database table.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	kusto "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/kusto"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own kusto data connection"})

    		ubx.Resource(kusto.DataConnection, "example", kusto.DataConnectionConfig{
    			Kind: "example",
    		})
    	}))
    }
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    import { intent, resource, stack } from "@ubx/sdk";
    ```

  </Tab>
  <Tab title="Python">
    ```python
    import ubx_sdk as ubx
    from ubx.azure.kusto.data_connection import DataConnection, DataConnectionConfig

    def describe():
        ubx.intent("example own kusto data connection")

        ubx.resource(DataConnection, "example", DataConnectionConfig(
            kind="example",
        ))

    if __name__ == "__main__":
        ubx.run("example", describe)
    ```
