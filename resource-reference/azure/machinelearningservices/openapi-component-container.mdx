`azure_machinelearningservices_openapi_component_container` is the named container in an Azure Machine Learning workspace that version(s) of a reusable pipeline component are registered under.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_machinelearningservices_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_machinelearningservices_openapi/sdk/go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_machinelearningservices_openapi/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi component container"})

    		ubx.Resource(machinelearningservices.OpenapiComponentContainer, "example", machinelearningservices.OpenapiComponentContainerConfig{
    			Properties: map[string]any{"provisioning_state": "example"},
    		})
    	}))
    }
    ```
  </Tab>
