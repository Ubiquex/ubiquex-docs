`azure_machinelearningservices_openapi_featurestore_entity_container` is the named container in an Azure Machine Learning workspace that version(s) of a feature store entity (e.g. `customer`) are registered under.

## Example

    ```go
    package main

    import (

    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi featurestore entity container"})

    		ubx.Resource(machinelearningservices.OpenapiFeaturestoreEntityContainer, "example", machinelearningservices.OpenapiFeaturestoreEntityContainerConfig{
    			Name:       "example",
    			Properties: map[string]any{"provisioning_state": "example"},
    		})
    	}))
    }
    ```
