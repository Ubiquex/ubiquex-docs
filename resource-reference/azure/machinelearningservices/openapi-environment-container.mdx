`azure_machinelearningservices_openapi_environment_container` is the named container in an Azure Machine Learning workspace that version(s) of a training/inference environment definition are registered under.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi environment container"})

    		ubx.Resource(machinelearningservices.OpenapiEnvironmentContainer, "example", machinelearningservices.OpenapiEnvironmentContainerConfig{
    			Properties: map[string]any{"provisioning_state": "example"},
    		})
    	}))
    }
    ```
  </Tab>
