`azure_machinelearningservices_openapi_environment_container` is the named container in an Azure Machine Learning workspace that version(s) of a training/inference environment definition are registered under.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_machinelearningservices_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_machinelearningservices_openapi/sdk/go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_machinelearningservices_openapi/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi environment container"})

    		ubx.Resource(machinelearningservices.OpenapiEnvironmentContainer, "ci-runner", machinelearningservices.OpenapiEnvironmentContainerConfig{
    			Properties: map[string]any{"provisioning_state": "example"},
    		})
    	}))
    }
    ```
