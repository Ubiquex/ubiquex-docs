`azure_machinelearningservices_openapi_code_version` is a specific, immutable version of source code registered in an Azure Machine Learning workspace, referenced by job(s) and component(s) that use it.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi code version"})

    		ubx.Resource(machinelearningservices.OpenapiCodeVersion, "example", machinelearningservices.OpenapiCodeVersionConfig{
    			Properties: map[string]any{"code_uri": "example"},
    			Version:    "example",
    		})
    	}))
    }
    ```
  </Tab>
