## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi component version"})

    		ubx.Resource(machinelearningservices.OpenapiComponentVersion, "example", machinelearningservices.OpenapiComponentVersionConfig{
    			Properties: map[string]any{"component_spec": "example"},
    			Version:    "example",
    		})
    	}))
    }
    ```
  </Tab>
