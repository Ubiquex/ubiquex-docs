`azure_machinelearningservices_openapi_environment_version` is a specific, immutable version of a training or inference software environment (a container image or Conda spec) registered in an Azure Machine Learning workspace.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi environment version"})

    		ubx.Resource(machinelearningservices.OpenapiEnvironmentVersion, "example", machinelearningservices.OpenapiEnvironmentVersionConfig{
    			Properties: map[string]any{"auto_rebuild": "example"},
    			Version:    "example",
    		})
    	}))
    }
    ```
