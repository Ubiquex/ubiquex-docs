`azure_machinelearningservices_openapi_datastore` registers a connection to an external storage account (Blob, Data Lake, or similar) with an Azure Machine Learning workspace, that dataset(s) are sourced from.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi datastore"})

    		ubx.Resource(machinelearningservices.OpenapiDatastore, "example", machinelearningservices.OpenapiDatastoreConfig{
    			Name:       "example",
    			Properties: map[string]any{"credentials": map[string]any{"credentials_type": "example"}},
    		})
    	}))
    }
    ```
  </Tab>

## Properties

<ResponseField name="name" type="string" required>
  path parameter, not part of the API's own resource representation Required.
</ResponseField>

<ResponseField name="properties" type="object" required>
  Base definition for datastore contents configuration. Required.
  <Expandable title="properties">
    <ResponseField name="credentials" type="object" required>
      Base definition for datastore credentials. Required.
      <Expandable title="properties">
        <ResponseField name="credentials_type" type="string" required>
          Enum to determine the datastore credentials type. Required.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="datastore_type" type="string" required>
      Enum to determine the datastore contents type. Required.
    </ResponseField>
    <ResponseField name="is_default" type="bool">
      Readonly property to indicate if datastore is the workspace default datastore Computed by Microsoft Azure after `ship`.
    </ResponseField>
  </Expandable>
</ResponseField>
