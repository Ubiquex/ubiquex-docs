`azure_machinelearningservices_openapi_serverless_endpoint` deploys a Models as a Service (MaaS) foundation model from the Azure AI Foundry Model Catalog to a pay-per-token serverless endpoint, without provisioning dedicated compute.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_machinelearningservices_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_machinelearningservices_openapi/sdk/go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_machinelearningservices_openapi/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi serverless endpoint"})

    		ubx.Resource(machinelearningservices.OpenapiServerlessEndpoint, "ci-runner", machinelearningservices.OpenapiServerlessEndpointConfig{
    			Name:       "ci-runner",
    			Properties: map[string]any{"auth_mode": "example"},
    		})
    	}))
    }
    ```
