`azure_machinelearningservices_openapi_model_version` is a specific, immutable version of a model registered in an Azure Machine Learning workspace, deployable to an online or batch endpoint.

## Example

    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_machinelearningservices_openapi/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("example", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "example own openapi model version"})

    		ubx.Resource(machinelearningservices.OpenapiModelVersion, "example", machinelearningservices.OpenapiModelVersionConfig{
    			Properties: map[string]any{"datasets": []map[string]any{{"name": "example-name"}}},
    			Version:    "example",
    		})
    	}))
    }
