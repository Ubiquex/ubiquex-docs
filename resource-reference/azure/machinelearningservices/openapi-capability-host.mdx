`azure_machinelearningservices_openapi_capability_host` provisions the compute and storage backing a specific AI capability (e.g. agent execution, RAG indexing) within an Azure AI Foundry hub or project.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_machinelearningservices_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_machinelearningservices_openapi/sdk/go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_machinelearningservices_openapi/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi capability host"})

    		ubx.Resource(machinelearningservices.OpenapiCapabilityHost, "ci-runner", machinelearningservices.OpenapiCapabilityHostConfig{
    			Name:       "ci-runner",
    			Properties: map[string]any{"aca_environment_connections": []string{"example"}},
    		})
    	}))
    }
    ```
