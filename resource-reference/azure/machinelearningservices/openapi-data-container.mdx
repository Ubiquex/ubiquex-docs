`azure_machinelearningservices_openapi_data_container` is the named container in an Azure Machine Learning workspace that version(s) of a registered dataset are grouped under.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_machinelearningservices_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_machinelearningservices_openapi/sdk/go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_machinelearningservices_openapi/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi data container"})

    		ubx.Resource(machinelearningservices.OpenapiDataContainer, "ci-runner", machinelearningservices.OpenapiDataContainerConfig{
    			Name:       "ci-runner",
    			Properties: map[string]any{"data_type": "example"},
    		})
    	}))
    }
    ```
