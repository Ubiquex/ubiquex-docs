`azure_machinelearningservices_openapi_schedule` triggers an Azure Machine Learning job or pipeline to run automatically on a recurring cron-style schedule.

## Example

<Tabs>
  <Tab title="Go">
    ```go
    // ubx sdk gen --only azure_machinelearningservices_openapi --lang go --out ./local-sdk
    // go.mod: replace github.com/ubiquex/ubx-sdk-azure/sdk/go => ./local-sdk/azure_machinelearningservices_openapi/sdk/go
    package main

    import (
    	machinelearningservices "github.com/ubiquex/ubx-sdk-azure/sdk/go/azure_machinelearningservices_openapi/machinelearningservices"
    	ubx "github.com/ubiquex/ubx-sdk-go/runtime"
    )

    func main() {
    	ubx.Main(ubx.Stack("payments", func() {
    		ubx.Intent(ubx.IntentInfo{Summary: "payments own openapi schedule"})

    		ubx.Resource(machinelearningservices.OpenapiSchedule, "ci-runner", machinelearningservices.OpenapiScheduleConfig{
    			Name:       "ci-runner",
    			Properties: map[string]any{"action": map[string]any{"action_type": "example"}},
    		})
    	}))
    }
    ```
